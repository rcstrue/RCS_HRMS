  1107→  1107→    // ── PT calculation (state-aware, slab-based) ─────────────────
  1108→  1108→    function calculatePT(grossSalary, ptApp) {
  1109→  1109→        if (!ptApp) return 0;
  1110→  1110→        // If DB slabs exist, use them
  1111→  1111→        if (PT_SLABS.length > 0) {
  1112→  1112→            for (var i = 0; i < PT_SLABS.length; i++) {
  1113→  1113→                var slab = PT_SLABS[i];
  1114→  1114→                var minG = parseFloat(slab.min_gross);
  1115→  1115→                var maxG = parseFloat(slab.max_gross);
  1116→  1116→                if (grossSalary > minG && (isNaN(maxG) || maxG === null || grossSalary <= maxG)) {
  1117→  1117→                    return parseFloat(slab.pt_amount);
  1118→  1118→                }
  1119→  1119→            }
  1120→  1120→            return 0;
  1121→  1121→        }
  1122→  1122→        // Fallback: state-code-based hardcoded slabs (same as class.payroll.php)
  1123→  1123→        var st = (UNIT_STATE || '').toUpperCase();
  1124→  1124→        if (st === 'MH' || st === 'MAHARASHTRA') {
  1125→  1125→            return grossSalary > 10000 ? 200 : 0;
  1126→  1126→        }
  1127→  1127→        if (st === 'KA' || st === 'KARNATAKA') {
  1128→  1128→            if (grossSalary > 15000) return 200;
  1129→  1129→            if (grossSalary > 10000) return 150;
  1130→  1130→            return 0;
  1131→  1131→        }
  1132→  1132→        if (st === 'TN' || st === 'TAMIL NADU') {
  1133→  1133→            if (grossSalary > 75000) return Math.round(1250/6 * 100) / 100;
  1134→  1134→            if (grossSalary > 50000) return Math.round(833/6 * 100) / 100;
  1135→  1135→            return 0;
  1136→  1136→        }
  1137→  1137→        if (st === 'DL' || st === 'DELHI') {
  1138→  1138→            return grossSalary > 25000 ? 200 : 0;
  1139→  1139→        }
  1140→  1140→        if (st === 'GJ' || st === 'GUJARAT') {
  1141→  1141→            return grossSalary > 12000 ? 200 : 0;
  1142→  1142→        }
  1143→  1143→        // Default: Maharashtra-style
  1144→  1144→        return grossSalary > 10000 ? 200 : 0;
  1145→  1145→    }

  1206→  1206→        var esiEmployer  = 0;
  1207→  1207→

  1214→  1214→

  1217→  1217→
  1218→  1218→        // LWF
