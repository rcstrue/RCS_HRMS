    75→// -- Build filter URL helper --
    76→function buildFilterUrl($overrides = []) {
    77→    $params = [
    78→        'employee_id' => $GLOBALS['filterEmployee'] ?? 0,
    79→        'my'          => $GLOBALS['filterMonthYear'] ?? '',
    80→    ];
    81→    $params = array_merge($params, $overrides);
    82→    $parts = [];
    83→    foreach ($params as $k => $v) {
    84→        if ($v !== '' && $v !== 0) {
    85→            $parts[] = $k . '=' . urlencode($v);
    86→        }
    87→    }
    88→    return $parts ? '&' . implode('&', $parts) : '';
    89→}
