   799→// Calculate row values
   800→function calculateRow(empId) {
   801→    // Get earnings
   802→    const basicDa = parseFloat(document.querySelector(`[name="basic_da[${empId}]"]`)?.value) || 0;
   803→    const hra = parseFloat(document.querySelector(`[name="hra[${empId}]"]`)?.value) || 0;
   804→    const leaveEncashment = parseFloat(document.querySelector(`[name="leave_encashment[${empId}]"]`)?.value) || 0;
   805→    const bonusEncashment = parseFloat(document.querySelector(`[name="bonus_encashment[${empId}]"]`)?.value) || 0;
   806→    const washingAllowance = parseFloat(document.querySelector(`[name="washing_allowance[${empId}]"]`)?.value) || 0;
   807→    
   808→    const gross = basicDa + hra + leaveEncashment + bonusEncashment + washingAllowance;
   809→    
   810→    // Get statutory checkboxes
   811→    const pfApplicable = document.querySelector(`[name="pf_applicable[${empId}]"]`)?.checked || false;
   812→    const esiApplicable = document.querySelector(`[name="esi_applicable[${empId}]"]`)?.checked || false;
   813→    const ptApplicable = document.querySelector(`[name="pt_applicable[${empId}]"]`)?.checked || false;
   814→    
   815→    // Calculate PF (12% of Basic+DA, max on 15000)
   816→    const pfBase = Math.min(basicDa, 15000);
   817→    const pf = pfApplicable ? Math.round(pfBase * 0.12 * 100) / 100 : 0;
   818→    
   819→    // Calculate ESI (0.75% of Gross if gross <= 21000)
   820→    const esi = esiApplicable && gross <= 21000 ? Math.round(gross * 0.0075 * 100) / 100 : 0;
   821→    
   822→    // PT (simplified)
   823→    const pt = ptApplicable ? (gross > 15000 ? 200 : 0) : 0;
   824→    
   825→    const totalDed = pf + esi + pt;
   826→    const netSalary = gross - totalDed;
   827→    
   828→    // Update display
   829→    const grossEl = document.querySelector(`.gross-${empId}`);
   830→    const pfEl = document.querySelector(`.pf-${empId}`);
   831→    const esiEl = document.querySelector(`.esi-${empId}`);
   832→    const totalDedEl = document.querySelector(`.totalded-${empId}`);
   833→    const netEl = document.querySelector(`.netsalary-${empId}`);
   834→    
   835→    if (grossEl) grossEl.textContent = gross.toFixed(2);
   836→    if (pfEl) pfEl.textContent = pf.toFixed(2);
   837→    if (esiEl) esiEl.textContent = esi.toFixed(2);
   838→    if (totalDedEl) totalDedEl.textContent = totalDed.toFixed(2);
   839→    if (netEl) netEl.textContent = netSalary.toFixed(2);
   840→    
   841→    // Update totals
   842→    calculateTotals();
   843→}
