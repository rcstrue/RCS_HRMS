   226→   226→    /**
   227→   227→     * Process payroll for period with selective filters
   228→   228→     * 
   229→   229→     * @param int $periodId The payroll period ID
   230→   230→     * @param array $filters Optional filters for selective processing
   231→   231→     *                       - client_id: Process only this client's employees
   232→   232→     *                       - unit_id: Process only this unit's employees
   233→   233→     *                       - employee_codes: Array of specific employee codes
   234→   234→     *                       - recalculate_dirty: Only recalculate dirty records
   235→   235→     * @return array Result with success status and processed count
   236→   236→     */
   237→   237→    public function processPayroll($periodId, $filters = []) {
   238→   238→        // Get period info
   239→   239→        $period = $this->db->fetch(
   240→   240→            "SELECT * FROM payroll_periods WHERE id = :id",
   241→   241→            ['id' => $periodId]
   242→   242→        );
   243→   243→
   244→   244→        if (!$period) {
   245→   245→            return ['success' => false, 'message' => 'Payroll period not found.'];
   246→   246→        }
   247→   247→
   248→   248→        // Check if period is frozen
   249→   249→        if ($period['status'] === 'Frozen' || $period['status'] === 'Locked') {
   250→   250→            return ['success' => false, 'message' => 'Payroll period is frozen. Cannot process.'];
   251→   251→        }
   252→   252→
   253→   253→        // Ensure payroll table has extra_days_amount column
   254→   254→        try {
   255→   255→            $checkColumn = $this->db->fetch("SHOW COLUMNS FROM payroll LIKE 'extra_days_amount'");
   256→   256→            if (!$checkColumn) {
   257→   257→                $this->db->exec("ALTER TABLE payroll ADD COLUMN extra_days_amount DECIMAL(10,2) DEFAULT 0.00 AFTER overtime_amount");
   258→   258→            }
   259→   259→        } catch (Exception $e) {
   260→   260→            // Column might already exist or table might not have records
   261→   261→        }
   262→   262→
   263→   263→        // Build employee query with filters
   264→   264→        $sql = "SELECT e.id, e.employee_code, e.full_name, e.date_of_joining,
   265→   265→                       e.date_of_leaving, e.client_id, e.unit_id, e.worker_category,
   266→   266→                       e.bank_name, e.account_number, e.ifsc_code,
   267→   267→                       c.name as client_name, u.name as unit_name,
   268→   268→                       ess.basic_da, ess.hra, ess.leave_encashment, ess.bonus_encashment, ess.washing_allowance,
   269→   269→                       ess.gross_salary, ess.pf_applicable, ess.esi_applicable,
   270→   270→                       ess.pt_applicable, ess.lwf_applicable, ess.overtime_applicable,
   271→   271→                       ess.bonus_applicable, ess.gratuity_applicable,
   272→   272→                       usf.pay_days_type, usf.ot_calculation_type, usf.ot_calculation_on, usf.ot_hours_per_day,
   273→   273→                       u.state as unit_state
   274→   274→                FROM employees e
   275→   275→                INNER JOIN employee_salary_structures ess ON e.id = ess.employee_id
   276→   276→                    AND (ess.effective_to IS NULL OR ess.effective_to >= CURDATE())
   277→   277→                LEFT JOIN clients c ON e.client_id = c.id
   278→   278→                LEFT JOIN units u ON e.unit_id = u.id
   279→   279→                LEFT JOIN unit_salary_formulas usf ON usf.unit_id = e.unit_id
   280→   280→                    AND usf.is_active = 1
   281→   281→                    AND (usf.effective_to IS NULL OR usf.effective_to >= CURDATE())
   282→   282→                WHERE e.status IN ('approved', 'active')";
   283→   283→
   284→   284→        $params = [];
   285→   285→
   286→   286→        // Apply filters for selective processing
   287→   287→        if (!empty($filters['client_id'])) {
   288→   288→            $sql .= " AND e.client_id = :client_id";
   289→   289→            $params['client_id'] = $filters['client_id'];
   290→   290→        }
   291→   291→
   292→   292→        if (!empty($filters['unit_id'])) {
   293→   293→            $sql .= " AND e.unit_id = :unit_id";
   294→   294→            $params['unit_id'] = $filters['unit_id'];
   295→   295→        }
   296→   296→
   297→   297→        // Employee codes filter - use named params to avoid mixing with positional
   298→   298→        if (!empty($filters['employee_codes']) && is_array($filters['employee_codes'])) {
   299→   299→            $empPlaceholders = [];
   300→   300→            $idx = 1;
   301→   301→            foreach ($filters['employee_codes'] as $code) {
   302→   302→                $key = 'emp_code_' . $idx++;
   303→   303→                $empPlaceholders[] = ':' . $key;
   304→   304→                $params[$key] = $code;
   305→   305→            }
   306→   306→            $sql .= " AND e.employee_code IN (" . implode(',', $empPlaceholders) . ")";
   307→   307→        }
   308→   308→
   309→   309→        // Only recalculate dirty records if specified
   310→   310→        if (!empty($filters['recalculate_dirty'])) {
   311→   311→            $sql .= " AND EXISTS (
   312→   312→                SELECT 1 FROM payroll p 
   313→   313→                WHERE p.employee_id = e.employee_code 
   314→   314→                AND p.payroll_period_id = :period_id_check
   315→   315→                AND p.payroll_dirty = 1
   316→   316→            )";
   317→   317→            $params['period_id_check'] = $periodId;
   318→   318→        }
   319→   319→
   320→   320→        $employees = $this->db->fetchAll($sql, $params);
   321→   321→
   322→   322→        $processed = 0;
   323→   323→        $errors = [];
   324→   324→        $exceptions = [];
   325→   325→        $totalGross = 0;
   326→   326→        $totalNet = 0;
   327→   327→        $totalPF = 0;
   328→   328→        $totalESI = 0;
   329→   329→
   330→   330→        $this->db->beginTransaction();
   331→   331→
   332→   332→        try {
   333→   333→            foreach ($employees as $emp) {
   334→   334→                try {
   335→   335→                    // Check for exceptions
   336→   336→                    $exceptionType = null;
   337→   337→                    
   338→   338→                    // Check if employee joined after period end
   339→   339→                    $periodEndDate = $period['end_date'];
   340→   340→                    if ($emp['date_of_joining'] && strtotime($emp['date_of_joining']) > strtotime($periodEndDate)) {
   341→   341→                        continue;
   342→   342→                    }
   343→   343→
   344→   344→                    // Check for missing salary structure
   345→   345→                    if (empty($emp['basic_da']) && empty($emp['gross_salary'])) {
   346→   346→                        $exceptionType = 'Undefined Salary';
   347→   347→                        $exceptions[] = [
   348→   348→                            'employee_id' => $emp['employee_code'],
   349→   349→                            'employee_name' => $emp['full_name'],
   350→   350→                            'type' => 'Undefined Salary',
   351→   351→                            'message' => 'No salary structure defined'
   352→   352→                        ];
   353→   353→                    }
   354→   354→
   355→   355→                    // Check for missing bank details
   356→   356→                    if (empty($emp['bank_name']) || empty($emp['account_number']) || empty($emp['ifsc_code'])) {
   357→   357→                        if (!$exceptionType) {
   358→   358→                            $exceptionType = 'Missing Bank Details';
   359→   359→                        }
   360→   360→                        $exceptions[] = [
   361→   361→                            'employee_id' => $emp['employee_code'],
   362→   362→                            'employee_name' => $emp['full_name'],
   363→   363→                            'type' => 'Missing Bank Details',
   364→   364→                            'message' => 'Bank account details incomplete'
   365→   365→                        ];
   366→   366→                    }
   367→   367→
   368→   368→                    // Get attendance summary — prefer unit_id match, fallback to any unit (backward compatible)
   369→   369→                    $attendance = $this->db->fetch(
   370→   370→                        "SELECT
   371→   371→                            total_present as present_days,
   372→   372→                            total_extra,
   373→   373→                            overtime_hours,
   374→   374→                            total_wo as weekly_offs,
   375→   375→                            total_paid_days
   376→   376→                        FROM attendance_summary
   377→   377→                        WHERE employee_id = :emp_id
   378→   378→                        AND month = :month
   379→   379→                        AND year = :year
   380→   380→                        ORDER BY unit_id = :unit_id DESC
   381→   381→                        LIMIT 1",
   382→   382→                        ['emp_id' => $emp['id'], 'unit_id' => $emp['unit_id'], 'month' => $period['month'], 'year' => $period['year']]
   383→   383→                    );
   384→   384→
   385→   385→                    // Calculate total days based on unit's pay_days_type setting
   386→   386→                    $payDaysType = $emp['pay_days_type'] ?? 'actual';
   387→   387→                    switch ($payDaysType) {
   388→   388→                        case 'fixed_30':
   389→   389→                            $totalDays = 30;
   390→   390→                            break;
   391→   391→                        case 'previous_month':
   392→   392→                            // For 26-25 cycle: use previous month's days
   393→   393→                            $prevMonth = $period['month'] - 1;
   394→   394→                            $prevYear = $period['year'];
   395→   395→                            if ($prevMonth < 1) { $prevMonth = 12; $prevYear--; }
   396→   396→                            $totalDays = (int)date('t', strtotime("$prevYear-$prevMonth-01"));
   397→   397→                            break;
   398→   398→                        case 'calendar_minus_sundays':
   399→   399→                            // Month days minus all Sundays
   400→   400→                            $calDays = (int)cal_days_in_month(CAL_GREGORIAN, $period['month'], $period['year']);
   401→   401→                            $sundays = 0;
   402→   402→                            for ($d = 1; $d <= $calDays; $d++) {
   403→   403→                                $dow = (int)date('N', strtotime(sprintf('%04d-%02d-%02d', $period['year'], $period['month'], $d)));
   404→   404→                                if ($dow === 7) $sundays++;
   405→   405→                            }
   406→   406→                            $totalDays = $calDays - $sundays;
   407→   407→                            break;
   408→   408→                        default: // 'actual'
   409→   409→                            $totalDays = $period['pay_days'] ?? 30;
   410→   410→                            break;
   411→   411→                    }
   412→   412→                    
   413→   413→                    // Use total_paid_days from attendance_summary if available, otherwise calculate
   414→   414→                    $preCalcPaidDays = floatval($attendance['total_paid_days'] ?? 0);
   415→   415→                    if ($preCalcPaidDays > 0) {
   416→   416→                        $paidDays = $preCalcPaidDays;
   417→   417→                    } else {
   418→   418→                        // Fallback: calculate from individual components
   419→   419→                        $paidDays = floatval($attendance['total_present'] ?? 0);
   420→   420→                        $paidDays += floatval($attendance['weekly_offs'] ?? 0);
   421→   421→                        $paidDays += floatval($attendance['total_extra'] ?? 0);
   422→   422→                    }
   423→   423→                    
   424→   424→                    // Check if attendance record exists (must check is_array first to avoid warnings)
   425→   425→                    // NOTE: SQL aliases total_present → present_days, total_wo → weekly_offs
   426→   426→                    $hasAttendance = is_array($attendance) && array_key_exists('present_days', $attendance) && $attendance['present_days'] !== null;
   427→   427→
   428→   428→                    // Check for missing attendance — SKIP employee, do NOT assume full month
   429→   429→                    if (!$hasAttendance) {
   430→   430→                        $exceptions[] = [
   431→   431→                            'employee_id' => $emp['employee_code'],
   432→   432→                            'employee_name' => $emp['full_name'],
   433→   433→                            'type' => 'Missing Attendance',
   434→   434→                            'message' => 'No attendance data found — employee skipped. Upload attendance first.'
   435→   435→                        ];
   436→   436→                        continue; // Skip this employee entirely
   437→   437→                    }
   438→   438→
   439→   439→                    $unpaidDays = max(0, $totalDays - $paidDays);
   440→   440→                    $overtimeHours = floatval($attendance['overtime_hours'] ?? 0);
   441→   441→
   442→   442→                    // Calculate earnings (pro-rated) - using basic_da as combined column
   443→   443→                    $basicDa = round(($emp['basic_da'] ?? 0) * $paidDays / $totalDays, 2);
   444→   444→                    $hra = round(($emp['hra'] ?? 0) * $paidDays / $totalDays, 2);

   453→   453→

  1257→  1257→     * Common PT slabs across India (monthly)
  1258→  1258→     */
  1259→  1259→    private function calculatePT($gross, $state = 'GJ') {
  1260→  1260→        // Read from professional_tax_rates DB table instead of hardcoded slabs
  1261→  1261→        try {
  1262→  1262→            $rate = $this->db->fetch(
  1263→  1263→                "SELECT ptr.pt_amount
  1264→  1264→                 FROM professional_tax_rates ptr
  1265→  1265→                 JOIN states s ON ptr.state_id = s.id
  1266→  1266→                 WHERE (s.state_code = :state OR s.state_name = :state)
  1267→  1267→                   AND ptr.salary_from <= :gross
  1268→  1268→                   AND (ptr.salary_to IS NULL OR ptr.salary_to >= :gross)
  1269→  1269→                   AND ptr.is_active = 1
  1270→  1270→                 ORDER BY ptr.salary_from DESC
  1271→  1271→                 LIMIT 1",
  1272→  1272→                ['state' => $state, 'gross' => $gross]
  1273→  1273→            );
  1274→  1274→            return $rate ? (float)$rate['pt_amount'] : 0;
  1275→  1275→        } catch (Exception $e) {
  1276→  1276→            // Fallback to simple default if DB query fails
  1277→  1277→            if ($gross > 12000) return 200;
  1278→  1278→            return 0;
  1279→  1279→        }
  1280→  1280→    }
