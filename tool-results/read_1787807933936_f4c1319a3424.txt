   484→function calcDays() {
   485→    const from = document.getElementById('fromDate').value;
   486→    const to = document.getElementById('toDate').value;
   487→    if (from && to) {
   488→        const d1 = new Date(from), d2 = new Date(to);
   489→        const diff = Math.ceil((d2 - d1) / (1000*60*60*24)) + 1;
   490→        document.getElementById('totalDays').value = Math.max(0.5, diff);
   491→    }
   492→}
