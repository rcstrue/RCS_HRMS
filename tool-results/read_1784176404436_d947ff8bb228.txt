    44→function _ensureTables(mysqli $conn): void
    45→{
    46→    // Main visits table
    47→    $conn->query("
    48→        CREATE TABLE IF NOT EXISTS ess_unit_visits (
    49→            id INT AUTO_INCREMENT PRIMARY KEY,
    50→            employee_id VARCHAR(50) NOT NULL,
    51→            unit_id INT NOT NULL,
    52→            visit_number TINYINT NOT NULL COMMENT '1=first visit, 2=second visit',
    53→            visit_month INT NOT NULL COMMENT '1-12',
    54→            visit_year INT NOT NULL,
    55→            document_url VARCHAR(500) DEFAULT '',
    56→            document_type VARCHAR(20) DEFAULT 'image' COMMENT 'image or pdf',
    57→            notes TEXT DEFAULT NULL,
    58→            status ENUM('submitted','approved','rejected') DEFAULT 'submitted',
    59→            rejection_reason TEXT DEFAULT NULL,
    60→            approved_by VARCHAR(50) DEFAULT NULL,
    61→            approved_at DATETIME DEFAULT NULL,
    62→            total_score DECIMAL(8,2) DEFAULT 0,
    63→            max_score DECIMAL(8,2) DEFAULT 0,
    64→            score_percent DECIMAL(5,2) DEFAULT 0,
    65→            created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    66→            updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    67→            INDEX idx_employee_month (employee_id, visit_month, visit_year),
    68→            INDEX idx_unit_month (unit_id, visit_month, visit_year),
    69→            INDEX idx_status (status)
    70→        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci
    71→    ");
    72→
    73→    // Add columns that may not exist in old schema
    74→    $cols = $conn->query("SHOW COLUMNS FROM ess_unit_visits LIKE 'rejection_reason'")->num_rows;
    75→    if ($cols == 0) {
    76→        $conn->query("ALTER TABLE ess_unit_visits ADD COLUMN rejection_reason TEXT DEFAULT NULL AFTER status");
    77→        $conn->query("ALTER TABLE ess_unit_visits ADD COLUMN approved_by VARCHAR(50) DEFAULT NULL AFTER rejection_reason");
    78→        $conn->query("ALTER TABLE ess_unit_visits ADD COLUMN approved_at DATETIME DEFAULT NULL AFTER approved_by");
    79→        $conn->query("ALTER TABLE ess_unit_visits ADD COLUMN total_score DECIMAL(8,2) DEFAULT 0 AFTER approved_at");
    80→        $conn->query("ALTER TABLE ess_unit_visits ADD COLUMN max_score DECIMAL(8,2) DEFAULT 0 AFTER total_score");
    81→        $conn->query("ALTER TABLE ess_unit_visits ADD COLUMN score_percent DECIMAL(5,2) DEFAULT 0 AFTER max_score");
    82→        $conn->query("ALTER TABLE ess_unit_visits ADD INDEX idx_status (status)");
    83→    }
    84→
    85→    // Drop old unique key if exists and recreate
    86→    try {
    87→        $conn->query("ALTER TABLE ess_unit_visits DROP INDEX uk_visit");
    88→    } catch (\Throwable $e) { /* index may not exist */ }
    89→    try {
    90→        $conn->query("ALTER TABLE ess_unit_visits ADD UNIQUE KEY uk_visit (employee_id, unit_id, visit_month, visit_year, visit_number)");
    91→    } catch (\Throwable $e) { /* may already exist */ }
    92→
    93→    // Checklist items table
    94→    $conn->query("
    95→        CREATE TABLE IF NOT EXISTS ess_visit_checklist_items (
    96→            id INT AUTO_INCREMENT PRIMARY KEY,
    97→            visit_id INT NOT NULL,
    98→            checklist_item_id INT NOT NULL,
    99→            category_id INT NOT NULL,
   100→            status ENUM('yes','no','na') DEFAULT 'yes',
   101→            remarks TEXT DEFAULT NULL,
   102→            photo_url VARCHAR(500) DEFAULT NULL,
   103→            created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

   111→            id INT AUTO_INCREMENT PRIMARY KEY,
   112→            visit_id INT NOT NULL,
   113→            action VARCHAR(50) NOT NULL,
   114→            performed_by VARCHAR(50) NOT NULL,
   115→            details TEXT DEFAULT NULL,
   116→            created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
   117→            INDEX idx_visit (visit_id),
   118→            INDEX idx_action (action)
   119→        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci

   127→function _calculateScore(mysqli $conn, int $visitId): array
   128→{
   129→    $stmt = $conn->prepare('
   130→        SELECT vci.status, ci.weight
   131→        FROM ess_visit_checklist_items vci
   132→        JOIN ess_checklist_items ci ON ci.id = vci.checklist_item_id
   133→        WHERE vci.visit_id = ?
   134→    ');
   135→    $stmt->bind_param('i', $visitId);
   136→    $stmt->execute();
   137→    $result = $stmt->get_result();
   138→
   139→    $total = 0;
   140→    $max = 0;
   141→    while ($row = $result->fetch_assoc()) {
   142→        $w = (float)($row['weight'] ?? 1);
   143→        if ($row['status'] !== 'na') {
   144→            $max += $w;
   145→            if ($row['status'] === 'yes') {
   146→                $total += $w;
   147→            }
   148→        }
   149→    }
   150→    $stmt->close();
   151→
   152→    $percent = $max > 0 ? round(($total / $max) * 100, 2) : 0;
   153→
   154→    // Update visit record
   155→    $upd = $conn->prepare('UPDATE ess_unit_visits SET total_score = ?, max_score = ?, score_percent = ? WHERE id = ?');
   156→    $upd->bind_param('dddi', $total, $max, $percent, $visitId);
   157→    $upd->execute();
   158→    $upd->close();
   159→
   160→    return ['total' => $total, 'max' => $max, 'percent' => $percent];
   161→}

   194→function _handleGet(): void
   195→{
   196→    $authId = requireAuth();
   197→    $conn = getDbConnection();
   198→    _ensureTables($conn);
   199→
   200→    $view = $_GET['view'] ?? '';

   211→function _handleGetList(mysqli $conn, string $authId): void
   212→{
   213→    $employeeId = $_GET['employee_id'] ?? $authId;
   214→    $month = (int)($_GET['month'] ?? 0);
   215→    $year = (int)($_GET['year'] ?? 0);
   216→    $unitId = (int)($_GET['unit_id'] ?? 0);
   217→    $status = $_GET['status'] ?? '';
   218→    $includeChecklist = ($_GET['include_checklist'] ?? '') === '1';
   219→    [$page, $limit, $offset] = getPaginationParams();
   220→
   221→    // Build where
   222→    $where = 'WHERE v.employee_id = ?';
   223→    $types = 's';
   224→    $params = [$employeeId];
   225→
   226→    if ($month > 0) { $where .= ' AND v.visit_month = ?'; $types .= 'i'; $params[] = $month; }
   227→    if ($year > 0) { $where .= ' AND v.visit_year = ?'; $types .= 'i'; $params[] = $year; }
   228→    if ($unitId > 0) { $where .= ' AND v.unit_id = ?'; $types .= 'i'; $params[] = $unitId; }
   229→    if ($status) { $where .= ' AND v.status = ?'; $types .= 's'; $params[] = $status; }
   230→
   231→    // Count
   232→    $countStmt = $conn->prepare("SELECT COUNT(*) AS total FROM ess_unit_visits v {$where}");
   233→    bindDynamicParams($countStmt, $types, $params);
   234→    $countStmt->execute();
   235→    $total = (int)$countStmt->get_result()->fetch_assoc()['total'];
   236→    $countStmt->close();
   237→
   238→    // Fetch
   239→    $dataQuery = "
   240→        SELECT v.id, v.employee_id, v.unit_id, v.visit_number, v.visit_month, v.visit_year,
   241→               v.document_url, v.document_type, v.notes, v.status, v.rejection_reason,
   242→               v.approved_by, v.approved_at, v.total_score, v.max_score, v.score_percent,
   243→               v.created_at, v.updated_at,
   244→               u.name AS unit_name, c.name AS client_name,
   245→               e.full_name AS employee_name, e.employee_code
   246→        FROM ess_unit_visits v
   247→        LEFT JOIN units u ON u.id = v.unit_id
   248→        LEFT JOIN clients c ON c.id = u.client_id
   249→        LEFT JOIN employees e ON e.id = v.employee_id
   250→        {$where}
   251→        ORDER BY v.visit_year DESC, v.visit_month DESC, v.created_at DESC
   252→        LIMIT ? OFFSET ?
   253→    ";
   254→    $dataTypes = $types . 'ii';
   255→    $dataParams = [...$params, $limit, $offset];
   256→
   257→    $stmt = $conn->prepare($dataQuery);
   258→    bindDynamicParams($stmt, $dataTypes, $dataParams);
   259→    $stmt->execute();
   260→    $result = $stmt->get_result();
   261→
   262→    $visits = [];
   263→    while ($row = $result->fetch_assoc()) {
   264→        $visit = [
   265→            'id' => (int)$row['id'],
   266→            'employee_id' => (int)$row['employee_id'],
   267→            'employee_name' => $row['employee_name'] ?? '',
   268→            'employee_code' => $row['employee_code'] ?? '',
   269→            'unit_id' => (int)$row['unit_id'],
   270→            'unit_name' => $row['unit_name'] ?? '',
   271→            'client_name' => $row['client_name'] ?? '',
   272→            'visit_number' => (int)$row['visit_number'],
   273→            'visit_month' => (int)$row['visit_month'],
   274→            'visit_year' => (int)$row['visit_year'],
   275→            'document_url' => $row['document_url'] ?? '',
   276→            'document_type' => $row['document_type'] ?? 'image',
   277→            'notes' => $row['notes'] ?? '',
   278→            'status' => $row['status'] ?? 'submitted',
   279→            'rejection_reason' => $row['rejection_reason'] ?? '',
   280→            'approved_by' => $row['approved_by'] ? (int)$row['approved_by'] : null,
   281→            'approved_at' => $row['approved_at'] ?? null,
   282→            'total_score' => (float)($row['total_score'] ?? 0),
   283→            'max_score' => (float)($row['max_score'] ?? 0),
   284→            'score_percent' => (float)($row['score_percent'] ?? 0),
   285→            'created_at' => $row['created_at'] ?? '',
   286→            'updated_at' => $row['updated_at'] ?? '',
   287→        ];
   288→
   289→        if ($includeChecklist) {
   290→            $visit['checklist_items'] = _fetchChecklistItems($conn, (int)$row['id']);
   291→        }
   292→
   293→        $visits[] = $visit;
   294→    }
   295→    $stmt->close();
   296→    $conn->close();
   297→
   298→    jsonOutput([
   299→        'success' => true,
   300→        'data' => [
   301→            'items' => $visits,
   302→            ...buildPagination($total, $page, $limit)
   303→        ]
   304→    ]);
   305→}

   307→function _handleGetDetail(mysqli $conn, string $authId): void
   308→{
   309→    $visitId = (int)($_GET['id'] ?? 0);
   310→    if ($visitId <= 0) {
   311→        jsonOutput(['success' => false, 'error' => 'Visit ID is required'], 400);
   312→    }
   313→
   314→    $stmt = $conn->prepare("
   315→        SELECT v.*, u.name AS unit_name, c.name AS client_name,
   316→               e.full_name AS employee_name, e.employee_code, e.mobile_number AS employee_mobile
   317→        FROM ess_unit_visits v
   318→        LEFT JOIN units u ON u.id = v.unit_id
   319→        LEFT JOIN clients c ON c.id = u.client_id
   320→        LEFT JOIN employees e ON e.id = v.employee_id
   321→        WHERE v.id = ?
   322→    ");
   323→    $stmt->bind_param('i', $visitId);
   324→    $stmt->execute();
   325→    $row = $stmt->get_result()->fetch_assoc();
   326→    $stmt->close();
   327→
   328→    if (!$row) {
   329→        $conn->close();
   330→        jsonOutput(['success' => false, 'error' => 'Visit not found'], 404);
   331→    }
   332→
   333→    // Recalculate score to ensure it's current
   334→    $scores = _calculateScore($conn, $visitId);
   335→
   336→    $visit = [
   337→        'id' => (int)$row['id'],
   338→        'employee_id' => (int)$row['employee_id'],
   339→        'employee_name' => $row['employee_name'] ?? '',
   340→        'employee_code' => $row['employee_code'] ?? '',
   341→        'employee_mobile' => $row['employee_mobile'] ?? '',
   342→        'unit_id' => (int)$row['unit_id'],
   343→        'unit_name' => $row['unit_name'] ?? '',
   344→        'client_name' => $row['client_name'] ?? '',
   345→        'visit_number' => (int)$row['visit_number'],
   346→        'visit_month' => (int)$row['visit_month'],
   347→        'visit_year' => (int)$row['visit_year'],
   348→        'document_url' => $row['document_url'] ?? '',
   349→        'document_type' => $row['document_type'] ?? 'image',
   350→        'notes' => $row['notes'] ?? '',
   351→        'status' => $row['status'] ?? 'submitted',
   352→        'rejection_reason' => $row['rejection_reason'] ?? '',
   353→        'approved_by' => $row['approved_by'] ? (int)$row['approved_by'] : null,
   354→        'approved_at' => $row['approved_at'] ?? null,
   355→        'total_score' => $scores['total'],
   356→        'max_score' => $scores['max'],
   357→        'score_percent' => $scores['percent'],
   358→        'checklist_items' => _fetchChecklistItems($conn, $visitId),
   359→        'created_at' => $row['created_at'] ?? '',
   360→        'updated_at' => $row['updated_at'] ?? '',
   361→    ];
   362→
   363→    // Fetch audit log
   364→    $auditStmt = $conn->prepare('SELECT * FROM ess_visit_audit_log WHERE visit_id = ? ORDER BY created_at ASC');
   365→    $auditStmt->bind_param('i', $visitId);
   366→    $auditStmt->execute();
   367→    $auditResult = $auditStmt->get_result();
   368→    $auditLog = [];
   369→    while ($aRow = $auditResult->fetch_assoc()) {
   370→        $auditLog[] = [
   371→            'id' => (int)$aRow['id'],
   372→            'action' => $aRow['action'],
   373→            'performed_by' => $aRow['performed_by'],
   374→            'details' => $aRow['details'] ?? '',
   375→            'created_at' => $aRow['created_at'],
   376→        ];
   377→    }
   378→    $auditStmt->close();
   379→    $visit['audit_log'] = $auditLog;
   380→
   381→    $conn->close();
   382→    jsonOutput(['success' => true, 'data' => $visit]);
   383→}

   385→function _handleGetDashboard(mysqli $conn, string $authId): void
   386→{
   387→    $employeeId = (int)($_GET['employee_id'] ?? $authId);
   388→    $now = new \DateTime('now', new \DateTimeZone('Asia/Kolkata'));
   389→    $currentMonth = (int)$now->format('n');
   390→    $currentYear = (int)$now->format('Y');
   391→
   392→    // Basic counts
   393→    $counts = $conn->query("
   394→        SELECT
   395→            COUNT(*) AS total_visits,
   396→            SUM(CASE WHEN status = 'submitted' THEN 1 ELSE 0 END) AS pending_visits,
   397→            SUM(CASE WHEN status = 'approved' THEN 1 ELSE 0 END) AS approved_visits,
   398→            SUM(CASE WHEN status = 'rejected' THEN 1 ELSE 0 END) AS rejected_visits,
   399→            SUM(CASE WHEN visit_month = {$currentMonth} AND visit_year = {$currentYear} THEN 1 ELSE 0 END) AS this_month_visits
   400→        FROM ess_unit_visits WHERE employee_id = {$employeeId}
   401→    ")->fetch_assoc();
   402→
   403→    // This month avg score
   404→    $monthAvg = $conn->query("
   405→        SELECT AVG(score_percent) AS avg_score
   406→        FROM ess_unit_visits
   407→        WHERE employee_id = {$employeeId} AND visit_month = {$currentMonth} AND visit_year = {$currentYear}
   408→            AND status IN ('approved','submitted') AND max_score > 0
   409→    ")->fetch_assoc();
   410→
   411→    // Score trend - last 6 months
   412→    $trend = [];
   413→    for ($i = 5; $i >= 0; $i--) {
   414→        $d = clone $now;
   415→        $d->modify("-{$i} months");
   416→        $m = (int)$d->format('n');
   417→        $y = (int)$d->format('Y');
   418→        $label = $d->format('M Y');
   419→        $avgRow = $conn->query("
   420→            SELECT AVG(score_percent) AS avg_score
   421→            FROM ess_unit_visits
   422→            WHERE employee_id = {$employeeId} AND visit_month = {$m} AND visit_year = {$y}
   423→                AND status IN ('approved','submitted') AND max_score > 0
   424→        ")->fetch_assoc();
   425→        $trend[] = ['month' => $label, 'avg_score' => round((float)($avgRow['avg_score'] ?? 0), 1)];
   426→    }
   427→
   428→    // Category scores
   429→    $catScores = $conn->query("
   430→        SELECT cc.name AS category, AVG(
   431→            CASE WHEN vci.status = 'yes' THEN ci.weight ELSE 0 END
   432→        ) / NULLIF(AVG(CASE WHEN vci.status != 'na' THEN ci.weight ELSE NULL END), 0) * 100 AS avg_score
   433→        FROM ess_visit_checklist_items vci
   434→        JOIN ess_checklist_items ci ON ci.id = vci.checklist_item_id
   435→        JOIN ess_checklist_categories cc ON cc.id = ci.category_id
   436→        JOIN ess_unit_visits v ON v.id = vci.visit_id
   437→        WHERE v.employee_id = {$employeeId} AND v.status IN ('approved','submitted')
   438→        GROUP BY cc.id, cc.name
   439→        ORDER BY cc.display_order
   440→    ");
   441→    $categoryScores = [];
   442→    while ($cr = $catScores->fetch_assoc()) {
   443→        $categoryScores[] = [
   444→            'category' => $cr['category'] ?? '',
   445→            'avg_score' => round((float)($cr['avg_score'] ?? 0), 1),
   446→        ];
   447→    }
   448→
   449→    // Unit compliance
   450→    $unitComp = $conn->query("
   451→        SELECT u.name AS unit_name, AVG(v.score_percent) AS score, COUNT(*) AS visits
   452→        FROM ess_unit_visits v
   453→        JOIN units u ON u.id = v.unit_id
   454→        WHERE v.employee_id = {$employeeId} AND v.status IN ('approved','submitted') AND v.max_score > 0
   455→        GROUP BY v.unit_id, u.name
   456→        ORDER BY score DESC
   457→        LIMIT 10
   458→    ");
   459→    $unitCompliance = [];
   460→    while ($ur = $unitComp->fetch_assoc()) {
   461→        $unitCompliance[] = [
   462→            'unit_name' => $ur['unit_name'] ?? '',
   463→            'score' => round((float)($ur['score'] ?? 0), 1),
   464→            'visits' => (int)$ur['visits'],
   465→        ];
   466→    }
   467→
   468→    $conn->close();
   469→
   470→    jsonOutput([
   471→        'success' => true,
   472→        'data' => [
   473→            'total_visits' => (int)($counts['total_visits'] ?? 0),
   474→            'pending_visits' => (int)($counts['pending_visits'] ?? 0),
   475→            'approved_visits' => (int)($counts['approved_visits'] ?? 0),
   476→            'rejected_visits' => (int)($counts['rejected_visits'] ?? 0),
   477→            'this_month_visits' => (int)($counts['this_month_visits'] ?? 0),
   478→            'this_month_score_avg' => round((float)($monthAvg['avg_score'] ?? 0), 1),
   479→            'score_trend' => $trend,
   480→            'category_scores' => $categoryScores,
   481→            'unit_compliance' => $unitCompliance,
   482→        ]
   483→    ]);
   484→}

   486→// ══════════════════════════════════════════════════════════════════════════
   487→// Fetch checklist items for a visit (with category/item names)
   488→// ══════════════════════════════════════════════════════════════════════════
   489→
   490→function _fetchChecklistItems(mysqli $conn, int $visitId): array
   491→{
   492→    $stmt = $conn->prepare("

   530→{
   531→    $employeeId = requireAuth();

   537→    if (($input['action'] ?? '') === 'send_email') {
   538→        $visitId = (int)($input['visit_id'] ?? 0);

   540→            jsonOutput(['success' => false, 'error' => 'Visit ID required'], 400);

   543→

   545→        try {
   546→            $emailSent = _sendVisitEmailInline($conn, $visitId);
   547→        } catch (\Throwable $e) {
   548→            $emailSent = false;
   549→        }

   552→        jsonOutput(['success' => true, 'message' => $emailSent ? 'Report email sent successfully' : 'Email will be sent shortly']);

   556→    // Validate

   558→    $visitMonth = (int)($input['visit_month'] ?? 0);
   559→    $visitYear = (int)($input['visit_year'] ?? 0);
   560→    $visitNumber = (int)($input['visit_number'] ?? 0);
   561→    $documentUrl = trim($input['document_url'] ?? '');
   562→    $documentType = trim($input['document_type'] ?? 'image');
   563→    $notes = trim($input['notes'] ?? '');
   564→    $checklistItems = $input['checklist_items'] ?? [];
   565→
   566→    if ($unitId <= 0) jsonOutput(['success' => false, 'error' => 'Unit is required'], 400);
   567→    if ($visitMonth < 1 || $visitMonth > 12) jsonOutput(['success' => false, 'error' => 'Invalid month (1-12 required)'], 400);
   568→    if ($visitYear < 2020 || $visitYear > 2099) jsonOutput(['success' => false, 'error' => 'Invalid year'], 400);
   569→    if ($visitNumber < 1 || $visitNumber > 2) jsonOutput(['success' => false, 'error' => 'Visit number must be 1 or 2'], 400);
   570→    if (empty($documentUrl)) jsonOutput(['success' => false, 'error' => 'Document (JPG/PDF) is required'], 400);
   571→
   572→    // Duplicate check
   573→    $dupStmt = $conn->prepare('
   574→        SELECT id FROM ess_unit_visits
   575→        WHERE employee_id = ? AND unit_id = ? AND visit_month = ? AND visit_year = ? AND visit_number = ?
   576→    ');

   584→            'success' => false,
   585→            'error' => 'A ' . _visitLabel($visitNumber) . ' for this unit in ' . _monthName($visitMonth) . ' ' . $visitYear . ' already exists',
   586→            'existing_id' => (int)$existing['id']
   587→        ], 409);
   588→    }
   589→
   590→    // Insert visit
   591→    $stmt = $conn->prepare('
   592→        INSERT INTO ess_unit_visits (employee_id, unit_id, visit_number, visit_month, visit_year, document_url, document_type, notes, status)
   593→        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
   594→    ');
   595→    bindDynamicParams($stmt, 'siiiissss', [
   596→        $employeeId, $unitId, $visitNumber, $visitMonth, $visitYear,
   597→        $documentUrl, $documentType, $notes, 'submitted'
   598→    ]);
   599→    $stmt->execute();
   600→    $newId = $stmt->insert_id;
   601→    $stmt->close();

   603→    // Insert checklist items
   604→    if (is_array($checklistItems) && count($checklistItems) > 0) {
   605→        $insStmt = $conn->prepare('
   606→            INSERT INTO ess_visit_checklist_items (visit_id, checklist_item_id, category_id, status, remarks, photo_url)
   607→            VALUES (?, ?, ?, ?, ?, ?)
   608→        ');
   609→        foreach ($checklistItems as $item) {
   610→            $itemId = (int)($item['checklist_item_id'] ?? 0);
   611→            $catId = (int)($item['category_id'] ?? 0);
   612→            $status = in_array($item['status'] ?? '', ['yes', 'no', 'na']) ? $item['status'] : 'yes';
   613→            $remarks = trim($item['remarks'] ?? '');
   614→            $photoUrl = trim($item['photo_url'] ?? '');
   615→            $insStmt->bind_param('iiisss', $newId, $itemId, $catId, $status, $remarks, $photoUrl);
   616→            $insStmt->execute();
   617→        }
   618→        $insStmt->close();
   619→    }
   620→
   621→    // Calculate score
   622→    $scores = _calculateScore($conn, $newId);
   623→
   624→    // Audit log
   625→    _logAudit($conn, $newId, 'created', $employeeId, "Visit submitted with {$scores['percent']}% score");
   626→
   627→    // Fetch unit info for response
   628→    $unitStmt = $conn->prepare('SELECT u.name AS unit_name, c.name AS client_name FROM units u LEFT JOIN clients c ON c.id = u.client_id WHERE u.id = ?');
   629→    $unitStmt->bind_param('i', $unitId);
   630→    $unitStmt->execute();
   631→    $unitInfo = $unitStmt->get_result()->fetch_assoc();
   632→    $unitStmt->close();
   633→
   634→    // Fetch employee info
   635→    $empStmt = $conn->prepare('SELECT full_name, employee_code FROM employees WHERE id = ?');
   636→    $empStmt->bind_param('s', $employeeId);
   637→    $empStmt->execute();
   638→    $empInfo = $empStmt->get_result()->fetch_assoc();
   639→    $empStmt->close();
   640→    $conn->close();
   641→
   642→    jsonOutput([
   643→        'success' => true,
   644→        'data' => [
   645→            'id' => $newId,
   646→            'employee_id' => (int)$employeeId,
   647→            'employee_name' => $empInfo['full_name'] ?? '',
   648→            'employee_code' => $empInfo['employee_code'] ?? '',
   649→            'unit_id' => $unitId,
   650→            'unit_name' => $unitInfo['unit_name'] ?? '',
   651→            'client_name' => $unitInfo['client_name'] ?? '',
   652→            'visit_number' => $visitNumber,
   653→            'visit_month' => $visitMonth,
   654→            'visit_year' => $visitYear,
   655→            'document_url' => $documentUrl,
   656→            'document_type' => $documentType,
   657→            'notes' => $notes,
   658→            'status' => 'submitted',
   659→            'total_score' => $scores['total'],
   660→            'max_score' => $scores['max'],
   661→            'score_percent' => $scores['percent'],
   662→            'message' => 'Unit visit ' . _visitLabel($visitNumber) . ' submitted successfully'
   663→        ]

   668→// PUT: Approve / Reject
   669→// ══════════════════════════════════════════════════════════════════════════
   670→
   671→function _handlePut(): void
   672→{
   673→    $employeeId = requireAuth();
   674→    $input = getInput();
   675→    $conn = getDbConnection();
   676→    _ensureTables($conn);
   677→
   678→    $visitId = (int)($input['id'] ?? 0);
   679→    $action = $input['action'] ?? '';
   680→
   681→    if ($visitId <= 0) jsonOutput(['success' => false, 'error' => 'Visit ID is required'], 400);
   682→    if (!in_array($action, ['approve', 'reject'])) jsonOutput(['success' => false, 'error' => 'Action must be approve or reject'], 400);
   683→
   684→    // Verify visit exists
   685→    $checkStmt = $conn->prepare('SELECT id, status, employee_id FROM ess_unit_visits WHERE id = ?');
   686→    $checkStmt->bind_param('i', $visitId);
   687→    $checkStmt->execute();
   688→    $visit = $checkStmt->get_result()->fetch_assoc();
   689→    $checkStmt->close();
   690→
   691→    if (!$visit) { $conn->close(); jsonOutput(['success' => false, 'error' => 'Visit not found'], 404); }
   692→    if ($visit['status'] !== 'submitted') { $conn->close(); jsonOutput(['success' => false, 'error' => 'Only submitted visits can be ' . $action . 'd'], 400); }
   693→
   694→    if ($action === 'approve') {
   695→        $stmt = $conn->prepare('UPDATE ess_unit_visits SET status = ?, approved_by = ?, approved_at = NOW() WHERE id = ?');
   696→        $stmt->bind_param('ssi', 'approved', $employeeId, $visitId);
   697→        $stmt->execute();
   698→        $stmt->close();
   699→        _logAudit($conn, $visitId, 'approved', $employeeId);
   700→    } else {
   701→        $rejectionReason = trim($input['rejection_reason'] ?? '');
   702→        $stmt = $conn->prepare('UPDATE ess_unit_visits SET status = ?, rejection_reason = ?, approved_by = ?, approved_at = NOW() WHERE id = ?');
   703→        $stmt->bind_param('sssi', 'rejected', $rejectionReason, $employeeId, $visitId);
   704→        $stmt->execute();
   705→        $stmt->close();
   706→        _logAudit($conn, $visitId, 'rejected', $employeeId, $rejectionReason);
   707→    }
   708→
   709→    // Fetch updated visit
   710→    $fetchStmt = $conn->prepare("
   711→        SELECT v.*, u.name AS unit_name, c.name AS client_name,
   712→               e.full_name AS employee_name, e.employee_code
   713→        FROM ess_unit_visits v
   714→        LEFT JOIN units u ON u.id = v.unit_id
   715→        LEFT JOIN clients c ON c.id = u.client_id
   716→        LEFT JOIN employees e ON e.id = v.employee_id
   717→        WHERE v.id = ?
   718→    ");
   719→    $fetchStmt->bind_param('i', $visitId);
   720→    $fetchStmt->execute();
   721→    $row = $fetchStmt->get_result()->fetch_assoc();
   722→    $fetchStmt->close();
   723→    $conn->close();
   724→
   725→    jsonOutput([
   726→        'success' => true,
   727→        'data' => [
   728→            'id' => (int)$row['id'],
   729→            'status' => $row['status'],
   730→            'message' => 'Visit ' . $action . 'd successfully'
   731→        ]
   732→    ]);
   733→}

   776→// Email Helper (inline)
   777→// ══════════════════════════════════════════════════════════════════════════
   778→
   779→function _sendVisitEmailInline(mysqli $conn, int $visitId): bool
   780→{
   781→    // Fetch visit with employee email
   782→    $stmt = $conn->prepare("
   783→        SELECT v.*, u.name AS unit_name, c.name AS client_name,
   784→               e.full_name AS employee_name, e.employee_code, e.email AS employee_email
   785→        FROM ess_unit_visits v
   786→        LEFT JOIN units u ON u.id = v.unit_id
   787→        LEFT JOIN clients c ON c.id = u.client_id
   788→        LEFT JOIN employees e ON e.id = v.employee_id
   789→        WHERE v.id = ?
   790→    ");
   791→    $stmt->bind_param('i', $visitId);
   792→    $stmt->execute();
   793→    $visit = $stmt->get_result()->fetch_assoc();
   794→    $stmt->close();
   795→
   796→    if (!$visit || empty($visit['employee_email'])) return false;
   797→
   798→    // Fetch checklist items
   799→    $itemsStmt = $conn->prepare('
   800→        SELECT ci.status, ci.remarks,
   801→               cmi.name AS item_name, cmc.name AS category_name
   802→        FROM ess_visit_checklist_items ci
   803→        LEFT JOIN ess_checklist_items cmi ON cmi.id = ci.checklist_item_id
   804→        LEFT JOIN ess_checklist_categories cmc ON cmc.id = ci.category_id
   805→        WHERE ci.visit_id = ?
   806→        ORDER BY cmc.display_order ASC, cmi.display_order ASC
   807→    ');
   808→    $itemsStmt->bind_param('i', $visitId);
   809→    $itemsStmt->execute();
   810→    $itemsResult = $itemsStmt->get_result();
   811→    $items = [];
   812→    while ($row = $itemsResult->fetch_assoc()) {
   813→        $items[] = $row;
   814→    }
   815→    $itemsResult->free();
   816→    $itemsStmt->close();
   817→
   818→    $score = (int)($visit['score_percent'] ?? 0);
   819→    $subject = sprintf('Unit Visit Report: %s - %s %s (%d%%)',
   820→        $visit['unit_name'] ?? 'Unknown',
   821→        _monthName((int)$visit['visit_month']),
   822→        $visit['visit_year'],
   823→        $score
   824→    );
   825→
   826→    // Build plain text email
   827→    $text = "UNIT VISIT CHECKLIST REPORT\nRCS TRUE FACILITIES PVT LTD\n" . str_repeat('=', 50) . "\n\n";
   828→    $text .= "Employee: " . ($visit['employee_name'] ?? '') . "\n";
   829→    $text .= "Client/Unit: " . ($visit['client_name'] ?? '') . " / " . ($visit['unit_name'] ?? '') . "\n";
   830→    $text .= "Visit: " . ((int)$visit['visit_number'] === 1 ? 'First' : 'Second') . " Visit - " . _monthName((int)$visit['visit_month']) . " " . $visit['visit_year'] . "\n";
   831→    $text .= "Score: {$score}% (" . ($visit['total_score'] ?? 0) . "/" . ($visit['max_score'] ?? 0) . " points)\n\n";
   832→
   833→    $grouped = [];
   834→    foreach ($items as $item) {
   835→        $cat = $item['category_name'] ?? 'Unknown';
   836→        if (!isset($grouped[$cat])) $grouped[$cat] = [];
   837→        $status = $item['status'] === 'na' ? 'N/A' : ($item['status'] === 'yes' ? '[YES]' : '[NO]');
   838→        $line = "  {$status} " . ($item['item_name'] ?? '');
   839→        if (!empty($item['remarks'])) $line .= " - " . $item['remarks'];
   840→        $grouped[$cat][] = $line;
   841→    }
   842→    foreach ($grouped as $cat => $lines) {
   843→        $text .= "\n--- " . strtoupper($cat) . " ---\n" . implode("\n", $lines) . "\n";
   844→    }
   845→
   846→    if (!empty($visit['notes'])) $text .= "\nNOTES: " . $visit['notes'] . "\n";
   847→
   848→    $headers = "From: RCS ESS Portal <noreply@rcsfacility.com>\r\nReply-To: support@rcsfacility.com\r\nX-Mailer: RCS-ESS/2.0";
   849→    return mail($visit['employee_email'], $subject, $text, $headers, '-fnoreply@rcsfacility.com');
   850→}
