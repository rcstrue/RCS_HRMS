   623→function recalcTemplate() {
   624→    var net = parseFloat(document.getElementById('tplNetSalary').value) || 0;
   625→    if (net <= 0) { clearCalcResults(); return; }
   626→
   627→    // Get selected worker category (single radio)
   628→    var catEl = document.querySelector('[name="worker_cat"]:checked');
   629→    var workerCategory = catEl ? catEl.value : '';
   630→    if (!workerCategory) {
   631→        clearCalcResults();
   632→        var w = document.getElementById('calcWarnings');
   633→        w.classList.remove('d-none');
   634→        w.innerHTML = '<div class="alert alert-info py-2 small mb-0"><i class="bi bi-info-circle me-1"></i>Please select a Worker Category to look up minimum wage.</div>';
   635→        return;
   636→    }
   637→
   638→    var params = {
   639→        action: 'reverse_calc',
   640→        net_salary: net,
   641→        unit_id: UNIT_ID,
   642→        worker_category: workerCategory,
   643→        // Send explicit overrides for the 4 statutory flags (Q1=A: defaults from unit, override allowed)
   644→        pf:  document.getElementById('stat_pf').checked  ? '1' : '0',
   645→        esi: document.getElementById('stat_esi').checked ? '1' : '0',
   646→        pt:  document.getElementById('stat_pt').checked  ? '1' : '0',
   647→        lwf: document.getElementById('stat_lwf').checked ? '1' : '0'
   648→        // No bonus_applicable — bonus is always auto (skipped only if it would overshoot)
   649→    };
   650→
   651→    fetch('index.php?page=api/salary-calc', {
   652→        method: 'POST',
   653→        headers: { 'Content-Type': 'application/json', 'X-CSRF-Token': CSRF_TOKEN },
   654→        body: JSON.stringify(params)
   655→    })
   656→    .then(function(r) { return r.json(); })
   657→    .then(function(data) {
   658→        // Handle hard error (e.g. target below min wage, target not achievable)
   659→        if (!data.success) {
   660→            clearCalcResults();
   661→            var w = document.getElementById('calcWarnings');
   662→            w.classList.remove('d-none');
   663→            var errType = (data.data && data.data.error) ? data.data.error : 'ERROR';
   664→            var alertClass = 'alert-danger';
   665→            if (errType === 'TARGET_BELOW_MIN_WAGE') alertClass = 'alert-warning';
   666→            w.innerHTML = '<div class="alert ' + alertClass + ' py-2 small mb-0"><i class="bi bi-exclamation-octagon me-1"></i><strong>Cannot calculate:</strong> ' + (data.error || 'Unknown error') + '</div>';
   667→            return;
   668→        }
   669→        var d = data.data;
   670→        document.getElementById('calcBasicDa').textContent = fmt(d.basic_da);
   671→        document.getElementById('calcLeave').textContent   = fmt(d.leave_encashment);
   672→        document.getElementById('calcBonus').textContent   = fmt(d.bonus_encashment);
   673→        document.getElementById('calcHra').textContent     = fmt(d.hra);
   674→        document.getElementById('calcGross').textContent   = fmt(d.gross_salary);
   675→        document.getElementById('calcPf').textContent      = fmt(d.deductions.pf);
   676→        document.getElementById('calcEsi').textContent     = fmt(d.deductions.esi);
   677→        document.getElementById('calcPt').textContent      = fmt(d.deductions.pt);
   678→        document.getElementById('calcLwf').textContent     = fmt(d.deductions.lwf);
   679→        document.getElementById('calcTotalDed').textContent = fmt(d.total_deductions);
   680→        document.getElementById('calcNet').textContent     = fmt(d.calculated_net);
   681→        document.getElementById('calcBasicPct').textContent = d.basic_percent + '% of Gross';
   682→
   683→        // Min wage + escalation level
   684→        document.getElementById('calcMinWage').textContent = d.min_wage > 0 ? fmt(d.min_wage) : 'Not found';
   685→
   686→        // Min wage diagnostic (explains why it's 0 / Not found)
   687→        var dbgEl = document.getElementById('calcMinWageDebug');
   688→        if (dbgEl) {
   689→            var dbg = d.min_wage_debug;
   690→            if (d.min_wage > 0 || !dbg) {
   691→                dbgEl.style.display = 'none';
   692→                dbgEl.textContent = '';
   693→            } else {
   694→                var msg = dbg.reason || 'Min wage not found.';
   695→                var extra = '';
   696→                if (dbg.categories_for_state && dbg.categories_for_state.length) {
   697→                    extra += ' | Categories in DB: ' + dbg.categories_for_state.join(', ');
   698→                }
   699→                if (dbg.zones_for_state && dbg.zones_for_state.length) {
   700→                    extra += ' | Zones: ' + dbg.zones_for_state.join(', ');
   701→                }
   702→                dbgEl.textContent = msg + extra;
   703→                dbgEl.style.display = 'block';

   709→        else if (d.level_reached === 3) levelClass = 'bg-warning text-dark';
