    26→// ============================================================================
    27→// Helper: Compute LIVE opening balance for a manager at the start of a month
    28→// Strategy: find the most recent settled ledger before this month.
    29→// If none exists, walk back through all previous months summing live data.
    30→// ============================================================================
    31→function computeLiveOpeningBalance($db, $managerId, $month, $year) {
    32→    // Look for the most recent settled ledger before this month/year
    33→    $prevSettled = $db->fetch(
    34→        "SELECT * FROM manager_ledger
    35→         WHERE manager_id = :mid
    36→           AND (year < :yr OR (year = :yr2 AND month < :m))
    37→           AND carried_forward = 1
    38→         ORDER BY year DESC, month DESC
    39→         LIMIT 1",
    40→        ['mid' => $managerId, 'yr' => $year, 'yr2' => $year, 'm' => $month]
    41→    );
    42→
    43→    if ($prevSettled) {
    44→        // We have a settled ledger. But there may be un-settled months between
    45→        // that settled month and the current month. Sum live data for those gap months.
    46→        $baseBalance = (float)$prevSettled['closing_balance'];
    47→        $gapStartMonth = (int)$prevSettled['month'] + 1;
    48→        $gapStartYear  = (int)$prevSettled['year'];
    49→
    50→        // Normalize: if gap start month > 12, rollover
    51→        if ($gapStartMonth > 12) {
    52→            $gapStartMonth = 1;
    53→            $gapStartYear++;
    54→        }
    55→
    56→        // Walk from gap start to the month BEFORE the target month
    57→        $bal = $baseBalance;
    58→        $cy = $gapStartYear;
    59→        $cm = $gapStartMonth;
    60→
    61→        while ($cy < $year || ($cy === $year && $cm < $month)) {
    62→            $live = computeLiveMonthTotals($db, $managerId, $cm, $cy);
    63→            $bal = $bal + $live['advances'] - $live['expenses'] - $live['employee_advances'];
    64→            $cm++;
    65→            if ($cm > 12) { $cm = 1; $cy++; }
    66→        }
    67→        return $bal;
    68→    }
    69→
    70→    // No settled ledger found at all. Compute from the very beginning by
    71→    // summing all transactions before the target month.
    72→    $startDate = '2000-01-01';
    73→    $endDate   = sprintf('%04d-%02d-01', $year, $month); // first day of target month, exclusive
    74→
    75→    // Sum all advances before this month
    76→    $totalAdvances = (float)$db->fetchColumn(
    77→        "SELECT COALESCE(SUM(amount), 0) FROM manager_advance_allocations
    78→         WHERE manager_id = :mid AND created_at < :ed",
    79→        ['mid' => $managerId, 'ed' => $endDate]
    80→    );
    81→
    82→    // Sum all approved expenses before this month
    83→    $totalExpenses = (float)$db->fetchColumn(
    84→        "SELECT COALESCE(SUM(amount), 0) FROM ess_expenses
    85→         WHERE (employee_id = :mid1 OR manager_id = :mid2)
    86→           AND category = 'expense' AND status = 'approved'
    87→           AND COALESCE(expense_date, DATE(created_at)) < :ed",
    88→        ['mid1' => $managerId, 'mid2' => $managerId, 'ed' => $endDate]
    89→    );
    90→
    91→    // Sum all approved employee advances before this month
    92→    $totalEmpAdvances = (float)$db->fetchColumn(
    93→        "SELECT COALESCE(SUM(amount), 0) FROM ess_expenses
    94→         WHERE (employee_id = :mid1 OR manager_id = :mid2)
    95→           AND category = 'employee_advance' AND status = 'approved'
    96→           AND COALESCE(expense_date, DATE(created_at)) < :ed",
    97→        ['mid1' => $managerId, 'mid2' => $managerId, 'ed' => $endDate]
    98→    );
    99→
   100→    return $totalAdvances - $totalExpenses - $totalEmpAdvances;
   101→}

   103→// ============================================================================
   104→// Helper: Compute LIVE totals for a specific month from actual transaction data

   106→function computeLiveMonthTotals($db, $managerId, $month, $year) {
   107→    $startDate = sprintf('%04d-%02d-01', $year, $month);
   108→    $endDate   = sprintf('%04d-%02d-31', $year, $month);
   109→
   110→    // Total advance allocations this month
   111→    $advances = (float)$db->fetchColumn(
   112→        "SELECT COALESCE(SUM(amount), 0) FROM manager_advance_allocations
   113→         WHERE manager_id = :mid
   114→           AND DATE(created_at) >= :sd AND DATE(created_at) <= :ed",
   115→        ['mid' => $managerId, 'sd' => $startDate, 'ed' => $endDate]
   116→    );
   117→
   118→    // Total approved expenses this month
   119→    $expenses = (float)$db->fetchColumn(
   120→        "SELECT COALESCE(SUM(amount), 0) FROM ess_expenses
   121→         WHERE (employee_id = :mid1 OR manager_id = :mid2)
   122→           AND category = 'expense' AND status = 'approved'
   123→           AND COALESCE(expense_date, DATE(created_at)) >= :sd
   124→           AND COALESCE(expense_date, DATE(created_at)) <= :ed",
   125→        ['mid1' => $managerId, 'mid2' => $managerId, 'sd' => $startDate, 'ed' => $endDate]
   126→    );
   127→
   128→    // Total approved employee advances this month
   129→    $employee_advances = (float)$db->fetchColumn(
   130→        "SELECT COALESCE(SUM(amount), 0) FROM ess_expenses
   131→         WHERE (employee_id = :mid1 OR manager_id = :mid2)
   132→           AND category = 'employee_advance' AND status = 'approved'
   133→           AND COALESCE(expense_date, DATE(created_at)) >= :sd
   134→           AND COALESCE(expense_date, DATE(created_at)) <= :ed",
   135→        ['mid1' => $managerId, 'mid2' => $managerId, 'sd' => $startDate, 'ed' => $endDate]
   136→    );
   137→
   138→    return [
   139→        'advances'          => $advances,
   140→        'expenses'          => $expenses,
   141→        'employee_advances' => $employee_advances,
   142→    ];
   143→}
