    65→                "INSERT INTO leave_applications 
    66→                 (employee_id, leave_type, from_date, to_date, total_days, reason, status, approved_by, approved_at, created_at)
    67→                 VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, NOW())"
    68→            );
    69→            $stmt->execute([$empId, $leaveType, $fromDate, $toDate, $totalDays, $reason, $status, $approvedBy, $approvedAt]);
    70→
    71→            // If approved, update leave balance
    72→            if ($autoApprove && $leaveType !== 'LWP') {
    73→                $year = date('Y', strtotime($fromDate));
    74→                try {
    75→                    $bal = $db->fetch(
    76→                        "SELECT id, used, closing_balance FROM leave_balances 
    77→                         WHERE employee_id = ? AND leave_type = ? AND year = ?",
    78→                        [$empId, $leaveType, $year]
    79→                    );

    81→                        $newUsed = floatval($bal['used']) + $totalDays;
    82→                        $newClosing = floatval($bal['closing_balance']) - $totalDays;
    83→                        $db->query(
    84→                            "UPDATE leave_balances SET used = ?, closing_balance = ? WHERE id = ?",
    85→                            [$newUsed, $newClosing, $bal['id']]
    86→                        );
    87→                    }
    88→                } catch (Exception $e) {
    89→                    // leave_balances table might not have a record for this type/year
    90→                }
    91→            }

   123→                        "UPDATE leave_balances SET used = ?, closing_balance = ? WHERE id = ?",
   124→                        [$newUsed, $newClosing, $bal['id']]
   125→                    );
   126→                }
   127→            } catch (Exception $e) {}
   128→        }
