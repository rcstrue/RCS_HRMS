    40→    40→function handleFileUpload($file, $uploadDir = 'uploads/profile/') {
    41→    41→    $response = null;
    42→    42→    $maxSize = 5 * 1024 * 1024; // 5MB
    43→    43→
    44→    44→    // Extension whitelist — NEVER trust the client-supplied filename extension
    45→    45→    $allowedImageExtensions = ['jpg', 'jpeg', 'png', 'gif'];
    46→    46→    $allowedPdfExtension    = ['pdf'];
    47→    47→    $allowedExtensions      = array_merge($allowedImageExtensions, $allowedPdfExtension);
    48→    48→
    49→    49→    // Map of real MIME type (from finfo) → safe forced extension
    50→    50→    $mimeToExtension = [
    51→    51→        'image/jpeg' => 'jpg',
    52→    52→        'image/png'  => 'png',
    53→    53→        'image/gif'  => 'gif',
    54→    54→        'application/pdf' => 'pdf',
    55→    55→    ];
    56→    56→
    57→    57→    // Check if file was uploaded via PHP
    58→    58→    if (!isset($file['tmp_name']) || empty($file['tmp_name'])) {
    59→    59→        return $response;
    60→    60→    }
    61→    61→
    62→    62→    // Check for upload errors
    63→    63→    if ($file['error'] !== UPLOAD_ERR_OK) {
    64→    64→        return $response;
    65→    65→    }
    66→    66→
    67→    67→    // Validate file size
    68→    68→    if ($file['size'] > $maxSize) {
    69→    69→        $response = ['error' => 'File size must be less than 5MB.'];
    70→    70→        return $response;
    71→    71→    }
    72→    72→
    73→    73→    // ─── Validate REAL file content with finfo (magic bytes), not client Content-Type ───
    74→    74→    $finfo = new finfo(FILEINFO_MIME_TYPE);
    75→    75→    $detectedMime = $finfo->file($file['tmp_name']);
    76→    76→
    77→    77→    if (!isset($mimeToExtension[$detectedMime])) {
    78→    78→        $response = ['error' => 'Invalid file type. Only JPG, PNG, GIF, PDF allowed.'];
    79→    79→        return $response;
    80→    80→    }
    81→    81→
    82→    82→    // Force the extension from the detected MIME type — ignore the uploaded filename entirely
    83→    83→    $safeExtension = $mimeToExtension[$detectedMime];
    84→    84→
    85→    85→    // For images, also validate with getimagesize() to confirm it's a real image
    86→    86→    if (in_array($safeExtension, $allowedImageExtensions)) {
    87→    87→        $imageInfo = @getimagesize($file['tmp_name']);
    88→    88→        if ($imageInfo === false) {
    89→    89→            $response = ['error' => 'Invalid image file. The content is not a valid image.'];
    90→    90→            return $response;
    91→    91→        }
    92→    92→        // Re-encode images server-side to strip any embedded payloads
    93→    93→        $destImage = null;
    94→    94→        switch ($safeExtension) {
    95→    95→            case 'jpg':
    96→    96→            case 'jpeg':
    97→    97→                $src = @imagecreatefromjpeg($file['tmp_name']);
    98→    98→                if ($src) { $destImage = $src; }
    99→    99→                break;
   100→   100→            case 'png':
   101→   101→                $src = @imagecreatefrompng($file['tmp_name']);
   102→   102→                if ($src) { $destImage = $src; }
   103→   103→                break;
   104→   104→            case 'gif':
   105→   105→                $src = @imagecreatefromgif($file['tmp_name']);
   106→   106→                if ($src) { $destImage = $src; }
   107→   107→                break;
   108→   108→        }
   109→   109→        if (!$destImage) {
   110→   110→            $response = ['error' => 'Failed to process image. Please upload a valid image file.'];
   111→   111→            return $response;
   112→   112→        }
   113→   113→    }
   114→   114→
   115→   115→    // Get web root (parent of APP_ROOT which is hrms folder)
   116→   116→    $webRoot = dirname(APP_ROOT);
   117→   117→
   118→   118→    // Create upload directory if not exists
   119→   119→    $fullPath = $webRoot . '/' . $uploadDir;
   120→   120→    if (!is_dir($fullPath)) {
   121→   121→        mkdir($fullPath, 0755, true);
   122→   122→    }
   123→   123→
   124→   124→    // Generate unique filename using ONLY the safe, forced extension
   125→   125→    $filename = uniqid() . '_' . time() . '.' . $safeExtension;
   126→   126→
   127→   127→    // Database path starts with / for web root
   128→   128→    $dbPath = '/' . $uploadDir . $filename;
   129→   129→
   130→   130→    // Save file: re-encode images via GD to strip payloads; copy PDFs directly
   131→   131→    $targetPath = $fullPath . $filename;
   132→   132→    if (in_array($safeExtension, $allowedImageExtensions)) {
   133→   133→        // Re-encode and save via GD (strips EXIF, comments, embedded scripts)
   134→   134→        $saveOk = false;
   135→   135→        switch ($safeExtension) {
   136→   136→            case 'jpg':
   137→   137→            case 'jpeg':
   138→   138→                $saveOk = imagejpeg($destImage, $targetPath, 90);
   139→   139→                break;
   140→   140→            case 'png':
   141→   141→                $saveOk = imagepng($destImage, $targetPath);
   142→   142→                break;
   143→   143→            case 'gif':
   144→   144→                $saveOk = imagegif($destImage, $targetPath);
   145→   145→                break;
   146→   146→        }
   147→   147→        imagedestroy($destImage);
   148→   148→        if (!$saveOk) {
   149→   149→            $response = ['error' => 'Failed to save processed image.'];
   150→   150→            return $response;
   151→   151→        }
   152→   152→    } else {
   153→   153→        // PDF: just move the validated file (finfo already confirmed it's a PDF)
   154→   154→        if (!move_uploaded_file($file['tmp_name'], $targetPath)) {
   155→   155→            $response = ['error' => 'Failed to upload file.'];
   156→   156→            return $response;
   157→   157→        }
   158→   158→    }
   159→   159→
   160→   160→    $response = ['path' => $dbPath];
   161→   161→    return $response;

   909→   909→// Auto-populate Worker Category from the selected Designation
   910→   910→// (the option carries a data-category attribute sourced from the
   911→   911→//  designations.worker_category column)
   912→   912→function autoFillWorkerCategory(designationSelect) {
   913→   913→    const opt = designationSelect.options[designationSelect.selectedIndex];
   914→   914→    const cat = opt ? (opt.getAttribute('data-category') || '') : '';
   915→   915→    const wcSelect = document.getElementById('worker_category');
   916→   916→    if (!wcSelect || !cat) return;
   917→   917→
   918→   918→    // Try exact match first
   919→   919→    let matched = false;
   920→   920→    for (const o of wcSelect.options) {
   921→   921→        if (o.value === cat) { o.selected = true; matched = true; break; }
   922→   922→    }
   923→   923→    // Fallback: case-insensitive match (handles "Semi-skilled" vs "Semi-Skilled")
   924→   924→    if (!matched) {
   925→   925→        const lower = cat.toLowerCase();
   926→   926→        for (const o of wcSelect.options) {
   927→   927→            if (o.value.toLowerCase() === lower) { o.selected = true; matched = true; break; }
   928→   928→        }
   929→   929→    }
   930→   930→    if (matched) {
   931→   931→        wcSelect.classList.add('is-valid');
   932→   932→        setTimeout(() => wcSelect.classList.remove('is-valid'), 1200);
   933→   933→    }
   934→   934→}
