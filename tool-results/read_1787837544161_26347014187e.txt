   692→            const managerId = document.getElementById('manager_id').value.trim();
   693→            const amount    = parseFloat(document.getElementById('amount').value);
   694→
   695→            if (!managerId) {
   696→                e.preventDefault();
   697→                alert('Please select a manager.');
   698→                return;
   699→            }
   700→
   701→            if (isNaN(amount) || amount <= 0) {
   702→                e.preventDefault();
   703→                alert('Please enter a valid amount greater than 0.');
   704→                return;
   705→            }
   706→
   707→            if (!confirm('Allocate \u20B9' + amount.toLocaleString('en-IN', {minimumFractionDigits:2}) + ' to this manager?')) {
   708→                e.preventDefault();
   709→            }
   710→        });
   711→    }
   712→
   713→    // Manager search filter
   714→    var searchInput = document.getElementById('managerSearch');
   715→    var managerSelect = document.getElementById('manager_id');
   716→    if (searchInput && managerSelect) {
   717→        // Store all original options
   718→        var allOptions = [];
   719→        for (var i = 0; i < managerSelect.options.length; i++) {
   720→            allOptions.push({
   721→                value: managerSelect.options[i].value,
   722→                text: managerSelect.options[i].text,
   723→                label: (managerSelect.options[i].getAttribute('data-label') || '').toLowerCase()
   724→            });
   725→        }
   726→
   727→        searchInput.addEventListener('input', function() {
   728→            var q = this.value.toLowerCase().trim();
   729→            var currentVal = managerSelect.value;
   730→            // Rebuild options
   731→            managerSelect.innerHTML = '';
   732→            for (var i = 0; i < allOptions.length; i++) {
   733→                if (i === 0 || !q || allOptions[i].label.indexOf(q) !== -1) {
   734→                    var opt = document.createElement('option');
   735→                    opt.value = allOptions[i].value;
   736→                    opt.textContent = allOptions[i].text;
   737→                    managerSelect.appendChild(opt);
   738→                }
   739→            }
   740→            // Re-select previous value if still visible
   741→            if (currentVal) managerSelect.value = currentVal;
   742→        });
   743→    }
