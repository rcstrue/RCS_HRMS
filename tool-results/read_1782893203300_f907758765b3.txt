   383→function _handleGetDashboard(mysqli $conn, string $authId): void
   384→{
   385→    $employeeId = (int)($_GET['employee_id'] ?? $authId);
   386→    $now = new \DateTime('now', new \DateTimeZone('Asia/Kolkata'));
   387→    $currentMonth = (int)$now->format('n');
   388→    $currentYear = (int)$now->format('Y');
   389→
   390→    // Basic counts
   391→    $counts = $conn->query("
   392→        SELECT
   393→            COUNT(*) AS total_visits,
   394→            SUM(CASE WHEN status = 'submitted' THEN 1 ELSE 0 END) AS pending_visits,
   395→            SUM(CASE WHEN status = 'approved' THEN 1 ELSE 0 END) AS approved_visits,
   396→            SUM(CASE WHEN status = 'rejected' THEN 1 ELSE 0 END) AS rejected_visits,
   397→            SUM(CASE WHEN visit_month = {$currentMonth} AND visit_year = {$currentYear} THEN 1 ELSE 0 END) AS this_month_visits
   398→        FROM ess_unit_visits WHERE employee_id = {$employeeId}
   399→    ")->fetch_assoc();
   400→
   401→    // This month avg score
   402→    $monthAvg = $conn->query("
   403→        SELECT AVG(score_percent) AS avg_score
   404→        FROM ess_unit_visits
   405→        WHERE employee_id = {$employeeId} AND visit_month = {$currentMonth} AND visit_year = {$currentYear}
   406→            AND status IN ('approved','submitted') AND max_score > 0
   407→    ")->fetch_assoc();
   408→
   409→    // Score trend - last 6 months
   410→    $trend = [];
   411→    for ($i = 5; $i >= 0; $i--) {
   412→        $d = clone $now;
   413→        $d->modify("-{$i} months");
   414→        $m = (int)$d->format('n');
   415→        $y = (int)$d->format('Y');
   416→        $label = $d->format('M Y');
   417→        $avgRow = $conn->query("
   418→            SELECT AVG(score_percent) AS avg_score
   419→            FROM ess_unit_visits
   420→            WHERE employee_id = {$employeeId} AND visit_month = {$m} AND visit_year = {$y}
   421→                AND status IN ('approved','submitted') AND max_score > 0
   422→        ")->fetch_assoc();
   423→        $trend[] = ['month' => $label, 'avg_score' => round((float)($avgRow['avg_score'] ?? 0), 1)];
   424→    }
   425→
   426→    // Category scores
   427→    $catScores = $conn->query("
   428→        SELECT cc.name AS category, AVG(
   429→            CASE WHEN vci.status = 'yes' THEN ci.weight ELSE 0 END
   430→        ) / NULLIF(AVG(CASE WHEN vci.status != 'na' THEN ci.weight ELSE NULL END), 0) * 100 AS avg_score
   431→        FROM ess_visit_checklist_items vci
   432→        JOIN ess_checklist_items ci ON ci.id = vci.checklist_item_id
   433→        JOIN ess_checklist_categories cc ON cc.id = ci.category_id
   434→        JOIN ess_unit_visits v ON v.id = vci.visit_id
   435→        WHERE v.employee_id = {$employeeId} AND v.status IN ('approved','submitted')
   436→        GROUP BY cc.id, cc.name
   437→        ORDER BY cc.display_order
   438→    ");
   439→    $categoryScores = [];
   440→    while ($cr = $catScores->fetch_assoc()) {
   441→        $categoryScores[] = [
   442→            'category' => $cr['category'] ?? '',
   443→            'avg_score' => round((float)($cr['avg_score'] ?? 0), 1),
   444→        ];
   445→    }
   446→
   447→    // Unit compliance
   448→    $unitComp = $conn->query("
   449→        SELECT u.name AS unit_name, AVG(v.score_percent) AS score, COUNT(*) AS visits
   450→        FROM ess_unit_visits v
   451→        JOIN units u ON u.id = v.unit_id
   452→        WHERE v.employee_id = {$employeeId} AND v.status IN ('approved','submitted') AND v.max_score > 0
   453→        GROUP BY v.unit_id, u.name
   454→        ORDER BY score DESC
   455→        LIMIT 10
   456→    ");
   457→    $unitCompliance = [];
   458→    while ($ur = $unitComp->fetch_assoc()) {
   459→        $unitCompliance[] = [
   460→            'unit_name' => $ur['unit_name'] ?? '',
   461→            'score' => round((float)($ur['score'] ?? 0), 1),
   462→            'visits' => (int)$ur['visits'],
   463→        ];
   464→    }
   465→
   466→    $conn->close();
   467→
   468→    jsonOutput([
   469→        'success' => true,
   470→        'data' => [
   471→            'total_visits' => (int)($counts['total_visits'] ?? 0),
   472→            'pending_visits' => (int)($counts['pending_visits'] ?? 0),
   473→            'approved_visits' => (int)($counts['approved_visits'] ?? 0),
   474→            'rejected_visits' => (int)($counts['rejected_visits'] ?? 0),
   475→            'this_month_visits' => (int)($counts['this_month_visits'] ?? 0),
   476→            'this_month_score_avg' => round((float)($monthAvg['avg_score'] ?? 0), 1),
   477→            'score_trend' => $trend,
   478→            'category_scores' => $categoryScores,
   479→            'unit_compliance' => $unitCompliance,
   480→        ]
   481→    ]);
   482→}

   680→    if (!in_array($action, ['approve', 'reject'])) jsonOutput(['success' => false, 'error' => 'Action must be approve or reject'], 400);
   681→
   682→    // Verify visit exists
   683→    $checkStmt = $conn->prepare('SELECT id, status, employee_id FROM ess_unit_visits WHERE id = ?');
   684→    $checkStmt->bind_param('i', $visitId);
   685→    $checkStmt->execute();
   686→    $visit = $checkStmt->get_result()->fetch_assoc();
   687→    $checkStmt->close();
   688→
   689→    if (!$visit) { $conn->close(); jsonOutput(['success' => false, 'error' => 'Visit not found'], 404); }
   690→    if ($visit['status'] !== 'submitted') { $conn->close(); jsonOutput(['success' => false, 'error' => 'Only submitted visits can be ' . $action . 'd'], 400); }
   691→
   692→    if ($action === 'approve') {
   693→        $stmt = $conn->prepare('UPDATE ess_unit_visits SET status = ?, approved_by = ?, approved_at = NOW() WHERE id = ?');
   694→        $stmt->bind_param('ssi', 'approved', $employeeId, $visitId);
   695→        $stmt->execute();
   696→        $stmt->close();
   697→        _logAudit($conn, $visitId, 'approved', $employeeId);
   698→    } else {
   699→        $rejectionReason = trim($input['rejection_reason'] ?? '');
   700→        $stmt = $conn->prepare('UPDATE ess_unit_visits SET status = ?, rejection_reason = ?, approved_by = ?, approved_at = NOW() WHERE id = ?');
   701→        $stmt->bind_param('sssi', 'rejected', $rejectionReason, $employeeId, $visitId);
   702→        $stmt->execute();
   703→        $stmt->close();
   704→        _logAudit($conn, $visitId, 'rejected', $employeeId, $rejectionReason);
   705→    }
   706→
   707→    // Fetch updated visit
   708→    $fetchStmt = $conn->prepare("
   709→        SELECT v.*, u.name AS unit_name, c.name AS client_name,
   710→               e.full_name AS employee_name, e.employee_code
   711→        FROM ess_unit_visits v
   712→        LEFT JOIN units u ON u.id = v.unit_id
   713→        LEFT JOIN clients c ON c.id = u.client_id
   714→        LEFT JOIN employees e ON e.id = v.employee_id
   715→        WHERE v.id = ?
   716→    ");
   717→    $fetchStmt->bind_param('i', $visitId);
   718→    $fetchStmt->execute();
   719→    $row = $fetchStmt->get_result()->fetch_assoc();
   720→    $fetchStmt->close();
   721→    $conn->close();
   722→
   723→    jsonOutput([
   724→        'success' => true,
   725→        'data' => [
   726→            'id' => (int)$row['id'],
   727→            'status' => $row['status'],
   728→            'message' => 'Visit ' . $action . 'd successfully'
   729→        ]
   730→    ]);
   731→}

   774→// Email Helper (inline)
   775→// ══════════════════════════════════════════════════════════════════════════
   776→
   777→function _sendVisitEmailInline(mysqli $conn, int $visitId): bool
   778→{
   779→    // Fetch visit with employee email
   780→    $stmt = $conn->prepare("
   781→        SELECT v.*, u.name AS unit_name, c.name AS client_name,
   782→               e.full_name AS employee_name, e.employee_code, e.email AS employee_email
   783→        FROM ess_unit_visits v
   784→        LEFT JOIN units u ON u.id = v.unit_id
   785→        LEFT JOIN clients c ON c.id = u.client_id
   786→        LEFT JOIN employees e ON e.id = v.employee_id
   787→        WHERE v.id = ?
   788→    ");
   789→    $stmt->bind_param('i', $visitId);
   790→    $stmt->execute();
   791→    $visit = $stmt->get_result()->fetch_assoc();
   792→    $stmt->close();
   793→
   794→    if (!$visit || empty($visit['employee_email'])) return false;
   795→
   796→    // Fetch checklist items
   797→    $itemsStmt = $conn->prepare('
   798→        SELECT ci.status, ci.remarks,
   799→               cmi.name AS item_name, cmc.name AS category_name
   800→        FROM ess_visit_checklist_items ci
   801→        LEFT JOIN ess_checklist_items cmi ON cmi.id = ci.checklist_item_id
   802→        LEFT JOIN ess_checklist_categories cmc ON cmc.id = ci.category_id
   803→        WHERE ci.visit_id = ?
   804→        ORDER BY cmc.display_order ASC, cmi.display_order ASC
   805→    ');
   806→    $itemsStmt->bind_param('i', $visitId);
   807→    $itemsStmt->execute();
   808→    $itemsResult = $itemsStmt->get_result();
   809→    $items = [];
   810→    while ($row = $itemsResult->fetch_assoc()) {
   811→        $items[] = $row;
   812→    }
   813→    $itemsResult->free();
   814→    $itemsStmt->close();
   815→
   816→    $score = (int)($visit['score_percent'] ?? 0);
   817→    $subject = sprintf('Unit Visit Report: %s - %s %s (%d%%)',
   818→        $visit['unit_name'] ?? 'Unknown',
   819→        _monthName((int)$visit['visit_month']),
   820→        $visit['visit_year'],
   821→        $score
   822→    );
   823→
   824→    // Build plain text email
   825→    $text = "UNIT VISIT CHECKLIST REPORT\nRCS TRUE FACILITIES PVT LTD\n" . str_repeat('=', 50) . "\n\n";
   826→    $text .= "Employee: " . ($visit['employee_name'] ?? '') . "\n";
   827→    $text .= "Client/Unit: " . ($visit['client_name'] ?? '') . " / " . ($visit['unit_name'] ?? '') . "\n";
   828→    $text .= "Visit: " . ((int)$visit['visit_number'] === 1 ? 'First' : 'Second') . " Visit - " . _monthName((int)$visit['visit_month']) . " " . $visit['visit_year'] . "\n";
   829→    $text .= "Score: {$score}% (" . ($visit['total_score'] ?? 0) . "/" . ($visit['max_score'] ?? 0) . " points)\n\n";
   830→
   831→    $grouped = [];
   832→    foreach ($items as $item) {
   833→        $cat = $item['category_name'] ?? 'Unknown';
   834→        if (!isset($grouped[$cat])) $grouped[$cat] = [];
   835→        $status = $item['status'] === 'na' ? 'N/A' : ($item['status'] === 'yes' ? '[YES]' : '[NO]');
   836→        $line = "  {$status} " . ($item['item_name'] ?? '');
   837→        if (!empty($item['remarks'])) $line .= " - " . $item['remarks'];
   838→        $grouped[$cat][] = $line;
   839→    }
   840→    foreach ($grouped as $cat => $lines) {
   841→        $text .= "\n--- " . strtoupper($cat) . " ---\n" . implode("\n", $lines) . "\n";
   842→    }
   843→
   844→    if (!empty($visit['notes'])) $text .= "\nNOTES: " . $visit['notes'] . "\n";
   845→
   846→    $headers = "From: RCS ESS Portal <noreply@rcsfacility.com>\r\nReply-To: support@rcsfacility.com\r\nX-Mailer: RCS-ESS/2.0";
   847→    return mail($visit['employee_email'], $subject, $text, $headers, '-fnoreply@rcsfacility.com');
   848→}
