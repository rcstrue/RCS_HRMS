    73→/**
    74→ * Deduct loan EMI for all employees in a payroll period + unit.
    75→ * Call this AFTER processPayroll() to add loan deductions.
    76→ */
    77→function deductLoansForPeriodUnit($db, $periodId, $unitId) {
    78→    // Get month/year from payroll_periods
    79→    $period = $db->fetch("SELECT month, year FROM payroll_periods WHERE id = ?", [$periodId]);
    80→    if (!$period) return 0;
    81→    
    82→    $month = (int)$period['month'];
    83→    $year  = (int)$period['year'];
    84→    $totalDeducted = 0;
    85→    
    86→    // Get all payroll records for this period + unit
    87→    $payrollRows = $db->fetchAll(
    88→        "SELECT p.id as payroll_id, p.employee_id as emp_code, p.loan_emi, p.total_deductions, p.net_pay,
    89→                e.id as employee_db_id
    90→         FROM payroll p
    91→         JOIN employees e ON p.employee_id = e.employee_code
    92→         WHERE p.payroll_period_id = ? AND e.unit_id = ?",
    93→        [$periodId, $unitId]
    94→    );
    95→    
    96→    foreach ($payrollRows as $pRow) {
    97→        $empDbId = (int)$pRow['employee_db_id'];
    98→        $empCode = $pRow['emp_code'];
    99→        $payrollId = (int)$pRow['payroll_id'];
   100→        $currentLoanEmi = (float)$pRow['loan_emi'];
   101→        
   102→        // Get active loans for this employee started on or before this month
   103→        $empLoans = $db->fetchAll(
   104→            "SELECT * FROM employee_loans
   105→             WHERE employee_id = ? AND status = 'Active'
   106→             AND (start_year < ? OR (start_year = ? AND start_month <= ?))",
   107→            [$empDbId, $year, $year, $month]
   108→        );
   109→        
   110→        $empLoanTotal = 0;
   111→        
   112→        foreach ($empLoans as $empLoan) {
   113→            // Check if EMI already deducted for this month
   114→            $alreadyDeducted = $db->fetch(
   115→                "SELECT id FROM loan_emi_log WHERE loan_id = ? AND month = ? AND year = ?",
   116→                [$empLoan['id'], $month, $year]
   117→            );
   118→            
   119→            if ($alreadyDeducted) {
   120→                // Already deducted — add existing EMI amount
   121→                $empLoanTotal += (float)$empLoan['emi_amount'];
   122→                continue;
   123→            }
   124→            
   125→            $emiAmount = (float)$empLoan['emi_amount'];
   126→            $balanceAmount = (float)$empLoan['balance_amount'];
   127→            
   128→            // Calculate interest/principal split
   129→            if ((float)$empLoan['interest_rate'] > 0) {
   130→                $monthlyRate = (float)$empLoan['interest_rate'] / 12 / 100;
   131→                $interestComponent = round($balanceAmount * $monthlyRate, 2);
   132→                $principalComponent = round($emiAmount - $interestComponent, 2);
   133→            } else {
   134→                $interestComponent = 0;
   135→                $principalComponent = $emiAmount;
   136→            }
   137→            
   138→            // Handle last EMI
   139→            if ($emiAmount > $balanceAmount) {
   140→                $emiAmount = $balanceAmount;
   141→                $principalComponent = $emiAmount - $interestComponent;
   142→                if ($principalComponent < 0) {
   143→                    $interestComponent = $emiAmount;
   144→                    $principalComponent = 0;
   145→                }
   146→            }
   147→            
   148→            $newBalance = round($balanceAmount - $emiAmount, 2);
   149→            if ($newBalance < 0) $newBalance = 0;
   150→            
   151→            // Record EMI log
   152→            $db->insert('loan_emi_log', [
   153→                'loan_id' => $empLoan['id'],
   154→                'employee_id' => $empDbId,
   155→                'month' => $month,
   156→                'year' => $year,
   157→                'emi_amount' => $emiAmount,
   158→                'principal_component' => $principalComponent,
   159→                'interest_component' => $interestComponent,
   160→                'balance_after' => $newBalance,
   161→                'deducted_via_payroll' => 1
   162→            ]);
   163→            
   164→            // Update loan balance and status
   165→            $newStatus = ($newBalance <= 0) ? 'Closed' : 'Active';
   166→            $db->update('employee_loans', [
   167→                'balance_amount' => $newBalance,
   168→                'emi_deducted' => (int)$empLoan['emi_deducted'] + 1,
   169→                'status' => $newStatus
   170→            ], 'id = :id', ['id' => $empLoan['id']]);
   171→            
   172→            $empLoanTotal += $emiAmount;
   173→        }
   174→        
   175→        // Update payroll record with total loan EMI for this employee
   176→        if ($empLoanTotal > 0 && abs($empLoanTotal - $currentLoanEmi) > 0.01) {
   177→            $currentTotalDed = (float)$pRow['total_deductions'];
   178→            $currentNetPay   = (float)$pRow['net_pay'];
   179→            
   180→            // Remove old loan EMI from deductions (if any), add new
   181→            $deductionAdjustment = $empLoanTotal - $currentLoanEmi;
   182→            
   183→            $newTotalDed = round($currentTotalDed + $deductionAdjustment, 2);
   184→            $newNetPay   = round($currentNetPay - $deductionAdjustment, 2);
   185→            
   186→            $db->update('payroll', [
   187→                'loan_emi' => round($empLoanTotal, 2),
   188→                'total_deductions' => $newTotalDed,
   189→                'net_pay' => $newNetPay,
   190→                'updated_at' => date('Y-m-d H:i:s')
   191→            ], 'id = :id', ['id' => $payrollId]);
   192→            
   193→            $totalDeducted += $deductionAdjustment;
   194→        } elseif ($empLoanTotal > 0 && $currentLoanEmi == 0) {
   195→            // First time setting loan EMI
   196→            $currentTotalDed = (float)$pRow['total_deductions'];
   197→            $currentNetPay   = (float)$pRow['net_pay'];
   198→            
   199→            $newTotalDed = round($currentTotalDed + $empLoanTotal, 2);
   200→            $newNetPay   = round($currentNetPay - $empLoanTotal, 2);
   201→            
   202→            $db->update('payroll', [
   203→                'loan_emi' => round($empLoanTotal, 2),
   204→                'total_deductions' => $newTotalDed,
   205→                'net_pay' => $newNetPay,
   206→                'updated_at' => date('Y-m-d H:i:s')
   207→            ], 'id = :id', ['id' => $payrollId]);
   208→            
   209→            $totalDeducted += $empLoanTotal;
   210→        }
   211→    }
   212→    
   213→    return $totalDeducted;
   214→}
