    29→    29→    private function checkSession() {
    30→    30→        if (isset($_SESSION['user_id'])) {
    31→    31→            $user = $this->db->fetch(
    32→    32→                "SELECT u.id, u.username, u.email, u.role_id, u.first_name, u.last_name, u.is_active,

    46→    46→    // Login user
    47→    47→    public function login($username, $password, $remember = false) {
    48→    48→        // ── Rate limiting / lockout check ──
    49→    49→        $ip = $_SERVER['REMOTE_ADDR'] ?? '0.0.0.0';
    50→    50→        $lockout = $this->_checkLoginLockout($username, $ip);
    51→    51→        if ($lockout) {
    52→    52→            return ['success' => false, 'error' => $lockout];
    53→    53→        }
    54→    54→
    55→    55→        // Get user with role info
    56→    56→        $user = $this->db->fetch(
    57→    57→            "SELECT u.id, u.username, u.email, u.password, u.role_id, u.first_name, u.last_name, u.is_active,
    58→    58→                    r.role_name, r.role_code
    59→    59→             FROM users u 
    60→    60→             LEFT JOIN roles r ON u.role_id = r.id 
    61→    61→             WHERE (u.username = :username OR u.email = :email)",
    62→    62→            ['username' => $username, 'email' => $username]
    63→    63→        );
    64→    64→        
    65→    65→        // Check if user exists
    66→    66→        if (!$user) {
    67→    67→            $this->_recordFailedLogin($username, $ip);
    68→    68→            return ['success' => false, 'error' => 'Invalid username or password.'];
    69→    69→        }
    70→    70→        
    71→    71→        // Check if active
    72→    72→        if (empty($user['is_active'])) {
    73→    73→            return ['success' => false, 'error' => 'Account is inactive.'];
    74→    74→        }
    75→    75→        
    76→    76→        // Verify password
    77→    77→        if (!password_verify($password, $user['password'])) {
    78→    78→            $this->_recordFailedLogin($username, $ip);
    79→    79→            return ['success' => false, 'error' => 'Invalid username or password.'];
    80→    80→        }
    81→    81→
    82→    82→        // Clear failed attempts on successful login
    83→    83→        $this->_clearFailedLogins($username, $ip);
    84→    84→        
    85→    85→        // Set session variables
    86→    86→        $_SESSION['user_id'] = $user['id'];
    87→    87→        $_SESSION['username'] = $user['username'];
    88→    88→        $_SESSION['email'] = $user['email'];
    89→    89→        $_SESSION['role_id'] = $user['role_id'];
    90→    90→        $_SESSION['role_code'] = $user['role_code'] ?? 'worker';
    91→    91→        $_SESSION['role_name'] = $user['role_name'] ?? 'User';
    92→    92→        $_SESSION['first_name'] = $user['first_name'] ?? '';
    93→    93→        $_SESSION['last_name'] = $user['last_name'] ?? '';
    94→    94→        
    95→    95→        // Update last login
    96→    96→        try {
    97→    97→            $this->db->update('users', [
    98→    98→                'last_login' => date('Y-m-d H:i:s')
    99→    99→            ], SQL_WHERE_USER_ID, ['id' => $user['id']]);
   100→   100→        } catch (Exception $e) {
   101→   101→            // Ignore if last_login column doesn't exist
   102→   102→        }
   103→   103→
   104→   104→        // Audit log — login
   105→   105→        if (function_exists('audit_log')) {
   106→   106→            audit_log('login', 'auth', $user['id'], "User '{$user['username']}' logged in");
   107→   107→        }
   108→   108→
   109→   109→        $this->user = $user;
   110→   110→        
   111→   111→        return [
   112→   112→            'success' => true,
   113→   113→            'message' => 'Login successful.',
   114→   114→            'user' => $user
   115→   115→        ];
   116→   116→    }

   118→   118→    // Logout user
   119→   119→    public function logout() {
   120→   120→        // Audit log — logout (session still active at this point)
   121→   121→        if (function_exists('audit_log') && isset($_SESSION['user_id'])) {
   122→   122→            $uid = $_SESSION['user_id'];
   123→   123→            $uname = $_SESSION['username'] ?? 'unknown';
   124→   124→            audit_log('logout', 'auth', $uid, "User '{$uname}' logged out");
   125→   125→        }
   126→   126→        $_SESSION = [];
   127→   127→        session_destroy();
   128→   128→        $this->user = null;
   129→   129→    }

   156→   156→    // Check if user has role level or higher
   157→   157→    public function hasRoleLevel($role) {
   158→   158→        if (!isset($_SESSION['role_code'])) {
   159→   159→            return false;
   160→   160→        }
   161→   161→        
   162→   162→        $userLevel = $this->roleHierarchy[$_SESSION['role_code']] ?? 0;
   163→   163→        $requiredLevel = $this->roleHierarchy[$role] ?? 0;
   164→   164→        
   165→   165→        return $userLevel >= $requiredLevel;
   166→   166→    }

   168→   168→    // Change password
   169→   169→    public function changePassword($userId, $currentPassword, $newPassword) {
   170→   170→        $user = $this->db->fetch(
   171→   171→            "SELECT password FROM users WHERE id = :id",
   172→   172→            ['id' => $userId]
   173→   173→        );
   174→   174→        
   175→   175→        if (!$user || !password_verify($currentPassword, $user['password'])) {
   176→   176→            return ['success' => false, 'message' => 'Current password is incorrect.'];
   177→   177→        }
   178→   178→        
   179→   179→        $hashedPassword = password_hash($newPassword, PASSWORD_BCRYPT, ['cost' => 12]);
   180→   180→        
   181→   181→        $this->db->update('users', [
   182→   182→            'password' => $hashedPassword
   183→   183→        ], SQL_WHERE_USER_ID, ['id' => $userId]);
   184→   184→        
   185→   185→        return ['success' => true, 'message' => 'Password changed successfully.'];
   186→   186→    }

   188→   188→    // Create new user
   189→   189→    public function createUser($data) {
   190→   190→        $exists = $this->db->fetch(
   191→   191→            "SELECT id FROM users WHERE username = :username OR email = :email",
   192→   192→            ['username' => $data['username'], 'email' => $data['email']]
   193→   193→        );
   194→   194→        
   195→   195→        if ($exists) {
   196→   196→            return ['success' => false, 'message' => 'Username or email already exists.'];
   197→   197→        }
   198→   198→        
   199→   199→        $hashedPassword = password_hash($data['password'], PASSWORD_BCRYPT, ['cost' => 12]);
   200→   200→        
   201→   201→        $userId = $this->db->insert('users', [
   202→   202→            'username' => $data['username'],
   203→   203→            'email' => $data['email'],
   204→   204→            'password' => $hashedPassword,
   205→   205→            'role_id' => $data['role_id'] ?? 5,
   206→   206→            'first_name' => $data['first_name'] ?? '',
   207→   207→            'last_name' => $data['last_name'] ?? '',
   208→   208→            'is_active' => 1
   209→   209→        ]);
   210→   210→        
   211→   211→        if ($userId) {
   212→   212→            return ['success' => true, 'message' => 'User created successfully.', 'user_id' => $userId];
   213→   213→        }
   214→   214→        
   215→   215→        return ['success' => false, 'message' => 'Failed to create user.'];
   216→   216→    }

   221→   221→        
   222→   222→        $result = $this->db->update('users', [

   225→   225→

   227→   227→            return ['success' => true, 'message' => 'Password reset successfully.'];
   228→   228→        }
   229→   229→        
   230→   230→        return ['success' => false, 'message' => 'Failed to reset password.'];
   231→   231→    }

   233→   233→    // Get all users (admin/hr)
   234→   234→    public function getUsers($filters = []) {
   235→   235→        $sql = "SELECT u.id, u.username, u.email, u.role_id, u.is_active, u.last_login, 
   236→   236→                       u.created_at, u.first_name, u.last_name, r.role_name, r.role_code
   237→   237→                FROM users u
   238→   238→                LEFT JOIN roles r ON u.role_id = r.id
   239→   239→                WHERE 1=1";
   240→   240→        $params = [];
   241→   241→        
   242→   242→        if (!empty($filters['role_id'])) {
   243→   243→            $sql .= " AND u.role_id = :role_id";
   244→   244→            $params['role_id'] = $filters['role_id'];
   245→   245→        }
   246→   246→        
   247→   247→        if (isset($filters['is_active'])) {
   248→   248→            $sql .= " AND u.is_active = :is_active";
   249→   249→            $params['is_active'] = $filters['is_active'];
   250→   250→        }
   251→   251→        
   252→   252→        $sql .= " ORDER BY u.created_at DESC";
   253→   253→        
   254→   254→        return $this->db->fetchAll($sql, $params);
   255→   255→    }

   278→   278→    // Update user
   279→   279→    public function updateUser($userId, $data) {
   280→   280→        $allowedFields = ['email', 'role_id', 'first_name', 'last_name', 'phone', 'is_active'];
   281→   281→        $updateData = [];
   282→   282→        
   283→   283→        foreach ($allowedFields as $field) {
   284→   284→            if (isset($data[$field])) {
   285→   285→                $updateData[$field] = $data[$field];
   286→   286→            }
   287→   287→        }
   288→   288→        
   289→   289→        if (empty($updateData)) {
   290→   290→            return ['success' => false, 'message' => 'No data to update.'];
   291→   291→        }
   292→   292→        
   293→   293→        $result = $this->db->update('users', $updateData, SQL_WHERE_USER_ID, ['id' => $userId]);
   294→   294→        
   295→   295→        return ['success' => true, 'message' => 'User updated successfully.'];
   296→   296→    }

   301→   301→        if (isset($_SESSION['user_id']) && $userId == $_SESSION['user_id']) {
   302→   302→            return ['success' => false, 'message' => 'Cannot delete your own account.'];

   304→   304→

   306→   306→        
   307→   307→        return ['success' => true, 'message' => 'User deleted successfully.'];
   308→   308→    }

   963→   963→    /**
   964→   964→     * Check if the username+IP is currently locked out.
   965→   965→     * Returns an error message string if locked, or null if allowed.
   966→   966→     */
   967→   967→    private function _checkLoginLockout($username, $ip): ?string
   968→   968→    {
   969→   969→        $this->_ensureLoginAttemptsTable();
   970→   970→
   971→   971→        $row = $this->db->fetch(
   972→   972→            "SELECT attempts, locked_until FROM login_attempts WHERE username = :u AND ip = :i",
   973→   973→            ['u' => $username, 'i' => $ip]
   974→   974→        );
   975→   975→
   976→   976→        if (!$row) {
   977→   977→            return null;
   978→   978→        }
   979→   979→
   980→   980→        // If there is a lockout window, check if it has expired
   981→   981→        if ($row['locked_until'] !== null) {
   982→   982→            $now  = new \DateTime('now');
   983→   983→            $lock = new \DateTime($row['locked_until']);
   984→   984→            if ($now < $lock) {
   985→   985→                $diff  = $now->diff($lock);
   986→   986→                $mins  = $diff->i + ($diff->h * 60) + ($diff->d * 1440);
   987→   987→                $hours = (int)floor($mins / 60);
   988→   988→                if ($hours >= 1) {
   989→   989→                    return "Account temporarily locked due to too many failed login attempts. Try again in {$hours} hour" . ($hours > 1 ? 's' : '') . ".";
   990→   990→                }
   991→   991→                return "Account temporarily locked due to too many failed login attempts. Try again in {$mins} minute" . ($mins !== 1 ? 's' : '') . ".";
   992→   992→            }
   993→   993→            // Lockout expired — reset
   994→   994→            $this->_clearFailedLogins($username, $ip);
   995→   995→        }
   996→   996→
   997→   997→        return null;
   998→   998→    }
