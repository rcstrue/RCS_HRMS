   634→function editIdCardPhoto() {
   635→    var photoImg = document.getElementById('idCardPhoto');
   636→    if (!photoImg) return;
   637→    var empId = document.querySelector('input[name="employee_id"]');
   638→    if (!empId) return;
   639→    
   640→    // Get the raw photo URL (with /uploads/ prefix) - safe_path() on server handles stripping
   641→    var rawUrl = photoImg.getAttribute('data-photo-path') || photoImg.src;
   642→    
   643→    openLiteEditor(photoImg.src, function(base64DataUrl) {
   644→        var formData = new FormData();
   645→        formData.append('action', 'save_canvas');
   646→        formData.append('file', rawUrl);
   647→        formData.append('image_data', base64DataUrl);
   648→        
   649→        fetch('?page=api/image-tool&ie_action=save_canvas', {
   650→            method: 'POST',
   651→            body: formData
   652→        })
   653→        .then(function(r) { return r.json(); })
   654→        .then(function(res) {
   655→            if (res.ok) {
   656→                // res.rel already contains uploads/ prefix (e.g. "uploads/profile/xxx.jpg")
   657→                photoImg.src = '/' + (res.rel || rawUrl.split('?')[0].replace(/^\//, '')) + '?t=' + Date.now();
   658→                refreshPreview();
   659→                alert('Photo saved successfully!');
   660→            } else {
   661→                alert('Save failed: ' + (res.msg || 'Unknown error'));
   662→            }
   663→        })
   664→        .catch(function(err) { alert('Error saving photo: ' + err.message); });
   665→    });
   666→}
