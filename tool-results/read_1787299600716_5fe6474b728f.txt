    28→$calendarDays = cal_days_in_month(CAL_GREGORIAN, $month, $year);
    29→
    30→// ── Fetch unit's pay_days_type from salary formula ──────────────
    31→$unitFormula = null;
    32→if ($unitId > 0) {
    33→    $unitFormula = $db->fetch(
    34→        "SELECT * FROM unit_salary_formulas
    35→         WHERE unit_id = ? AND is_active = 1
    36→         ORDER BY effective_from DESC LIMIT 1",
    37→        [$unitId]
    38→    );
    39→}
    40→$payDaysType = $unitFormula['pay_days_type'] ?? 'actual';
    41→
    42→// Calculate totalDays based on pay_days_type
    43→$totalDays = $calendarDays;
    44→switch ($payDaysType) {
    45→    case 'fixed_30':
    46→        $totalDays = 30;
    47→        break;
    48→    case 'previous_month':
    49→        $prevMonth = $month - 1;
    50→        $prevYear  = $year;
    51→        if ($prevMonth < 1) { $prevMonth = 12; $prevYear--; }
    52→        $totalDays = (int)cal_days_in_month(CAL_GREGORIAN, $prevMonth, $prevYear);
    53→        break;
    54→    case 'calendar_minus_sundays':
    55→        // Count all Sundays in the selected month
    56→        $sundays = 0;
    57→        for ($d = 1; $d <= $calendarDays; $d++) {
    58→            $dow = (int)date('N', strtotime("$year-$month-$d"));
    59→            if ($dow === 7) $sundays++;
    60→        }
    61→        $totalDays = $calendarDays - $sundays;
    62→        break;
    63→    default: // 'actual'
    64→        $totalDays = $calendarDays;
    65→        break;
    66→}
    67→
    68→// Also get OT settings from unit formula
    69→$unitOtCalcType = $unitFormula['ot_calculation_type'] ?? 'double_pay';
    70→$unitOtCalcOn   = $unitFormula['ot_calculation_on'] ?? 'basic_da';
    71→$unitOtHrsPerDay = (float)($unitFormula['ot_hours_per_day'] ?? 8);
    72→

    74→

    76→$pfEmployeeShare = 12.00;
    77→$pfWageCeiling = 15000;
    78→$pfEmployerShare = 3.67;
    79→$pfEmployerEps = 8.33;
    80→$pfEmployerEdlis = 0.50;

    83→$esiEmployerShare = 3.25;

    89→        $pfWageCeiling  = (float)$pfRate['wage_ceiling'];
    90→        $pfEmployerShare = (float)$pfRate['employer_share_pf'];

    97→    $esiRate = $db->fetch("SELECT * FROM esi_rates WHERE is_active = 1 ORDER BY effective_from DESC LIMIT 1");
    98→    if ($esiRate) {
    99→        $esiEmployeeShare = (float)$esiRate['employee_share'];
   100→        $esiEmployerShare = (float)$esiRate['employer_share'];
   101→        $esiWageCeiling   = (float)$esiRate['wage_ceiling'];

   112→        "SELECT u.state, s.pt_applicable as state_pt_applicable, s.id as state_id
   113→         FROM units u
   114→         LEFT JOIN states s ON (u.state = s.state_code OR u.state = s.state_name)
   115→         WHERE u.id = ?",
   116→        [$unitId]
   117→    );
   118→    if ($unitInfo) {
   119→        $unitState = $unitInfo['state'] ?? '';
   120→        $unitStatePtApplicable = !empty($unitInfo['state_pt_applicable']);
   121→    }
   122→    // Fetch PT slabs for this state (professional_tax_rates joined to states —
   123→    // the professional_tax_slabs table does not exist in the schema)
   124→    if ($unitStatePtApplicable && $unitState) {
   125→        try {
   126→            $ptSlabs = $db->fetchAll(
   127→                "SELECT ptr.salary_from AS min_gross, ptr.salary_to AS max_gross, ptr.pt_amount
   128→                 FROM professional_tax_rates ptr
   129→                 JOIN states s ON s.id = ptr.state_id
   130→                 WHERE (LOWER(s.state_code) = LOWER(?) OR LOWER(s.state_name) = LOWER(?))
   131→                 AND ptr.is_active = 1
   132→                 AND ptr.effective_from <= CURDATE()
   133→                 AND ptr.gender_specific = 'All'
   134→                 ORDER BY ptr.salary_from",
   135→                [$unitState, $unitState]
   136→            );
   137→        } catch (Exception $e) {}
   138→    }
   139→    // Fetch LWF rate for this state (from lwf_rates → lwf_state_rates fallback)
   140→    if ($unitState) {
   141→        try {
   142→            $lwfRate = $db->fetch(
   143→                "SELECT lr.employee_share, lr.contribution_months
   144→                 FROM lwf_rates lr
   145→                 JOIN states s ON lr.state_id = s.id
   146→                 WHERE (s.state_code = ? OR s.state_name = ?)
   147→                 AND lr.is_active = 1 AND lr.effective_from <= CURDATE()
   148→                 ORDER BY lr.effective_from DESC LIMIT 1",
   149→                [$unitState, $unitState]
   150→            );
   151→            if (!$lwfRate) {
   152→                $lwfRate = $db->fetch(
   153→                    "SELECT employee_share, contribution_months
   154→                     FROM lwf_state_rates
   155→                     WHERE (state_name = ? OR state_name = ?) AND is_active = 1
   156→                     AND effective_from <= CURDATE()
   157→                     ORDER BY effective_from DESC LIMIT 1",
   158→                    [$unitState, ucfirst(strtolower($unitState))]
   159→                );
   160→            }
   161→            if ($lwfRate) {
   162→                $contribMonths = array_filter(array_map('intval', explode(',', $lwfRate['contribution_months'] ?? '')));
   163→                if (in_array($month, $contribMonths)) {
   164→                    $lwfAmount = floatval($lwfRate['employee_share']);
   165→                }
   166→            }
   167→        } catch (Exception $e) {}

   172→$clients = $db->fetchAll("SELECT id, name FROM clients WHERE is_active = 1 ORDER BY name");

   175→$units = [];
   176→if ($clientId > 0) {
   177→    $units = $db->fetchAll(

   179→        [$clientId]

   190→    // Ensure loan tables exist before querying them in the loop
   191→    try {
   192→        $db->exec("CREATE TABLE IF NOT EXISTS `employee_loans` (
   193→            `id` int(11) NOT NULL AUTO_INCREMENT,
   194→            `employee_id` int(11) NOT NULL,
   195→            `unit_id` int(11) DEFAULT NULL,
   196→            `loan_type` varchar(50) DEFAULT 'Personal',
   197→            `amount` decimal(12,2) NOT NULL,
   198→            `interest_rate` decimal(5,2) DEFAULT 0.00,
   199→            `tenure_months` int(11) NOT NULL,
   200→            `emi_amount` decimal(12,2) NOT NULL,
   201→            `total_interest` decimal(12,2) DEFAULT 0.00,
   202→            `total_repayable` decimal(12,2) NOT NULL,
   203→            `balance_amount` decimal(12,2) NOT NULL,
   204→            `emi_deducted` int(11) DEFAULT 0,
   205→            `start_month` int(2) NOT NULL,
   206→            `start_year` int(4) NOT NULL,
   207→            `status` enum('Active','Closed','Settled','Written Off') DEFAULT 'Active',
   208→            `remarks` text DEFAULT NULL,
   209→            `created_at` timestamp DEFAULT CURRENT_TIMESTAMP,
   210→            `updated_at` timestamp DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
   211→            PRIMARY KEY (`id`),
   212→            KEY `idx_employee` (`employee_id`),
   213→            KEY `idx_unit` (`unit_id`),
   214→            KEY `idx_status` (`status`)
   215→        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4");
   216→        $db->exec("CREATE TABLE IF NOT EXISTS `loan_emi_log` (
   217→            `id` int(11) NOT NULL AUTO_INCREMENT,
   218→            `loan_id` int(11) NOT NULL,
   219→            `employee_id` int(11) NOT NULL,
   220→            `month` int(2) NOT NULL,
   221→            `year` int(4) NOT NULL,
   222→            `emi_amount` decimal(12,2) NOT NULL,
   223→            `principal_component` decimal(12,2) DEFAULT 0.00,
   224→            `interest_component` decimal(12,2) DEFAULT 0.00,
   225→            `balance_after` decimal(12,2) NOT NULL,
   226→            `deducted_via_payroll` tinyint(1) DEFAULT 1,
   227→            `created_at` timestamp DEFAULT CURRENT_TIMESTAMP,
   228→            PRIMARY KEY (`id`),
   229→            UNIQUE KEY `uniq_loan_month_year` (`loan_id`, `month`, `year`),
   230→            KEY `idx_employee_month` (`employee_id`, `month`, `year`)
   231→        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4");
   232→    } catch (Exception $e) {}
   233→
   234→    $employees = $db->fetchAll(
   235→        "SELECT id, employee_code, full_name, designation
   236→         FROM employees
   237→         WHERE unit_id = ? AND status = 'approved'
   238→         ORDER BY employee_code",
   239→        [$unitId]
   240→    );
   241→
   242→    // Use last day of SELECTED month — not CURDATE() — so editing a past month
   243→    // loads the salary structure that was actually active during that month,
   244→    // not the latest one created for a future month.
   245→    $effDateStr = sprintf('%04d-%02d-%02d', $year, $month, cal_days_in_month(CAL_GREGORIAN, $month, $year));
   246→
   247→    foreach ($employees as $emp) {
   248→        $empId  = (int)$emp['id'];
   249→        $empCode = $emp['employee_code'];
   250→
   251→        // Salary structure (latest effective for selected month)
   252→        $salary = $db->fetch(
   253→            "SELECT * FROM employee_salary_structures
   254→             WHERE employee_id = ? AND effective_from <= ?

   263→                "SELECT * FROM attendance_summary
   264→                 WHERE employee_id = ? AND month = ? AND year = ?",

   267→        } catch (Exception $e) {
   268→            $att = [];
   269→        }
   270→
   271→        // Advances
   272→        $adv = [];
   273→        try {
   274→            $adv = $db->fetch(
   275→                "SELECT COALESCE(adv1,0)+COALESCE(adv2,0)+COALESCE(office_advance,0)+COALESCE(dress_advance,0) as total_advance,
   276→                        COALESCE(adv1,0) as adv1,
   277→                        COALESCE(adv2,0) as adv2,
   278→                        COALESCE(office_advance,0) as office_advance,
   279→                        COALESCE(dress_advance,0) as dress_advance
   280→                 FROM employee_advances
   281→                 WHERE employee_id = ? AND month = ? AND year = ?",
   282→                [$empId, $month, $year]
   283→            );
   284→        } catch (Exception $e) {
   285→            $adv = [];
   286→        }
   287→
   288→        // Existing payroll record (if already saved for this month/year)
   289→        $existing = [];
   290→        try {
   291→            // Try direct month/year lookup first (new schema)
   292→            $existing = $db->fetch(
   293→                "SELECT * FROM payroll
   294→                 WHERE employee_id = ? AND month = ? AND year = ?",
   295→                [$empCode, $month, $year]
   296→            );
   297→        } catch (Exception $e) {
   298→            $existing = [];
   299→        }
   300→
   301→        // Active loan EMI for this employee (with balance > 0, not already deducted this month)
   302→        $loanEmi = 0;
   303→        try {
   304→            // Sum EMI from ALL active loans
   305→            $loans = $db->fetchAll(
   306→                "SELECT el.emi_amount, el.id as loan_id
   307→                 FROM employee_loans el
   308→                 WHERE el.employee_id = ? AND el.status = 'Active'
   309→                 AND el.balance_amount > 0
   310→                 AND (el.start_year < ? OR (el.start_year = ? AND el.start_month <= ?))
   311→                 AND el.id NOT IN (
   312→                     SELECT lem.loan_id FROM loan_emi_log lem
   313→                     WHERE lem.month = ? AND lem.year = ?
   314→                 )",
   315→                [$empId, $year, $year, $month, $month, $year]
   316→            );
   317→            foreach ($loans as $loan) {
   318→                $loanEmi += floatval($loan['emi_amount']);
   319→            }
   320→        } catch (Exception $e) {
   321→            // Fallback: try without emi_log check (in case table has different structure)
   322→            try {
   323→                $loans2 = $db->fetchAll(
   324→                    "SELECT el.emi_amount, el.id as loan_id
   325→                     FROM employee_loans el
   326→                     WHERE el.employee_id = ? AND el.status = 'Active'
   327→                     AND el.balance_amount > 0
   328→                     AND (el.start_year < ? OR (el.start_year = ? AND el.start_month <= ?))",
   329→                    [$empId, $year, $year, $month]
   330→                );
   331→                foreach ($loans2 as $loan2) {
   332→                    $loanEmi += floatval($loan2['emi_amount']);
   333→                }
   334→            } catch (Exception $e2) {
   335→                $loanEmi = 0;
   336→            }
   337→        }
   338→
   339→        $employeeData[] = [
   340→            'emp'      => $emp,
   341→            'salary'   => $salary ?: [],
   342→            'att'      => $att ?: [],
   343→            'adv'      => $adv ?: [],
   344→            'existing' => $existing ?: [],
   345→            'loan_emi' => $loanEmi,
   346→        ];
   347→    }
   348→}

   352→               7=>'July','August','September','October','November','December'];

  1432→    function previewModalCalc() {

  1439→
  1440→        var wBasicDa   = parseFloat($('#modalWBasicDa').val()) || 0;
  1441→        var wHra       = parseFloat($('#modalWHra').val()) || 0;
  1442→        var wLeaveEnc  = parseFloat($('#modalWLeaveEnc').val()) || 0;
  1443→        var wBonusEnc  = parseFloat($('#modalWBonusEnc').val()) || 0;
  1444→        var wWash      = parseFloat($('#modalWWash').val()) || 0;

  1446→
  1447→        var advance    = parseFloat($('#modalAdvance').val()) || 0;
  1448→        var officeDed  = parseFloat($('#modalOfficeDed').val()) || 0;
  1449→        var loanEmiM   = parseFloat($('#modalLoanEmi').val()) || 0;

  1452→            $('#modalPreviewGross').text('₹0.00');
  1453→            $('#modalPreviewDed').text('₹0.00');
  1454→            $('#modalPreviewNet').text('₹0.00');

  1458→        var pfApp   = $('#modalPfApp').is(':checked');
  1459→        var esiApp  = $('#modalEsiApp').is(':checked');
  1460→        var ptApp   = $('#modalPtApp').is(':checked');
  1461→        var lwfApp  = $('#modalLwfApp').is(':checked');
  1462→
  1463→        var otMult  = modalCurrentRow ? (parseFloat(modalCurrentRow.data('ot-multiplier')) || 1) : 1;
  1464→        var lwfAmt  = modalCurrentRow ? (parseFloat(modalCurrentRow.data('lwf-amount')) || 0) : 0;

  1467→        var basicDa  = round2(wBasicDa * paidDays / totalDays);

  1472→        var otAmt    = round2(wOtRate * otHours * otMult);
