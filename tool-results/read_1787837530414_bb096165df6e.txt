    40→    // 2. Reject single entry (rejection_reason required)

    42→        $id     = (int) ($_POST['id'] ?? 0);
    43→        $reason = sanitize(trim($_POST['rejection_reason'] ?? ''));
    44→        if ($id > 0 && $reason !== '') {

    46→                'ess_expenses',

    52→                'id = :id',
    53→                ['id' => $id]
    54→            );
    55→            setFlash('success', 'Expense entry rejected.');
    56→        } elseif ($id > 0 && $reason === '') {
    57→            setFlash('danger', 'Rejection reason is required.');
    58→        }
    59→        redirect('index.php?page=expense/approvals');
    60→    }
    61→
    62→    // 3. Edit entry (amount & description) – stay on same page
    63→    if ($action === 'edit_entry') {
    64→        $id   = (int) ($_POST['id'] ?? 0);

   215→function categoryBadge($cat): string
   216→{
   217→    if (empty($cat)) $cat = 'expense';

   220→        'expense'          => 'info',

   258→function filterUrl($baseUrl, $override = []): string
   259→{
   260→    $params = $_GET;
   261→    foreach ($override as $k => $v) {
   262→        if ($v === '' || $v === 'all') {
   263→            unset($params[$k]);
   264→        } else {
   265→            $params[$k] = $v;
   266→        }
   267→    }
   268→    return $baseUrl . (empty($params) ? '' : '&' . http_build_query($params));

   623→document.addEventListener('DOMContentLoaded', function () {
   624→
   625→    const allCheckboxes    = document.querySelectorAll('.row-checkbox');
   626→    const selectAllHeader  = document.getElementById('selectAllHeader');
   627→    const selectAllBelow   = document.getElementById('selectAll');
   628→    const selectedCountEl  = document.getElementById('selectedCount');
   629→    const bulkApproveBtn   = document.getElementById('bulkApproveBtn');
   630→    const selectedIdsInput = document.getElementById('selectedIdsInput');
   631→    const bulkApproveForm  = document.getElementById('bulkApproveForm');
   632→
   633→    function updateSelectedCount() {
   634→        const checked = document.querySelectorAll('.row-checkbox:checked');
   635→        const count   = checked.length;
   636→        selectedCountEl.textContent = count + ' selected';
   637→        bulkApproveBtn.disabled = (count === 0);
   638→        const ids = Array.from(checked).map(cb => cb.value);
   639→        selectedIdsInput.value = ids.join(',');
   640→    }
   641→
   642→    function syncSelectAll() {
   643→        const total  = allCheckboxes.length;
   644→        const checked = document.querySelectorAll('.row-checkbox:checked').length;
   645→        const state  = (checked === total && total > 0);
   646→        if (selectAllHeader) selectAllHeader.checked = state;
   647→        if (selectAllBelow)  selectAllBelow.checked  = state;
   648→    }
   649→
   650→    if (selectAllHeader) {
   651→        selectAllHeader.addEventListener('change', function () {
   652→            allCheckboxes.forEach(cb => { cb.checked = this.checked; });
   653→            syncSelectAll();
   654→            updateSelectedCount();
   655→        });
   656→    }
   657→
   658→    if (selectAllBelow) {
   659→        selectAllBelow.addEventListener('change', function () {
   660→            allCheckboxes.forEach(cb => { cb.checked = this.checked; });
   661→            syncSelectAll();
   662→            updateSelectedCount();
   663→        });
   664→    }
   665→
   666→    allCheckboxes.forEach(cb => {
   667→        cb.addEventListener('change', function () {
   668→            syncSelectAll();
   669→            updateSelectedCount();
   670→        });
   671→    });
   672→
   673→    if (bulkApproveBtn) {
   674→        bulkApproveBtn.addEventListener('click', function () {
   675→            const count = document.querySelectorAll('.row-checkbox:checked').length;
   676→            if (count === 0) return;
   677→            if (!confirm('Approve ' + count + ' selected expense(s)?')) return;
   678→            const existing = bulkApproveForm.querySelectorAll('input[name="selected_ids[]"]');
   679→            existing.forEach(el => el.remove());
   680→            document.querySelectorAll('.row-checkbox:checked').forEach(cb => {
   681→                const input = document.createElement('input');
   682→                input.type  = 'hidden';
   683→                input.name  = 'selected_ids[]';
   684→                input.value = cb.value;
   685→                bulkApproveForm.appendChild(input);
   686→            });
   687→            selectedIdsInput.remove();
   688→            bulkApproveForm.submit();
   689→        });
   690→    }
