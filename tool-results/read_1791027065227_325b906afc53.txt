   567→function markDirty(empId, fieldKey, td) {
   568→    var cellKey = empId + '|' + fieldKey;
   569→    var original = originalData[empId];

   575→        currentValue = input ? input.value : '';

   604→function saveAllChanges() {
   605→    if (dirtyCells.size === 0) return;
   606→    var uniqueEmps = new Set();
   607→    dirtyCells.forEach(function(key) { uniqueEmps.add(key.split('|')[0]); });
   608→    var msg = 'Update ' + uniqueEmps.size + ' employee(s) with ' + dirtyCells.size + ' changed field(s). Are you sure?';
   609→    if (!confirm(msg)) return;
   610→
   611→    var btn = document.getElementById('btnSaveAll');
   612→    btn.disabled = true;
   613→    btn.innerHTML = '<i class="bi bi-spinner me-1"></i><span class="spinner-border spinner-border-sm me-1"></span>Saving...';
   614→
   615→    var employeesData = {};
   616→    dirtyCells.forEach(function(cellKey) {
   617→        var parts = cellKey.split('|');
   618→        var empId = parts[0];
   619→        var fieldName = parts[1];
   620→        var td = document.querySelector('td[data-emp-id="' + empId + '"][data-field="' + fieldName + '"]');
   621→        if (!td) return;
   622→        var input = td.querySelector('.bulk-input');
   623→        if (!input) return;
   624→        if (!employeesData[empId]) employeesData[empId] = { id: parseInt(empId), fields: {} };
   625→        if (input.type === 'checkbox') {
   626→            employeesData[empId].fields[fieldName] = input.checked ? '1' : '0';
   627→        } else if (input.type === 'number' || (ALL_COLUMNS[fieldName] && ALL_COLUMNS[fieldName].type === 'number')) {
   628→            employeesData[empId].fields[fieldName] = parseFloat(input.value) || 0;
   629→        } else {
   630→            employeesData[empId].fields[fieldName] = input.value;
   631→        }
   632→    });
   633→
   634→    fetch('index.php?page=api/bulk-edit', {
   635→        method: 'POST',
   636→        headers: {
   637→            'Content-Type': 'application/json',
   638→            'X-CSRF-Token': CSRF_TOKEN
   639→        },
   640→        body: JSON.stringify({ employees: Object.values(employeesData) })
   641→    })
   642→    .then(function(r) { return r.json(); })
   643→    .then(function(data) {
   644→        if (data.success) {
   645→            Object.keys(employeesData).forEach(function(empId) {
   646→                Object.keys(employeesData[empId].fields).forEach(function(key) {
   647→                    originalData[empId][key] = employeesData[empId].fields[key];
   648→                });
   649→            });
   650→            dirtyCells.clear();
   651→            document.querySelectorAll('.cell-changed').forEach(function(el) { el.classList.remove('cell-changed'); });
   652→            updateSaveButton();
   653→            showSaveStatus('success', data.message || 'Updated ' + data.updated + ' employee(s)');
   654→        } else {
   655→            showSaveStatus('error', data.message || 'Failed to save changes');
   656→            btn.disabled = false;
   657→            updateSaveButton();
   658→        }
   659→    })
   660→    .catch(function(err) {
   661→        showSaveStatus('error', 'Network error. Please try again.');
   662→        btn.disabled = false;
   663→        updateSaveButton();
   664→    });
   665→}

   667→function undoAllChanges() {
   668→    if (dirtyCells.size === 0) return;
   669→    if (!confirm('Undo all changes? This will restore all cells to their original values.')) return;
   670→    dirtyCells.forEach(function(cellKey) {
   671→        var parts = cellKey.split('|');
   672→        var empId = parts[0];
   673→        var fieldName = parts[1];
   674→        var td = document.querySelector('td[data-emp-id="' + empId + '"][data-field="' + fieldName + '"]');
   675→        if (!td) return;
   676→        var input = td.querySelector('.bulk-input');
   677→        if (!input) return;
   678→        var original = originalData[empId][fieldName];
   679→        if (input.type === 'checkbox') {
   680→            input.checked = !!(original == 1 || original === '1' || original === true);
   681→        } else {
   682→            input.value = original || '';
   683→        }
   684→        td.classList.remove('cell-changed');
   685→    });
   686→    dirtyCells.clear();
   687→    updateSaveButton();
   688→    showSaveStatus('info', 'All changes undone.');
   689→}
