   651→<script>
   652→(function() {
   653→    // Select all checkbox
   654→    const selectAll = document.getElementById('selectAll');
   655→    if (selectAll) {
   656→        selectAll.addEventListener('change', function() {
   657→            document.querySelectorAll('.row-checkbox').forEach(function(cb) { cb.checked = this.checked; }.bind(this));
   658→            updateSelectedCount();
   659→        });
   660→    }
   661→
   662→    // Individual checkboxes
   663→    document.querySelectorAll('.row-checkbox').forEach(function(cb) {
   664→        cb.addEventListener('change', updateSelectedCount);
   665→    });
   666→
   667→    function updateSelectedCount() {
   668→        const count = document.querySelectorAll('.row-checkbox:checked').length;
   669→        const el = document.getElementById('selectedCount');
   670→        if (el) el.textContent = count > 0 ? count + ' selected' : '';
   671→    }
   672→
   673→    // Approve single request
   674→    window.approveRequest = function(id) {
   675→        if (!confirm('Are you sure you want to approve this change request? The employee record will be updated.')) return;
   676→
   677→        const form = document.createElement('form');
   678→        form.method = 'POST';
   679→        form.action = '?page=employee/change-requests<?= $fStatus !== 'all' ? '&status=' . $fStatus : '' ?>';
   680→
   681→        form.appendChild(createInput('action', 'approve'));
   682→        form.appendChild(createInput('id', id));
   683→        form.appendChild(createInput('csrf_token', '<?= htmlspecialchars($csrfToken) ?>'));
   684→
   685→        document.body.appendChild(form);
   686→        form.submit();
   687→    };
   688→
   689→    // Show reject modal
   690→    window.showRejectModal = function(id) {
   691→        document.getElementById('rejectId').value = id;
   692→        document.getElementById('rejection_reason').value = '';
   693→        new bootstrap.Modal(document.getElementById('rejectModal')).show();
   694→    };
   695→
   696→    // Bulk approve
   697→    window.bulkApprove = function() {
   698→        const checked = document.querySelectorAll('.row-checkbox:checked');
   699→        if (checked.length === 0) {
   700→            alert('Please select at least one request.');
   701→            return;
   702→        }
   703→        if (!confirm('Approve ' + checked.length + ' selected change request(s)? Employee records will be updated.')) return;
   704→        document.getElementById('bulkForm').submit();
   705→    };
   706→
   707→    function createInput(name, value) {
   708→        const input = document.createElement('input');
   709→        input.type = 'hidden';
   710→        input.name = name;
   711→        input.value = value;
   712→        return input;
   713→    }
   714→})();
   715→</script>
