    28→// Override from GET params (when form is submitted)
    29→if (isset($_GET['client_id'])) $clientId = (int)$_GET['client_id'];
    30→if (isset($_GET['unit_id']))   $unitId   = (int)$_GET['unit_id'];
    31→if (isset($_GET['month']))     $month    = (int)$_GET['month'];
    32→if (isset($_GET['year']))      $year     = (int)$_GET['year'];
    33→$payrollPeriodId = (int)($_GET['period_id'] ?? 0);
    34→
    35→// Save back to session for persistence
    36→$_SESSION['filter_client_id'] = $clientId;
    37→$_SESSION['filter_unit_id']   = $unitId;

    41→$calendarDays = cal_days_in_month(CAL_GREGORIAN, $month, $year);
    42→
    43→// ── Fetch unit's pay_days_type from salary formula ──────────────
    44→$unitFormula = null;
    45→if ($unitId > 0) {
    46→    $unitFormula = $db->fetch(
    47→        "SELECT * FROM unit_salary_formulas
    48→         WHERE unit_id = ? AND is_active = 1

    50→        [$unitId]
    51→    );
    52→}
    53→$payDaysType = $unitFormula['pay_days_type'] ?? 'actual';
    54→
    55→// Calculate totalDays based on pay_days_type
    56→$totalDays = $calendarDays;
    57→switch ($payDaysType) {
    58→    case 'fixed_30':
    59→        $totalDays = 30;
    60→        break;
    61→    case 'previous_month':
    62→        $prevMonth = $month - 1;
    63→        $prevYear  = $year;
    64→        if ($prevMonth < 1) { $prevMonth = 12; $prevYear--; }
    65→        $totalDays = (int)cal_days_in_month(CAL_GREGORIAN, $prevMonth, $prevYear);
    66→        break;
    67→    case 'calendar_minus_sundays':
    68→        // Count all Sundays in the selected month
    69→        $sundays = 0;
    70→        for ($d = 1; $d <= $calendarDays; $d++) {
    71→            $dow = (int)date('N', strtotime("$year-$month-$d"));
    72→            if ($dow === 7) $sundays++;
    73→        }
    74→        $totalDays = $calendarDays - $sundays;
    75→        break;
    76→    default: // 'actual'
    77→        $totalDays = $calendarDays;
    78→        break;
    79→}

    91→} elseif ($unitId > 0) {
    92→    // Fallback: look up period by month/year
    93→    $periodInfo = $db->fetch(
    94→        "SELECT id, pay_days FROM payroll_periods WHERE month = ? AND year = ? ORDER BY id DESC LIMIT 1",
    95→        [$month, $year]
    96→    );
    97→    if ($periodInfo) {
    98→        $payrollPeriodId = (int)$periodInfo['id'];
    99→        $periodPayDays = (int)$periodInfo['pay_days'];
   100→    }
   101→}
   102→// For 'actual' type, prefer payroll_periods.pay_days over calendar days
   103→if ($payDaysType === 'actual' && $periodPayDays > 0) {
   104→    $totalDays = $periodPayDays;

   115→$esiEmployerShare = 3.25;

   118→    $pfRate = $db->fetch("SELECT * FROM pf_rates WHERE is_active = 1 ORDER BY effective_from DESC LIMIT 1");
   119→    if ($pfRate) {
   120→        $pfEmployeeShare = (float)$pfRate['employee_share'];
   121→        $pfWageCeiling  = (float)$pfRate['wage_ceiling'];
   122→        $pfEmployerShare = (float)$pfRate['employer_share_pf'];
   123→        $pfEmployerEps  = (float)$pfRate['employer_share_eps'];
   124→        $pfEmployerEdlis = (float)$pfRate['employer_share_edlis'];
   125→        $pfEpfAdmin     = (float)$pfRate['epf_admin_charges'];
   126→    }
   127→} catch (Exception $e) {}
   128→try {
   129→    $esiRate = $db->fetch("SELECT * FROM esi_rates WHERE is_active = 1 ORDER BY effective_from DESC LIMIT 1");
   130→    if ($esiRate) {
   131→        $esiEmployeeShare = (float)$esiRate['employee_share'];
   132→        $esiEmployerShare = (float)$esiRate['employer_share'];
   133→        $esiWageCeiling   = (float)$esiRate['wage_ceiling'];
   134→    }
   135→} catch (Exception $e) {}

   137→// ── Fetch unit state + PT + LWF configuration ────────────────
   138→$unitState = '';
   139→$unitStatePtApplicable = false;
   140→$ptSlabs = [];
   141→$lwfAmount = 0; // Default 0, will be set per-state below
   142→if ($unitId > 0) {
   143→    $unitInfo = $db->fetch(
   144→        "SELECT u.state, s.pt_applicable as state_pt_applicable, s.id as state_id
   145→         FROM units u
   146→         LEFT JOIN states s ON (u.state = s.state_code OR u.state = s.state_name)
   147→         WHERE u.id = ?",
   148→        [$unitId]
   149→    );
   150→    if ($unitInfo) {
   151→        $unitState = $unitInfo['state'] ?? '';
   152→        $unitStatePtApplicable = !empty($unitInfo['state_pt_applicable']);
   153→    }
   154→    // Fetch PT slabs for this state
   155→    if ($unitStatePtApplicable && $unitState) {
   156→        try {
   157→            $ptSlabs = $db->fetchAll(
   158→                "SELECT min_gross, max_gross, pt_amount
   159→                 FROM professional_tax_slabs
   160→                 WHERE (state_name = ? OR state_name = ?) AND is_active = 1
   161→                 ORDER BY min_gross",
   162→                [$unitState, ucfirst(strtolower($unitState))]
   163→            );
   164→        } catch (Exception $e) {}
   165→    }
   166→    // Fetch LWF rate for this state (from lwf_rates → lwf_state_rates fallback)
   167→    if ($unitState) {
   168→        try {
   169→            $lwfRate = $db->fetch(
   170→                "SELECT lr.employee_share, lr.contribution_months
   171→                 FROM lwf_rates lr
   172→                 JOIN states s ON lr.state_id = s.id
   173→                 WHERE (s.state_code = ? OR s.state_name = ?)
   174→                 AND lr.is_active = 1 AND lr.effective_from <= CURDATE()
   175→                 ORDER BY lr.effective_from DESC LIMIT 1",
   176→                [$unitState, $unitState]
   177→            );
   178→            if (!$lwfRate) {
   179→                $lwfRate = $db->fetch(
   180→                    "SELECT employee_share, contribution_months
   181→                     FROM lwf_state_rates
   182→                     WHERE (state_name = ? OR state_name = ?) AND is_active = 1
   183→                     AND effective_from <= CURDATE()
   184→                     ORDER BY effective_from DESC LIMIT 1",
   185→                    [$unitState, ucfirst(strtolower($unitState))]
   186→                );
   187→            }
   188→            if ($lwfRate) {
   189→                $contribMonths = array_filter(array_map('intval', explode(',', $lwfRate['contribution_months'] ?? '')));
   190→                if (in_array($month, $contribMonths)) {
   191→                    $lwfAmount = floatval($lwfRate['employee_share']);
   192→                }
   193→            }
   194→        } catch (Exception $e) {}
   195→    }

   217→        "SELECT id, employee_code, full_name, designation
   218→         FROM employees
   219→         WHERE unit_id = ? AND status IN ('approved', 'active')
   220→         ORDER BY employee_code",
   221→        [$unitId]

   225→
   226→    foreach ($employees as $emp) {
   227→        $empId  = (int)$emp['id'];
   228→        $empCode = $emp['employee_code'];
   229→
   230→        // Salary structure (latest effective for selected month)
   231→        $salary = $db->fetch(
   232→            "SELECT * FROM employee_salary_structures
   233→             WHERE employee_id = ? AND effective_from <= ?
   234→             ORDER BY effective_from DESC LIMIT 1",
   235→            [$empId, $effDateStr]
   236→        );
   237→
   238→        // Attendance
   239→        $att = [];
   240→        try {
   241→            $att = $db->fetch(
   242→                "SELECT * FROM attendance_summary
   243→                 WHERE employee_id = ? AND month = ? AND year = ?",
   244→                [$empId, $month, $year]
   245→            );
   246→        } catch (Exception $e) {
   247→            $att = [];
   248→        }
   249→
   250→        // Advances
   251→        $adv = [];
   252→        try {
   253→            $adv = $db->fetch(
   254→                "SELECT COALESCE(adv1,0)+COALESCE(adv2,0)+COALESCE(office_advance,0)+COALESCE(dress_advance,0) as total_advance,
   255→                        COALESCE(adv1,0) as adv1,
   256→                        COALESCE(adv2,0) as adv2,
   257→                        COALESCE(office_advance,0) as office_advance,
   258→                        COALESCE(dress_advance,0) as dress_advance
   259→                 FROM employee_advances
   260→                 WHERE employee_id = ? AND month = ? AND year = ?",
   261→                [$empId, $month, $year]
   262→            );
   263→        } catch (Exception $e) {
   264→            $adv = [];
   265→        }
   266→
   267→        // Existing payroll record (if already processed) — prefer period_id lookup
   268→        $existing = [];
   269→        try {
   270→            if ($payrollPeriodId > 0) {
   271→                $existing = $db->fetch(
   272→                    "SELECT * FROM payroll
   273→                     WHERE employee_id = ? AND payroll_period_id = ?",
   274→                    [$empCode, $payrollPeriodId]
   275→                );
   276→            }
   277→            if (empty($existing)) {
   278→                // Fallback: join via payroll_periods to find by month/year
   279→                $existing = $db->fetch(
   280→                    "SELECT p.* FROM payroll p
   281→                     JOIN payroll_periods pp ON p.payroll_period_id = pp.id
   282→                     WHERE p.employee_id = ? AND pp.month = ? AND pp.year = ?
   283→                     ORDER BY p.id DESC LIMIT 1",
   284→                    [$empCode, $month, $year]
   285→                );
   286→            }
   287→        } catch (Exception $e) {
   288→            $existing = [];
   289→        }
   290→
   291→        // Active loan EMI for this employee (with balance > 0, not already deducted this month)
   292→        $loanEmi = 0;
   293→        try {
   294→            // Sum EMI from ALL active loans (match process.php logic)
   295→            $loans = $db->fetchAll(
   296→                "SELECT el.emi_amount, el.id as loan_id
   297→                 FROM employee_loans el
   298→                 WHERE el.employee_id = ? AND el.status = 'Active'
   299→                 AND el.balance_amount > 0
   300→                 AND (el.start_year < ? OR (el.start_year = ? AND el.start_month <= ?))
   301→                 AND el.id NOT IN (
   302→                     SELECT lem.loan_id FROM loan_emi_log lem
   303→                     WHERE lem.month = ? AND lem.year = ?
   304→                 )",
   305→                [$empId, $year, $year, $month, $month, $year]
   306→            );
   307→            foreach ($loans as $loan) {
   308→                $loanEmi += floatval($loan['emi_amount']);
   309→            }
   310→        } catch (Exception $e) {
   311→            // Fallback: try without emi_log check (in case table has different structure)
   312→            try {
   313→                $loans2 = $db->fetchAll(
   314→                    "SELECT el.emi_amount, el.id as loan_id
   315→                     FROM employee_loans el
   316→                     WHERE el.employee_id = ? AND el.status = 'Active'
   317→                     AND el.balance_amount > 0
   318→                     AND (el.start_year < ? OR (el.start_year = ? AND el.start_month <= ?))",
   319→                    [$empId, $year, $year, $month]
   320→                );
   321→                foreach ($loans2 as $loan2) {
   322→                    $loanEmi += floatval($loan2['emi_amount']);
   323→                }
   324→            } catch (Exception $e2) {
   325→                $loanEmi = 0;
   326→            }
   327→        }
   328→
   329→        $employeeData[] = [
   330→            'emp'      => $emp,
   331→            'salary'   => $salary ?: [],
   332→            'att'      => $att ?: [],
   333→            'adv'      => $adv ?: [],
   334→            'existing' => $existing ?: [],
   335→            'loan_emi' => $loanEmi,
   336→        ];
   337→    }
   338→}

   342→               7=>'July','August','September','October','November','December'];

  1426→    function previewModalCalc() {
  1427→        var totalDays = TOTAL_DAYS;
  1428→        var attPresent = parseFloat($('#modalAttPresent').val()) || 0;
  1429→        var attWO      = parseFloat($('#modalAttWO').val()) || 0;
  1430→        var attExtra   = parseFloat($('#modalAttExtra').val()) || 0;
  1431→        var otHours    = parseFloat($('#modalOtHours').val()) || 0;
  1432→        var paidDays   = attPresent + attWO + attExtra;
  1433→
  1434→        var wBasicDa   = parseFloat($('#modalWBasicDa').val()) || 0;
  1435→        var wHra       = parseFloat($('#modalWHra').val()) || 0;
  1436→        var wLeaveEnc  = parseFloat($('#modalWLeaveEnc').val()) || 0;
  1437→        var wBonusEnc  = parseFloat($('#modalWBonusEnc').val()) || 0;
  1438→        var wWash      = parseFloat($('#modalWWash').val()) || 0;
  1439→        var wOtRate    = parseFloat($('#modalWOtRate').val()) || 0;
  1440→
  1441→        var advance    = parseFloat($('#modalAdvance').val()) || 0;
  1442→        var officeDed  = parseFloat($('#modalOfficeDed').val()) || 0;
  1443→        var loanEmiM   = parseFloat($('#modalLoanEmi').val()) || 0;
  1444→
  1445→        if (totalDays === 0 || paidDays === 0) {
  1446→            $('#modalPreviewGross').text('₹0.00');
  1447→            $('#modalPreviewDed').text('₹0.00');
  1448→            $('#modalPreviewNet').text('₹0.00');
  1449→            return;
  1450→        }
  1451→
  1452→        var pfApp   = $('#modalPfApp').is(':checked');
  1453→        var esiApp  = $('#modalEsiApp').is(':checked');
  1454→        var ptApp   = $('#modalPtApp').is(':checked');
  1455→        var lwfApp  = $('#modalLwfApp').is(':checked');
  1456→
  1457→        var otMult  = modalCurrentRow ? (parseFloat(modalCurrentRow.data('ot-multiplier')) || 1) : 1;
  1458→        var lwfAmt  = modalCurrentRow ? (parseFloat(modalCurrentRow.data('lwf-amount')) || 0) : 0;
  1459→        var existGr = modalCurrentRow ? (parseFloat(modalCurrentRow.data('existing-gross')) || 0) : 0;
  1460→
  1461→        var basicDa  = round2(wBasicDa * paidDays / totalDays);
  1462→        var hra      = round2(wHra * paidDays / totalDays);
  1463→        var leaveEnc = round2(wLeaveEnc * paidDays / totalDays);
  1464→        var bonusEnc = round2(wBonusEnc * paidDays / totalDays);
  1465→        var wash     = round2(wWash * paidDays / totalDays);
  1466→        var otAmt    = round2(wOtRate * otHours * otMult);
  1467→
  1468→        var gross = round2(basicDa + hra + leaveEnc + bonusEnc + wash + otAmt);
  1469→        var pf    = pfApp ? round2(Math.min(basicDa, PF_WAGE_CEILING) * PF_EMPLOYEE_SHARE / 100) : 0;
  1470→        var esiG  = existGr > 0 ? existGr : 0; // full monthly gross
  1471→        var esi   = (esiApp && esiG > 0 && esiG <= ESI_WAGE_CEILING) ? round2(gross * ESI_EMPLOYEE_SHARE / 100) : 0;
  1472→        var pt    = calculatePT(gross, ptApp);
  1473→        var lwf   = lwfApp ? lwfAmt : 0;
  1474→        var ded   = round2(pf + esi + pt + lwf + advance + officeDed + loanEmiM);
  1475→        var net   = Math.round(gross - ded); // Round to nearest ₹.00
  1476→
  1477→        $('#modalPreviewGross').text(fmt(gross));
  1478→        $('#modalPreviewDed').text(fmt(ded));
  1479→        $('#modalPreviewNet').text(fmt(net));
  1480→    }
