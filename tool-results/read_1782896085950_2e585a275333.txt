    79→// ── Helpers ────────────────────────────────────────────
    80→function getInitials(name: string): string {
    81→  return name
    82→    .split(' ')
    83→    .map((w) => w[0])
    84→    .filter(Boolean)
    85→    .slice(0, 2)
    86→    .join('')
    87→    .toUpperCase();
    88→}
    89→
    90→function maskMobile(mobile: string): string {
    91→  // Show full mobile number — no masking
    92→  return mobile || '';
    93→}
    94→
    95→function maskSensitive(value: string | number | undefined): string {
    96→  const str = value != null ? String(value) : '';
    97→  if (!str) return '****';
    98→  if (str.length <= 4) return '****';
    99→  const first = str.slice(0, 2);
   100→  const last = str.slice(-2);
   101→  const middleLen = str.length - 4;
   102→  return first + '*'.repeat(middleLen) + last;
   103→}
   104→
   105→function formatDate(dateStr?: string): string {
   106→  if (!dateStr) return '—';
   107→  const d = new Date(dateStr);
   108→  return d.toLocaleDateString('en-IN', {
   109→    day: 'numeric',
   110→    month: 'short',
   111→    year: 'numeric',
   112→  });
   113→}
   114→
   115→function isActive(employee: Employee): boolean {
   116→  return !employee.date_of_leaving && employee.status !== 'inactive' && employee.status !== 'resigned';
   117→}

   133→  const [page, setPage] = useState(1);

   167→    } catch (err) {

   170→      setFiltersLoading(false);

   182→        requester_id: employeeId,

   187→        unit_id: selectedUnit ? Number(selectedUnit) : undefined,

   189→        unit_ids: unitIds.length > 0 ? unitIds : undefined,

   192→        toast.error(fetchError);
   193→        return;

   204→      setTotal(effectiveTotal);
   205→      setTotalPages(effectiveTotalPages);

   207→      console.error('Failed to fetch employees:', err);
   208→      setError('Failed to load directory. Please try again.');
   209→      toast.error('Failed to load employee directory');
   210→    } finally {
   211→      setLoading(false);
   212→    }
   213→  }, [scope, employeeId, page, searchQuery, selectedClient, selectedUnit, unitIds]);

   216→  const pullRefresh = usePullToRefresh<HTMLDivElement>({
   217→    onRefresh: loadEmployees,
   218→  });
   219→
   220→  useEffect(() => {
   221→    loadFilters();
   222→  }, [loadFilters]);
   223→
   224→  // Auto-load employees on mount for non-employee roles (supervisor/manager/admin)

   244→  }, [filterKey]);

   260→    setSearchInput('');
   261→    setSelectedClient('');
   262→    setSelectedUnit('');

   264→
   265→  const hasActiveFilters = searchQuery || (selectedClient && selectedClient !== 'all_clients') || (selectedUnit && selectedUnit !== 'all_units');

   275→      e.designation || '',

   288→    setProfileLoading(true);

   290→      const { data: fullEmp } = await fetchEmployeeById(emp.id);
   291→      if (fullEmp) {
   292→        setSelectedEmployee(fullEmp);
   293→      }
   294→    } catch (err) {
   295→      console.error('Failed to fetch full employee details:', err);
   296→      // Keep the basic data from the list
   297→    } finally {
   298→      setProfileLoading(false);
   299→    }
   300→  };
   301→
   302→  const emp = selectedEmployee;
   303→
   304→  // Pull-to-refresh wrapper props
   305→  const pullRefreshProps = {
   306→    ref: pullRefresh.containerRef,
   307→    onTouchStart: pullRefresh.handleTouchStart,
   308→    onTouchMove: pullRefresh.handleTouchMove,
   309→    onTouchEnd: pullRefresh.handleTouchEnd,
   310→  };
   311→
   312→  // ── Render ──
   313→  return (
   314→    <div {...pullRefreshProps} className="flex flex-col gap-4 pb-6" style={{ touchAction: 'pan-y' }}>
   315→      {/* Pull-to-refresh indicator */}
   316→      <div style={pullRefresh.pullIndicatorStyle} className="flex items-center justify-center">

   322→          <h2 className="text-xl font-bold tracking-tight">Directory</h2>
   323→          <p className="text-sm text-muted-foreground">
   324→          {searchedOnce && (total > 0 || employees.length > 0) ? `${total || employees.length} employee${(total || employees.length) > 1 ? 's' : ''} found` :
   325→           searchedOnce && employees.length === 0 ? 'No employees found' :
   326→           'Search employees by name, client, or unit'}

   330→          <Button variant="outline" size="sm" className="gap-1.5 shrink-0 mt-0.5" onClick={handleExport}>
   331→            <Download className="h-4 w-4" />
   332→            <span className="text-xs font-medium hidden sm:inline">Export</span>
   333→          </Button>
   334→        )}
   335→      </div>
   336→
   337→      {/* Search */}
   338→      <div className="relative">
   339→        <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   340→        <Input
   341→          placeholder="Search by name or employee code..."
   342→          className="pl-9 pr-9"
   343→          value={searchInput}
   344→          onChange={(e) => setSearchInput(e.target.value)}
   345→          onKeyDown={(e) => {
   346→            if (e.key === 'Enter') {
   347→              setSearchQuery(searchInput.trim());

   350→        />

   352→          <button
   353→            onClick={() => { setSearchInput(''); setSearchQuery(''); }}
   354→            className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   355→          >
   356→            <X className="h-4 w-4" />

   359→      </div>

   361→      {/* Filter dropdowns */}
   362→      <div className="grid grid-cols-2 gap-2">

   365→            <SelectValue placeholder={filtersLoading ? 'Loading...' : 'All Clients'} />
   366→          </SelectTrigger>

   370→              <SelectItem key={c.id} value={String(c.id)}>
   371→                {c.name}

   375→        </Select>
   376→
   377→        <Select value={selectedUnit} onValueChange={setSelectedUnit}>
   378→          <SelectTrigger className="h-9 text-sm">
   379→            <SelectValue placeholder={filtersLoading ? 'Loading...' : 'All Units'} />
   380→          </SelectTrigger>
   381→          <SelectContent>
   382→            <SelectItem value="all_units">All Units</SelectItem>
   383→            {filteredUnits.map((u) => (
   384→              <SelectItem key={u.id} value={String(u.id)}>

   392→      {/* Pagination - under dropdowns, before list */}

   395→          <p className="text-xs text-muted-foreground">
   396→            {(page - 1) * PAGE_SIZE + 1}–{Math.min(page * PAGE_SIZE, total)} of {total}

   523→                {/* Avatar */}
   524→                <div className="relative shrink-0">

   550→                  <div className="flex items-center gap-3 mt-0.5">
   551→                    {(emp.client_name || emp.unit_name) && (
   552→                      <span className="text-xs text-muted-foreground/80 truncate">
   553→                        {[emp.client_name, emp.unit_name].filter(Boolean).join(' / ')}
   554→                      </span>
   555→                    )}
   556→                  </div>
   557→                </div>
   558→
   559→                {/* Employee Code + Mobile */}
   560→                <div className="flex flex-col items-end shrink-0 gap-0.5">
   561→                  {emp.employee_code && (
   562→                    <span className="text-xs font-bold font-mono text-gray-700">{emp.employee_code}</span>
   563→                  )}
   564→                  {emp.mobile_number && (
   565→                    <div className="flex items-center gap-1.5 text-xs text-muted-foreground">
   566→                      <Phone className="h-3 w-3" />
   567→                      <span>{maskMobile(emp.mobile_number)}</span>
   568→                    </div>
   569→                  )}
   570→                </div>
   571→              </button>
   572→            ))}
   573→          </div>
   574→
   575→          {/* Pagination at bottom of list */}
   576→          {totalPages > 1 && (
   577→            <div className="flex items-center justify-center gap-2 pt-2">
   578→              <Button
   579→                variant="outline"
   580→                size="icon"
   581→                className="h-8 w-8"
   582→                disabled={page <= 1}
   583→                onClick={() => setPage((p) => Math.max(1, p - 1))}
   584→              >
   585→                <ChevronLeft className="h-4 w-4" />
   586→              </Button>
   587→              <span className="text-sm text-muted-foreground">Page {page} of {totalPages}</span>
   588→              <Button
   589→                variant="outline"
   590→                size="icon"
   591→                className="h-8 w-8"
   592→                disabled={page >= totalPages}
   593→                onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   594→              >
   595→                <ChevronRight className="h-4 w-4" />
   596→              </Button>
   597→            </div>
   598→          )}
   599→        </>
   600→      )}
   601→
   602→      {/* ── Profile Dialog (Full Detail) ── */}
   603→      <Dialog open={profileOpen} onOpenChange={setProfileOpen}>
   604→        <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-md p-0">
   605→          <DialogTitle className="sr-only">Employee Profile — {emp?.full_name || 'Employee'}</DialogTitle>
   606→          {emp && (
   607→            <>
   608→              {profileLoading ? (
   609→                <div className="flex items-center justify-center py-20">
   610→                  <Loader2 className="h-8 w-8 animate-spin text-primary" />
   611→                </div>
   612→              ) : (
   613→                <>
   614→                  {/* Header with avatar */}
   615→                  <div className="flex flex-col items-center gap-3 pt-6 pb-4 px-4">
   616→                    <div className="relative">
   617→                      <Avatar className="h-20 w-20 border-2 border-primary/20">
   618→                        <AvatarImage src={getFileUrl(emp.profile_pic_cropped_url || emp.profile_pic_url) || undefined} alt={emp.full_name} />
   619→                        <AvatarFallback className="bg-primary/10 text-primary text-xl font-semibold">
   620→                          {getInitials(emp.full_name || 'U')}
   621→                        </AvatarFallback>
   622→                      </Avatar>
   623→                      <span
   624→                        className={cn(
   625→                          'absolute bottom-1 right-1 h-4 w-4 rounded-full border-2 border-background',
   626→                          isActive(emp) ? 'bg-emerald-500' : 'bg-slate-400'
   627→                        )}
   628→                      />
   629→                    </div>
   630→                    <div className="text-center">
   631→                      <h3 className="text-lg font-bold">{emp.full_name}</h3>
   632→                      <p className="text-sm text-muted-foreground">
   633→                        {emp.employee_code || `EMP-${emp.id}`}
   634→                      </p>
   635→                      <div className="flex items-center justify-center gap-2 mt-1">
   636→                        <Badge
   637→                          variant="outline"
   638→                          className={cn(
   639→                            isActive(emp)
   640→                              ? 'bg-emerald-100 text-emerald-700 border-emerald-200'
   641→                              : 'bg-slate-100 text-slate-500 border-slate-200'
   642→                          )}
   643→                        >
   644→                          {isActive(emp) ? 'Active' : 'Inactive'}
   645→                        </Badge>
   646→                        {emp.designation && (
   647→                          <Badge variant="secondary" className="text-[10px]">
   648→                            {emp.designation}
   649→                          </Badge>
   650→                        )}
   651→                      </div>
   652→                    </div>
   653→                  </div>
   654→
   655→                  <div className="px-4 pb-6 space-y-4">
   656→                    {/* Personal Information */}
   657→                    <ProfileSection title="Personal">
   658→                      <ProfileRow icon={User} label="Father/Husband Name" value={emp.father_name} />
   659→                      <ProfileRow icon={Calendar} label="Date of Birth" value={emp.date_of_birth ? formatDate(emp.date_of_birth) : undefined} />
   660→                      <ProfileRow icon={User} label="Gender" value={emp.gender} />
   661→                      <ProfileRow icon={Heart} label="Marital Status" value={emp.marital_status} />
   662→                      <ProfileRow icon={Heart} label="Blood Group" value={emp.blood_group} />
   663→                    </ProfileSection>
   664→
   665→                    {/* Contact */}
   666→                    <ProfileSection title="Contact">
   667→                      <ProfileRow icon={Phone} label="Mobile" value={emp.mobile_number ? maskMobile(emp.mobile_number) : undefined} />
   668→                      <SensitiveProfileRow icon={Phone} label="Alternate Mobile" value={emp.alternate_mobile} />
   669→                      <ProfileRow icon={Mail} label="Email" value={emp.email} />
   670→                      <ProfileRow icon={MapPin} label="Address" value={emp.address} />
   671→                      <ProfileRow
   672→                        icon={MapPin}
   673→                        label="Location"
   674→                        value={[emp.city, emp.district, emp.state].filter(Boolean).join(', ') || undefined}
   675→                      />
   676→                      <ProfileRow icon={Hash} label="PIN Code" value={emp.pin_code} />
   677→                    </ProfileSection>
   678→
   679→                    {/* Employment */}
   680→                    <ProfileSection title="Employment">
   681→                      <ProfileRow
   682→                        icon={Building2}
   683→                        label="Client / Unit"
   684→                        value={[emp.client_name, emp.unit_name].filter(Boolean).join(' / ') || undefined}
   685→                      />
   686→                      <ProfileRow icon={Building2} label="Department" value={emp.department} />
   687→                      <ProfileRow icon={Briefcase} label="Employment Type" value={emp.employment_type} />
   688→                      <ProfileRow icon={Shield} label="Worker Category" value={emp.worker_category} />
   689→                      <ProfileRow icon={IdCard} label="Role" value={emp.employee_role} />
   690→                      <ProfileRow icon={Clock} label="Date of Joining" value={emp.date_of_joining ? formatDate(emp.date_of_joining) : undefined} />
   691→                      <ProfileRow icon={Calendar} label="Confirmation Date" value={emp.confirmation_date ? formatDate(emp.confirmation_date) : undefined} />
   692→                      <ProfileRow icon={Clock} label="Probation Period" value={emp.probation_period ? `${emp.probation_period} months` : undefined} />
   693→                      <ProfileRow icon={Calendar} label="Date of Leaving" value={emp.date_of_leaving ? formatDate(emp.date_of_leaving) : undefined} />
   694→                    </ProfileSection>
   695→
   696→                    {/* Government IDs — always show section */}
   697→                    <ProfileSection title="Government IDs">
   698→                      <SensitiveProfileRow icon={IdCard} label="Aadhaar Number" value={emp.aadhaar_number} />
   699→                      <SensitiveProfileRow icon={Hash} label="UAN Number" value={emp.uan_number} />
   700→                      <SensitiveProfileRow icon={Hash} label="ESIC Number" value={emp.esic_number} />
   701→                    </ProfileSection>
   702→
   703→                    {/* Bank Details — always show section */}
   704→                    <ProfileSection title="Bank Details">
   705→                      <ProfileRow icon={CreditCard} label="Bank Name" value={emp.bank_name} />
   706→                      <ProfileRow icon={User} label="Account Holder" value={emp.account_holder_name} />
   707→                      <SensitiveProfileRow icon={CreditCard} label="Account Number" value={emp.account_number} />
   708→                      <SensitiveProfileRow icon={Hash} label="IFSC Code" value={emp.ifsc_code} />
   709→                    </ProfileSection>
   710→
   711→                    {/* Emergency Contact — always show section */}
   712→                    <ProfileSection title="Emergency Contact">
   713→                      <ProfileRow icon={UsersRound} label="Contact Name" value={emp.emergency_contact_name} />
   714→                      <ProfileRow icon={UsersRound} label="Relation" value={emp.emergency_contact_relation} />
   715→                    </ProfileSection>
   716→
   717→                    {/* Nominee — always show section */}
   718→                    <ProfileSection title="Nominee">
   719→                      <ProfileRow icon={UsersRound} label="Nominee Name" value={emp.nominee_name} />
   720→                      <ProfileRow icon={UsersRound} label="Relationship" value={emp.nominee_relationship} />
   721→                      <ProfileRow icon={Calendar} label="Nominee DOB" value={emp.nominee_dob ? formatDate(emp.nominee_dob) : undefined} />
   722→                      <SensitiveProfileRow icon={Phone} label="Nominee Contact" value={emp.nominee_contact} />
   723→                    </ProfileSection>
   724→
   725→                    {/* Documents — always show section */}
   726→                    <ProfileSection title="Documents">
   727→                      {emp.aadhaar_front_url ? (
   728→                        <DocThumbnail
   729→                          label="Aadhaar Front"
   730→                          url={getFileUrl(emp.aadhaar_front_url)}
   731→                          onClick={() => setViewingDoc({ url: getFileUrl(emp.aadhaar_front_url), title: 'Aadhaar Front' })}
   732→                        />
   733→                      ) : (
   734→                        <ProfileRow icon={FileText} label="Aadhaar Front" value={undefined} />
   735→                      )}
   736→                      {emp.aadhaar_back_url ? (
   737→                        <DocThumbnail
   738→                          label="Aadhaar Back"
   739→                          url={getFileUrl(emp.aadhaar_back_url)}
   740→                          onClick={() => setViewingDoc({ url: getFileUrl(emp.aadhaar_back_url), title: 'Aadhaar Back' })}
   741→                        />
   742→                      ) : (
   743→                        <ProfileRow icon={FileText} label="Aadhaar Back" value={undefined} />
   744→                      )}
   745→                      {emp.bank_document_url ? (
   746→                        <DocThumbnail
   747→                          label="Bank Document"
   748→                          url={getFileUrl(emp.bank_document_url)}
   749→                          onClick={() => setViewingDoc({ url: getFileUrl(emp.bank_document_url), title: 'Bank Document' })}
   750→                        />
   751→                      ) : (
   752→                        <ProfileRow icon={FileText} label="Bank Document" value={undefined} />
   753→                      )}
   754→                    </ProfileSection>
   755→                  </div>
   756→                </>
   757→              )}
   758→            </>
   759→          )}
   760→        </DialogContent>
   761→      </Dialog>
   762→
   763→      {/* ── Document Viewer Dialog ── */}
   764→      <Dialog open={!!viewingDoc} onOpenChange={() => setViewingDoc(null)}>
   765→        <DialogContent className="sm:max-w-lg p-0">
   766→          <DialogHeader className="p-3 border-b">
   767→            <DialogTitle className="text-sm">{viewingDoc?.title}</DialogTitle>
   768→          </DialogHeader>
   769→          <div className="p-2">
   770→            {viewingDoc?.url && (
   771→              <img
   772→                src={viewingDoc.url}
   773→                alt={viewingDoc.title}
   774→                className="w-full rounded-lg"
   775→              />
   776→            )}
   777→          </div>
   778→        </DialogContent>
   779→      </Dialog>
   780→    </div>
   781→  );
   782→}

   817→// ── Sensitive Profile Row (with eye toggle) ──────────
   818→function SensitiveProfileRow({
   819→  icon: Icon,
   820→  label,
   821→  value,
   822→}: {
   823→  icon: typeof User;
   824→  label: string;
   825→  value?: string | number;
   826→}) {
   827→  const [revealed, setRevealed] = useState(false);
   828→  const displayValue = value != null ? String(value) : '';
   829→
   830→  if (!displayValue) {
   831→    return (
   832→      <div className="flex items-start gap-3">
   833→        <Icon className="h-4 w-4 text-muted-foreground mt-0.5 shrink-0" />
   834→        <div className="flex flex-col">
   835→          <span className="text-xs text-muted-foreground">{label}</span>
   836→          <span className="text-sm font-medium text-muted-foreground/50 italic">—</span>
   837→        </div>
   838→      </div>
   839→    );
   840→  }
   841→
   842→  return (
   843→    <div className="flex items-start gap-3">
   844→      <Icon className="h-4 w-4 text-muted-foreground mt-0.5 shrink-0" />
   845→      <div className="flex flex-col flex-1 min-w-0">
   846→        <span className="text-xs text-muted-foreground">{label}</span>
   847→        <span className="text-sm font-medium font-mono">
   848→          {revealed ? displayValue : maskSensitive(displayValue)}
   849→        </span>
   850→      </div>
   851→      <button
   852→        type="button"
   853→        onClick={() => setRevealed((r) => !r)}
   854→        className="mt-0.5 p-1 rounded text-muted-foreground hover:text-foreground hover:bg-muted/50 transition-colors shrink-0"
   855→        aria-label={revealed ? `Hide ${label}` : `Reveal ${label}`}
   856→      >
   857→        {revealed ? <EyeOff className="h-3.5 w-3.5" /> : <Eye className="h-3.5 w-3.5" />}
   858→      </button>
   859→    </div>
   860→  );
   861→}
