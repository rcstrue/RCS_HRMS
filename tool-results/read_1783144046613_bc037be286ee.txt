    36→                    break;
    37→                case 'notif_sms_provider':

    40→                case 'notif_email_host':
    41→                    $this->emailConfig['host'] = $s['setting_value'];
    42→                    break;
    43→                case 'notif_email_user':
    44→                    $this->emailConfig['user'] = $s['setting_value'];
    45→                    break;
    46→                case 'notif_email_pass':
    47→                    $this->emailConfig['pass'] = $s['setting_value'];

    66→    /**
    67→     * Send SMS using Fast2SMS (Free tier available)
    68→     * Get API key from: https://docs.fast2sms.com
    69→     */
    70→    public function sendSMS($mobile, $message, $templateId = null) {
    71→        // Clean mobile number
    72→        $mobile = preg_replace(REGEX_NON_NUMERIC, '', $mobile);
    73→        
    74→        if (strlen($mobile) == 10) {
    75→            $mobile = '91' . $mobile;
    76→        }
    77→        
    78→        // Fast2SMS API
    79→        $apiKey = $this->smsApiKey ?? 'YOUR_FAST2SMS_API_KEY';
    80→        
    81→        $url = "https://www.fast2sms.com/dev/bulkV2";
    82→        
    83→        $data = [
    84→            'route' => 'q', // Quick transactional route
    85→            'message' => $message,
    86→            'language' => 'english',
    87→            'flash' => 0,
    88→            'numbers' => substr($mobile, 2) // Remove country code for Fast2SMS
    89→        ];
    90→        
    91→        if ($templateId) {
    92→            $data['route'] = 'dlt'; // DLT route
    93→            $data['template_id'] = $templateId;
    94→        }
    95→        
    96→        $ch = curl_init();
    97→        curl_setopt($ch, CURLOPT_URL, $url);
    98→        curl_setopt($ch, CURLOPT_POST, true);
    99→        curl_setopt($ch, CURLOPT_POSTFIELDS, http_build_query($data));
   100→        curl_setopt($ch, CURLOPT_HTTPHEADER, [
   101→            'authorization: ' . $apiKey,
   102→            'Content-Type: application/x-www-form-urlencoded'
   103→        ]);
   104→        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
   105→        
   106→        $response = curl_exec($ch);
   107→        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
   108→        curl_close($ch);
   109→        
   110→        $result = json_decode($response, true);
   111→        
   112→        // Log the SMS
   113→        $this->logNotification('sms', $mobile, $message, $httpCode == 200 ? 'sent' : 'failed', $response);
   114→        
   115→        return [
   116→            'success' => $httpCode == 200 && ($result['return'] ?? false),
   117→            'message' => $result['message'] ?? 'SMS sent',
   118→            'response' => $result
   119→        ];
   120→    }

   165→    /**
   166→     * Send Email using PHP's mail() or SMTP
   167→     */
   168→    public function sendEmail($to, $subject, $body, $attachments = [], $isHTML = true) {
   169→        $fromEmail = $this->emailConfig['user'] ?? 'noreply@rcshrms.com';
   170→        $fromName = 'RCS HRMS Pro';
   171→        
   172→        // Headers
   173→        $headers = [
   174→            'From' => $fromName . ' <' . $fromEmail . '>',
   175→            'Reply-To' => $fromEmail,
   176→            'X-Mailer' => 'PHP/' . phpversion()
   177→        ];
   178→        
   179→        if ($isHTML) {
   180→            $headers['MIME-Version'] = '1.0';
   181→            $headers['Content-Type'] = 'text/html; charset=UTF-8';
   182→        }
   183→        
   184→        // Convert headers array to string
   185→        $headerStr = '';
   186→        foreach ($headers as $key => $value) {
   187→            $headerStr .= $key . ': ' . $value . "\r\n";
   188→        }
   189→        
   190→        // Handle attachments (simple implementation)
   191→        if (!empty($attachments)) {
   192→            $boundary = md5(time());
   193→            $headerStr = "MIME-Version: 1.0\r\n";
   194→            $headerStr .= "Content-Type: multipart/mixed; boundary=\"$boundary\"\r\n";
   195→            
   196→            $mimeBody = "--$boundary\r\n";
   197→            $mimeBody .= "Content-Type: text/html; charset=UTF-8\r\n";
   198→            $mimeBody .= "Content-Transfer-Encoding: 7bit\r\n\r\n";
   199→            $mimeBody .= $body . "\r\n";
   200→            
   201→            foreach ($attachments as $attachment) {
   202→                if (file_exists($attachment['path'])) {
   203→                    $content = chunk_split(base64_encode(file_get_contents($attachment['path'])));
   204→                    $mimeBody .= "--$boundary\r\n";
   205→                    $mimeBody .= "Content-Type: " . ($attachment['type'] ?? 'application/octet-stream') . "; name=\"" . $attachment['name'] . "\"\r\n";
   206→                    $mimeBody .= "Content-Transfer-Encoding: base64\r\n";
   207→                    $mimeBody .= "Content-Disposition: attachment; filename=\"" . $attachment['name'] . "\"\r\n\r\n";
   208→                    $mimeBody .= $content . "\r\n";
   209→                }
   210→            }
   211→            $mimeBody .= "--$boundary--";
   212→            $body = $mimeBody;
   213→        }
   214→        
   215→        // Send email
   216→        $result = mail($to, $subject, $body, $headerStr);
   217→        
   218→        // Log
   219→        $this->logNotification('email', $to, $subject, $result ? 'sent' : 'failed', '');
   220→        
   221→        return [
   222→            'success' => $result,
   223→            'message' => $result ? 'Email sent successfully' : 'Failed to send email'
   224→        ];
   225→    }

   227→    /**
   228→     * Send email using Gmail SMTP (more reliable)
   229→     */
   230→    public function sendEmailSMTP($to, $subject, $body, $attachments = []) {
   231→        // Check if PHPMailer is available
   232→        $phpmailerPath = APP_ROOT . '/vendor/phpmailer/phpmailer/src/PHPMailer.php';
   233→        
   234→        if (!file_exists($phpmailerPath)) {
   235→            // Fall back to basic mail
   236→            return $this->sendEmail($to, $subject, $body, $attachments);
   237→        }
   238→        
   239→        require_once $phpmailerPath;
   240→        require_once APP_ROOT . '/vendor/phpmailer/phpmailer/src/SMTP.php';
   241→        
   242→        $mail = new PHPMailer\PHPMailer\PHPMailer(true);
   243→        
   244→        try {
   245→            // SMTP Configuration
   246→            $mail->isSMTP();
   247→            $mail->Host = $this->emailConfig['host'] ?? 'smtp.gmail.com';
   248→            $mail->SMTPAuth = true;
   249→            $mail->Username = $this->emailConfig['user'] ?? 'your-email@gmail.com';
   250→            $mail->Password = $this->emailConfig['pass'] ?? 'your-app-password';
   251→            $mail->SMTPSecure = 'tls';
   252→            $mail->Port = 587;
   253→            
   254→            $mail->setFrom($this->emailConfig['user'] ?? 'noreply@rcshrms.com', 'RCS HRMS Pro');
   255→            $mail->addAddress($to);
   256→            
   257→            $mail->isHTML(true);
   258→            $mail->Subject = $subject;
   259→            $mail->Body = $body;
   260→            
   261→            // Attachments
   262→            foreach ($attachments as $attachment) {
   263→                if (file_exists($attachment['path'])) {
   264→                    $mail->addAttachment($attachment['path'], $attachment['name']);
   265→                }
   266→            }
   267→            
   268→            $mail->send();
   269→            
   270→            $this->logNotification('email', $to, $subject, 'sent', '');
   271→            
   272→            return ['success' => true, 'message' => 'Email sent via SMTP'];
   273→            
   274→        } catch (Exception $e) {
   275→            $this->logNotification('email', $to, $subject, 'failed', $e->getMessage());
   276→            
   277→            return ['success' => false, 'message' => $e->getMessage()];
   278→        }
   279→    }

   303→     * Send WhatsApp message automatically via WhatsApp Bot API
   304→     * (Scan QR once, messages sent from your WhatsApp)
   305→     * Falls back to link generation if bot not configured
   306→     */
   307→    public function sendWhatsApp($mobile, $message, $autoSend = true) {
   308→        $mobile = preg_replace(REGEX_NON_NUMERIC, '', $mobile);
   309→        if (strlen($mobile) == 10) {
   310→            $mobile = '91' . $mobile;
   311→        }
   312→        
   313→        // Try WhatsApp Bot API (auto-send via Baileys)
   314→        if ($autoSend && !empty($this->whatsappConfig['api_url']) && !empty($this->whatsappConfig['api_key'])) {
   315→            $botUrl = rtrim($this->whatsappConfig['api_url'], '/') . '/send-message';
   316→            $apiKey = $this->whatsappConfig['api_key'];
   317→            
   318→            $ch = curl_init();
   319→            curl_setopt_array($ch, [
   320→                CURLOPT_URL => $botUrl,
   321→                CURLOPT_POST => true,
   322→                CURLOPT_RETURNTRANSFER => true,
   323→                CURLOPT_TIMEOUT => 30,
   324→                CURLOPT_CONNECTTIMEOUT => 5,
   325→                CURLOPT_HTTPHEADER => [
   326→                    'Content-Type: application/json',
   327→                    'X-API-Key: ' . $apiKey
   328→                ],
   329→                CURLOPT_POSTFIELDS => json_encode([
   330→                    'mobile' => $mobile,
   331→                    'message' => $message
   332→                ])
   333→            ]);
   334→            
   335→            $response = curl_exec($ch);
   336→            $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
   337→            $curlError = curl_error($ch);
   338→            curl_close($ch);
   339→            
   340→            $result = json_decode($response, true);
   341→            
   342→            if ($httpCode == 200 && ($result['success'] ?? false)) {
   343→                $status = ($result['queued'] ?? false) ? 'queued' : 'sent';
   344→                $this->logNotification('whatsapp', $mobile, $message, $status, $response);
   345→                
   346→                return [
   347→                    'success' => true,
   348→                    'message' => $result['message'] ?? 'WhatsApp message sent via bot',
   349→                    'queued' => $result['queued'] ?? false
   350→                ];
   351→            }
   352→            
   353→            // Bot API failed — log and fall back to link
   354→            $this->logNotification('whatsapp', $mobile, $message, 'failed', $response . ' | Error: ' . $curlError);
   355→        }
   356→        
   357→        // Fall back to WhatsApp link generation (manual send)
   358→        $link = $this->generateWhatsAppLink($mobile, $message);
   359→        $this->logNotification('whatsapp', $mobile, $message, 'link_generated', json_encode($link));
   360→        
   361→        return [
   362→            'success' => true,
   363→            'manual' => true,
   364→            'link' => $link['link'],
   365→            'qr_code' => $link['qr_code'],
   366→            'message' => 'Bot not configured — WhatsApp link generated for manual send'
   367→        ];
   368→    }

   548→        
   549→        // Generate email body
   550→        $monthYear = date('F Y', mktime(0, 0, 0, $data['month'], 1, $data['year']));

   552→        $body = $this->getEmailTemplate('payslip', [

   730→    /**
   731→     * Send salary credit WhatsApp to ALL employees in a payroll batch
   732→     */
   733→    public function sendBulkSalaryCreditWhatsApp($payrollPeriodId, $unitId = null) {
   734→        $where = "p.payroll_period_id = :pid AND p.net_pay > 0";
   735→        $params = ['pid' => $payrollPeriodId];
   736→        if ($unitId) {
   737→            $where .= " AND p.unit_id = :uid";
   738→            $params['uid'] = $unitId;
   739→        }
   740→
   741→        $rows = $this->db->fetchAll(
   742→            "SELECT e.full_name, e.mobile, e.employee_code,
   743→                    p.gross_earnings, p.total_deductions, p.net_pay,
   744→                    pp.month, pp.year
   745→             FROM payroll p
   746→             JOIN employees e ON p.employee_id = e.employee_code
   747→             JOIN payroll_periods pp ON p.payroll_period_id = pp.id
   748→             WHERE $where",
   749→            $params
   750→        );
   751→
   752→        if (empty($rows)) {
   753→            return ['success' => false, 'message' => 'No payroll records found', 'sent' => 0];
   754→        }
   755→
   756→        $monthYear = date('F Y', mktime(0, 0, 0, $rows[0]['month'], 1, $rows[0]['year']));
   757→        $recipients = [];
   758→        $sent = 0;
   759→        $failed = 0;
   760→
   761→        foreach ($rows as $row) {
   762→            $mobile = $row['mobile'];
   763→            if (empty($mobile)) { $failed++; continue; }
   764→
   765→            $msg = $this->getWhatsAppTemplate('salary_credit', [
   766→                'name' => $row['full_name'],
   767→                'month_year' => $monthYear,
   768→                'gross' => number_format($row['gross_earnings'] ?? 0, 2),
   769→                'deductions' => number_format($row['total_deductions'] ?? 0, 2),
   770→                'net_pay' => number_format($row['net_pay'] ?? 0, 2)
   771→            ]);
   772→
   773→            $recipients[] = ['to' => $mobile, 'message' => $msg];
   774→        }
   775→
   776→        $result = $this->sendWhatsAppBulk($recipients, '');
   777→        return $result;
   778→    }

   825→        // Check for pending compliance
   826→        $pendingCompliance = $this->db->fetchColumn(
   827→            "SELECT COUNT(*) FROM compliance_filings WHERE status = 'Pending' AND due_date <= DATE_ADD(CURDATE(), INTERVAL 7 DAY)"
   828→        );
   829→        
   830→        if ($pendingCompliance > 0) {
   831→            $alerts[] = [
   832→                'type' => 'warning',
   833→                'icon' => 'exclamation-triangle',
   834→                'title' => 'Compliance Pending',
   835→                'message' => "$pendingCompliance compliance filings due within 7 days",
   836→                'link' => 'index.php?page=compliance/dashboard'
   837→            ];
   838→        }
   839→        
   840→        // Check for pending salary below minimum wage
   841→        $belowMinWage = $this->db->fetchColumn(
   842→            "SELECT COUNT(*) FROM employees e
   843→             LEFT JOIN employee_salary_structures ess ON e.id = ess.employee_id
   844→             LEFT JOIN minimum_wages mw ON e.state = mw.state
   845→             WHERE e.status = 'approved' AND ess.gross_salary < mw.total_per_month"
   846→        );

   850→                'type' => 'danger',

   852→                'title' => 'Minimum Wage Violation',
   853→                'message' => "$belowMinWage employees paid below minimum wage",
   854→                'link' => 'index.php?page=compliance/minimum-wage-check'
   855→            ];
   856→        }
   857→        
   858→        // Check for pending F&F settlements
   859→        try {
   860→            $pendingFF = $this->db->fetchColumn(
   861→                "SELECT COUNT(*) FROM employee_settlements WHERE status = 'pending'"
   862→            );
   863→

   866→                    'type' => 'info',

   869→                    'message' => "$pendingFF F&F settlements pending approval",

   872→            }
   873→        } catch (Exception $e) {
   874→            // employee_settlements table may not exist
   875→        }
   876→        
   877→        // Check for pending approvals
   878→        $pendingApprovals = $this->db->fetchColumn(
   879→            "SELECT COUNT(*) FROM employees WHERE status LIKE 'pending%'"
   880→        );

   903→     * @param string $link Link to navigate when clicked
   904→     * @param string $type Notification type (info, warning, danger, success)
   905→     * @param int|null $userId Specific user ID, or null for all admins

   907→     */
   908→    public function createNotification($title, $message, $link = '', $type = 'info', $userId = null) {
   909→        try {
   910→            // If no specific user, notify all admin/HR users
   911→            if ($userId === null) {
   912→                $users = $this->db->fetchAll(
   913→                    "SELECT u.id FROM users u 
   914→                     JOIN roles r ON u.role_id = r.id 
   915→                     WHERE r.role_code IN ('admin', 'hr_executive') AND u.is_active = 1"
   916→                );
   917→                
   918→                foreach ($users as $user) {
   919→                    $this->db->insert('notifications', [
   920→                        'user_id' => $user['id'],
   921→                        'title' => $title,
   922→                        'message' => $message,
   923→                        'link' => $link,
   924→                        'type' => $type,
   925→                        'is_read' => 0,
   926→                        'created_at' => date('Y-m-d H:i:s')
   927→                    ]);
   928→                }
   929→            } else {
   930→                $this->db->insert('notifications', [
   931→                    'user_id' => $userId,
   932→                    'title' => $title,
   933→                    'message' => $message,
   934→                    'link' => $link,

   942→        } catch (Exception $e) {
