    33→    if (empty($designation)) return false;
    34→    $d = strtolower(trim($designation));
    35→    return (strpos($d, 'hk supervisor') !== false || strpos($d, 'forklift driver') !== false || strpos($d, 'fork lift driver') !== false);

    38→function ensureOwnUnitAllocated($db, $empCode, $unitName) {
    39→    if (empty($empCode) || empty($unitName)) return;
    40→    $existing = $db->fetch(
    41→        "SELECT id FROM user_access WHERE user_id = ? AND access_type = 'unit' AND access_id = ?",
    42→        [$empCode, $unitName]

    44→    if (!$existing) {
    45→        try {
    46→            $db->insert('user_access', [
    47→                'user_id' => $empCode,
    48→                'access_type' => 'unit',
    49→                'access_id' => $unitName
    50→            ]);
    51→        } catch (Exception $e) { /* skip dup */ }
    52→    }

    81→                        'access_type' => 'unit',

   536→function updateCounts() {
   537→    var allChecks = document.querySelectorAll('.unitCheck');
   538→    var selectedTotal = 0;
   539→    var cityCounts = {};
   540→    
   541→    allChecks.forEach(function(cb) {
   542→        var city = cb.getAttribute('data-city');
   543→        if (!cityCounts[city]) cityCounts[city] = {total: 0, checked: 0};
   544→        cityCounts[city].total++;
   545→        if (cb.checked) {
   546→            selectedTotal++;
   547→            cityCounts[city].checked++;
   548→        }
   549→    });
   550→    
   551→    var countEl = document.getElementById('selectedCount');
   552→    if (countEl) countEl.textContent = selectedTotal;
   553→    
   554→    // Update city checkboxes and counts
   555→    document.querySelectorAll('.cityCheck').forEach(function(cityCb) {
   556→        var city = cityCb.getAttribute('data-city');
   557→        if (cityCounts[city]) {
   558→            var info = cityCounts[city];
   559→            cityCb.checked = (info.checked === info.total);
   560→            var label = cityCb.nextElementSibling;
   561→            if (label) {
   562→                var smallEl = label.querySelector('small');
   563→                if (smallEl) smallEl.textContent = info.checked + '/' + info.total;
   564→            }
   565→        }
   566→    });
   567→}
