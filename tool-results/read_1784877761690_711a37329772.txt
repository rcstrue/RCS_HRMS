   418→// ── File handling ──
   419→function handleFileSelect(files) {
   420→    for (const f of files) {
   421→        if (!f.name.toLowerCase().endsWith('.csv')) continue;
   422→        if (selectedFiles.length >= 500) break;
   423→        if (!selectedFiles.some(x => x.name === f.name && x.size === f.size)) {
   424→            selectedFiles.push(f);
   425→        }
   426→    }
   427→    renderFileList();
   428→}
   429→function handleDrop(e) {
   430→    e.preventDefault();
   431→    e.currentTarget.classList.remove('border-primary', 'bg-light');
   432→    handleFileSelect(e.dataTransfer.files);
   433→}
   434→function renderFileList() {
   435→    const container = document.getElementById('fileListContainer');
   436→    const list = document.getElementById('fileList');
   437→    const count = document.getElementById('fileCount');
   438→    const btn = document.getElementById('importBtn');
   439→    count.textContent = selectedFiles.length;
   440→    btn.disabled = selectedFiles.length === 0;
   441→    if (selectedFiles.length === 0) { container.classList.add('d-none'); return; }
   442→    container.classList.remove('d-none');
   443→    let html = '';
   444→    selectedFiles.forEach((f, i) => {
   445→        const sizeKB = (f.size / 1024).toFixed(1);

   457→function startImport() {
   458→    if (selectedFiles.length === 0) return;
   459→    const btn = document.getElementById('importBtn');
   460→    btn.disabled = true;
   461→    btn.innerHTML = '<span class="spinner-border spinner-border-sm me-1"></span>Importing...';
   462→    const progressContainer = document.getElementById('progressContainer');
   463→    const progressBar = document.getElementById('progressBar');
   464→    const progressLabel = document.getElementById('progressLabel');
   465→    const progressPercent = document.getElementById('progressPercent');
   466→    progressContainer.classList.remove('d-none');
   467→    progressBar.style.width = '0%';
   468→    progressBar.classList.remove('bg-danger');
   469→    progressBar.classList.add('bg-primary', 'progress-bar-animated');
   470→    progressLabel.textContent = 'Uploading files...';
   471→    progressPercent.textContent = '0%';
   472→    const formData = new FormData();
   473→    selectedFiles.forEach((f) => formData.append('csv_files[]', f));
   474→    const xhr = new XMLHttpRequest();
   475→    xhr.open('POST', 'index.php?page=employee/esic-import&ajax=1', true);
   476→    xhr.upload.onprogress = function(e) {
   477→        if (e.lengthComputable) {
   478→            const pct = Math.round((e.loaded / e.total) * 100);
   479→            progressBar.style.width = pct + '%';
   480→            progressPercent.textContent = pct + '%';
   481→            progressLabel.textContent = 'Uploading files...';
   482→        }
   483→    };
   484→    xhr.onload = function() {
   485→        btn.disabled = false;
   486→        btn.innerHTML = '<i class="bi bi-play-fill me-1"></i>Import';
   487→        if (xhr.status === 200) {
   488→            try {
   489→                const res = JSON.parse(xhr.responseText);
   490→                if (res.success) {
   491→                    showSummary(res.data);
   492→                    progressBar.style.width = '100%';
   493→                    progressBar.classList.remove('progress-bar-animated');
   494→                    progressLabel.textContent = 'Import complete!';
   495→                    progressPercent.textContent = '100%';
   496→                } else {
   497→                    progressLabel.textContent = 'Error: ' + (res.error || 'Unknown error');
   498→                    progressBar.style.width = '100%';
   499→                    progressBar.classList.remove('progress-bar-animated', 'bg-primary');
   500→                    progressBar.classList.add('bg-danger');
   501→                    alert('Import failed: ' + (res.error || 'Unknown error'));
   502→                }
   503→            } catch(e) {
   504→                alert('Invalid server response.');
   505→                console.error(xhr.responseText);
   506→            }
   507→        } else {
   508→            alert('Server error (HTTP ' + xhr.status + ')');
   509→        }
   510→    };
   511→    xhr.onerror = function() {
   512→        btn.disabled = false;
   513→        btn.innerHTML = '<i class="bi bi-play-fill me-1"></i>Import';
   514→        progressLabel.textContent = 'Upload failed';
   515→        progressBar.classList.remove('bg-primary');
   516→        progressBar.classList.add('bg-danger');
   517→    };
   518→    xhr.send(formData);
   519→}
   520→function showSummary(d) {
   521→    document.getElementById('sumFiles').textContent = nFmt(d.files_uploaded);
   522→    document.getElementById('sumRead').textContent = nFmt(d.rows_read);
   523→    document.getElementById('sumInserted').textContent = nFmt(d.rows_inserted);
   524→    document.getElementById('sumUpdated').textContent = nFmt(d.rows_updated);
   525→    document.getElementById('sumSkipped').textContent = nFmt(d.rows_skipped);
   526→    document.getElementById('sumErrors').textContent = nFmt(d.error_count);
   527→    document.getElementById('summaryCard').classList.remove('d-none');
   528→    if (d.error_count > 0) {
   529→        document.getElementById('downloadErrorBtn').classList.remove('d-none');
   530→        lastImportId = d.import_id;
   531→    } else {
   532→        document.getElementById('downloadErrorBtn').classList.add('d-none');
   533→    }
   534→    document.getElementById('totalRecords').textContent = nFmt(d.total_records);
   535→    document.getElementById('lastImportDate').textContent = d.import_date || 'Just now';
   536→    document.getElementById('totalImports').textContent = nFmt(d.total_imports);
   537→}
