    88→function getInitials(name: string): string {
    89→  return name
    90→    .split(' ')
    91→    .map((w) => w[0])
    92→    .filter(Boolean)
    93→    .slice(0, 2)
    94→    .join('')
    95→    .toUpperCase();
    96→}
    97→
    98→function maskMobile(mobile: string): string {
    99→  // Show full mobile number — no masking
   100→  return mobile || '';
   101→}
   102→
   103→function maskSensitive(value: string | number | undefined): string {
   104→  const str = value != null ? String(value) : '';
   105→  if (!str) return '****';
   106→  if (str.length <= 4) return '****';
   107→  const first = str.slice(0, 2);
   108→  const last = str.slice(-2);
   109→  const middleLen = str.length - 4;
   110→  return first + '*'.repeat(middleLen) + last;
   111→}
   112→
   113→function formatDate(dateStr?: string): string {
   114→  if (!dateStr) return '—';
   115→  const d = new Date(dateStr);
   116→  return d.toLocaleDateString('en-IN', {
   117→    day: 'numeric',
   118→    month: 'short',
   119→    year: 'numeric',
   120→  });
   121→}
   122→
   123→function isActive(employee: Employee): boolean {
   124→  return !employee.date_of_leaving && employee.status !== 'inactive' && employee.status !== 'resigned';
   125→}
