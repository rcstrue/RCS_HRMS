   991→    if (!imgEl) return;
   992→    var empId = <?php echo (int)$employeeId; ?>;
   993→    var currentUrl = imgEl.getAttribute('src');
   994→    if (!currentUrl) return;
   995→
   996→    openLiteEditor(currentUrl, function(base64DataUrl) {
   997→        var formData = new FormData();
   998→        formData.append('action', 'crop_save_doc');
   999→        formData.append('employee_id', empId);
  1000→        formData.append('field', field);
  1001→        formData.append('image_data', base64DataUrl);
  1002→        formData.append('original_url', currentUrl);
  1003→
  1004→        fetch(window.location.pathname + '?page=api/crop-save', {
  1005→            method: 'POST',
  1006→            body: formData
  1007→        })
  1008→        .then(function(r) { return r.json(); })
  1009→        .then(function(res) {
  1010→            if (res.ok) {
  1011→                imgEl.src = res.url + '?t=' + Date.now();
  1012→                var link = imgEl.closest('a');
  1013→                if (link) link.href = res.url;
  1014→                // Also update the sidebar profile photo if it was the profile pic
  1015→                if (field.indexOf('profile_pic') !== -1) {
  1016→                    var sidePhoto = document.querySelector('.card-body.text-center img.rounded-circle');
  1017→                    if (sidePhoto) sidePhoto.src = res.url + '?t=' + Date.now();
  1018→                }
  1019→                alert('Image saved successfully!');
  1020→            } else {
  1021→                alert('Save failed: ' + (res.msg || 'Unknown error'));
  1022→            }
  1023→        })
  1024→        .catch(function(err) { alert('Error: ' + err.message); });
  1025→    });
