    85→function getInitials(name: string): string {
    86→  return name
    87→    .split(' ')
    88→    .map((w) => w[0])
    89→    .filter(Boolean)
    90→    .slice(0, 2)
    91→    .join('')
    92→    .toUpperCase();
    93→}
    94→
    95→function maskMobile(mobile: string): string {
    96→  // Show full mobile number — no masking
    97→  return mobile || '';
    98→}
    99→
   100→function maskSensitive(value: string | number | undefined): string {
   101→  const str = value != null ? String(value) : '';
   102→  if (!str) return '****';
   103→  if (str.length <= 4) return '****';
   104→  const first = str.slice(0, 2);
   105→  const last = str.slice(-2);
   106→  const middleLen = str.length - 4;
   107→  return first + '*'.repeat(middleLen) + last;
   108→}
   109→
   110→function formatDate(dateStr?: string): string {
   111→  if (!dateStr) return '—';
   112→  const d = new Date(dateStr);
   113→  return d.toLocaleDateString('en-IN', {
   114→    day: 'numeric',
   115→    month: 'short',
   116→    year: 'numeric',
   117→  });
   118→}
   119→
   120→function isActive(employee: Employee): boolean {
   121→  return !employee.date_of_leaving && employee.status !== 'inactive' && employee.status !== 'resigned';
   122→}

   985→// ── Sensitive Profile Row (with eye toggle) ──────────
   986→function SensitiveProfileRow({
   987→  icon: Icon,
   988→  label,
   989→  value,
   990→}: {
   991→  icon: typeof User;
   992→  label: string;
   993→  value?: string | number;

   998→  if (!displayValue) {
   999→    return (
  1000→      <div className="flex items-start gap-3">
  1001→        <Icon className="h-4 w-4 text-muted-foreground mt-0.5 shrink-0" />
  1002→        <div className="flex flex-col">
  1003→          <span className="text-xs text-muted-foreground">{label}</span>

  1005→        </div>
  1006→      </div>
  1007→    );
  1008→  }
  1009→
  1010→  return (
  1011→    <div className="flex items-start gap-3">
  1012→      <Icon className="h-4 w-4 text-muted-foreground mt-0.5 shrink-0" />
  1013→      <div className="flex flex-col flex-1 min-w-0">
  1014→        <span className="text-xs text-muted-foreground">{label}</span>

  1018→      </div>

  1020→        type="button"
  1021→        onClick={() => setRevealed((r) => !r)}
  1022→        className="mt-0.5 p-1 rounded text-muted-foreground hover:text-foreground hover:bg-muted/50 transition-colors shrink-0"
  1023→        aria-label={revealed ? `Hide ${label}` : `Reveal ${label}`}

  1026→      </button>
  1027→    </div>
  1028→  );
  1029→}
