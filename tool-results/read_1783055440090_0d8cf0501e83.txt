   545→    // ── EMI Preview Calculator ──
   546→    function calculateEmiPreview() {
   547→        var amount = parseFloat($('#loanAmount').val()) || 0;
   548→        var rate = parseFloat($('#loanInterestRate').val()) || 0;
   549→        var tenure = parseInt($('#loanTenure').val()) || 0;
   550→
   551→        if (amount > 0 && tenure > 0) {
   552→            var emi, totalInterest, totalRepayable;
   553→            if (rate <= 0) {
   554→                emi = amount / tenure;
   555→                totalInterest = 0;
   556→                totalRepayable = amount;
   557→            } else {
   558→                var monthlyRate = rate / 12 / 100;
   559→                var onePlusR = 1 + monthlyRate;
   560→                var power = Math.pow(onePlusR, tenure);
   561→                emi = amount * monthlyRate * power / (power - 1);
   562→                totalRepayable = emi * tenure;
   563→                totalInterest = totalRepayable - amount;
   564→            }
   565→            $('#previewEmi').text('₹' + emi.toFixed(2));
   566→            $('#previewInterest').text('₹' + totalInterest.toFixed(2));
   567→            $('#previewTotal').text('₹' + totalRepayable.toFixed(2));
   568→            $('#emiPreview').slideDown();
   569→        } else {
   570→            $('#emiPreview').slideUp();
   571→        }
   572→    }
   573→
   574→    $('#loanAmount, #loanInterestRate, #loanTenure').on('input', calculateEmiPreview);
