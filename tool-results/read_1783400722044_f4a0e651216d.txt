   668→function toggleColumn(columnKey, isVisible) {
   669→    const table = document.getElementById('employees-table');
   670→    
   671→    // Toggle header
   672→    const header = table.querySelector('th[data-column="' + columnKey + '"]');

   675→    }

   678→    const cells = table.querySelectorAll('td[data-column="' + columnKey + '"]');
   679→    cells.forEach(cell => {
   680→        cell.style.display = isVisible ? '' : 'none';
   681→    });
   682→    
   683→    // Save preference
   684→    saveColumnPreferences();
   685→}

   687→function saveColumnPreferences() {
   688→    const preferences = {};
   689→    document.querySelectorAll('.column-toggle').forEach(checkbox => {
   690→        preferences[checkbox.dataset.column] = checkbox.checked;
   691→    });
   692→    localStorage.setItem('employeeListColumnPrefs', JSON.stringify(preferences));
   693→}
   694→
   695→function loadColumnPreferences() {
   696→    const saved = localStorage.getItem('employeeListColumnPrefs');
   697→    if (saved) {
   698→        try {
   699→            const preferences = JSON.parse(saved);
   700→            Object.keys(preferences).forEach(columnKey => {
   701→                const checkbox = document.querySelector('.column-toggle[data-column="' + columnKey + '"]');
   702→                if (checkbox) {
   703→                    checkbox.checked = preferences[columnKey];
   704→                    toggleColumn(columnKey, preferences[columnKey]);
   705→                }
   706→            });
   707→        } catch (e) {
   708→            console.error('Error loading column preferences:', e);
   709→        }
   710→    }
   711→}
