   968→// Crop document image (Aadhaar / Bank / Profile)
   969→function cropDocImage(field, imgEl) {
   970→    if (!imgEl) return;
   971→    var empId = <?php echo (int)$employeeId; ?>;
   972→    var currentUrl = imgEl.getAttribute('src');
   973→    if (!currentUrl) return;
   974→
   975→    openLiteEditor(currentUrl, function(base64DataUrl) {
   976→        var formData = new FormData();
   977→        formData.append('action', 'crop_save_doc');
   978→        formData.append('employee_id', empId);
   979→        formData.append('field', field);
   980→        formData.append('image_data', base64DataUrl);
   981→        formData.append('original_url', currentUrl);
   982→
   983→        fetch(window.location.pathname + '?page=api/crop-save', {
   984→            method: 'POST',
   985→            body: formData
   986→        })
   987→        .then(function(r) { return r.json(); })
   988→        .then(function(res) {
   989→            if (res.ok) {
   990→                imgEl.src = res.url + '?t=' + Date.now();
   991→                var link = imgEl.closest('a');
   992→                if (link) link.href = res.url;
   993→                // Also update the sidebar profile photo if it was the profile pic
   994→                if (field.indexOf('profile_pic') !== -1) {
   995→                    var sidePhoto = document.querySelector('.card-body.text-center img.rounded-circle');
   996→                    if (sidePhoto) sidePhoto.src = res.url + '?t=' + Date.now();
   997→                }
   998→                alert('Image saved successfully!');
   999→            } else {
  1000→                alert('Save failed: ' + (res.msg || 'Unknown error'));
  1001→            }
  1002→        })
  1003→        .catch(function(err) { alert('Error: ' + err.message); });
  1004→    });
  1005→}
