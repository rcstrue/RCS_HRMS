  1169→ * Parse a CSV salary revision file.
  1170→ * Handles both old format (10 cols: code,basic,hra,leave,bonus,wash,pf,esi,pt,lwf)
  1171→ * and new format (13 cols: code,name,client,unit,basic,hra,leave,bonus,wash,pf,esi,pt,lwf)
  1172→ */
  1173→function parseSalaryRevisionCSV($filePath) {
  1174→    $rows = [];
  1175→    $handle = fopen($filePath, 'r');
  1176→    if (!$handle) return $rows;
  1177→    
  1178→    $header = fgetcsv($handle); // skip header
  1179→    if (!$header) { fclose($handle); return $rows; }
  1180→    
  1181→    // Detect format by column count
  1182→    $colCount = count($header);
  1183→    $isOldFormat = ($colCount <= 11); // old: 10-11 cols (last is optional name ref)
  1184→    
  1185→    while (($row = fgetcsv($handle)) !== false) {
  1186→        if (empty(array_filter($row))) continue; // skip blank rows
  1187→        
  1188→        if ($isOldFormat) {
  1189→            // Old format: [code, basic_da, hra, leave_encash, bonus_encash, wash, pf, esi, pt, lwf, (name?)]
  1190→            $rows[] = [
  1191→                'employee_code'    => trim($row[0] ?? ''),
  1192→                'basic_da'          => (float)($row[1] ?? 0),
  1193→                'hra'               => (float)($row[2] ?? 0),
  1194→                'leave_encashment'  => (float)($row[3] ?? 0),
  1195→                'bonus_encashment'  => (float)($row[4] ?? 0),
  1196→                'washing_allowance' => (float)($row[5] ?? 0),
  1197→                'pf_applicable'     => (int)($row[6] ?? 1),
  1198→                'esi_applicable'     => (int)($row[7] ?? 1),
  1199→                'pt_applicable'     => (int)($row[8] ?? 1),
  1200→                'lwf_applicable'     => (int)($row[9] ?? 1),
  1201→            ];
  1202→        } else {
  1203→            // New format: [code, name(ref), client(ref), unit(ref), basic_da, hra, leave, bonus, wash, pf, esi, pt, lwf]
  1204→            $rows[] = [
  1205→                'employee_code'    => trim($row[0] ?? ''),
  1206→                'basic_da'          => (float)($row[4] ?? 0),
  1207→                'hra'               => (float)($row[5] ?? 0),
  1208→                'leave_encashment'  => (float)($row[6] ?? 0),
  1209→                'bonus_encashment'  => (float)($row[7] ?? 0),
  1210→                'washing_allowance' => (float)($row[8] ?? 0),

  1214→                'lwf_applicable'     => (int)($row[12] ?? 1),
  1215→            ];
  1216→        }
  1217→    }
  1218→    
  1219→    fclose($handle);
  1220→    return $rows;

  1223→/**
  1224→ * Parse a SpreadsheetML XML (.xls) salary revision file.
  1225→ * Expects the format generated by download_salary_template.php.
  1226→ */
  1227→function parseSalaryRevisionXML($filePath) {
  1228→    $rows = [];
  1229→    $content = file_get_contents($filePath);
  1230→    if (empty($content)) return $rows;
  1231→    
  1232→    // Register namespace for SpreadsheetML
  1233→    // Handle both with and without namespace prefix
  1234→    libxml_use_internal_errors(true);
  1235→    $xml = @simplexml_load_string($content);
  1236→    if (!$xml) {
  1237→        // Try with namespace
  1238→        $xml = @simplexml_load_string($content, 'SimpleXMLElement', LIBXML_NOCDATA, 'urn:schemas-microsoft-com:office:spreadsheet');
  1239→    }
  1240→    
  1241→    if (!$xml) {
  1242→        // Fallback: regex parse (robust against namespace issues)
  1243→        return parseSalaryRevisionXMLFallback($content);
  1244→    }
  1245→    
  1246→    // Register namespaces
  1247→    $namespaces = $xml->getNamespaces(true);
  1248→    $ss = $namespaces['ss'] ?? 'urn:schemas-microsoft-com:office:spreadsheet';
  1249→    $xml->registerXPathNamespace('ss', $ss);
  1250→    
  1251→    // Find all Rows in the Table
  1252→    $tableRows = $xml->xpath('//ss:Row');
  1253→    if (empty($tableRows)) {
  1254→        // Try without namespace
  1255→        $tableRows = $xml->xpath('//Row');
  1256→    }
  1257→    
  1258→    $rowIdx = 0;
  1259→    foreach ($tableRows as $xmlRow) {
  1260→        $rowIdx++;
  1261→        if ($rowIdx <= 2) continue; // skip header + instruction rows
  1262→        
  1263→        // Get cells
  1264→        $cells = [];
  1265→        $cellNodes = $xmlRow->xpath('ss:Cell') ?: $xmlRow->xpath('Cell');
  1266→        
  1267→        // Handle ss:Index attribute (sparse cells)
  1268→        $lastIndex = 0;
  1269→        foreach ($cellNodes as $cellNode) {
  1270→            $indexAttr = $cellNode->attributes($ss)['Index'] ?? null;
  1271→            if ($indexAttr !== null) {
  1272→                $index = (int)$indexAttr - 1; // 0-based
  1273→            } else {
  1274→                $index = $lastIndex;
  1275→            }
  1276→            // Fill gaps
  1277→            while (count($cells) < $index) {
  1278→                $cells[] = '';
  1279→            }
  1280→            $cellValue = (string)($cellNode->xpath('ss:Data')[0] ?? $cellNode->Data[0] ?? '');
  1281→            $cells[] = $cellValue;
  1282→            $lastIndex = count($cells);
  1283→        }
  1284→        
  1285→        if (empty(array_filter($cells))) continue; // skip blank rows
  1286→        
  1287→        // New format: [code, name(ref), client(ref), unit(ref), basic_da, hra, leave, bonus, wash, pf, esi, pt, lwf]
  1288→        $rows[] = [
  1289→            'employee_code'    => trim($cells[0] ?? ''),
  1290→            'basic_da'          => (float)($cells[4] ?? 0),
  1291→            'hra'               => (float)($cells[5] ?? 0),
  1292→            'leave_encashment'  => (float)($cells[6] ?? 0),
  1293→            'bonus_encashment'  => (float)($cells[7] ?? 0),
  1294→            'washing_allowance' => (float)($cells[8] ?? 0),
  1295→            'pf_applicable'     => (int)($cells[9] ?? 1),
  1296→            'esi_applicable'     => (int)($cells[10] ?? 1),
  1297→            'pt_applicable'     => (int)($cells[11] ?? 1),
  1298→            'lwf_applicable'     => (int)($cells[12] ?? 1),
  1299→        ];
  1300→    }
  1301→    
  1302→    libxml_clear_errors();
  1303→    return $rows;
  1304→}

  1306→/**
  1307→ * Fallback XML parser using regex for when SimpleXML fails with namespaces.
  1308→ */
  1309→function parseSalaryRevisionXMLFallback($content) {
  1310→    $rows = [];
  1311→    
  1312→    // Extract all <Row> blocks
  1313→    preg_match_all('/<Row[^>]*>(.*?)<\/Row>/si', $content, $rowMatches);
  1314→    
  1315→    $rowIdx = 0;
  1316→    foreach ($rowMatches[1] as $rowContent) {
  1317→        $rowIdx++;
  1318→        if ($rowIdx <= 2) continue; // skip header + instruction rows
  1319→        
  1320→        // Extract all <Data> values (cell values)
  1321→        preg_match_all('/<Data[^>]*>(.*?)<\/Data>/si', $rowContent, $cellMatches);
  1322→        $cells = $cellMatches[1];
  1323→        
  1324→        // Also track ss:Index for sparse cells
  1325→        preg_match_all('/<Cell[^>]*ss:Index="(\d+)"[^>]*>/si', $rowContent, $indexMatches);
  1326→        
  1327→        if (empty(array_filter($cells))) continue;
  1328→        
  1329→        // Build sparse array with proper indexing
  1330→        $fullCells = [];
  1331→        $cellIdx = 0;
  1332→        foreach ($cells as $ci => $val) {
  1333→            if (isset($indexMatches[1][$ci])) {
  1334→                $targetIdx = (int)$indexMatches[1][$ci] - 1; // 1-based to 0-based
  1335→                while (count($fullCells) < $targetIdx) {
  1336→                    $fullCells[] = '';
  1337→                }
  1338→            }
  1339→            $fullCells[] = trim($val);
  1340→        }
  1341→        
  1342→        $rows[] = [
  1343→            'employee_code'    => trim($fullCells[0] ?? ''),
  1344→            'basic_da'          => (float)($fullCells[4] ?? 0),
  1345→            'hra'               => (float)($fullCells[5] ?? 0),
  1346→            'leave_encashment'  => (float)($fullCells[6] ?? 0),
  1347→            'bonus_encashment'  => (float)($fullCells[7] ?? 0),
  1348→            'washing_allowance' => (float)($fullCells[8] ?? 0),
  1349→            'pf_applicable'     => (int)($fullCells[9] ?? 1),
  1350→            'esi_applicable'     => (int)($fullCells[10] ?? 1),
  1351→            'pt_applicable'     => (int)($fullCells[11] ?? 1),
  1352→            'lwf_applicable'     => (int)($fullCells[12] ?? 1),
  1353→        ];
  1354→    }
  1355→    
  1356→    return $rows;
  1357→}
