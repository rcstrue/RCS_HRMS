   982→function cropDocImage(field, imgEl) {
   983→    if (!imgEl) return;
   984→    var empId = <?php echo (int)$employeeId; ?>;
   985→    var currentUrl = imgEl.getAttribute('src');
   986→    if (!currentUrl) return;
   987→
   988→    openLiteEditor(currentUrl, function(base64DataUrl) {
   989→        var formData = new FormData();
   990→        formData.append('action', 'crop_save_doc');
   991→        formData.append('employee_id', empId);
   992→        formData.append('field', field);
   993→        formData.append('image_data', base64DataUrl);
   994→        formData.append('original_url', currentUrl);
   995→
   996→        fetch(window.location.pathname + '?page=api/crop-save', {
   997→            method: 'POST',
   998→            body: formData
   999→        })
  1000→        .then(function(r) { return r.json(); })
  1001→        .then(function(res) {
  1002→            if (res.ok) {
  1003→                imgEl.src = res.url + '?t=' + Date.now();
  1004→                var link = imgEl.closest('a');
  1005→                if (link) link.href = res.url;
  1006→                // Also update the sidebar profile photo if it was the profile pic
  1007→                if (field.indexOf('profile_pic') !== -1) {
  1008→                    var sidePhoto = document.querySelector('.card-body.text-center img.rounded-circle');
  1009→                    if (sidePhoto) sidePhoto.src = res.url + '?t=' + Date.now();
  1010→                }
  1011→                alert('Image saved successfully!');
  1012→            } else {
  1013→                alert('Save failed: ' + (res.msg || 'Unknown error'));
  1014→            }
  1015→        })
  1016→        .catch(function(err) { alert('Error: ' + err.message); });
  1017→    });
  1018→}
