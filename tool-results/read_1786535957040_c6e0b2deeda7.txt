    80→    80→        // Check if employee_salary_structures table exists
    81→    81→        $useSalaryTable = $this->checkSalaryTableExists();
    82→    82→        $baseColumns = $this->getBaseColumns();
    83→    83→

    86→    86→                           ess.basic_da, ess.hra, ess.leave_encashment, ess.bonus_encashment,
    87→    87→                           ess.washing_allowance, ess.gross_salary,
    88→    88→                           ess.pf_applicable, ess.esi_applicable, ess.pt_applicable,
    89→    89→                           c.name as client_name_display,

    93→    93→                        SELECT employee_id, basic_da, hra, leave_encashment, bonus_encashment,
    94→    94→                               washing_allowance, gross_salary, pf_applicable, esi_applicable, pt_applicable
    95→    95→                        FROM employee_salary_structures
    96→    96→                        WHERE effective_to IS NULL OR effective_to >= CURDATE()
    97→    97→                        GROUP BY employee_id
    98→    98→                    ) ess ON e.id = ess.employee_id
    99→    99→                    LEFT JOIN clients c ON e.client_id = c.id
   100→   100→                    LEFT JOIN units u ON e.unit_id = u.id
   101→   101→                    WHERE 1=1";
   102→   102→        } else {
   103→   103→            $sql = "SELECT $baseColumns,
   104→   104→                           c.name as client_name_display,
   105→   105→                           u.name as unit_name_display
   106→   106→                    FROM employees e
   107→   107→                    LEFT JOIN clients c ON e.client_id = c.id
   108→   108→                    LEFT JOIN units u ON e.unit_id = u.id
   109→   109→                    WHERE 1=1";
   110→   110→        }
   111→   111→        $params = [];
   112→   112→
   113→   113→        // Status filter - use strtolower for case-insensitive comparison
   114→   114→        if (!empty($filters['status'])) {
   115→   115→            $statusLower = strtolower($filters['status']);
   116→   116→            if ($statusLower === 'active' || $statusLower === 'approved') {
   117→   117→                $sql .= " AND e.status = :status";
   118→   118→                $params['status'] = STATUS_APPROVED;
   119→   119→            } elseif ($statusLower === 'pending') {
   120→   120→                $sql .= " AND e.status LIKE :status";
   121→   121→                $params['status'] = STATUS_PENDING . '%';
   122→   122→            } else {
   123→   123→                $sql .= " AND e.status = :status";
   124→   124→                $params['status'] = $filters['status'];
   125→   125→            }
   126→   126→        }
   127→   127→
   128→   128→        // Client filter
   129→   129→        if (!empty($filters['client_id'])) {
   130→   130→            $sql .= " AND e.client_id = :client_id";
   131→   131→            $params['client_id'] = $filters['client_id'];
   132→   132→        }
   133→   133→
   134→   134→        // Unit filter
   135→   135→        if (!empty($filters['unit_id'])) {
   136→   136→            $sql .= " AND e.unit_id = :unit_id";
   137→   137→            $params['unit_id'] = $filters['unit_id'];
   138→   138→        }
   139→   139→
   140→   140→        // Worker category filter
   141→   141→        if (!empty($filters['worker_category'])) {
   142→   142→            $sql .= " AND e.worker_category = :worker_category";
   143→   143→            $params['worker_category'] = $filters['worker_category'];
   144→   144→        }
   145→   145→
   146→   146→        // Search filter - use unique parameter names for PDO with emulated prepares disabled
   147→   147→        if (!empty($filters['search'])) {
   148→   148→            $sql .= " AND (e.employee_code LIKE :search1
   149→   149→                      OR e.full_name LIKE :search2
   150→   150→                      OR e.mobile_number LIKE :search3
   151→   151→                      OR e.aadhaar_number LIKE :search4)";
   152→   152→            $searchTerm = '%' . $filters['search'] . '%';
   153→   153→            $params['search1'] = $searchTerm;
   154→   154→            $params['search2'] = $searchTerm;
   155→   155→            $params['search3'] = $searchTerm;
   156→   156→            $params['search4'] = $searchTerm;
   157→   157→        }
   158→   158→
   159→   159→        // Count total
   160→   160→        $countSql = "SELECT COUNT(*) as total FROM ($sql) as t";
   161→   161→        $total = $this->db->fetchColumn($countSql, $params);
   162→   162→
   163→   163→        // Pagination
   164→   164→        $offset = ($page - 1) * $perPage;
   165→   165→        $sql .= " ORDER BY e.created_at DESC LIMIT $perPage OFFSET $offset";
   166→   166→
   167→   167→        $employees = $this->db->fetchAll($sql, $params);
   168→   168→
   169→   169→        return [
   170→   170→            'data' => $employees,
   171→   171→            'total' => $total ?: 0,
   172→   172→            'page' => $page,
   173→   173→            'per_page' => $perPage,
   174→   174→            'total_pages' => ceil(($total ?: 0) / $perPage)
   175→   175→        ];
   176→   176→    }

   304→   304→    // Create new employee
   305→   305→    public function create($data) {
   306→   306→        // Generate employee code if not provided
   307→   307→        if (empty($data['employee_code'])) {
   308→   308→            $data['employee_code'] = $this->generateEmployeeCode();
   309→   309→        }
   310→   310→        
   311→   311→        // Set default status
   312→   312→        $data['status'] = $data['status'] ?? STATUS_PENDING_HR;
   313→   313→        // Set default app_role to 'employee' if not provided
   314→   314→        $data['app_role'] = $data['app_role'] ?? 'employee';
   315→   315→        $data['created_at'] = date(DATETIME_FORMAT_DB);
   316→   316→
   317→   317→        // Map form fields to database columns
   318→   318→        $dbData = $this->mapFormDataToDb($data);
   319→   319→
   320→   320→        // Check if employee_salary_structures table exists
   321→   321→        $useSalaryTable = $this->checkSalaryTableExists();
   322→   322→
   323→   323→        try {
   324→   324→            $this->db->beginTransaction();
   325→   325→            
   326→   326→            // Insert employee
   327→   327→            $this->db->insert('employees', $dbData);
   328→   328→            
   329→   329→            // Get the newly created employee ID
   330→   330→            $employeeId = $this->db->lastInsertId();
   331→   331→            
   332→   332→            // Insert salary structure if table exists and any salary/statutory data provided
   333→   333→            $hasSalaryData = !empty($data['basic_da']) || !empty($data['basic_salary']) ||
   334→   334→                isset($data['pf_applicable']) || isset($data['esi_applicable']) ||
   335→   335→                isset($data['pt_applicable']) || isset($data['lwf_applicable']) ||
   336→   336→                isset($data['bonus_applicable']) || isset($data['gratuity_applicable']) ||
   337→   337→                isset($data['overtime_applicable']);
   338→   338→            
   339→   339→            if ($useSalaryTable && $hasSalaryData && $employeeId) {
   340→   340→                $salaryData = $this->buildSalaryData($data, $employeeId);
   341→   341→                $this->db->insert('employee_salary_structures', $salaryData);
   342→   342→            }
   343→   343→            
   344→   344→            $this->db->commit();
   345→   345→            
   346→   346→            // Send notification for pending employee registration
   347→   347→            if (strpos($data['status'], 'pending') !== false) {
   348→   348→                $notification = new Notification();
   349→   349→                $notification->notifyNewEmployeeRegistration([
   350→   350→                    'id' => $employeeId,
   351→   351→                    'full_name' => $data['full_name'] ?? '',
   352→   352→                    'employee_code' => $data['employee_code']
   353→   353→                ]);
   354→   354→            }
   355→   355→            
   356→   356→            return ['success' => true, 'message' => MSG_RECORD_CREATED, 'employee_id' => $employeeId];
   357→   357→        } catch (Exception $e) {
   358→   358→            $this->db->rollBack();
   359→   359→            return ['success' => false, 'message' => 'Failed to create employee: ' . $e->getMessage()];
   360→   360→        }
   361→   361→    }

   363→   363→    // Map form data to database columns
   364→   364→    private function mapFormDataToDb($data) {
   365→   365→        $mapped = [];
   366→   366→        
   367→   367→        // Direct mapping - form fields match database columns (removed client_name/unit_name as they don't exist)
   368→   368→        $directFields = [
   369→   369→            'full_name', 'father_name', 'employee_code', 'mobile_number', 'alternate_mobile',
   370→   370→            'email', 'gender', 'date_of_birth', 'marital_status', 'blood_group',
   371→   371→            'aadhaar_number', 'uan_number', 'esic_number',
   372→   372→            'address', 'pin_code', 'state', 'district',
   373→   373→            'bank_name', 'account_number', 'ifsc_code', 'account_holder_name',
   374→   374→            'client_id', 'unit_id', 'app_role',
   375→   375→            'designation', 'department',
   376→   376→            'worker_category', 'employment_type', 'date_of_joining', 'probation_period',
   377→   377→            'nominee_name', 'nominee_relationship', 'nominee_dob', 'nominee_contact',
   378→   378→            'emergency_contact_name', 'emergency_contact_relation',
   379→   379→            'status', 'profile_pic_url', 'profile_pic_cropped_url',
   380→   380→            'aadhaar_front_url', 'aadhaar_back_url', 'bank_document_url'
   381→   381→        ];
   382→   382→        
   383→   383→        foreach ($directFields as $field) {
   384→   384→            if (isset($data[$field])) {
   385→   385→                $mapped[$field] = $data[$field];
   386→   386→            }
   387→   387→        }
   388→   388→        
   389→   389→        // Handle middle_name -> father_name mapping (for backward compatibility)
   390→   390→        if (isset($data['middle_name']) && !isset($data['father_name'])) {
   391→   391→            $mapped['father_name'] = $data['middle_name'];
   392→   392→        }
   393→   393→        
   394→   394→        // Look up client_id from client_name if not provided (for import compatibility)
   395→   395→        if (!empty($data['client_name']) && empty($data['client_id'])) {
   396→   396→            $client = $this->db->fetch(
   397→   397→                "SELECT id FROM clients WHERE name = :name LIMIT 1",
   398→   398→                ['name' => $data['client_name']]
   399→   399→            );
   400→   400→            if ($client) {
   401→   401→                $mapped['client_id'] = $client['id'];
   402→   402→            }
   403→   403→        }
   404→   404→        
   405→   405→        // Look up unit_id from unit_name if not provided (for import compatibility)
   406→   406→        if (!empty($data['unit_name']) && empty($data['unit_id'])) {
   407→   407→            $unit = $this->db->fetch(
   408→   408→                "SELECT id FROM units WHERE name = :name LIMIT 1",
   409→   409→                ['name' => $data['unit_name']]
   410→   410→            );
   411→   411→            if ($unit) {
   412→   412→                $mapped['unit_id'] = $unit['id'];
   413→   413→            }
   414→   414→        }
   415→   415→        
   416→   416→        return $mapped;
   417→   417→    }

   429→   429→        unset($dbData['id'], $dbData['employee_code'], $dbData['created_at']);

   438→   438→            // Check if any salary or statutory field is present
   439→   439→            $hasSalaryData = isset($data['basic_da']) || isset($data['basic_salary']) ||
   440→   440→                isset($data['pf_applicable']) || isset($data['esi_applicable']) ||
   441→   441→                isset($data['pt_applicable']) || isset($data['lwf_applicable']) ||
   442→   442→                isset($data['bonus_applicable']) || isset($data['gratuity_applicable']);
   443→   443→
   444→   444→            if ($hasSalaryData) {
   445→   445→                $salaryData = [
   446→   446→                    'basic_da' => floatval($data['basic_da'] ?? $data['basic_salary'] ?? 0),
   447→   447→                    'hra' => floatval($data['hra'] ?? 0),
   448→   448→                    'leave_encashment' => floatval($data['leave_encashment'] ?? 0),
   449→   449→                    'bonus_encashment' => floatval($data['bonus_encashment'] ?? 0),
   450→   450→                    'washing_allowance' => floatval($data['washing_allowance'] ?? 0),
   451→   451→                    'gross_salary' => floatval($data['gross_salary'] ?? 0),
   452→   452→                    'pf_applicable' => !empty($data['pf_applicable']) ? 1 : 0,
   453→   453→                    'esi_applicable' => !empty($data['esi_applicable']) ? 1 : 0,
   454→   454→                    'pt_applicable' => !empty($data['pt_applicable']) ? 1 : 0,
   455→   455→                    'lwf_applicable' => !empty($data['lwf_applicable']) ? 1 : 0,
   456→   456→                    'bonus_applicable' => !empty($data['bonus_applicable']) ? 1 : 0,
   457→   457→                    'gratuity_applicable' => !empty($data['gratuity_applicable']) ? 1 : 0,
   458→   458→                    'overtime_applicable' => !empty($data['overtime_applicable']) ? 1 : 0,
   459→   459→                    'updated_at' => date(DATETIME_FORMAT_DB),
   460→   460→                ];
   461→   461→            }
   462→   462→        }
   463→   463→
   464→   464→        try {
   465→   465→            $this->db->beginTransaction();
   466→   466→            
   467→   467→            // Update employee
   468→   468→            $this->db->update('employees', $dbData, SQL_WHERE_ID, ['id' => $id]);
   469→   469→            
   470→   470→            // Update salary structure if provided and table exists (with versioning)
   471→   471→            if (!empty($salaryData)) {
   472→   472→                // Check if salary structure exists
   473→   473→                $existing = $this->db->fetch(
   474→   474→                    "SELECT id, effective_from FROM employee_salary_structures 
   475→   475→                     WHERE employee_id = :id AND (effective_to IS NULL OR effective_to >= CURDATE())",
   476→   476→                    ['id' => $id]
   477→   477→                );
   478→   478→                
   479→   479→                if ($existing) {
   480→   480→                    // Close the old active record to preserve history
   481→   481→                    $this->db->query(
   482→   482→                        "UPDATE employee_salary_structures 
   483→   483→                         SET effective_to = DATE_SUB(CURDATE(), INTERVAL 1 DAY), updated_at = NOW() 
   484→   484→                         WHERE id = ?",
   485→   485→                        [$existing['id']]
   486→   486→                    );
   487→   487→                }
   488→   488→                // Insert new versioned record
   489→   489→                $salaryData['employee_id']   = $id;
   490→   490→                $salaryData['effective_from'] = date(DATE_FORMAT_DB);
   491→   491→                $this->db->insert('employee_salary_structures', $salaryData);
   492→   492→            }
   493→   493→            
   494→   494→            $this->db->commit();
   495→   495→            return ['success' => true, 'message' => MSG_RECORD_UPDATED];
   496→   496→        } catch (Exception $e) {
   497→   497→            $this->db->rollBack();
   498→   498→            return ['success' => false, 'message' => 'Failed to update employee: ' . $e->getMessage()];
   499→   499→        }
   500→   500→    }

   653→   653→    // Import employees from array
   654→   654→    public function importFromData($data, $skipDuplicates = true) {
   655→   655→        $imported = 0;
   656→   656→        $errors = [];
   657→   657→        $duplicates = [];
   658→   658→
   659→   659→        foreach ($data as $row) {
   660→   660→            try {
   661→   661→                // Check for existing employee by mobile or aadhaar
   662→   662→                if (!empty($row['mobile_number'])) {
   663→   663→                    $exists = $this->db->fetch(
   664→   664→                        "SELECT id FROM employees WHERE mobile_number = :mobile",
   665→   665→                        ['mobile' => $row['mobile_number']]
   666→   666→                    );
   667→   667→                    if ($exists && $skipDuplicates) {
   668→   668→                        $duplicates[] = $row['mobile_number'];
   669→   669→                        continue;
   670→   670→                    }
   671→   671→                }
   672→   672→
   673→   673→                $result = $this->create($row);
   674→   674→                if ($result['success']) {
   675→   675→                    $imported++;
   676→   676→                } else {
   677→   677→                    $errors[] = $result['message'];
   678→   678→                }
   679→   679→            } catch (Exception $e) {
   680→   680→                $errors[] = "Error importing row: " . $e->getMessage();
   681→   681→            }
   682→   682→        }
   683→   683→
   684→   684→        return [
   685→   685→            'success' => true,
   686→   686→            'imported' => $imported,
   687→   687→            'duplicates' => count($duplicates),
   688→   688→            'errors' => $errors
   689→   689→        ];
   690→   690→    }
