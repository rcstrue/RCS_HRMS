    89→// ── Helpers ────────────────────────────────────────────
    90→function getInitials(name: string): string {
    91→  return name
    92→    .split(' ')
    93→    .map((w) => w[0])
    94→    .filter(Boolean)
    95→    .slice(0, 2)
    96→    .join('')
    97→    .toUpperCase();
    98→}
    99→
   100→function maskMobile(mobile: string): string {
   101→  // Show full mobile number — no masking
   102→  return mobile || '';
   103→}
   104→
   105→function maskSensitive(value: string | number | undefined): string {
   106→  const str = value != null ? String(value) : '';
   107→  if (!str) return '****';
   108→  if (str.length <= 4) return '****';
   109→  const first = str.slice(0, 2);
   110→  const last = str.slice(-2);
   111→  const middleLen = str.length - 4;
   112→  return first + '*'.repeat(middleLen) + last;
   113→}
   114→
   115→function formatDate(dateStr?: string): string {
   116→  if (!dateStr) return '—';
   117→  const d = new Date(dateStr);
   118→  return d.toLocaleDateString('en-IN', {
   119→    day: 'numeric',
   120→    month: 'short',
   121→    year: 'numeric',
   122→  });
   123→}
   124→
   125→function isActive(employee: Employee): boolean {
   126→  return !employee.date_of_leaving && employee.status !== 'inactive' && employee.status !== 'resigned';
   127→}
