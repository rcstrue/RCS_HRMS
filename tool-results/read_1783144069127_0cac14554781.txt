    85→    85→            'message' => $message,

   166→   166→     * Send Email using PHP's mail() or SMTP

   168→   168→    public function sendEmail($to, $subject, $body, $attachments = [], $isHTML = true) {
   169→   169→        $fromEmail = $this->emailConfig['user'] ?? 'noreply@rcshrms.com';
   170→   170→        $fromName = 'RCS HRMS Pro';
   171→   171→        
   172→   172→        // Headers
   173→   173→        $headers = [
   174→   174→            'From' => $fromName . ' <' . $fromEmail . '>',
   175→   175→            'Reply-To' => $fromEmail,
   176→   176→            'X-Mailer' => 'PHP/' . phpversion()
   177→   177→        ];
   178→   178→        
   179→   179→        if ($isHTML) {
   180→   180→            $headers['MIME-Version'] = '1.0';
   181→   181→            $headers['Content-Type'] = 'text/html; charset=UTF-8';
   182→   182→        }
   183→   183→        
   184→   184→        // Convert headers array to string
   185→   185→        $headerStr = '';
   186→   186→        foreach ($headers as $key => $value) {
   187→   187→            $headerStr .= $key . ': ' . $value . "\r\n";
   188→   188→        }
   189→   189→        
   190→   190→        // Handle attachments (simple implementation)
   191→   191→        if (!empty($attachments)) {
   192→   192→            $boundary = md5(time());
   193→   193→            $headerStr = "MIME-Version: 1.0\r\n";
   194→   194→            $headerStr .= "Content-Type: multipart/mixed; boundary=\"$boundary\"\r\n";
   195→   195→            
   196→   196→            $mimeBody = "--$boundary\r\n";
   197→   197→            $mimeBody .= "Content-Type: text/html; charset=UTF-8\r\n";
   198→   198→            $mimeBody .= "Content-Transfer-Encoding: 7bit\r\n\r\n";
   199→   199→            $mimeBody .= $body . "\r\n";
   200→   200→            
   201→   201→            foreach ($attachments as $attachment) {
   202→   202→                if (file_exists($attachment['path'])) {
   203→   203→                    $content = chunk_split(base64_encode(file_get_contents($attachment['path'])));
   204→   204→                    $mimeBody .= "--$boundary\r\n";
   205→   205→                    $mimeBody .= "Content-Type: " . ($attachment['type'] ?? 'application/octet-stream') . "; name=\"" . $attachment['name'] . "\"\r\n";
   206→   206→                    $mimeBody .= "Content-Transfer-Encoding: base64\r\n";
   207→   207→                    $mimeBody .= "Content-Disposition: attachment; filename=\"" . $attachment['name'] . "\"\r\n\r\n";
   208→   208→                    $mimeBody .= $content . "\r\n";
   209→   209→                }
   210→   210→            }
   211→   211→            $mimeBody .= "--$boundary--";
   212→   212→            $body = $mimeBody;
   213→   213→        }
   214→   214→        
   215→   215→        // Send email
   216→   216→        $result = mail($to, $subject, $body, $headerStr);
   217→   217→        
   218→   218→        // Log
   219→   219→        $this->logNotification('email', $to, $subject, $result ? 'sent' : 'failed', '');
   220→   220→        
   221→   221→        return [
   222→   222→            'success' => $result,
   223→   223→            'message' => $result ? 'Email sent successfully' : 'Failed to send email'
   224→   224→        ];
   225→   225→    }

   228→   228→     * Send email using Gmail SMTP (more reliable)

   230→   230→    public function sendEmailSMTP($to, $subject, $body, $attachments = []) {
   231→   231→        // Check if PHPMailer is available
   232→   232→        $phpmailerPath = APP_ROOT . '/vendor/phpmailer/phpmailer/src/PHPMailer.php';
   233→   233→        
   234→   234→        if (!file_exists($phpmailerPath)) {
   235→   235→            // Fall back to basic mail
   236→   236→            return $this->sendEmail($to, $subject, $body, $attachments);
   237→   237→        }
   238→   238→        
   239→   239→        require_once $phpmailerPath;
   240→   240→        require_once APP_ROOT . '/vendor/phpmailer/phpmailer/src/SMTP.php';
   241→   241→        
   242→   242→        $mail = new PHPMailer\PHPMailer\PHPMailer(true);
   243→   243→        
   244→   244→        try {
   245→   245→            // SMTP Configuration
   246→   246→            $mail->isSMTP();
   247→   247→            $mail->Host = $this->emailConfig['host'] ?? 'smtp.gmail.com';
   248→   248→            $mail->SMTPAuth = true;
   249→   249→            $mail->Username = $this->emailConfig['user'] ?? 'your-email@gmail.com';
   250→   250→            $mail->Password = $this->emailConfig['pass'] ?? 'your-app-password';
   251→   251→            $mail->SMTPSecure = 'tls';
   252→   252→            $mail->Port = 587;
   253→   253→            
   254→   254→            $mail->setFrom($this->emailConfig['user'] ?? 'noreply@rcshrms.com', 'RCS HRMS Pro');
   255→   255→            $mail->addAddress($to);
   256→   256→            
   257→   257→            $mail->isHTML(true);
   258→   258→            $mail->Subject = $subject;
   259→   259→            $mail->Body = $body;
   260→   260→            
   261→   261→            // Attachments
   262→   262→            foreach ($attachments as $attachment) {
   263→   263→                if (file_exists($attachment['path'])) {
   264→   264→                    $mail->addAttachment($attachment['path'], $attachment['name']);
   265→   265→                }
   266→   266→            }
   267→   267→            
   268→   268→            $mail->send();
   269→   269→            
   270→   270→            $this->logNotification('email', $to, $subject, 'sent', '');
   271→   271→            
   272→   272→            return ['success' => true, 'message' => 'Email sent via SMTP'];
   273→   273→            
   274→   274→        } catch (Exception $e) {
   275→   275→            $this->logNotification('email', $to, $subject, 'failed', $e->getMessage());
   276→   276→            
   277→   277→            return ['success' => false, 'message' => $e->getMessage()];
   278→   278→        }
   279→   279→    }

   450→   450→        ]);
   451→   451→        
   452→   452→        $response = curl_exec($ch);

   455→   455→        
   456→   456→        $result = json_decode($response, true);
   457→   457→        
   458→   458→        return [
   459→   459→            'success' => ($httpCode == 200 && ($result['success'] ?? false)),

   899→   899→    /**
   900→   900→     * Create in-app notification for users
   901→   901→     * @param string $title Notification title
   902→   902→     * @param string $message Notification message
   903→   903→     * @param string $link Link to navigate when clicked
   904→   904→     * @param string $type Notification type (info, warning, danger, success)
   905→   905→     * @param int|null $userId Specific user ID, or null for all admins
   906→   906→     * @return bool
   907→   907→     */
   908→   908→    public function createNotification($title, $message, $link = '', $type = 'info', $userId = null) {
   909→   909→        try {
   910→   910→            // If no specific user, notify all admin/HR users
   911→   911→            if ($userId === null) {
   912→   912→                $users = $this->db->fetchAll(
   913→   913→                    "SELECT u.id FROM users u 
   914→   914→                     JOIN roles r ON u.role_id = r.id 
   915→   915→                     WHERE r.role_code IN ('admin', 'hr_executive') AND u.is_active = 1"
   916→   916→                );
   917→   917→                
   918→   918→                foreach ($users as $user) {
   919→   919→                    $this->db->insert('notifications', [
   920→   920→                        'user_id' => $user['id'],
   921→   921→                        'title' => $title,
   922→   922→                        'message' => $message,
   923→   923→                        'link' => $link,
   924→   924→                        'type' => $type,
   925→   925→                        'is_read' => 0,
   926→   926→                        'created_at' => date('Y-m-d H:i:s')
   927→   927→                    ]);
   928→   928→                }
   929→   929→            } else {
   930→   930→                $this->db->insert('notifications', [
   931→   931→                    'user_id' => $userId,
   932→   932→                    'title' => $title,
   933→   933→                    'message' => $message,
   934→   934→                    'link' => $link,
   935→   935→                    'type' => $type,
   936→   936→                    'is_read' => 0,
   937→   937→                    'created_at' => date('Y-m-d H:i:s')
   938→   938→                ]);
   939→   939→            }
   940→   940→            
   941→   941→            return true;
   942→   942→        } catch (Exception $e) {
   943→   943→            error_log('Notification creation error: ' . $e->getMessage());
   944→   944→            return false;
   945→   945→        }
   946→   946→    }
