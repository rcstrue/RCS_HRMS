  1233→ * Parse a SpreadsheetML XML (.xls) salary revision file.
  1234→ * Expects the format generated by download_salary_template.php.
  1235→ */
  1236→function parseSalaryRevisionXML($filePath) {
  1237→    $rows = [];
  1238→    $content = file_get_contents($filePath);
  1239→    if (empty($content)) return $rows;
  1240→    
  1241→    // Register namespace for SpreadsheetML
  1242→    // Handle both with and without namespace prefix
  1243→    libxml_use_internal_errors(true);
  1244→    $xml = @simplexml_load_string($content);
  1245→    if (!$xml) {
  1246→        // Try with namespace
  1247→        $xml = @simplexml_load_string($content, 'SimpleXMLElement', LIBXML_NOCDATA, 'urn:schemas-microsoft-com:office:spreadsheet');
  1248→    }
  1249→    
  1250→    if (!$xml) {
  1251→        // Fallback: regex parse (robust against namespace issues)
  1252→        return parseSalaryRevisionXMLFallback($content);
  1253→    }
  1254→    
  1255→    // Register namespaces
  1256→    $namespaces = $xml->getNamespaces(true);
  1257→    $ss = $namespaces['ss'] ?? 'urn:schemas-microsoft-com:office:spreadsheet';
  1258→    $xml->registerXPathNamespace('ss', $ss);
  1259→    
  1260→    // Find all Rows in the Table
  1261→    $tableRows = $xml->xpath('//ss:Row');
  1262→    if (empty($tableRows)) {
  1263→        // Try without namespace
  1264→        $tableRows = $xml->xpath('//Row');
  1265→    }
  1266→    
  1267→    $rowIdx = 0;
  1268→    foreach ($tableRows as $xmlRow) {
  1269→        $rowIdx++;
  1270→        if ($rowIdx <= 2) continue; // skip header + instruction rows
  1271→        
  1272→        // Get cells
  1273→        $cells = [];
  1274→        $cellNodes = $xmlRow->xpath('ss:Cell') ?: $xmlRow->xpath('Cell');
  1275→        
  1276→        // Handle ss:Index attribute (sparse cells)
  1277→        $lastIndex = 0;
  1278→        foreach ($cellNodes as $cellNode) {
  1279→            $indexAttr = $cellNode->attributes($ss)['Index'] ?? null;
  1280→            if ($indexAttr !== null) {
  1281→                $index = (int)$indexAttr - 1; // 0-based
  1282→            } else {
  1283→                $index = $lastIndex;
  1284→            }
  1285→            // Fill gaps
  1286→            while (count($cells) < $index) {
  1287→                $cells[] = '';
  1288→            }
  1289→            $cellValue = (string)($cellNode->xpath('ss:Data')[0] ?? $cellNode->Data[0] ?? '');
  1290→            $cells[] = $cellValue;
  1291→            $lastIndex = count($cells);
  1292→        }
  1293→        
  1294→        if (empty(array_filter($cells))) continue; // skip blank rows
  1295→        
  1296→        // New format: [code, name(ref), client(ref), unit(ref), basic_da, hra, leave, bonus, wash, pf, esi, pt, lwf]
  1297→        $rows[] = [
  1298→            'employee_code'    => trim($cells[0] ?? ''),
  1299→            'basic_da'          => (float)($cells[4] ?? 0),
  1300→            'hra'               => (float)($cells[5] ?? 0),
  1301→            'leave_encashment'  => (float)($cells[6] ?? 0),
  1302→            'bonus_encashment'  => (float)($cells[7] ?? 0),
  1303→            'washing_allowance' => (float)($cells[8] ?? 0),
  1304→            'pf_applicable'     => (int)($cells[9] ?? 1),
  1305→            'esi_applicable'     => (int)($cells[10] ?? 1),
  1306→            'pt_applicable'     => (int)($cells[11] ?? 1),
  1307→            'lwf_applicable'     => (int)($cells[12] ?? 1),
  1308→        ];
  1309→    }
  1310→    
  1311→    libxml_clear_errors();
  1312→    return $rows;
  1313→}

  1315→/**
  1316→ * Fallback XML parser using regex for when SimpleXML fails with namespaces.
  1317→ */
  1318→function parseSalaryRevisionXMLFallback($content) {
  1319→    $rows = [];
  1320→    
  1321→    // Extract all <Row> blocks
  1322→    preg_match_all('/<Row[^>]*>(.*?)<\/Row>/si', $content, $rowMatches);
  1323→    
  1324→    $rowIdx = 0;
  1325→    foreach ($rowMatches[1] as $rowContent) {
  1326→        $rowIdx++;
  1327→        if ($rowIdx <= 2) continue; // skip header + instruction rows
  1328→        
  1329→        // Extract all <Data> values (cell values)
  1330→        preg_match_all('/<Data[^>]*>(.*?)<\/Data>/si', $rowContent, $cellMatches);
  1331→        $cells = $cellMatches[1];
  1332→        
  1333→        // Also track ss:Index for sparse cells
  1334→        preg_match_all('/<Cell[^>]*ss:Index="(\d+)"[^>]*>/si', $rowContent, $indexMatches);
  1335→        
  1336→        if (empty(array_filter($cells))) continue;
  1337→        
  1338→        // Build sparse array with proper indexing
  1339→        $fullCells = [];
  1340→        $cellIdx = 0;
  1341→        foreach ($cells as $ci => $val) {
  1342→            if (isset($indexMatches[1][$ci])) {
  1343→                $targetIdx = (int)$indexMatches[1][$ci] - 1; // 1-based to 0-based
  1344→                while (count($fullCells) < $targetIdx) {
  1345→                    $fullCells[] = '';
  1346→                }
  1347→            }
  1348→            $fullCells[] = trim($val);
  1349→        }
  1350→        
  1351→        $rows[] = [
  1352→            'employee_code'    => trim($fullCells[0] ?? ''),
  1353→            'basic_da'          => (float)($fullCells[4] ?? 0),
  1354→            'hra'               => (float)($fullCells[5] ?? 0),
  1355→            'leave_encashment'  => (float)($fullCells[6] ?? 0),
  1356→            'bonus_encashment'  => (float)($fullCells[7] ?? 0),
  1357→            'washing_allowance' => (float)($fullCells[8] ?? 0),
  1358→            'pf_applicable'     => (int)($fullCells[9] ?? 1),
