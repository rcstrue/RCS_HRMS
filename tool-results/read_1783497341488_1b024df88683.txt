    85→    85→function getInitials(name: string): string {
    86→    86→  return name
    87→    87→    .split(' ')
    88→    88→    .map((w) => w[0])
    89→    89→    .filter(Boolean)
    90→    90→    .slice(0, 2)

    92→    92→    .toUpperCase();
    93→    93→}

   100→   100→function maskSensitive(value: string | number | undefined): string {
   101→   101→  const str = value != null ? String(value) : '';
   102→   102→  if (!str) return '****';
   103→   103→  if (str.length <= 4) return '****';
   104→   104→  const first = str.slice(0, 2);
   105→   105→  const last = str.slice(-2);
   106→   106→  const middleLen = str.length - 4;
   107→   107→  return first + '*'.repeat(middleLen) + last;
   108→   108→}

   110→   110→function formatDate(dateStr?: string): string {
   111→   111→  if (!dateStr) return '—';
   112→   112→  const d = new Date(dateStr);
   113→   113→  return d.toLocaleDateString('en-IN', {
   114→   114→    day: 'numeric',
   115→   115→    month: 'short',
   116→   116→    year: 'numeric',
   117→   117→  });
   118→   118→}
