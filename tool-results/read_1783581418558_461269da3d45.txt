    46→         WHERE unit_id = ? AND is_active = 1

    69→            $dow = (int)date('N', strtotime("$year-$month-$d"));
    70→            if ($dow === 7) $sundays++;
    71→        }
    72→        $totalDays = $calendarDays - $sundays;
    73→        break;
    74→    default: // 'actual'

   102→        $pfEmployerEps  = (float)$pfRate['employer_share_eps'];
   103→        $pfEmployerEdlis = (float)$pfRate['employer_share_edlis'];

   111→        $esiEmployerShare = (float)$esiRate['employer_share'];

   123→        "SELECT u.state, s.pt_applicable as state_pt_applicable, s.id as state_id
   124→         FROM units u
   125→         LEFT JOIN states s ON (u.state = s.state_code OR u.state = s.state_name)
   126→         WHERE u.id = ?",

   130→        $unitState = $unitInfo['state'] ?? '';
   131→        $unitStatePtApplicable = !empty($unitInfo['state_pt_applicable']);
   132→    }

   136→            $ptSlabs = $db->fetchAll(
   137→                "SELECT min_gross, max_gross, pt_amount
   138→                 FROM professional_tax_slabs
   139→                 WHERE (state_name = ? OR state_name = ?) AND is_active = 1
   140→                 ORDER BY min_gross",
   141→                [$unitState, ucfirst(strtolower($unitState))]
   142→            );
   143→        } catch (Exception $e) {}
   144→    }
   145→    // Fetch LWF rate for this state (from lwf_rates → lwf_state_rates fallback)
   146→    if ($unitState) {
   147→        try {
   148→            $lwfRate = $db->fetch(
   149→                "SELECT lr.employee_share, lr.contribution_months
   150→                 FROM lwf_rates lr
   151→                 JOIN states s ON lr.state_id = s.id
   152→                 WHERE (s.state_code = ? OR s.state_name = ?)
   153→                 AND lr.is_active = 1 AND lr.effective_from <= CURDATE()
   154→                 ORDER BY lr.effective_from DESC LIMIT 1",
   155→                [$unitState, $unitState]
   156→            );
   157→            if (!$lwfRate) {
   158→                $lwfRate = $db->fetch(
   159→                    "SELECT employee_share, contribution_months
   160→                     FROM lwf_state_rates
   161→                     WHERE (state_name = ? OR state_name = ?) AND is_active = 1
   162→                     AND effective_from <= CURDATE()
   163→                     ORDER BY effective_from DESC LIMIT 1",
   164→                    [$unitState, ucfirst(strtolower($unitState))]
   165→                );
   166→            }
   167→            if ($lwfRate) {
   168→                $contribMonths = array_filter(array_map('intval', explode(',', $lwfRate['contribution_months'] ?? '')));
   169→                if (in_array($month, $contribMonths)) {

   171→                }

   212→             WHERE employee_id = ? AND effective_from <= ?
   213→             ORDER BY effective_from DESC LIMIT 1",
   214→            [$empId, $effDateStr]

   216→
   217→        // Attendance
   218→        $att = [];
   219→        try {
   220→            $att = $db->fetch(
   221→                "SELECT * FROM attendance_summary
   222→                 WHERE employee_id = ? AND month = ? AND year = ?",
   223→                [$empId, $month, $year]

   225→        } catch (Exception $e) {
   226→            $att = [];
   227→        }
   228→
   229→        // Advances
   230→        $adv = [];
   231→        try {
   232→            $adv = $db->fetch(
   233→                "SELECT COALESCE(adv1,0)+COALESCE(adv2,0)+COALESCE(office_advance,0)+COALESCE(dress_advance,0) as total_advance,
   234→                        COALESCE(adv1,0) as adv1,
   235→                        COALESCE(adv2,0) as adv2,
   236→                        COALESCE(office_advance,0) as office_advance,
   237→                        COALESCE(dress_advance,0) as dress_advance
   238→                 FROM employee_advances
   239→                 WHERE employee_id = ? AND month = ? AND year = ?",
   240→                [$empId, $month, $year]
   241→            );
   242→        } catch (Exception $e) {
   243→            $adv = [];
   244→        }
   245→
   246→        // Existing payroll record (if already saved for this month/year)
   247→        $existing = [];
   248→        try {
   249→            // Try direct month/year lookup first (new schema)
   250→            $existing = $db->fetch(
   251→                "SELECT * FROM payroll
   252→                 WHERE employee_id = ? AND month = ? AND year = ?",
   253→                [$empCode, $month, $year]
   254→            );
   255→            if (empty($existing)) {
   256→                // Fallback: via payroll_periods for old data
   257→                $existing = $db->fetch(
   258→                    "SELECT p.* FROM payroll p
   259→                     JOIN payroll_periods pp ON p.payroll_period_id = pp.id
   260→                     WHERE p.employee_id = ? AND pp.month = ? AND pp.year = ?
   261→                     ORDER BY p.id DESC LIMIT 1",
   262→                    [$empCode, $month, $year]
   263→                );
   264→            }
   265→        } catch (Exception $e) {
   266→            $existing = [];
   267→        }
   268→
   269→        // Active loan EMI for this employee (with balance > 0, not already deducted this month)
   270→        $loanEmi = 0;
   271→        try {
   272→            // Sum EMI from ALL active loans (match process.php logic)
   273→            $loans = $db->fetchAll(
   274→                "SELECT el.emi_amount, el.id as loan_id
   275→                 FROM employee_loans el
   276→                 WHERE el.employee_id = ? AND el.status = 'Active'
   277→                 AND el.balance_amount > 0
   278→                 AND (el.start_year < ? OR (el.start_year = ? AND el.start_month <= ?))
   279→                 AND el.id NOT IN (
   280→                     SELECT lem.loan_id FROM loan_emi_log lem
   281→                     WHERE lem.month = ? AND lem.year = ?
   282→                 )",
   283→                [$empId, $year, $year, $month, $month, $year]
   284→            );
   285→            foreach ($loans as $loan) {
   286→                $loanEmi += floatval($loan['emi_amount']);
   287→            }
   288→        } catch (Exception $e) {
   289→            // Fallback: try without emi_log check (in case table has different structure)
   290→            try {
   291→                $loans2 = $db->fetchAll(
   292→                    "SELECT el.emi_amount, el.id as loan_id
   293→                     FROM employee_loans el
   294→                     WHERE el.employee_id = ? AND el.status = 'Active'
   295→                     AND el.balance_amount > 0
   296→                     AND (el.start_year < ? OR (el.start_year = ? AND el.start_month <= ?))",
   297→                    [$empId, $year, $year, $month]
   298→                );
   299→                foreach ($loans2 as $loan2) {
   300→                    $loanEmi += floatval($loan2['emi_amount']);
   301→                }
   302→            } catch (Exception $e2) {
   303→                $loanEmi = 0;
   304→            }
   305→        }
   306→
   307→        $employeeData[] = [
   308→            'emp'      => $emp,
   309→            'salary'   => $salary ?: [],
   310→            'att'      => $att ?: [],
   311→            'adv'      => $adv ?: [],
   312→            'existing' => $existing ?: [],
   313→            'loan_emi' => $loanEmi,
   314→        ];
   315→    }

  1403→    function previewModalCalc() {
  1404→        var totalDays = TOTAL_DAYS;
  1405→        var attPresent = parseFloat($('#modalAttPresent').val()) || 0;
  1406→        var attWO      = parseFloat($('#modalAttWO').val()) || 0;
  1407→        var attExtra   = parseFloat($('#modalAttExtra').val()) || 0;
  1408→        var otHours    = parseFloat($('#modalOtHours').val()) || 0;
  1409→        var paidDays   = attPresent + attWO + attExtra;
  1410→
  1411→        var wBasicDa   = parseFloat($('#modalWBasicDa').val()) || 0;
  1412→        var wHra       = parseFloat($('#modalWHra').val()) || 0;
  1413→        var wLeaveEnc  = parseFloat($('#modalWLeaveEnc').val()) || 0;
  1414→        var wBonusEnc  = parseFloat($('#modalWBonusEnc').val()) || 0;
  1415→        var wWash      = parseFloat($('#modalWWash').val()) || 0;
  1416→        var wOtRate    = parseFloat($('#modalWOtRate').val()) || 0;
  1417→
  1418→        var advance    = parseFloat($('#modalAdvance').val()) || 0;
  1419→        var officeDed  = parseFloat($('#modalOfficeDed').val()) || 0;
  1420→        var loanEmiM   = parseFloat($('#modalLoanEmi').val()) || 0;
  1421→
  1422→        if (totalDays === 0 || paidDays === 0) {
  1423→            $('#modalPreviewGross').text('₹0.00');
  1424→            $('#modalPreviewDed').text('₹0.00');
  1425→            $('#modalPreviewNet').text('₹0.00');
  1426→            return;
  1427→        }
  1428→
  1429→        var pfApp   = $('#modalPfApp').is(':checked');
  1430→        var esiApp  = $('#modalEsiApp').is(':checked');
  1431→        var ptApp   = $('#modalPtApp').is(':checked');
  1432→        var lwfApp  = $('#modalLwfApp').is(':checked');
  1433→
  1434→        var otMult  = modalCurrentRow ? (parseFloat(modalCurrentRow.data('ot-multiplier')) || 1) : 1;
  1435→        var lwfAmt  = modalCurrentRow ? (parseFloat(modalCurrentRow.data('lwf-amount')) || 0) : 0;
  1436→        var existGr = modalCurrentRow ? (parseFloat(modalCurrentRow.data('existing-gross')) || 0) : 0;
  1437→
  1438→        var basicDa  = round2(wBasicDa * paidDays / totalDays);
  1439→        var hra      = round2(wHra * paidDays / totalDays);
  1440→        var leaveEnc = round2(wLeaveEnc * paidDays / totalDays);
  1441→        var bonusEnc = round2(wBonusEnc * paidDays / totalDays);
  1442→        var wash     = round2(wWash * paidDays / totalDays);
  1443→        var otAmt    = round2(wOtRate * otHours * otMult);
  1444→
  1445→        var gross = round2(basicDa + hra + leaveEnc + bonusEnc + wash + otAmt);
  1446→        var pf    = pfApp ? round2(Math.min(basicDa, PF_WAGE_CEILING) * PF_EMPLOYEE_SHARE / 100) : 0;
  1447→        var esiG  = existGr > 0 ? existGr : 0; // full monthly gross
  1448→        var esi   = (esiApp && esiG > 0 && esiG <= ESI_WAGE_CEILING) ? round2(gross * ESI_EMPLOYEE_SHARE / 100) : 0;
  1449→        var pt    = calculatePT(gross, ptApp);
  1450→        var lwf   = lwfApp ? lwfAmt : 0;
  1451→        var ded   = round2(pf + esi + pt + lwf + advance + officeDed + loanEmiM);
  1452→        var net   = Math.round(gross - ded); // Round to nearest ₹.00
  1453→
  1454→        $('#modalPreviewGross').text(fmt(gross));
  1455→        $('#modalPreviewDed').text(fmt(ded));
  1456→        $('#modalPreviewNet').text(fmt(net));
  1457→    }
