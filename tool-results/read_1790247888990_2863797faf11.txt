    14→// ══════════════════════════════════════════════════════════════
    15→// unwrap - PHP wraps responses in { success, message, data }
    16→// apiRequest gives { data: <PHP envelope>, error }
    17→// This helper strips the envelope so callers get the actual payload
    18→// Handles: { success: true, data: ... } → { data: ..., error: null }
    19→//          { success: false, message/error: "..." } → { data: null, error: "..." }
    20→//          Raw responses (no envelope) → pass through unchanged
    21→// ══════════════════════════════════════════════════════════════
    22→function unwrap<T>(result: Promise<{ data: T | null; error: string | null }>): Promise<{ data: T | null; error: string | null }> {
    23→  return result.then((res) => {
    24→    if (res.error) return res;
    25→    const d = res.data as Record<string, unknown> | null;
    26→    if (d && typeof d === 'object' && 'success' in d) {
    27→      // PHP returned a success:false envelope — extract error message
    28→      if (d.success === false) {
    29→        const msg = (d.message as string) || (d.error as string) || 'Request failed';
    30→        return { data: null, error: msg };
    31→      }
    32→      // PHP returned a success:true envelope — extract inner data
    33→      // Guard: data key may exist but be null (e.g. { success: true, data: null })
    34→      if ('data' in d && d.data != null) {
    35→        return { data: d.data as T, error: null };
    36→      }
    37→      // success:true but data is null/missing → treat as empty result
    38→      if (d.success === true) {
    39→        return { data: null, error: null };
    40→      }
    41→    }
    42→    return res;
    43→  });
    44→}
    45→
    46→// ===== Auth =====
    47→// Login is special: PHP may return { success: false, is_locked: true, lockout_remaining: ... }
    48→// alongside the error. We must NOT unwrap success:false into data:null for login,
    49→// because the caller needs access to is_locked, rate_limit_remaining, etc.
    50→export async function essLogin(mobileNumber: string, pin: string) {
    51→  return apiRequest<LoginResponse>('/ess/login', {
    52→    method: 'POST',
    53→    body: JSON.stringify({ mobile_number: mobileNumber, pin }),
    54→  }).then((res) => {
    55→    if (res.error) return res;
    56→    const d = res.data as Record<string, unknown> | null;
    57→    if (d && typeof d === 'object' && 'success' in d) {
    58→      if (d.success === false) {
    59→        // Return the FULL envelope as data so LoginScreen can read is_locked, etc.
    60→        return { data: d as unknown as LoginResponse, error: (d.message as string) || (d.error as string) || 'Login failed' };
    61→      }
    62→      if ('data' in d && d.data != null) {
    63→        return { data: d.data as LoginResponse, error: null };
    64→      }
    65→      // success:true but data is null/missing → treat as empty result
    66→      if (d.success === true) {
    67→        return { data: null, error: null };
    68→      }
    69→    }
    70→    return res;
    71→  });
    72→}

   296→export async function fetchUnitVisits(params: {
   297→  employee_id?: number;
   298→  /** When true, fetch checklists from ALL managers (not just the caller's own).
   299→   *  Backend ignores employee_id and returns every manager's visits.
   300→   *  Only honored for manager+ roles; lower roles are silently scoped to self. */
   301→  all?: boolean;
   302→  month?: number;
   303→  year?: number;
   304→  unit_id?: number;
   305→  status?: string;
   306→  page?: number;
   307→  limit?: number;
   308→  include_checklist?: boolean;
   309→}) {
   310→  const searchParams = new URLSearchParams();
   311→  if (params.all) {
   312→    searchParams.set('all', '1');
   313→  } else if (params.employee_id != null) {
   314→    searchParams.set('employee_id', String(params.employee_id));
   315→  }
   316→  if (params.month) searchParams.set('month', String(params.month));
   317→  if (params.year) searchParams.set('year', String(params.year));
   318→  if (params.unit_id) searchParams.set('unit_id', String(params.unit_id));
   319→  if (params.status) searchParams.set('status', params.status);
   320→  if (params.page) searchParams.set('page', String(params.page));
   321→  if (params.limit) searchParams.set('limit', String(params.limit));
   322→  if (params.include_checklist) searchParams.set('include_checklist', '1');
   323→  return unwrap(apiRequest(`/ess/unit-visits?${searchParams}`));
   324→}

   629→export function verifyCertificate(code: string) {
   630→  // Public endpoint — NO auth token, but X-API-KEY is required by server .htaccess
   631→  // Direct fetch with cache:'no-store' + timestamp to bypass service worker cache
   632→  const API_KEY = (import.meta as Record<string, Record<string, string>>).env?.VITE_API_KEY ?? '';
   633→  const url = `${API_BASE_URL}/api/ess/verify-certificate.php?cert=${encodeURIComponent(code)}&_t=${Date.now()}`;
   634→  return fetch(url, {
   635→    method: 'GET',
   636→    headers: { 'Content-Type': 'application/json', 'X-API-KEY': API_KEY },
   637→    cache: 'no-store',
   638→  }).then(async (resp) => {
   639→    const text = await resp.text();
   640→    if (!text.trim()) return { data: null, error: 'Empty response' };
   641→    try {
   642→      const json = JSON.parse(text);
   643→      if (!resp.ok) return { data: null, error: json?.error || json?.message || 'Verification failed' };
   644→      // Unwrap PHP envelope: { success, data }
   645→      return { data: (json?.success ? json.data : json) as VerifyCertificateResponse, error: null };
   646→    } catch {
   647→      return { data: null, error: 'Invalid response' };
   648→    }
   649→  }).catch(() => ({ data: null, error: 'Network error. Please check your connection.' }));
   650→}
