    50→

    54→                ['status' => STATUS_PENDING . '%']
    55→            );
    56→            $counts['pending'] = (int)($result['count'] ?? 0);
    57→            
    58→            // Inactive employees
    59→            $result = $this->db->fetch(
    60→                "SELECT COUNT(*) as count FROM employees WHERE status = :status",
    61→                ['status' => STATUS_INACTIVE]
    62→            );
    63→            $counts['inactive'] = (int)($result['count'] ?? 0);
    64→            
    65→            // Left employees
    66→            $result = $this->db->fetch(
    67→                "SELECT COUNT(*) as count FROM employees WHERE date_of_leaving IS NOT NULL"

    71→            // Table might not exist yet

    73→        }

    78→    // Get all employees with filters
    79→    public function getAll($filters = [], $page = 1, $perPage = DEFAULT_PAGE_SIZE) {
    80→        // Check if employee_salary_structures table exists
    81→        $useSalaryTable = $this->checkSalaryTableExists();
    82→        $baseColumns = $this->getBaseColumns();
    83→        
    84→        if ($useSalaryTable) {
    85→            $sql = "SELECT $baseColumns,
    86→                           ess.basic_da, ess.hra, ess.leave_encashment, ess.bonus_encashment,
    87→                           ess.washing_allowance, ess.gross_salary,
    88→                           ess.pf_applicable, ess.esi_applicable, ess.pt_applicable,
    89→                           c.name as client_name_display,
    90→                           u.name as unit_name_display
    91→                    FROM employees e
    92→                    LEFT JOIN (
    93→                        SELECT employee_id, basic_da, hra, leave_encashment, bonus_encashment,
    94→                               washing_allowance, gross_salary, pf_applicable, esi_applicable, pt_applicable
    95→                        FROM employee_salary_structures
    96→                        WHERE effective_to IS NULL OR effective_to >= CURDATE()
    97→                        GROUP BY employee_id
    98→                    ) ess ON e.id = ess.employee_id
    99→                    LEFT JOIN clients c ON e.client_id = c.id
   100→                    LEFT JOIN units u ON e.unit_id = u.id
   101→                    WHERE 1=1";
   102→        } else {
   103→            $sql = "SELECT $baseColumns,
   104→                           c.name as client_name_display,
   105→                           u.name as unit_name_display
   106→                    FROM employees e
   107→                    LEFT JOIN clients c ON e.client_id = c.id
   108→                    LEFT JOIN units u ON e.unit_id = u.id
   109→                    WHERE 1=1";
   110→        }
   111→        $params = [];
   112→
   113→        // Status filter - use strtolower for case-insensitive comparison
   114→        if (!empty($filters['status'])) {
   115→            $statusLower = strtolower($filters['status']);
   116→            if ($statusLower === 'active' || $statusLower === 'approved') {
   117→                $sql .= " AND e.status = :status";
   118→                $params['status'] = STATUS_APPROVED;
   119→            } elseif ($statusLower === 'pending') {
   120→                $sql .= " AND e.status LIKE :status";
   121→                $params['status'] = STATUS_PENDING . '%';
   122→            } else {
   123→                $sql .= " AND e.status = :status";
   124→                $params['status'] = $filters['status'];
   125→            }
   126→        }
   127→
   128→        // Client filter
   129→        if (!empty($filters['client_id'])) {
   130→            $sql .= " AND e.client_id = :client_id";
   131→            $params['client_id'] = $filters['client_id'];
   132→        }
   133→
   134→        // Unit filter
   135→        if (!empty($filters['unit_id'])) {
   136→            $sql .= " AND e.unit_id = :unit_id";
   137→            $params['unit_id'] = $filters['unit_id'];
   138→        }
   139→
   140→        // Worker category filter
   141→        if (!empty($filters['worker_category'])) {
   142→            $sql .= " AND e.worker_category = :worker_category";
   143→            $params['worker_category'] = $filters['worker_category'];
   144→        }
   145→
   146→        // Search filter - use unique parameter names for PDO with emulated prepares disabled
   147→        if (!empty($filters['search'])) {
   148→            $sql .= " AND (e.employee_code LIKE :search1
   149→                      OR e.full_name LIKE :search2
   150→                      OR e.mobile_number LIKE :search3
   151→                      OR e.aadhaar_number LIKE :search4)";
   152→            $searchTerm = '%' . $filters['search'] . '%';
   153→            $params['search1'] = $searchTerm;
   154→            $params['search2'] = $searchTerm;
   155→            $params['search3'] = $searchTerm;
   156→            $params['search4'] = $searchTerm;
   157→        }
   158→
   159→        // Count total
   160→        $countSql = "SELECT COUNT(*) as total FROM ($sql) as t";
   161→        $total = $this->db->fetchColumn($countSql, $params);
   162→
   163→        // Pagination
   164→        $offset = ($page - 1) * $perPage;
   165→        $sql .= " ORDER BY e.created_at DESC LIMIT $perPage OFFSET $offset";
   166→
   167→        $employees = $this->db->fetchAll($sql, $params);
   168→
   169→        return [
   170→            'data' => $employees,
   171→            'total' => $total ?: 0,
   172→            'page' => $page,
   173→            'per_page' => $perPage,
   174→            'total_pages' => ceil(($total ?: 0) / $perPage)
   175→        ];
   176→    }

   311→    // Resolve employee by either id (numeric) or code (alphanumeric)
   312→    // Returns the full employee record or null
   313→    public function getByIdOrCode($idOrCode) {
   314→        // If it looks like a numeric id, try getById first
   315→        if (is_numeric($idOrCode) && (int)$idOrCode > 0) {
   316→            $emp = $this->getById((int)$idOrCode);
   317→            if ($emp) return $emp;
   318→        }
   319→        // Try as employee_code
   320→        $emp = $this->getByCode($idOrCode);
   321→        if ($emp) return $emp;
   322→        // Last resort: if numeric, could be a code that happens to be numeric
   323→        if (is_numeric($idOrCode)) {
   324→            return $this->getByCode((string)$idOrCode);
   325→        }
   326→        return null;
   327→    }

   351→    // Create new employee
   352→    public function create($data) {
   353→        // Generate employee code if not provided
   354→        if (empty($data['employee_code'])) {
   355→            $data['employee_code'] = $this->generateEmployeeCode();
   356→        }
   357→        
   358→        // Set default status
   359→        $data['status'] = $data['status'] ?? STATUS_PENDING_HR;
   360→        // Set default app_role to 'employee' if not provided
   361→        $data['app_role'] = $data['app_role'] ?? 'employee';
   362→        $data['created_at'] = date(DATETIME_FORMAT_DB);
   363→
   364→        // Map form fields to database columns
   365→        $dbData = $this->mapFormDataToDb($data);
   366→
   367→        // Check if employee_salary_structures table exists
   368→        $useSalaryTable = $this->checkSalaryTableExists();
   369→
   370→        try {
   371→            $this->db->beginTransaction();
   372→            
   373→            // Insert employee
   374→            $this->db->insert('employees', $dbData);
   375→            
   376→            // Get the newly created employee ID
   377→            $employeeId = $this->db->lastInsertId();
   378→            
   379→            // Insert salary structure if table exists and any salary/statutory data provided
   380→            $hasSalaryData = !empty($data['basic_da']) || !empty($data['basic_salary']) ||
   381→                isset($data['pf_applicable']) || isset($data['esi_applicable']) ||
   382→                isset($data['pt_applicable']) || isset($data['lwf_applicable']) ||
   383→                isset($data['bonus_applicable']) || isset($data['gratuity_applicable']) ||
   384→                isset($data['overtime_applicable']);
   385→            
   386→            if ($useSalaryTable && $hasSalaryData && $employeeId) {
   387→                $salaryData = $this->buildSalaryData($data, $employeeId);
   388→                $this->db->insert('employee_salary_structures', $salaryData);
   389→            }
   390→            
   391→            $this->db->commit();
   392→            
   393→            // Send notification for pending employee registration
   394→            if (strpos($data['status'], 'pending') !== false) {
   395→                $notification = new Notification();
   396→                $notification->notifyNewEmployeeRegistration([
   397→                    'id' => $employeeId,
   398→                    'full_name' => $data['full_name'] ?? '',
   399→                    'employee_code' => $data['employee_code']
   400→                ]);
   401→            }
   402→            
   403→            return ['success' => true, 'message' => MSG_RECORD_CREATED, 'employee_id' => $employeeId];
   404→        } catch (Exception $e) {
   405→            $this->db->rollBack();
   406→            return ['success' => false, 'message' => 'Failed to create employee: ' . $e->getMessage()];
   407→        }
   408→    }

   410→    // Map form data to database columns
   411→    private function mapFormDataToDb($data) {
   412→        $mapped = [];
   413→        
   414→        // Direct mapping - form fields match database columns (removed client_name/unit_name as they don't exist)
   415→        $directFields = [
   416→            'full_name', 'father_name', 'employee_code', 'mobile_number', 'alternate_mobile',
   417→            'email', 'gender', 'date_of_birth', 'marital_status', 'blood_group',
   418→            'aadhaar_number', 'uan_number', 'esic_number',
   419→            'address', 'pin_code', 'state', 'district',
   420→            'bank_name', 'account_number', 'ifsc_code', 'account_holder_name',
   421→            'client_id', 'unit_id', 'app_role', 'employee_role',
   422→            'designation', 'department',
   423→            'worker_category', 'employment_type', 'date_of_joining', 'probation_period',
   424→            'nominee_name', 'nominee_relationship', 'nominee_dob', 'nominee_contact',
   425→            'emergency_contact_name', 'emergency_contact_relation',
   426→            'status', 'profile_pic_url', 'profile_pic_cropped_url',
   427→            'aadhaar_front_url', 'aadhaar_back_url', 'bank_document_url'
   428→        ];
   429→        
   430→        foreach ($directFields as $field) {
   431→            if (isset($data[$field])) {
   432→                $mapped[$field] = $data[$field];
   433→            }
   434→        }
   435→        
   436→        // Handle middle_name -> father_name mapping (for backward compatibility)
   437→        if (isset($data['middle_name']) && !isset($data['father_name'])) {
   438→            $mapped['father_name'] = $data['middle_name'];
   439→        }
   440→        
   441→        // Look up client_id from client_name if not provided (for import compatibility)
   442→        if (!empty($data['client_name']) && empty($data['client_id'])) {
   443→            $client = $this->db->fetch(
   444→                "SELECT id FROM clients WHERE name = :name LIMIT 1",
   445→                ['name' => $data['client_name']]
   446→            );
   447→            if ($client) {
   448→                $mapped['client_id'] = $client['id'];
   449→            }
   450→        }
   451→        
   452→        // Look up unit_id from unit_name if not provided (for import compatibility)
   453→        if (!empty($data['unit_name']) && empty($data['unit_id'])) {
   454→            $unit = $this->db->fetch(
   455→                "SELECT id FROM units WHERE name = :name LIMIT 1",
   456→                ['name' => $data['unit_name']]
   457→            );
   458→            if ($unit) {
   459→                $mapped['unit_id'] = $unit['id'];
   460→            }
   461→        }
   462→        
   463→        return $mapped;
   464→    }

   466→    // Update employee
   467→    public function update($id, $data) {
   468→        $employee = $this->getById($id);
   469→
   470→        if (!$employee) {
   471→            return ['success' => false, 'message' => MSG_RECORD_NOT_FOUND];
   472→        }
   473→
   474→        // Map form fields to database columns
   475→        $dbData = $this->mapFormDataToDb($data);
   476→        unset($dbData['id'], $dbData['created_at']);
   477→        // employee_code is now mutable — can be changed via update()
   478→        // Validate uniqueness if code is being changed
   479→        if (isset($dbData['employee_code']) && $dbData['employee_code'] !== $employee['employee_code']) {
   480→            $existing = $this->db->fetch(
   481→                "SELECT id FROM employees WHERE employee_code = :code AND id != :id",
   482→                ['code' => $dbData['employee_code'], 'id' => $id]
   483→            );
   484→            if ($existing) {
   485→                return ['success' => false, 'message' => 'Employee code already exists. Please use a unique code.'];
   486→            }
   487→        }
   488→        $dbData['updated_at'] = date(DATETIME_FORMAT_DB);
   489→
   490→        // Check if employee_salary_structures table exists
   491→        $useSalaryTable = $this->checkSalaryTableExists();
   492→
   493→        // Extract salary data - include all statutory fields even if salary is not set
   494→        $salaryData = [];
   495→        if ($useSalaryTable) {
   496→            // Check if any salary or statutory field is present
   497→            $hasSalaryData = isset($data['basic_da']) || isset($data['basic_salary']) ||
   498→                isset($data['pf_applicable']) || isset($data['esi_applicable']) ||
   499→                isset($data['pt_applicable']) || isset($data['lwf_applicable']) ||
   500→                isset($data['bonus_applicable']) || isset($data['gratuity_applicable']);
   501→
   502→            if ($hasSalaryData) {
   503→                $salaryData = [
   504→                    'basic_da' => floatval($data['basic_da'] ?? $data['basic_salary'] ?? 0),
   505→                    'hra' => floatval($data['hra'] ?? 0),
   506→                    'leave_encashment' => floatval($data['leave_encashment'] ?? 0),
   507→                    'bonus_encashment' => floatval($data['bonus_encashment'] ?? 0),
   508→                    'washing_allowance' => floatval($data['washing_allowance'] ?? 0),
   509→                    'gross_salary' => floatval($data['gross_salary'] ?? 0),
   510→                    'pf_applicable' => !empty($data['pf_applicable']) ? 1 : 0,
   511→                    'esi_applicable' => !empty($data['esi_applicable']) ? 1 : 0,
   512→                    'pt_applicable' => !empty($data['pt_applicable']) ? 1 : 0,
   513→                    'lwf_applicable' => !empty($data['lwf_applicable']) ? 1 : 0,
   514→                    'bonus_applicable' => !empty($data['bonus_applicable']) ? 1 : 0,
   515→                    'gratuity_applicable' => !empty($data['gratuity_applicable']) ? 1 : 0,
   516→                    'overtime_applicable' => !empty($data['overtime_applicable']) ? 1 : 0,
   517→                    'updated_at' => date(DATETIME_FORMAT_DB),
   518→                ];
   519→            }
   520→        }
   521→
   522→        try {
   523→            $this->db->beginTransaction();
   524→            
   525→            // Update employee
   526→            $this->db->update('employees', $dbData, SQL_WHERE_ID, ['id' => $id]);
   527→            
   528→            // Update salary structure if provided and table exists (with versioning)
   529→            if (!empty($salaryData)) {
   530→                // Check if salary structure exists
   531→                $existing = $this->db->fetch(
   532→                    "SELECT id, effective_from FROM employee_salary_structures 
   533→                     WHERE employee_id = :id AND (effective_to IS NULL OR effective_to >= CURDATE())",
   534→                    ['id' => $id]
   535→                );
   536→                
   537→                if ($existing) {
   538→                    // Close the old active record to preserve history
   539→                    $this->db->query(
   540→                        "UPDATE employee_salary_structures 
   541→                         SET effective_to = DATE_SUB(CURDATE(), INTERVAL 1 DAY), updated_at = NOW() 
   542→                         WHERE id = ?",
   543→                        [$existing['id']]
   544→                    );
   545→                }
   546→                // Insert new versioned record
   547→                $salaryData['employee_id']   = $id;
   548→                $salaryData['effective_from'] = date(DATE_FORMAT_DB);
   549→                $this->db->insert('employee_salary_structures', $salaryData);
   550→            }
   551→            
   552→            $this->db->commit();
   553→            return ['success' => true, 'message' => MSG_RECORD_UPDATED];
   554→        } catch (Exception $e) {
   555→            $this->db->rollBack();
   556→            return ['success' => false, 'message' => 'Failed to update employee: ' . $e->getMessage()];
   557→        }
   558→    }

   632→    // Bulk update employee codes (array of [id => newCode])
   633→    public function bulkUpdateEmployeeCodes($codeMap) {
   634→        $results = ['success' => 0, 'failed' => 0, 'errors' => []];
   635→        
   636→        $this->db->beginTransaction();
   637→        try {
   638→            foreach ($codeMap as $id => $newCode) {
   639→                $newCode = trim($newCode);
   640→                if (empty($newCode)) {
   641→                    $results['failed']++;
   642→                    $results['errors'][] = "ID {$id}: Empty code";
   643→                    continue;
   644→                }
   645→                // Check uniqueness
   646→                $existing = $this->db->fetch(
   647→                    "SELECT id, employee_code FROM employees WHERE employee_code = :code AND id != :id",
   648→                    ['code' => $newCode, 'id' => $id]
   649→                );
   650→                if ($existing) {
   651→                    $results['failed']++;
   652→                    $results['errors'][] = "ID {$id}: Code '{$newCode}' already belongs to employee {$existing['employee_code']}";
   653→                    continue;
   654→                }
   655→                $this->db->update('employees',
   656→                    ['employee_code' => $newCode, 'updated_at' => date(DATETIME_FORMAT_DB)],
   657→                    SQL_WHERE_ID, ['id' => $id]
   658→                );
   659→                // Sync cache
   660→                try {
   661→                    $this->db->query(
   662→                        "UPDATE ess_employee_cache SET employee_code = ?, updated_at = NOW() WHERE id = ?",
   663→                        [$newCode, $id]
   664→                    );
   665→                } catch (Exception $e) {}
   666→                $results['success']++;
   667→            }
   668→            $this->db->commit();
   669→        } catch (Exception $e) {
   670→            $this->db->rollBack();
   671→            $results['errors'][] = 'Transaction failed: ' . $e->getMessage();
   672→        }
   673→        
   674→        return $results;
   675→    }

   788→    // Import employees from array
   789→    public function importFromData($data, $skipDuplicates = true) {
   790→        $imported = 0;
   791→        $errors = [];
   792→        $duplicates = [];
   793→
   794→        foreach ($data as $row) {
   795→            try {
   796→                // Check for existing employee by mobile or aadhaar
   797→                if (!empty($row['mobile_number'])) {
   798→                    $exists = $this->db->fetch(
   799→                        "SELECT id FROM employees WHERE mobile_number = :mobile",
   800→                        ['mobile' => $row['mobile_number']]
   801→                    );
   802→                    if ($exists && $skipDuplicates) {
   803→                        $duplicates[] = $row['mobile_number'];
   804→                        continue;
   805→                    }
   806→                }
   807→
   808→                $result = $this->create($row);
   809→                if ($result['success']) {
   810→                    $imported++;
   811→                } else {
   812→                    $errors[] = $result['message'];
   813→                }
   814→            } catch (Exception $e) {
   815→                $errors[] = "Error importing row: " . $e->getMessage();
   816→            }
   817→        }
   818→
   819→        return [
   820→            'success' => true,
   821→            'imported' => $imported,
   822→            'duplicates' => count($duplicates),
   823→            'errors' => $errors
   824→        ];
   825→    }
