    13→ */
    14→

    20→$bonusWageCeiling = floatval($db->fetchColumn("SELECT setting_value FROM settings WHERE setting_key = 'bonus_wage_ceiling'") ?: 7000);
    21→
    22→// Handle bonus calculation
    23→if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
    24→    
    25→    if ($_POST['action'] === 'calculate_all') {
    26→        $financialYear = sanitize($_POST['financial_year']); // Format: 2023-2024
    27→        
    28→        // Get all eligible employees
    29→        $eligibleEmployees = $db->fetchAll(
    30→            "SELECT e.id, e.employee_code, e.full_name, e.date_of_joining, e.date_of_leaving,
    31→                    e.is_bonus_applicable, c.name as client_name,
    32→                    ess.basic_da, ess.gross_salary
    33→             FROM employees e
    34→             LEFT JOIN employee_salary_structures ess ON e.id = ess.employee_id 
    35→                AND (ess.effective_to IS NULL OR ess.effective_to >= CURDATE())
    36→             LEFT JOIN clients c ON e.client_id = c.id
    37→             WHERE e.status IN ('active', 'resigned', 'terminated')
    38→                AND e.is_bonus_applicable = 1
    39→                AND (ess.basic_da <= 21000 OR ess.gross_salary <= 21000)
    40→             ORDER BY e.full_name"
    41→        );
    42→        
    43→        // Parse financial year
    44→        $fyParts = explode('-', $financialYear);
    45→        $fyStartYear = intval($fyParts[0]);
    46→        $fyEndYear = $fyStartYear + 1;
    47→        
    48→        $fyStart = "$fyStartYear-04-01";
    49→        $fyEnd = "$fyEndYear-03-31";
    50→        
    51→        $bonusRecords = [];
    52→        $totalBonus = 0;
    53→        
    54→        foreach ($eligibleEmployees as $emp) {
    55→            // Calculate days worked in FY
    56→            $doj = new DateTime($emp['date_of_joining']);
    57→            $dol = $emp['date_of_leaving'] ? new DateTime($emp['date_of_leaving']) : new DateTime($fyEnd);
    58→            $fyStartDate = new DateTime($fyStart);
    59→            $fyEndDate = new DateTime($fyEnd);
    60→            
    61→            // Adjust start date if joined after FY start
    62→            if ($doj > $fyStartDate) {
    63→                $fyStartDate = $doj;
    64→            }
    65→            
    66→            // Adjust end date if left before FY end
    67→            if ($dol < $fyEndDate) {
    68→                $fyEndDate = $dol;
    69→            }
    70→            
    71→            // Calculate months worked
    72→            $interval = $fyStartDate->diff($fyEndDate);
    73→            $monthsWorked = ($interval->y * 12) + $interval->m + ($interval->d > 0 ? 1 : 0);
    74→            
    75→            // Minimum 30 days worked required
    76→            if ($monthsWorked < 1 && $interval->d < 30) {
    77→                continue;
    78→            }
    79→            
    80→            // Calculate bonus
    81→            $basicDa = floatval($emp['basic_da'] ?? 0);
    82→            $wageForBonus = min($basicDa, $bonusWageCeiling); // Ceiling of ₹7,000 or actual
    83→            
    84→            // Calculate payable bonus (pro-rata for partial year)
    85→            $annualWage = $wageForBonus * 12;
    86→            $proRataFactor = min($monthsWorked, 12) / 12;
    87→            
    88→            // Default bonus rate (can be adjusted based on profits)
    89→            $bonusRate = floatval($_POST['bonus_rate'] ?? $bonusMinRate);
    90→            
    91→            $bonusAmount = ($annualWage * $bonusRate / 100) * $proRataFactor;
    92→

    95→                'employee_code' => $emp['employee_code'],
    96→                'full_name' => $emp['full_name'],
    97→                'client_name' => $emp['client_name'],
    98→                'basic_da' => $basicDa,
    99→                'months_worked' => min($monthsWorked, 12),
   100→                'bonus_rate' => $bonusRate,
   101→                'bonus_amount' => $bonusAmount
   102→            ];
   103→            
   104→            $totalBonus += $bonusAmount;
   105→        }

   112→            'rate' => $bonusRate
   113→        ];

   149→                    // Insert
   150→                    $db->insert('employee_bonus', [
   151→                        'employee_id' => $record['employee_id'],
   152→                        'financial_year' => $preview['financial_year'],
   153→                        'basic_da' => $record['basic_da'],

   222→            ], SQL_WHERE_ID, ['id' => $existingPayroll['id']]);

   224→            $db->insert('payroll', [
   225→                'payroll_period_id' => $periodId,
   226→                'employee_id' => $bonus['employee_id'],
   227→                'bonus' => $bonus['bonus_amount'],
   228→                'net_salary' => $bonus['bonus_amount'],
   229→                'payment_status' => 'pending',
   230→                'created_at' => date(DATETIME_FORMAT_DB)
   231→            ]);
   232→        }

   234→        // Update bonus status
   235→        $db->update('employee_bonus', [
   236→            'status' => 'disbursed',
   237→            'disbursed_at' => date(DATETIME_FORMAT_DB),
   238→            'payment_period_id' => $periodId
   239→        ], SQL_WHERE_ID, ['id' => $bonusId]);

   256→// Get preview data
