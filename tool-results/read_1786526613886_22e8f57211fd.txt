   544→                      value={editData.esic_number}
   545→                      onChange={(e) => setEditData(prev => ({ ...prev, esic_number: e.target.value }))}
   546→                      placeholder="Enter ESIC number"
   547→                    />
   548→                  </div>
   549→                  <div className="space-y-2">
   550→                    <Label>Marital Status</Label>
   551→                    <Select
   552→                      value={editData.marital_status}
   553→                      onValueChange={(value) => setEditData(prev => ({ ...prev, marital_status: value }))}
   554→                    >
   555→                      <SelectTrigger>
   556→                        <SelectValue placeholder="Select marital status" />
   557→                      </SelectTrigger>
   558→                      <SelectContent>
   559→                        {MARITAL_STATUS_OPTIONS.map((option) => (
   560→                          <SelectItem key={option.value} value={option.value}>
   561→                            {option.label}
   562→                          </SelectItem>
   563→                        ))}
   564→                      </SelectContent>
   565→                    </Select>
   566→                  </div>
   567→                </div>
   568→              ) : (
   569→                <div className="space-y-3">
   570→                  <InfoRow label="Email" value={employee.email} />
   571→                  <InfoRow label="UAN Number" value={employee.uan_number} />
   572→                  <InfoRow label="ESIC Number" value={employee.esic_number} />
   573→                  <InfoRow 
   574→                    label="Marital Status" 
   575→                    value={employee.marital_status ? employee.marital_status.charAt(0).toUpperCase() + employee.marital_status.slice(1) : null} 
   576→                  />
   577→                </div>
   578→              )}
   579→            </div>
   580→
   581→            {/* Personal Details - Read Only */}
   582→            <div className="form-section">
   583→              <h3 className="font-semibold flex items-center gap-2 mb-4">
   584→                <User className="w-4 h-4 text-muted-foreground" />
   585→                Personal Details
   586→              </h3>
   587→              <div className="space-y-3">
   588→                <InfoRow label="Full Name" value={employee.full_name} />
   589→                <InfoRow label="Date of Birth" value={employee.date_of_birth ? formatDateDDMMYYYY(employee.date_of_birth) : null} />
   590→                <InfoRow label="Gender" value={employee.gender ? employee.gender.charAt(0).toUpperCase() + employee.gender.slice(1) : null} />
   591→                <InfoRow label="Aadhaar Number" value={employee.aadhaar_number} masked />
   592→              </div>
   593→            </div>
   594→
   595→            {/* Address */}
   596→            <div className="form-section">
   597→              <h3 className="font-semibold flex items-center gap-2 mb-4">
   598→                <MapPin className="w-4 h-4 text-muted-foreground" />
   599→                Address
   600→              </h3>
   601→              <div className="space-y-3">
   602→                <InfoRow label="Address" value={employee.address} />
   603→                <InfoRow label="PIN Code" value={employee.pin_code} />
   604→                <InfoRow label="District" value={employee.district} />
   605→                <InfoRow label="State" value={employee.state} />
   606→              </div>
   607→            </div>
   608→
   609→            {/* Bank Details */}
   610→            <div className="form-section">
   611→              <h3 className="font-semibold flex items-center gap-2 mb-4">
   612→                <CreditCard className="w-4 h-4 text-muted-foreground" />
   613→                Bank Details
   614→              </h3>
   615→              <div className="space-y-3">
   616→                <InfoRow label="Bank Name" value={employee.bank_name} />
   617→                <InfoRow label="Account Number" value={employee.account_number} masked />
   618→                <InfoRow label="IFSC Code" value={employee.ifsc_code} />
   619→                <InfoRow label="Account Holder" value={employee.account_holder_name} />
   620→              </div>
   621→            </div>
   622→
   623→            {/* Assignment */}
   624→            <div className="form-section">
   625→              <h3 className="font-semibold flex items-center gap-2 mb-4">
   626→                <Building className="w-4 h-4 text-muted-foreground" />
   627→                Assignment
   628→              </h3>
   629→              <div className="space-y-3">
   630→                <InfoRow label="Client" value={employee.client_name} />
   631→                <InfoRow label="Unit/Location" value={employee.unit_name} />
   632→                {employee.designation && (
   633→                  <InfoRow label="Designation" value={employee.designation} />
   634→                )}
   635→              </div>
   636→            </div>
   637→          </>
   638→        )}
   639→      </main>
   640→
   641→      {/* Image Preview Dialog */}
   642→      <Dialog open={!!selectedImage} onOpenChange={() => setSelectedImage(null)}>
   643→        <DialogContent className="max-w-3xl">
   644→          <DialogHeader>
   645→            <DialogTitle>Document Preview</DialogTitle>
   646→          </DialogHeader>
   647→          {selectedImage && (
   648→            <div className="relative">
   649→              <img 
   650→                src={getFileUrl(selectedImage) || undefined} 
   651→                alt="Document"
   652→                className="w-full h-auto max-h-[70vh] object-contain rounded-lg"
   653→              />
   654→              <Button
   655→                variant="outline"
   656→                size="sm"
   657→                className="absolute top-2 right-2"
   658→                onClick={() => {
   659→                  const url = getFileUrl(selectedImage);
   660→                  if (url) window.open(url, '_blank');
   661→                }}
   662→              >
   663→                <ExternalLink className="w-4 h-4 mr-1" />
   664→                Open Full
   665→              </Button>
   666→            </div>
   667→          )}
   668→        </DialogContent>
   669→      </Dialog>
   670→    </div>
   671→  );
   672→}
   673→
   674→function InfoRow({ label, value, masked }: { label: string; value: string | null; masked?: boolean }) {
   675→  const displayValue = value 
   676→    ? (masked ? value.slice(0, 4) + ' **** ' + value.slice(-4) : value)
   677→    : null;
   678→
   679→  return (
   680→    <div className="flex justify-between items-center py-2 border-b border-border/50 last:border-0">
   681→      <span className="text-sm text-muted-foreground">{label}</span>
   682→      {displayValue ? (
   683→        <span className="text-sm font-medium">{displayValue}</span>
   684→      ) : (
   685→        <span className="text-xs text-orange-500 flex items-center gap-1">
   686→          <AlertCircle className="w-3 h-3" />
   687→          Not provided
   688→        </span>
   689→      )}
   690→    </div>
   691→  );
   692→}
