   707→<!-- ── Page-specific JS ────────────────────────────────────────────────── -->
   708→<script>
   709→(function() {
   710→    // Select all checkbox
   711→    const selectAll = document.getElementById('selectAll');
   712→    if (selectAll) {
   713→        selectAll.addEventListener('change', function() {
   714→            document.querySelectorAll('.row-checkbox').forEach(cb => cb.checked = this.checked);
   715→            updateSelectedCount();
   716→        });
   717→    }
   718→
   719→    // Individual checkboxes
   720→    document.querySelectorAll('.row-checkbox').forEach(cb => {
   721→        cb.addEventListener('change', updateSelectedCount);
   722→    });
   723→
   724→    function updateSelectedCount() {
   725→        const count = document.querySelectorAll('.row-checkbox:checked').length;
   726→        const el = document.getElementById('selectedCount');
   727→        if (el) el.textContent = count > 0 ? count + ' selected' : '';
   728→    }
   729→
   730→    // Approve single request
   731→    window.approveRequest = function(id) {
   732→        if (!confirm('Are you sure you want to approve this change request? The employee record will be updated.')) return;
   733→
   734→        const form = document.createElement('form');
   735→        form.method = 'POST';
   736→        form.action = '?page=employee/change-requests<?= $fStatus !== 'all' ? '&status=' . $fStatus : '' ?>';
   737→
   738→        form.appendChild(createInput('action', 'approve'));
   739→        form.appendChild(createInput('id', id));
   740→        form.appendChild(createInput('csrf_token', '<?= htmlspecialchars(getCSRFToken()) ?>'));
   741→
   742→        document.body.appendChild(form);
   743→        form.submit();
   744→    };
   745→
   746→    // Show reject modal
   747→    window.showRejectModal = function(id) {
   748→        document.getElementById('rejectId').value = id;
   749→        document.getElementById('rejection_reason').value = '';
   750→        new bootstrap.Modal(document.getElementById('rejectModal')).show();
   751→    };
   752→
   753→    // Bulk approve
   754→    window.bulkApprove = function() {
   755→        const checked = document.querySelectorAll('.row-checkbox:checked');
   756→        if (checked.length === 0) {
   757→            alert('Please select at least one request.');
   758→            return;
   759→        }
   760→        if (!confirm('Approve ' + checked.length + ' selected change request(s)? Employee records will be updated.')) return;
   761→        document.getElementById('bulkForm').submit();
   762→    };
   763→
   764→    function createInput(name, value) {
   765→        const input = document.createElement('input');
   766→        input.type = 'hidden';
   767→        input.name = name;
   768→        input.value = value;
   769→        return input;
   770→    }
   771→})();
   772→</script>
