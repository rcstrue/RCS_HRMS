   489→function toggleChecks(className, checkAll) {
   490→    document.querySelectorAll('.' + className).forEach(function(cb) {
   491→        if (!cb.disabled) cb.checked = checkAll;
   492→    });
   493→    updateCounts();

   496→function toggleCityFromCheckbox(cityCheckbox) {
   497→    var city = cityCheckbox.getAttribute('data-city');
   498→    var checkAll = cityCheckbox.checked;
   499→    document.querySelectorAll('.unitCheck[data-city="' + city + '"]').forEach(function(cb) {
   500→        cb.checked = checkAll;
   501→    });
   502→    updateCounts();
   503→}
   504→
   505→function updateCounts() {
   506→    var allChecks = document.querySelectorAll('.unitCheck');
   507→    var selectedTotal = 0;
   508→    var cityCounts = {};
   509→    
   510→    allChecks.forEach(function(cb) {
   511→        var city = cb.getAttribute('data-city');
   512→        if (!cityCounts[city]) cityCounts[city] = {total: 0, checked: 0};
   513→        cityCounts[city].total++;
   514→        if (cb.checked) {
   515→            selectedTotal++;
   516→            cityCounts[city].checked++;
   517→        }
   518→    });
   519→    
   520→    var countEl = document.getElementById('selectedCount');
   521→    if (countEl) countEl.textContent = selectedTotal;
   522→    
   523→    // Update city checkboxes and counts
   524→    document.querySelectorAll('.cityCheck').forEach(function(cityCb) {
   525→        var city = cityCb.getAttribute('data-city');
   526→        if (cityCounts[city]) {
   527→            var info = cityCounts[city];
   528→            cityCb.checked = (info.checked === info.total);
   529→            var label = cityCb.nextElementSibling;
   530→            if (label) {
   531→                var smallEl = label.querySelector('small');
   532→                if (smallEl) smallEl.textContent = info.checked + '/' + info.total;
   533→            }
   534→        }
   535→    });
   536→}
