   111→        }

   125→        }

   142→        }

   146→        $otherRecoveries = 0;

   150→        $totalDeductions = $noticeRecovery + $pendingAdvance + $otherRecoveries;
   151→        $netPayable = $totalEarnings - $totalDeductions;
   152→        
   153→        // PF Withdrawal Info

   159→        ];

   163→            'employee_id' => $emp['id'],
   164→            'last_working_day' => $lastWorkingDay,
   165→            'leaving_reason' => $leavingReason,
   166→            'service_years' => round($serviceYears, 2),
   167→            'salary_days' => $daysWorked,
   168→            'salary_amount' => $salaryPayable,
   169→            'leave_encashment_days' => $earnedLeaveBalance,
   170→            'leave_encashment_amount' => $leaveEncashment,
   171→            'gratuity_years' => floor($serviceYears),
   172→            'gratuity_amount' => $gratuity,
   173→            'bonus_amount' => $bonusPayable,
   174→            'notice_shortfall' => $noticeShortfall,
   175→            'notice_recovery' => $noticeRecovery,
   176→            'advance_recovery' => $pendingAdvance,
   177→            'total_earnings' => $totalEarnings,
   178→            'total_deductions' => $totalDeductions,
   179→            'net_payable' => $netPayable,
   180→            'status' => 'pending',
   181→            'created_by' => $_SESSION['user_id'],
   182→            'created_at' => date('Y-m-d H:i:s')
   183→        ];
