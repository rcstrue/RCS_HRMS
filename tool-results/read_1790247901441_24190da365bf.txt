     4→/**
     5→ * RCS HRMS Pro — Employee Change Requests Approval Page
     6→ *
     7→ * Allows HR/Admin to view, approve, or reject change requests
     8→ * submitted by employees via the ESS mobile app.
     9→ *
    10→ * Approve: updates the employee record with the new value.
    11→ * Reject: stores rejection reason; employee sees it in ESS.
    12→ * Both actions send email + WhatsApp notification to the employee.
    13→ */
    14→
    15→$pageTitle = 'Change Request Approvals';

    17→// ─── In-App Notification Helper (INSERT into ess_notifications) ──────────
    18→function sendInAppNotification(int $employeeId, string $title, string $message, string $type = 'info', string $link = '') {
    19→    global $db;
    20→    try {
    21→        // Use the same format as ESS notifications.php: employee_id, title, message, type, link, created_at
    22→        $linkVal = $link ?: '';
    23→        $sql = "INSERT INTO ess_notifications (employee_id, title, message, type, link, created_at)
    24→                VALUES (:employee_id, :title, :message, :type, :link, NOW())";
    25→        $db->exec($sql, [
    26→            'employee_id' => (string)$employeeId,
    27→            'title'       => $title,
    28→            'message'     => $message,
    29→            'type'        => $type,
    30→            'link'        => $linkVal,
    31→        ]);
    32→    } catch (Exception $e) {
    33→        error_log('[change-request in-app notif] ' . $e->getMessage());
    34→    }
    35→}
    36→
    37→// ─── POST Actions ────────────────────────────────────────────────────────────
    38→
    39→if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
    40→    // CSRF check
    41→    $csrfToken = $_POST['csrf_token'] ?? '';
    42→    if (!validateCSRFToken($csrfToken)) {
    43→        setFlash('error', 'Invalid request. Please refresh the page and try again.');
    44→        redirect($_SERVER['REQUEST_URI'] ?? 'index.php?page=employee/change-requests');
    45→    }
    46→    $action = $_POST['action'] ?? '';
    47→
    48→    // ── Whitelist of fields that can be updated on approval ──
    49→    $APPROVAL_FIELDS = [
    50→        'full_name', 'father_name', 'date_of_birth', 'gender',
    51→        'designation', 'department', 'profile_pic_url',
    52→        // Bank & statutory details — manager/employee change requests for the
    53→        // ESS Manager Edit flow (see RCS_HRMS_Manager_Edit_Button.md). Sensitive
    54→        // columns on employees; applied to the employee record on approval.
    55→        'uan_number', 'esic_number', 'aadhaar_number',
    56→        'bank_name', 'account_holder_name', 'account_number', 'ifsc_code',
    57→    ];
    58→
    59→    // ── 1. Approve ─────────────────────────────────────────────
    60→    if ($action === 'approve') {
    61→        $id = (int) ($_POST['id'] ?? 0);
    62→        if ($id > 0) {
    63→            // Fetch request WITHOUT JOIN, then get employee separately
    64→            $request = $db->fetch(
    65→                "SELECT * FROM employee_change_requests
    66→                 WHERE id = :id AND status = 'pending' LIMIT 1",
    67→                ['id' => $id]
    68→            );
    69→
    70→            if ($request) {
    71→                // Fetch employee info separately
    72→                $empInfo = $db->fetch(
    73→                    "SELECT full_name, email, mobile_number, employee_code, designation
    74→                     FROM employees WHERE id = :eid LIMIT 1",
    75→                    ['eid' => (int)$request['employee_id']]
    76→                );
    77→                if ($empInfo) {
    78→                    $request['emp_name']      = $empInfo['full_name'];
    79→                    $request['email']         = $empInfo['email'];
    80→                    $request['mobile_number'] = $empInfo['mobile_number'];
    81→                    $request['employee_code'] = $empInfo['employee_code'];
    82→                    $request['designation']   = $empInfo['designation'];
    83→                } else {
    84→                    $request['emp_name'] = 'Employee #' . $request['employee_id'];
    85→                }
    86→                $field = $request['field_name'];
    87→                $newValue = $request['new_value'];
    88→
    89→                // Apply the change to employee record if it's a whitelisted field
    90→                if (in_array($field, $APPROVAL_FIELDS, true)) {
    91→                    try {
    92→                        $db->update(
    93→                            'employees',
    94→                            [$field => $newValue],
    95→                            'id = :id',
    96→                            ['id' => $request['employee_id']]
    97→                        );
    98→                    } catch (Exception $e) {
    99→                        setFlash('error', 'Failed to update employee record: ' . $e->getMessage());
   100→                        redirect('index.php?page=employee/change-requests');
   101→                    }
   102→                }
   103→
   104→                // Mark request as approved
   105→                $db->update(
   106→                    'employee_change_requests',
   107→                    [
   108→                        'status'      => 'approved',
   109→                        'reviewed_at' => date('Y-m-d H:i:s'),
   110→                        'reviewed_by' => $_SESSION['user_id'] ?? null,
   111→                    ],
   112→                    'id = :id',
   113→                    ['id' => $id]
   114→                );
   115→
   116→                // ── In-App Notification to employee ──
   117→                $fieldLabel = ucfirst(str_replace('_', ' ', $field));
   118→                sendInAppNotification(
   119→                    (int)$request['employee_id'],
   120→                    "Change Request Approved — {$fieldLabel}",
   121→                    "Your change request for {$fieldLabel} has been APPROVED. New value: {$newValue}",
   122→                    'success',
   123→                    '/profile/change-requests'
   124→                );
   125→
   126→                // Audit log
   127→                if (function_exists('logActivity')) {
   128→                    logActivity('approve_change_request', 'employee', $request['employee_id'],
   129→                        "Approved change request #{$id}: {$field} = " . substr($newValue, 0, 50));
   130→                }
   131→
   132→                setFlash('success', 'Change request approved. Employee has been notified.');
   133→            } else {
   134→                setFlash('error', 'Request not found or already processed.');
   135→            }
   136→        }
   137→        redirect('index.php?page=employee/change-requests');
   138→    }
   139→
   140→    // ── 2. Reject ──────────────────────────────────────────────
   141→    if ($action === 'reject') {
   142→        $id     = (int) ($_POST['id'] ?? 0);
   143→        $reason = trim($_POST['rejection_reason'] ?? '');
   144→
   145→        if ($id > 0 && $reason !== '') {
   146→            // Fetch request WITHOUT JOIN
   147→            $request = $db->fetch(
   148→                "SELECT * FROM employee_change_requests
   149→                 WHERE id = :id AND status = 'pending' LIMIT 1",
   150→                ['id' => $id]
   151→            );
   152→
   153→            if ($request) {
   154→                // Fetch employee info separately
   155→                $empInfo = $db->fetch(
   156→                    "SELECT full_name, email, mobile_number, employee_code, designation
   157→                     FROM employees WHERE id = :eid LIMIT 1",
   158→                    ['eid' => (int)$request['employee_id']]
   159→                );
   160→                if ($empInfo) {
   161→                    $request['emp_name']      = $empInfo['full_name'];
   162→                    $request['email']         = $empInfo['email'];
   163→                    $request['mobile_number'] = $empInfo['mobile_number'];
   164→                    $request['employee_code'] = $empInfo['employee_code'];
   165→                    $request['designation']   = $empInfo['designation'];
   166→                } else {
   167→                    $request['emp_name'] = 'Employee #' . $request['employee_id'];
   168→                }
   169→                $field = $request['field_name'];
   170→
   171→                // Mark request as rejected
   172→                $db->update(
   173→                    'employee_change_requests',
   174→                    [
   175→                        'status'           => 'rejected',
   176→                        'reviewed_at'      => date('Y-m-d H:i:s'),
   177→                        'reviewed_by'      => $_SESSION['user_id'] ?? null,
   178→                        'rejection_reason' => $reason,
   179→                    ],
   180→                    'id = :id',
   181→                    ['id' => $id]
   182→                );
   183→
   184→                // ── In-App Notification to employee ──
   185→                $fieldLabel = ucfirst(str_replace('_', ' ', $field));
   186→                sendInAppNotification(
   187→                    (int)$request['employee_id'],
   188→                    "Change Request Rejected — {$fieldLabel}",
   189→                    "Your change request for {$fieldLabel} has been REJECTED. Reason: {$reason}",
   190→                    'danger',
   191→                    '/profile/change-requests'
   192→                );
   193→
   194→                if (function_exists('logActivity')) {
   195→                    logActivity('reject_change_request', 'employee', $request['employee_id'],
   196→                        "Rejected change request #{$id}: {$field} — " . substr($reason, 0, 50));
   197→                }
   198→
   199→                setFlash('success', 'Change request rejected. Employee has been notified.');
   200→            } else {
   201→                setFlash('error', 'Request not found or already processed.');
   202→            }
   203→        } elseif ($id > 0 && $reason === '') {
   204→            setFlash('danger', 'Rejection reason is required.');
   205→        }
   206→        redirect('index.php?page=employee/change-requests');
   207→    }
   208→
   209→    // ── 3. Bulk approve ───────────────────────────────────────
   210→    if ($action === 'bulk_approve') {
   211→        $ids = $_POST['selected_ids'] ?? [];
   212→        if (!empty($ids) && is_array($ids)) {
   213→            $count = 0;
   214→            $errors = 0;
   215→
   216→            foreach ($ids as $id) {
   217→                $id = (int) $id;
   218→                if ($id <= 0) continue;
   219→
   220→                $request = $db->fetch(
   221→                    "SELECT * FROM employee_change_requests
   222→                     WHERE id = :id AND status = 'pending'
   223→                     LIMIT 1",
   224→                    ['id' => $id]
   225→                );
   226→
   227→                if (!$request) continue;
   228→
   229→                $field = $request['field_name'];
   230→                $newValue = $request['new_value'];
   231→
   232→                if (in_array($field, $APPROVAL_FIELDS, true)) {
   233→                    try {
   234→                        $db->update('employees', [$field => $newValue], 'id = :id', ['id' => $request['employee_id']]);
   235→                    } catch (Exception $e) {
   236→                        $errors++;
   237→                        continue;
   238→                    }
   239→                }
   240→
   241→                $db->update(
   242→                    'employee_change_requests',
   243→                    [
   244→                        'status'      => 'approved',
   245→                        'reviewed_at' => date('Y-m-d H:i:s'),
   246→                        'reviewed_by' => $_SESSION['user_id'] ?? null,
   247→                    ],
   248→                    'id = :id',
   249→                    ['id' => $id]
   250→                );
   251→                $count++;
   252→            }
   253→
   254→            $msg = "{$count} change request(s) approved.";
   255→            if ($errors > 0) $msg .= " ({$errors} error(s))";
   256→            setFlash('success', $msg);
   257→        } else {
   258→            setFlash('warning', 'No requests selected.');
   259→        }
   260→        redirect('index.php?page=employee/change-requests');
   261→    }
   262→}

   264→// ─── GET Filters ────────────────────────────────────────────────────────────
   265→
   266→// Raw GET values — do NOT use sanitize() (htmlspecialchars breaks PDO binding)
   267→$fStatusRaw = trim($_GET['status'] ?? 'all');
   268→$fSearchRaw = trim($_GET['search'] ?? '');
   269→
   270→// Validate status against whitelist
   271→$fStatus = in_array($fStatusRaw, ['all', 'pending', 'approved', 'rejected'], true) ? $fStatusRaw : 'all';
   272→$fSearch = substr($fSearchRaw, 0, 100);
   273→
   274→// ─── Summary counts (independent query) ────────────────────────────────────
   275→
   276→$pendingCount  = 0;
   277→$approvedCount = 0;
   278→$rejectedCount = 0;
   279→try {
   280→    $pendingCount  = (int)$db->fetchColumn("SELECT COUNT(*) FROM employee_change_requests WHERE status = 'pending'") ?: 0;
   281→    $approvedCount = (int)$db->fetchColumn("SELECT COUNT(*) FROM employee_change_requests WHERE status = 'approved'") ?: 0;
   282→    $rejectedCount = (int)$db->fetchColumn("SELECT COUNT(*) FROM employee_change_requests WHERE status = 'rejected'") ?: 0;
   283→} catch (Exception $e) {
   284→    error_log('[change-requests] Count queries failed: ' . $e->getMessage());
   285→}
   286→
   287→// ─── Fetch requests — 2-step approach (no JOINs) ──────────────────────────
   288→
   289→$requests = [];
   290→$dbError  = '';
   291→$diagInfo = '';
   292→
   293→try {
   294→    // Step 1: Fetch change requests WITHOUT any JOINs
   295→    if ($fStatus !== 'all') {
   296→        $rawRequests = $db->fetchAll(
   297→            "SELECT * FROM employee_change_requests WHERE status = :crstatus ORDER BY created_at DESC",
   298→            ['crstatus' => $fStatus]
   299→        );
   300→    } else {
   301→        $rawRequests = $db->fetchAll(
   302→            "SELECT * FROM employee_change_requests ORDER BY created_at DESC"
   303→        );
   304→    }
   305→
   306→    $diagInfo .= 'Step1: fetched ' . count($rawRequests) . ' requests. ';
   307→
   308→    // Step 2: Get unique employee IDs and batch-fetch
   309→    $empIds = [];
   310→    foreach ($rawRequests as $r) {
   311→        $eid = (int)$r['employee_id'];
   312→        if ($eid > 0 && !in_array($eid, $empIds)) {
   313→            $empIds[] = $eid;
   314→        }
   315→    }
   316→    $empMap = [];
   317→
   318→    if (!empty($empIds)) {
   319→        $idList = implode(',', array_map('intval', $empIds));
   320→        try {
   321→            $empRows = $db->fetchAll(
   322→                "SELECT id, full_name, employee_code, mobile_number, email, designation
   323→                 FROM employees WHERE id IN ({$idList})"
   324→            );
   325→            foreach ($empRows as $emp) {
   326→                $empMap[$emp['id']] = $emp;
   327→            }
   328→        } catch (Exception $e) {
   329→            $diagInfo .= 'Step2 ERROR: ' . $e->getMessage();
   330→        }
   331→    }
   332→
   333→    $diagInfo .= 'Step2: found ' . count($empMap) . ' employees. ';
   334→
   335→    // Step 3: Merge employee data into requests
   336→    foreach ($rawRequests as $r) {
   337→        $eid = (int)$r['employee_id'];
   338→        $emp = $empMap[$eid] ?? [];
   339→
   340→        // Get reviewer name
   341→        $reviewerName = null;
   342→        $revId = (int)($r['reviewed_by'] ?? 0);
   343→        if ($revId > 0) {
   344→            try {
   345→                $rev = $db->fetch(
   346→                    "SELECT full_name FROM employees WHERE id = :rid LIMIT 1",
   347→                    ['rid' => $revId]
   348→                );
   349→                if ($rev) {
   350→                    $reviewerName = $rev['full_name'];
   351→                }
   352→            } catch (Exception $e) {
   353→                /* ignore */
   354→            }
   355→        }
   356→
   357→        $requests[] = array_merge($r, [
   358→            'emp_name'          => $emp['full_name'] ?? ('Employee #' . $eid),
   359→            'employee_code'     => $emp['employee_code'] ?? '',
   360→            'mobile_number'     => $emp['mobile_number'] ?? '',
   361→            'email'             => $emp['email'] ?? '',
   362→            'emp_designation'   => $emp['designation'] ?? '',
   363→            'reviewed_by_name'  => $reviewerName,
   364→        ]);
   365→    }
   366→
   367→} catch (Exception $e) {
   368→    $dbError = $e->getMessage();
   369→    error_log('[change-requests] Fetch failed: ' . $dbError);
   370→    $diagInfo .= 'FATAL: ' . $dbError;
   371→}
   372→
   373→// ─── Apply search filter (post-fetch) ──
   374→if ($fSearch !== '' && !empty($requests)) {
   375→    $searchLower = strtolower($fSearch);
   376→    $filtered = [];
   377→    foreach ($requests as $r) {
   378→        if (strpos(strtolower($r['emp_name'] ?? ''), $searchLower) !== false
   379→            || strpos(strtolower($r['employee_code'] ?? ''), $searchLower) !== false
   380→            || strpos(strtolower($r['field_name'] ?? ''), $searchLower) !== false) {
   381→            $filtered[] = $r;
   382→        }
   383→    }
   384→    $requests = $filtered;
   385→}

   390→    'full_name'       => 'Full Name',
   391→    'father_name'     => "Father's Name",
   392→    'date_of_birth'   => 'Date of Birth',
   393→    'gender'          => 'Gender',
   394→    'designation'     => 'Designation',
   395→    'department'      => 'Department',
   396→    'profile_pic_url' => 'Profile Photo',

   398→    'esic_number'         => 'ESIC Number',

   663→<script>
   664→(function() {
   665→    // Select all checkbox
   666→    const selectAll = document.getElementById('selectAll');
   667→    if (selectAll) {
   668→        selectAll.addEventListener('change', function() {
   669→            document.querySelectorAll('.row-checkbox').forEach(function(cb) { cb.checked = this.checked; }.bind(this));
   670→            updateSelectedCount();
   671→        });
   672→    }
   673→
   674→    // Individual checkboxes
   675→    document.querySelectorAll('.row-checkbox').forEach(function(cb) {
   676→        cb.addEventListener('change', updateSelectedCount);
   677→    });
   678→
   679→    function updateSelectedCount() {
   680→        const count = document.querySelectorAll('.row-checkbox:checked').length;
   681→        const el = document.getElementById('selectedCount');
   682→        if (el) el.textContent = count > 0 ? count + ' selected' : '';
   683→    }
   684→
   685→    // Approve single request
   686→    window.approveRequest = function(id) {
   687→        if (!confirm('Are you sure you want to approve this change request? The employee record will be updated.')) return;
   688→
   689→        const form = document.createElement('form');
   690→        form.method = 'POST';
   691→        form.action = '?page=employee/change-requests<?= $fStatus !== 'all' ? '&status=' . $fStatus : '' ?>';
   692→
   693→        form.appendChild(createInput('action', 'approve'));
   694→        form.appendChild(createInput('id', id));
   695→        form.appendChild(createInput('csrf_token', '<?= htmlspecialchars($csrfToken) ?>'));
   696→
   697→        document.body.appendChild(form);
   698→        form.submit();
   699→    };
   700→
   701→    // Show reject modal
   702→    window.showRejectModal = function(id) {
   703→        document.getElementById('rejectId').value = id;
   704→        document.getElementById('rejection_reason').value = '';
   705→        new bootstrap.Modal(document.getElementById('rejectModal')).show();
   706→    };
   707→
   708→    // Bulk approve
   709→    window.bulkApprove = function() {
   710→        const checked = document.querySelectorAll('.row-checkbox:checked');
   711→        if (checked.length === 0) {
   712→            alert('Please select at least one request.');
   713→            return;
   714→        }
   715→        if (!confirm('Approve ' + checked.length + ' selected change request(s)? Employee records will be updated.')) return;
   716→        document.getElementById('bulkForm').submit();
   717→    };
   718→
   719→    function createInput(name, value) {
   720→        const input = document.createElement('input');
   721→        input.type = 'hidden';
   722→        input.name = name;
   723→        input.value = value;
   724→        return input;
   725→    }
   726→})();
