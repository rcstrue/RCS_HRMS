    74→    74→ * Deduct loan EMI for all employees in a payroll period + unit.
    75→    75→ * Call this AFTER processPayroll() to add loan deductions.
    76→    76→ */
    77→    77→function deductLoansForPeriodUnit($db, $periodId, $unitId) {
    78→    78→    // Get month/year from payroll_periods
    79→    79→    $period = $db->fetch("SELECT month, year FROM payroll_periods WHERE id = ?", [$periodId]);
    80→    80→    if (!$period) return 0;
    81→    81→    
    82→    82→    $month = (int)$period['month'];
    83→    83→    $year  = (int)$period['year'];
    84→    84→    $totalDeducted = 0;
    85→    85→    
    86→    86→    // Get all payroll records for this period + unit
    87→    87→    $payrollRows = $db->fetchAll(
    88→    88→        "SELECT p.id as payroll_id, p.employee_id as emp_code, p.loan_emi, p.total_deductions, p.net_pay,
    89→    89→                e.id as employee_db_id
    90→    90→         FROM payroll p
    91→    91→         JOIN employees e ON p.employee_id = e.employee_code
    92→    92→         WHERE p.payroll_period_id = ? AND e.unit_id = ?",
    93→    93→        [$periodId, $unitId]
    94→    94→    );
    95→    95→    
    96→    96→    foreach ($payrollRows as $pRow) {
    97→    97→        $empDbId = (int)$pRow['employee_db_id'];
    98→    98→        $empCode = $pRow['emp_code'];
    99→    99→        $payrollId = (int)$pRow['payroll_id'];
   100→   100→        $currentLoanEmi = (float)$pRow['loan_emi'];
   101→   101→        
   102→   102→        // Get active loans for this employee started on or before this month
   103→   103→        $empLoans = $db->fetchAll(
   104→   104→            "SELECT * FROM employee_loans
   105→   105→             WHERE employee_id = ? AND status = 'Active'
   106→   106→             AND (start_year < ? OR (start_year = ? AND start_month <= ?))",
   107→   107→            [$empDbId, $year, $year, $month]
   108→   108→        );
   109→   109→        
   110→   110→        $empLoanTotal = 0;
   111→   111→        
   112→   112→        foreach ($empLoans as $empLoan) {
   113→   113→            // Check if EMI already deducted for this month
   114→   114→            $alreadyDeducted = $db->fetch(
   115→   115→                "SELECT id FROM loan_emi_log WHERE loan_id = ? AND month = ? AND year = ?",
   116→   116→                [$empLoan['id'], $month, $year]
   117→   117→            );
   118→   118→            
   119→   119→            if ($alreadyDeducted) {
   120→   120→                // Already deducted — add existing EMI amount
   121→   121→                $empLoanTotal += (float)$empLoan['emi_amount'];
   122→   122→                continue;
   123→   123→            }
   124→   124→            
   125→   125→            $emiAmount = (float)$empLoan['emi_amount'];
   126→   126→            $balanceAmount = (float)$empLoan['balance_amount'];
   127→   127→            
   128→   128→            // Calculate interest/principal split
   129→   129→            if ((float)$empLoan['interest_rate'] > 0) {
   130→   130→                $monthlyRate = (float)$empLoan['interest_rate'] / 12 / 100;
   131→   131→                $interestComponent = round($balanceAmount * $monthlyRate, 2);
   132→   132→                $principalComponent = round($emiAmount - $interestComponent, 2);
   133→   133→            } else {
   134→   134→                $interestComponent = 0;
   135→   135→                $principalComponent = $emiAmount;
   136→   136→            }
   137→   137→            
   138→   138→            // Handle last EMI
   139→   139→            if ($emiAmount > $balanceAmount) {
   140→   140→                $emiAmount = $balanceAmount;
   141→   141→                $principalComponent = $emiAmount - $interestComponent;
   142→   142→                if ($principalComponent < 0) {
   143→   143→                    $interestComponent = $emiAmount;
   144→   144→                    $principalComponent = 0;
   145→   145→                }
   146→   146→            }
   147→   147→            
   148→   148→            $newBalance = round($balanceAmount - $emiAmount, 2);
   149→   149→            if ($newBalance < 0) $newBalance = 0;
   150→   150→            
   151→   151→            // Record EMI log
   152→   152→            $db->insert('loan_emi_log', [
   153→   153→                'loan_id' => $empLoan['id'],
   154→   154→                'employee_id' => $empDbId,
   155→   155→                'month' => $month,
   156→   156→                'year' => $year,
   157→   157→                'emi_amount' => $emiAmount,
   158→   158→                'principal_component' => $principalComponent,
   159→   159→                'interest_component' => $interestComponent,
   160→   160→                'balance_after' => $newBalance,
   161→   161→                'deducted_via_payroll' => 1
   162→   162→            ]);
   163→   163→            
   164→   164→            // Update loan balance and status
   165→   165→            $newStatus = ($newBalance <= 0) ? 'Closed' : 'Active';
   166→   166→            $db->update('employee_loans', [
   167→   167→                'balance_amount' => $newBalance,
   168→   168→                'emi_deducted' => (int)$empLoan['emi_deducted'] + 1,
   169→   169→                'status' => $newStatus
   170→   170→            ], 'id = :id', ['id' => $empLoan['id']]);
   171→   171→            
   172→   172→            $empLoanTotal += $emiAmount;
   173→   173→        }
   174→   174→        
   175→   175→        // Update payroll record with total loan EMI for this employee
   176→   176→        if ($empLoanTotal > 0 && abs($empLoanTotal - $currentLoanEmi) > 0.01) {
   177→   177→            $currentTotalDed = (float)$pRow['total_deductions'];
   178→   178→            $currentNetPay   = (float)$pRow['net_pay'];
   179→   179→            
   180→   180→            // Remove old loan EMI from deductions (if any), add new
   181→   181→            $deductionAdjustment = $empLoanTotal - $currentLoanEmi;
   182→   182→            
   183→   183→            $newTotalDed = round($currentTotalDed + $deductionAdjustment, 2);
   184→   184→            $newNetPay   = round($currentNetPay - $deductionAdjustment, 2);
   185→   185→            
   186→   186→            $db->update('payroll', [
   187→   187→                'loan_emi' => round($empLoanTotal, 2),
   188→   188→                'total_deductions' => $newTotalDed,
   189→   189→                'net_pay' => $newNetPay,
   190→   190→                'updated_at' => date('Y-m-d H:i:s')
   191→   191→            ], 'id = :id', ['id' => $payrollId]);
   192→   192→            
   193→   193→            $totalDeducted += $deductionAdjustment;
   194→   194→        } elseif ($empLoanTotal > 0 && $currentLoanEmi == 0) {
   195→   195→            // First time setting loan EMI
   196→   196→            $currentTotalDed = (float)$pRow['total_deductions'];
   197→   197→            $currentNetPay   = (float)$pRow['net_pay'];
   198→   198→            
   199→   199→            $newTotalDed = round($currentTotalDed + $empLoanTotal, 2);
   200→   200→            $newNetPay   = round($currentNetPay - $empLoanTotal, 2);
   201→   201→            
   202→   202→            $db->update('payroll', [
   203→   203→                'loan_emi' => round($empLoanTotal, 2),
   204→   204→                'total_deductions' => $newTotalDed,
   205→   205→                'net_pay' => $newNetPay,
   206→   206→                'updated_at' => date('Y-m-d H:i:s')
   207→   207→            ], 'id = :id', ['id' => $payrollId]);
   208→   208→            
   209→   209→            $totalDeducted += $empLoanTotal;
   210→   210→        }
   211→   211→    }
   212→   212→    
   213→   213→    return $totalDeducted;
   214→   214→}
