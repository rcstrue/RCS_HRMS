    78→// ══════════════════════════════════════════════════════════════
    79→// unwrap helper — PHP wraps in { success, data } envelope
    80→// ══════════════════════════════════════════════════════════════
    81→
    82→function unwrap<T>(result: Promise<{ data: T | null; error: string | null }>): Promise<{ data: T | null; error: string | null }> {
    83→  return result.then((res) => {
    84→    if (res.error) return res;
    85→    const d = res.data as Record<string, unknown> | null;
    86→    if (d && typeof d === 'object' && 'success' in d) {
    87→      if (d.success === false) {
    88→        const msg = (d.message as string) || (d.error as string) || 'Request failed';
    89→        return { data: null, error: msg };
    90→      }
    91→      if ('data' in d && d.data != null) {
    92→        return { data: d.data as T, error: null };
    93→      }
    94→      if (d.success === true) {
    95→        return { data: null, error: null };
    96→      }
    97→    }
    98→    return res;
    99→  });
   100→}

   174→// ══════════════════════════════════════════════════════════════
   175→
   176→function ComposeTab({ onSent }: { onSent: () => void }) {
   177→  // ── Target type ──
   178→  const [targetType, setTargetType] = useState<TargetType>('all');
   179→
   180→  // ── Filters data ──
   181→  const [filters, setFilters] = useState<FilterData | null>(null);
   182→  const [filtersLoading, setFiltersLoading] = useState(true);
   183→
   184→  // ── Multi-select selections ──
   185→  const [selectedUnits, setSelectedUnits] = useState<number[]>([]);
   186→  const [selectedClients, setSelectedClients] = useState<number[]>([]);
   187→  const [selectedCities, setSelectedCities] = useState<string[]>([]);
   188→  const [selectedStates, setSelectedStates] = useState<string[]>([]);
   189→
   190→  // ── Individual search ──
   191→  const [searchQuery, setSearchQuery] = useState('');
   192→  const [searchResults, setSearchResults] = useState<SearchedEmployee[]>([]);
   193→  const [searchLoading, setSearchLoading] = useState(false);
   194→  const [selectedEmployees, setSelectedEmployees] = useState<SearchedEmployee[]>([]);
   195→  const searchTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   196→
   197→  // ── Form ──
   198→  const [title, setTitle] = useState('');
   199→  const [message, setMessage] = useState('');
   200→  const [sending, setSending] = useState(false);
   201→
   202→  // ── Load filters on mount ──
   203→  useEffect(() => {
   204→    let cancelled = false;
   205→    setFiltersLoading(true);
   206→    unwrap(apiRequest<{ units: FilterData['units']; clients: FilterData['clients']; cities: FilterData['cities']; states: FilterData['states'] }>(
   207→      '/ess/admin-notifications?view=filters'
   208→    )).then(({ data, error }) => {
   209→      if (cancelled) return;
   210→      if (error) {
   211→        toast.error(error);
   212→        setFiltersLoading(false);
   213→        return;
   214→      }
   215→      if (data) {
   216→        setFilters({
   217→          units: Array.isArray(data.units) ? data.units : [],
   218→          clients: Array.isArray(data.clients) ? data.clients : [],
   219→          cities: Array.isArray(data.cities) ? data.cities : [],
   220→          states: Array.isArray(data.states) ? data.states : [],
   221→        });
   222→      }
   223→      setFiltersLoading(false);
   224→    });
   225→    return () => { cancelled = true; };
   226→  }, []);
   227→
   228→  // ── Individual search with debounce ──
   229→  useEffect(() => {
   230→    if (searchTimerRef.current) clearTimeout(searchTimerRef.current);
   231→    if (!searchQuery.trim()) {
   232→      setSearchResults([]);
   233→      return;
   234→    }
   235→    searchTimerRef.current = setTimeout(async () => {
   236→      setSearchLoading(true);
   237→      const { data, error } = await unwrap(apiRequest<SearchedEmployee[]>(
   238→        `/ess/admin-notifications?view=search-employees&q=${encodeURIComponent(searchQuery.trim())}`
   239→      ));
   240→      setSearchLoading(false);
   241→      if (error) { toast.error(error); return; }
   242→      setSearchResults(data ? data.slice(0, 5) : []);
   243→    }, 300);
   244→    return () => { if (searchTimerRef.current) clearTimeout(searchTimerRef.current); };
   245→  }, [searchQuery]);
   246→
   247→  // ── Selection helpers ──
   248→  const toggleUnit = useCallback((id: number) => {
   249→    setSelectedUnits((prev) => prev.includes(id) ? prev.filter((u) => u !== id) : [...prev, id]);
   250→  }, []);
   251→
   252→  const toggleClient = useCallback((id: number) => {
   253→    setSelectedClients((prev) => prev.includes(id) ? prev.filter((c) => c !== id) : [...prev, id]);
   254→  }, []);
   255→
   256→  const toggleCity = useCallback((city: string) => {
   257→    setSelectedCities((prev) => prev.includes(city) ? prev.filter((c) => c !== city) : [...prev, city]);
   258→  }, []);
   259→
   260→  const toggleState = useCallback((state: string) => {
   261→    setSelectedStates((prev) => prev.includes(state) ? prev.filter((s) => s !== state) : [...prev, state]);
   262→  }, []);
   263→
   264→  const addEmployee = useCallback((emp: SearchedEmployee) => {
   265→    if (selectedEmployees.some((e) => e.id === emp.id)) return;
   266→    setSelectedEmployees((prev) => [...prev, emp]);
   267→    setSearchQuery('');
   268→    setSearchResults([]);
   269→  }, [selectedEmployees]);
   270→
   271→  const removeEmployee = useCallback((id: string) => {
   272→    setSelectedEmployees((prev) => prev.filter((e) => e.id !== id));
   273→  }, []);
   274→
   275→  // ── Validation ──
   276→  const isFormValid = (() => {
   277→    if (!title.trim() || !message.trim()) return false;
   278→    switch (targetType) {
   279→      case 'all':
   280→      case 'managers':
   281→        return true;
   282→      case 'unit':
   283→        return selectedUnits.length > 0;
   284→      case 'client':
   285→        return selectedClients.length > 0;
   286→      case 'city':
   287→        return selectedCities.length > 0;
   288→      case 'state':
   289→        return selectedStates.length > 0;
   290→      case 'individual':
   291→        return selectedEmployees.length > 0;
   292→      default:
   293→        return false;
   294→    }
   295→  })();
   296→
   297→  // ── Build target_ids for submission ──
   298→  const getTargetIds = useCallback((): (number | string)[] => {
   299→    switch (targetType) {
   300→      case 'all':
   301→      case 'managers':
   302→        return [];
   303→      case 'unit':
   304→        return selectedUnits;
   305→      case 'client':
   306→        return selectedClients;
   307→      case 'city':
   308→        return selectedCities;
   309→      case 'state':
   310→        return selectedStates;
   311→      case 'individual':
   312→        return selectedEmployees.map((e) => e.id);
   313→      default:
   314→        return [];
   315→    }
   316→  }, [targetType, selectedUnits, selectedClients, selectedCities, selectedStates, selectedEmployees]);
   317→
   318→  // ── Confirm & send ──
   319→  const handleSend = useCallback(async () => {
   320→    if (!isFormValid || sending) return;
   321→
   322→    const targetIds = getTargetIds();
   323→    const targetLabel = TARGET_TYPES.find((t) => t.value === targetType)?.label ?? targetType;
   324→    let recipientDesc = '';
   325→    if (targetType === 'all') recipientDesc = 'all employees';
   326→    else if (targetType === 'managers') recipientDesc = 'all managers';
   327→    else if (targetType === 'individual') recipientDesc = `${targetIds.length} employee(s)`;
   328→    else recipientDesc = `${targetIds.length} ${targetLabel.toLowerCase()}(s)`;
   329→
   330→    if (!window.confirm(`Send this notification to ${recipientDesc}?`)) return;
   331→
   332→    setSending(true);
   333→    try {
   334→      const { data, error } = await unwrap(apiRequest<{ broadcast_id: number; recipient_count: number; message: string }>(
   335→        '/ess/admin-notifications',
   336→        {
   337→          method: 'POST',
   338→          body: JSON.stringify({
   339→            title: title.trim(),
   340→            message: message.trim(),
   341→            target_type: targetType,
   342→            target_ids: targetIds,
   343→          }),
   344→        }
   345→      ));
   346→
   347→      if (error) {
   348→        toast.error(error);
   349→        return;
   350→      }
   351→
   352→      toast.success(`Notification sent to ${data?.recipient_count ?? 0} recipient(s)`);
   353→
   354→      // Reset form
   355→      setTitle('');
   356→      setMessage('');
   357→      setTargetType('all');
   358→      setSelectedUnits([]);
   359→      setSelectedClients([]);
   360→      setSelectedCities([]);
   361→      setSelectedStates([]);
   362→      setSelectedEmployees([]);
   363→      setSearchQuery('');
   364→      setSearchResults([]);
   365→
   366→      // Switch to sent tab
   367→      onSent();
   368→    } catch (err) {
   369→      logger.error('Failed to send notification:', err);
   370→      toast.error('Failed to send notification');
   371→    } finally {
   372→      setSending(false);
   373→    }
   374→  }, [isFormValid, sending, getTargetIds, targetType, title, message, onSent]);
   375→
   376→  // ── Reset target selections when type changes ──
   377→  const handleTargetTypeChange = useCallback((newType: TargetType) => {
   378→    setTargetType(newType);
   379→    setSelectedUnits([]);
   380→    setSelectedClients([]);
   381→    setSelectedCities([]);
   382→    setSelectedStates([]);
   383→    setSelectedEmployees([]);
   384→    setSearchQuery('');
   385→    setSearchResults([]);
   386→  }, []);
   387→
   388→  return (
   389→    <div className="flex flex-col gap-4">
   390→      {/* Purple gradient header */}
   391→      <div className="bg-gradient-to-r from-violet-600 to-purple-500 rounded-2xl px-4 py-3 text-white flex items-center gap-2 shadow-lg">
   392→        <Send className="w-5 h-5 text-white/80" />
   393→        <h1 className="text-[15px] font-bold">Send Notification</h1>
   394→      </div>
   395→
   396→      {/* Target Type Selector — horizontal scroll */}
   397→      <div className="flex gap-2 overflow-x-auto pb-1 -mx-1 px-1 scrollbar-none">
   398→        {TARGET_TYPES.map((t) => {
   399→          const Icon = t.icon;
   400→          const isActive = targetType === t.value;
   401→          return (
   402→            <button
   403→              key={t.value}
   404→              onClick={() => handleTargetTypeChange(t.value)}
   405→              className={cn(
   406→                'flex items-center gap-1.5 px-3 py-1.5 rounded-full text-[12px] font-semibold whitespace-nowrap shrink-0 transition-all border',
   407→                isActive
   408→                  ? 'bg-violet-600 text-white border-violet-600 shadow-sm'
   409→                  : 'bg-white text-muted-foreground border-gray-200 hover:border-violet-300 hover:text-violet-700'
   410→              )}
   411→            >
   412→              <Icon className="w-3.5 h-3.5" />
   413→              {t.label}
   414→            </button>
   415→          );
   416→        })}
   417→      </div>
   418→
   419→      {/* ── Target-specific content ── */}
   420→
   421→      {/* All employees info */}
   422→      {targetType === 'all' && (
   423→        <div className="flex items-start gap-2.5 bg-emerald-50 border border-emerald-200 rounded-xl px-3.5 py-3">
   424→          <Users className="w-4 h-4 text-emerald-600 shrink-0 mt-0.5" />
   425→          <div className="min-w-0">
   426→            <p className="text-[12px] font-semibold text-emerald-800">Broadcast to all employees</p>
   427→            <p className="text-[11px] text-emerald-600 mt-0.5">
   428→              This notification will be sent to all active employees across the organization.
   429→            </p>
   430→          </div>
   431→        </div>
   432→      )}
   433→
   434→      {/* Managers info */}
   435→      {targetType === 'managers' && (
   436→        <div className="flex items-start gap-2.5 bg-emerald-50 border border-emerald-200 rounded-xl px-3.5 py-3">
   437→          <Users className="w-4 h-4 text-emerald-600 shrink-0 mt-0.5" />
   438→          <div className="min-w-0">
   439→            <p className="text-[12px] font-semibold text-emerald-800">Broadcast to all managers</p>
   440→            <p className="text-[11px] text-emerald-600 mt-0.5">
   441→              This notification will be sent to all active managers across the organization.
   442→            </p>
   443→          </div>
   444→        </div>

   450→          <CardContent className="p-3 space-y-2">

   457→                </Badge>
   458→              )}

   475→                      onClick={() => toggleUnit(unit.id)}

   477→                        'w-full flex items-center gap-2.5 px-3 py-2 rounded-lg text-[13px] transition-colors text-left',
   478→                        isSelected
   479→                          ? 'bg-violet-50 border border-violet-200'
   480→                          : 'hover:bg-gray-50 border border-transparent'

   482→                    >
   483→                      <div className={cn(
   484→                        'w-5 h-5 rounded border-2 flex items-center justify-center shrink-0 transition-colors',

   486→                          ? 'bg-violet-600 border-violet-600'

   502→                {selectedUnits.map((id) => {
   503→                  const unit = filters?.units.find((u) => u.id === id);
   504→                  if (!unit) return null;

   506→                    <span
   507→                      key={id}
   508→                      className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full bg-violet-100 text-violet-700 text-[11px] font-medium border border-violet-200"
   509→                    >
   510→                      {unit.name}
   511→                      <button onClick={() => toggleUnit(id)} className="hover:text-violet-900">
   512→                        <X className="w-3 h-3" />
   513→                      </button>
   514→                    </span>
   515→                  );
   516→                })}
   517→              </div>
   518→            )}
   519→          </CardContent>
   520→        </Card>
   521→      )}
   522→
   523→      {/* By Client — multi-select */}
   524→      {targetType === 'client' && (
   525→        <Card className="border-0 shadow-sm">
   526→          <CardContent className="p-3 space-y-2">
   527→            <p className="text-[12px] font-semibold text-gray-700 flex items-center gap-1.5">
   528→              <Building2 className="w-3.5 h-3.5 text-violet-500" />

   531→                <Badge variant="secondary" className="ml-auto text-[11px] bg-violet-100 text-violet-700 border-violet-200">

   533→                </Badge>

   538→                {Array.from({ length: 4 }).map((_, i) => (
   539→                  <Skeleton key={i} className="h-9 w-full rounded-lg" />
   540→                ))}
   541→              </div>
   542→            ) : !filters || filters.clients.length === 0 ? (
   543→              <p className="text-[12px] text-muted-foreground py-2 text-center">No clients available</p>
   544→            ) : (
   545→              <div className="max-h-48 overflow-y-auto space-y-1">

   987→// ══════════════════════════════════════════════════════════════
   988→
   989→function BroadcastCard({ item }: { item: BroadcastItem }) {
   990→  const readPercent = Math.round(item.read_percent) || 0;
   991→
   992→  const timeAgo = (dateStr: string): string => {
   993→    const now = new Date();
   994→    const date = new Date(dateStr);
   995→    const diffMs = now.getTime() - date.getTime();
   996→    const diffSec = Math.floor(diffMs / 1000);
   997→    const diffMin = Math.floor(diffSec / 60);
   998→    const diffHr = Math.floor(diffMin / 60);
   999→    const diffDay = Math.floor(diffHr / 24);
  1000→
  1001→    if (diffSec < 60) return 'Just now';
  1002→    if (diffMin < 60) return `${diffMin}m ago`;
  1003→    if (diffHr < 24) return `${diffHr}h ago`;
  1004→    if (diffDay < 7) return `${diffDay}d ago`;
  1005→    return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
  1006→  };
  1007→
  1008→  const progressColor = readPercent >= 80
  1009→    ? 'bg-emerald-500'
  1010→    : readPercent >= 50
  1011→      ? 'bg-amber-500'
  1012→      : readPercent > 0
  1013→        ? 'bg-violet-500'
  1014→        : 'bg-gray-300';
  1015→
  1016→  return (
  1017→    <div className="rounded-xl border bg-card p-4 transition-colors hover:bg-accent/30">
  1018→      {/* Title + target badge */}
  1019→      <div className="flex items-start justify-between gap-2 mb-1.5">
  1020→        <h3 className="font-semibold text-[13px] text-gray-900 leading-snug line-clamp-1">
  1021→          {item.title}
  1022→        </h3>
  1023→        <Badge
  1024→          variant="outline"
  1025→          className="shrink-0 text-[10px] bg-violet-50 text-violet-700 border-violet-200"
  1026→        >
  1027→          {item.target_label}
  1028→        </Badge>
  1029→      </div>
  1030→
  1031→      {/* Message (truncated) */}
  1032→      <p className="text-[12px] text-muted-foreground leading-relaxed line-clamp-2 mb-3">
  1033→        {item.message}
  1034→      </p>
  1035→
  1036→      {/* Read progress bar */}
  1037→      <div className="flex items-center gap-2.5 mb-2.5">
  1038→        <div className="flex-1 h-2 rounded-full bg-gray-100 overflow-hidden">
  1039→          <div
  1040→            className={cn('h-full rounded-full transition-all', progressColor)}
  1041→            style={{ width: `${Math.min(readPercent, 100)}%` }}
  1042→          />
  1043→        </div>
  1044→        <span className="text-[11px] font-semibold text-gray-500 tabular-nums shrink-0">
  1045→          {readPercent}%
  1046→        </span>
  1047→      </div>
  1048→
  1049→      {/* Meta: recipients, read count, time */}
  1050→      <div className="flex items-center justify-between text-[11px] text-muted-foreground">
  1051→        <div className="flex items-center gap-3">
  1052→          <span className="flex items-center gap-1">
  1053→            <Users className="w-3 h-3" />
  1054→            {item.total_recipients}
  1055→          </span>
  1056→          <span className="flex items-center gap-1">
  1057→            <Eye className="w-3 h-3" />
  1058→            {item.read_count}
  1059→          </span>
  1060→        </div>
  1061→        <span className="flex items-center gap-1">
  1062→          <Clock className="w-3 h-3" />
  1063→          {timeAgo(item.created_at)}
  1064→        </span>
  1065→      </div>
  1066→    </div>
  1067→  );
  1068→}
