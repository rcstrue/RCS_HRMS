     8→ * IMPORTANT NOTES FOR DEVELOPERS:
     9→ * =================================
    10→ * 1. Always use client_id and unit_id for filtering (NOT client_name/unit_name)
    11→ * 2. Use JOINs to get client/unit names from their respective tables
    12→ * 3. clients table uses 'name' column (not 'client_name')

    83→    // Get payroll periods
    84→    public function getPeriods($status = null) {
    85→        $sql = "SELECT pp.*, 
    86→                       COUNT(p.id) as employee_count,
    87→                       SUM(CASE WHEN p.salary_hold = 1 THEN 1 ELSE 0 END) as hold_count
    88→                FROM payroll_periods pp
    89→                LEFT JOIN payroll p ON pp.id = p.payroll_period_id
    90→                WHERE 1=1";
    91→        $params = [];
    92→
    93→        if ($status) {
    94→            $sql .= " AND pp.status = :status";
    95→            $params['status'] = $status;
    96→        }
    97→
    98→        $sql .= " GROUP BY pp.id ORDER BY pp.year DESC, pp.month DESC";
    99→
   100→        return $this->db->fetchAll($sql, $params);
   101→    }

   156→    // Get payroll for period with filters
   157→    public function getPayrollReport($periodId, $filters = []) {
   158→        $sql = "SELECT p.*,
   159→                       e.employee_code, e.full_name, e.designation, e.department,
   160→                       e.bank_name, e.account_number, e.ifsc_code,
   161→                       c.name as client_name, u.name as unit_name
   162→                FROM payroll p
   163→                JOIN employees e ON p.employee_id = e.employee_code
   164→                LEFT JOIN clients c ON e.client_id = c.id
   165→                LEFT JOIN units u ON e.unit_id = u.id
   166→                WHERE p.payroll_period_id = :period_id";
   167→        $params = ['period_id' => $periodId];
   168→
   169→        // Apply filters
   170→        if (!empty($filters['unit_id'])) {
   171→            $sql .= " AND e.unit_id = :unit_id";
   172→            $params['unit_id'] = $filters['unit_id'];
   173→        }
   174→
   175→        if (!empty($filters['client_id'])) {
   176→            $sql .= " AND e.client_id = :client_id";
   177→            $params['client_id'] = $filters['client_id'];
   178→        }
   179→
   180→        if (!empty($filters['status'])) {
   181→            $sql .= " AND p.status = :pstatus";
   182→            $params['pstatus'] = $filters['status'];
   183→        }
   184→
   185→        if (!empty($filters['salary_hold'])) {
   186→            $sql .= " AND p.salary_hold = :salary_hold";
   187→            $params['salary_hold'] = $filters['salary_hold'];
   188→        }
   189→
   190→        if (!empty($filters['search'])) {
   191→            $sql .= " AND (e.full_name LIKE :search1 OR e.employee_code LIKE :search2)";
   192→            $params['search1'] = '%' . $filters['search'] . '%';
   193→            $params['search2'] = '%' . $filters['search'] . '%';
   194→        }
   195→
   196→        $sql .= " ORDER BY c.name, u.name, e.employee_code";
   197→
   198→        return $this->db->fetchAll($sql, $params);

   227→    /**
   228→     * Process payroll for period with selective filters
   229→     * 
   230→     * @param int $periodId The payroll period ID
   231→     * @param array $filters Optional filters for selective processing
   232→     *                       - client_id: Process only this client's employees
   233→     *                       - unit_id: Process only this unit's employees
   234→     *                       - employee_codes: Array of specific employee codes
   235→     *                       - recalculate_dirty: Only recalculate dirty records
   236→     * @return array Result with success status and processed count
   237→     */
   238→    public function processPayroll($periodId, $filters = []) {
   239→        // Get period info
   240→        $period = $this->db->fetch(
   241→            "SELECT * FROM payroll_periods WHERE id = :id",
   242→            ['id' => $periodId]
   243→        );
   244→
   245→        if (!$period) {
   246→            return ['success' => false, 'message' => 'Payroll period not found.'];
   247→        }
   248→
   249→        // Check if period is frozen
   250→        if ($period['status'] === 'Frozen' || $period['status'] === 'Locked') {
   251→            return ['success' => false, 'message' => 'Payroll period is frozen. Cannot process.'];
   252→        }
   253→
   254→        // Ensure payroll table has extra_days_amount column
   255→        try {
   256→            $checkColumn = $this->db->fetch("SHOW COLUMNS FROM payroll LIKE 'extra_days_amount'");
   257→            if (!$checkColumn) {
   258→                $this->db->exec("ALTER TABLE payroll ADD COLUMN extra_days_amount DECIMAL(10,2) DEFAULT 0.00 AFTER overtime_amount");
   259→            }
   260→        } catch (Exception $e) {
   261→            // Column might already exist or table might not have records
   262→        }
   263→
   264→        // Build employee query with filters
   265→        $sql = "SELECT e.id, e.employee_code, e.full_name, e.date_of_joining,
   266→                       e.date_of_leaving, e.client_id, e.unit_id, e.worker_category,
   267→                       e.bank_name, e.account_number, e.ifsc_code,
   268→                       c.name as client_name, u.name as unit_name,
   269→                       ess.basic_da, ess.hra, ess.leave_encashment, ess.bonus_encashment, ess.washing_allowance,
   270→                       ess.gross_salary, ess.pf_applicable, ess.esi_applicable,
   271→                       ess.pt_applicable, ess.lwf_applicable, ess.overtime_applicable,
   272→                       ess.bonus_applicable, ess.gratuity_applicable,
   273→                       usf.pay_days_type, usf.ot_calculation_type, usf.ot_calculation_on, usf.ot_hours_per_day,
   274→                       u.state as unit_state
   275→                FROM employees e
   276→                INNER JOIN employee_salary_structures ess ON e.id = ess.employee_id
   277→                    AND (ess.effective_to IS NULL OR ess.effective_to >= CURDATE())
   278→                LEFT JOIN clients c ON e.client_id = c.id
   279→                LEFT JOIN units u ON e.unit_id = u.id
   280→                LEFT JOIN unit_salary_formulas usf ON usf.unit_id = e.unit_id
   281→                    AND usf.is_active = 1
   282→                    AND (usf.effective_to IS NULL OR usf.effective_to >= CURDATE())
   283→                WHERE e.status = 'approved'";
   284→
   285→        $params = [];
   286→
   287→        // Apply filters for selective processing
   288→        if (!empty($filters['client_id'])) {
   289→            $sql .= " AND e.client_id = :client_id";
   290→            $params['client_id'] = $filters['client_id'];
   291→        }
   292→
   293→        if (!empty($filters['unit_id'])) {
   294→            $sql .= " AND e.unit_id = :unit_id";
   295→            $params['unit_id'] = $filters['unit_id'];
   296→        }
   297→
   298→        // Employee codes filter - use named params to avoid mixing with positional
   299→        if (!empty($filters['employee_codes']) && is_array($filters['employee_codes'])) {
   300→            $empPlaceholders = [];
   301→            $idx = 1;
   302→            foreach ($filters['employee_codes'] as $code) {
   303→                $key = 'emp_code_' . $idx++;
   304→                $empPlaceholders[] = ':' . $key;
   305→                $params[$key] = $code;
   306→            }
   307→            $sql .= " AND e.employee_code IN (" . implode(',', $empPlaceholders) . ")";
   308→        }
   309→
   310→        // Only recalculate dirty records if specified
   311→        if (!empty($filters['recalculate_dirty'])) {
   312→            $sql .= " AND EXISTS (
   313→                SELECT 1 FROM payroll p 
   314→                WHERE p.employee_id = e.employee_code 
   315→                AND p.payroll_period_id = :period_id_check
   316→                AND p.payroll_dirty = 1
   317→            )";
   318→            $params['period_id_check'] = $periodId;
   319→        }
   320→
   321→        $employees = $this->db->fetchAll($sql, $params);
   322→
   323→        $processed = 0;
   324→        $errors = [];
   325→        $exceptions = [];
   326→        $totalGross = 0;
   327→        $totalNet = 0;
   328→        $totalPF = 0;
   329→        $totalESI = 0;
   330→
   331→        $this->db->beginTransaction();
   332→
   333→        try {
   334→            foreach ($employees as $emp) {
   335→                try {
   336→                    // Check for exceptions
   337→                    $exceptionType = null;
   338→                    
   339→                    // Check if employee joined after period end
   340→                    $periodEndDate = $period['end_date'];
   341→                    if ($emp['date_of_joining'] && strtotime($emp['date_of_joining']) > strtotime($periodEndDate)) {
   342→                        continue;
   343→                    }
   344→
   345→                    // Check for missing salary structure
   346→                    if (empty($emp['basic_da']) && empty($emp['gross_salary'])) {
   347→                        $exceptionType = 'Undefined Salary';
   348→                        $exceptions[] = [
   349→                            'employee_id' => $emp['employee_code'],
   350→                            'employee_name' => $emp['full_name'],
   351→                            'type' => 'Undefined Salary',
   352→                            'message' => 'No salary structure defined'
   353→                        ];
   354→                    }
   355→
   356→                    // Check for missing bank details
   357→                    if (empty($emp['bank_name']) || empty($emp['account_number']) || empty($emp['ifsc_code'])) {
   358→                        if (!$exceptionType) {
   359→                            $exceptionType = 'Missing Bank Details';
   360→                        }
   361→                        $exceptions[] = [
   362→                            'employee_id' => $emp['employee_code'],
   363→                            'employee_name' => $emp['full_name'],
   364→                            'type' => 'Missing Bank Details',
   365→                            'message' => 'Bank account details incomplete'
   366→                        ];
   367→                    }
   368→
   369→                    // Get attendance summary — prefer unit_id match, fallback to any unit (backward compatible)
   370→                    $attendance = $this->db->fetch(
   371→                        "SELECT
   372→                            total_present as present_days,
   373→                            total_extra,
   374→                            overtime_hours,
   375→                            total_wo as weekly_offs,
   376→                            total_paid_days
   377→                        FROM attendance_summary
   378→                        WHERE employee_id = :emp_id
   379→                        AND month = :month
   380→                        AND year = :year
   381→                        ORDER BY unit_id = :unit_id DESC
   382→                        LIMIT 1",
   383→                        ['emp_id' => $emp['id'], 'unit_id' => $emp['unit_id'], 'month' => $period['month'], 'year' => $period['year']]
   384→                    );
   385→
   386→                    // Calculate total days based on unit's pay_days_type setting
   387→                    $payDaysType = $emp['pay_days_type'] ?? 'actual';
   388→                    switch ($payDaysType) {
   389→                        case 'fixed_30':
   390→                            $totalDays = 30;
   391→                            break;
   392→                        case 'previous_month':
   393→                            // For 26-25 cycle: use previous month's days
   394→                            $prevMonth = $period['month'] - 1;
   395→                            $prevYear = $period['year'];
   396→                            if ($prevMonth < 1) { $prevMonth = 12; $prevYear--; }
   397→                            $totalDays = (int)date('t', strtotime("$prevYear-$prevMonth-01"));
   398→                            break;
   399→                        case 'calendar_minus_sundays':
   400→                            // Month days minus all Sundays
   401→                            $calDays = (int)cal_days_in_month(CAL_GREGORIAN, $period['month'], $period['year']);
   402→                            $sundays = 0;
   403→                            for ($d = 1; $d <= $calDays; $d++) {
   404→                                $dow = (int)date('N', strtotime(sprintf('%04d-%02d-%02d', $period['year'], $period['month'], $d)));
   405→                                if ($dow === 7) $sundays++;
   406→                            }
   407→                            $totalDays = $calDays - $sundays;
   408→                            break;
   409→                        default: // 'actual'
   410→                            $totalDays = $period['pay_days'] ?? 30;
   411→                            break;
   412→                    }
   413→                    
   414→                    // Use total_paid_days from attendance_summary if available, otherwise calculate
   415→                    $preCalcPaidDays = floatval($attendance['total_paid_days'] ?? 0);
   416→                    if ($preCalcPaidDays > 0) {
   417→                        $paidDays = $preCalcPaidDays;
   418→                    } else {
   419→                        // Fallback: calculate from individual components
   420→                        $paidDays = floatval($attendance['total_present'] ?? 0);
   421→                        $paidDays += floatval($attendance['weekly_offs'] ?? 0);
   422→                        $paidDays += floatval($attendance['total_extra'] ?? 0);
   423→                    }
   424→                    
   425→                    // Check if attendance record exists (must check is_array first to avoid warnings)
   426→                    // NOTE: SQL aliases total_present → present_days, total_wo → weekly_offs
   427→                    $hasAttendance = is_array($attendance) && array_key_exists('present_days', $attendance) && $attendance['present_days'] !== null;
   428→
   429→                    // Check for missing attendance — SKIP employee, do NOT assume full month
   430→                    if (!$hasAttendance) {
   431→                        $exceptions[] = [
   432→                            'employee_id' => $emp['employee_code'],
   433→                            'employee_name' => $emp['full_name'],
   434→                            'type' => 'Missing Attendance',
   435→                            'message' => 'No attendance data found — employee skipped. Upload attendance first.'
   436→                        ];
   437→                        continue; // Skip this employee entirely
   438→                    }
   439→
   440→                    $unpaidDays = max(0, $totalDays - $paidDays);
   441→                    $overtimeHours = floatval($attendance['overtime_hours'] ?? 0);
   442→
   443→                    // Calculate earnings (pro-rated) - using basic_da as combined column
   444→                    $basicDa = round(($emp['basic_da'] ?? 0) * $paidDays / $totalDays, 2);
   445→                    $hra = round(($emp['hra'] ?? 0) * $paidDays / $totalDays, 2);
   446→                    $leaveEncashment = round(($emp['leave_encashment'] ?? 0) * $paidDays / $totalDays, 2);
   447→                    $bonusEncashment = round(($emp['bonus_encashment'] ?? 0) * $paidDays / $totalDays, 2);
   448→                    $washingAllowance = round(($emp['washing_allowance'] ?? 0) * $paidDays / $totalDays, 2);
   449→
   450→                    $grossEarnings = round($basicDa + $hra + $leaveEncashment + $bonusEncashment + $washingAllowance, 2);
   451→
   452→                    // Extra days are now included in paid_days, so no separate calculation needed
   453→                    $extraDaysAmount = 0;
   454→
   455→                    // Calculate overtime amount based on unit salary formula settings
   456→                    $overtimeAmount = 0;
   457→                    if ($overtimeHours > 0 && ($emp['overtime_applicable'] ?? 0)) {
   458→                        $otCalcType = $emp['ot_calculation_type'] ?? 'single_pay'; // single_pay or double_pay
   459→                        $otCalcOn = $emp['ot_calculation_on'] ?? 'basic_da';      // basic, basic_da, basic_hra, gross
   460→                        $otHoursPerDay = floatval($emp['ot_hours_per_day'] ?? 8);
   461→
   462→                        // Determine OT base amount
   463→                        switch ($otCalcOn) {
   464→                            case 'gross':
   465→                                $otBase = $emp['gross_salary'] ?? 0;
   466→                                break;
   467→                            case 'basic_hra':
   468→                                $otBase = ($emp['basic_da'] ?? 0) + ($emp['hra'] ?? 0);
   469→                                break;
   470→                            case 'basic':
   471→                                // Basic only (without DA) — approximate from basic_da
   472→                                $otBase = ($emp['basic_da'] ?? 0) * 0.5; // rough split
   473→                                break;
   474→                            default: // basic_da
   475→                                $otBase = $emp['basic_da'] ?? 0;
   476→                                break;
   477→                        }
   478→
   479→                        $hourlyRate = $otBase / $totalDays / $otHoursPerDay;
   480→                        $otMultiplier = ($otCalcType === 'double_pay') ? 2 : 1;
   481→                        $overtimeAmount = round($hourlyRate * $overtimeHours * $otMultiplier, 2);
   482→                    }
   483→
   484→                    $grossWithOT = round($grossEarnings + $overtimeAmount, 2);
   485→
   486→                    // Calculate PF
   487→                    $pfEmployee = 0;
   488→                    $pfEmployer = 0;
   489→                    $epsEmployer = 0;
   490→                    $edlisEmployer = 0;
   491→                    $epfAdmin = 0;
   492→
   493→                    if ($emp['pf_applicable'] ?? 0) {
   494→                        $pfBase = min($basicDa, $this->pfRates['wage_ceiling']);
   495→                        $pfEmployee = round($pfBase * $this->pfRates['employee_share'] / 100, 2);
   496→                        $pfEmployer = round($pfBase * $this->pfRates['employer_pf'] / 100, 2);
   497→                        $epsEmployer = round($pfBase * $this->pfRates['employer_eps'] / 100, 2);
   498→                        $edlisEmployer = round($pfBase * $this->pfRates['employer_edlis'] / 100, 2);
   499→                        $epfAdmin = round($pfBase * $this->pfRates['epf_admin'] / 100, 2);
   500→                    }
   501→
   502→                    // Calculate ESI - Check against actual gross salary (not pro-rated)
   503→                    $esiEmployee = 0;
   504→                    $esiEmployer = 0;
   505→                    $actualGrossSalary = floatval($emp['gross_salary'] ?? 0);
   506→
   507→                    if (($emp['esi_applicable'] ?? 0) && $actualGrossSalary <= $this->esiRates['wage_ceiling']) {
   508→                        $esiEmployee = round($grossWithOT * $this->esiRates['employee_share'] / 100, 2);
   509→                        $esiEmployer = round($grossWithOT * $this->esiRates['employer_share'] / 100, 2);
   510→                    }
   511→
   512→                    // Calculate Professional Tax
   513→                    $pt = 0;
   514→                    if ($emp['pt_applicable'] ?? 0) {
   515→                        $pt = $this->calculatePT($grossWithOT, $emp['unit_state'] ?? '');
   516→                    }
   517→
   518→                    // Calculate LWF (Labour Welfare Fund) — state-specific from lwf_rates table
   519→                    $lwfEmployee = 0;
   520→                    $lwfEmployer = 0;
   521→                    if (($emp['lwf_applicable'] ?? 0) && !empty($emp['unit_state'])) {
   522→                        $stateCode = $emp['unit_state'];
   523→                        $lwfRate = $this->db->fetch(
   524→                            "SELECT lr.employee_share, lr.employer_share, lr.contribution_months
   525→                             FROM lwf_rates lr
   526→                             JOIN states s ON lr.state_id = s.id
   527→                             WHERE (s.state_code = :st OR s.state_name = :st)
   528→                             AND lr.is_active = 1
   529→                             AND lr.effective_from <= CURDATE()
   530→                             ORDER BY lr.effective_from DESC LIMIT 1",
   531→                            ['st' => $stateCode]
   532→                        );
   533→                        if (!$lwfRate) {
   534→                            // Fallback: try lwf_state_rates by state_name
   535→                            $lwfRate = $this->db->fetch(
   536→                                "SELECT employee_share, employer_share, contribution_months
   537→                                 FROM lwf_state_rates
   538→                                 WHERE state_name = :st AND is_active = 1
   539→                                 AND effective_from <= CURDATE()
   540→                                 ORDER BY effective_from DESC LIMIT 1",
   541→                                ['st' => $stateCode]
   542→                            );
   543→                        }
   544→                        if ($lwfRate) {
   545→                            // Check if current month is a contribution month
   546→                            $contribMonths = array_filter(array_map('intval', explode(',', $lwfRate['contribution_months'] ?? '')));
   547→                            if (in_array((int)$period['month'], $contribMonths)) {
   548→                                $lwfEmployee = floatval($lwfRate['employee_share']);
   549→                                $lwfEmployer = floatval($lwfRate['employer_share']);
   550→                            }
   551→                        }
   552→                    }
   553→
   554→                    // Get advances for the period (EXCLUDE office_advance to avoid double-counting)
   555→                    $advance = $this->db->fetch(
   556→                        "SELECT COALESCE(SUM(adv1 + adv2 + dress_advance), 0) as total_advance
   557→                        FROM employee_advances
   558→                        WHERE employee_id = :emp_id AND month = :month AND year = :year",
   559→                        ['emp_id' => $emp['id'], 'month' => $period['month'], 'year' => $period['year']]
   560→                    );
   561→                    $salaryAdvance = $advance['total_advance'] ?? 0;
   562→
   563→                    // Get office deduction for the period (separate from advances)
   564→                    $officeDeduction = 0;
   565→                    $officeAdv = $this->db->fetch(
   566→                        "SELECT COALESCE(office_advance, 0) as office_ded
   567→                         FROM employee_advances

   570→                    );
   571→                    $officeDeduction = floatval($officeAdv['office_ded'] ?? 0);
   572→
   573→                    // Get active loan EMI for this employee (Audit Issue #8 — include in total_deductions)
   574→                    $loanEmi = 0;
   575→                    $activeLoans = $this->db->fetchAll(
   576→                        "SELECT el.emi_amount
   577→                         FROM employee_loans el
   578→                         WHERE el.employee_id = :emp_id AND el.status = 'Active'
   579→                         AND el.balance_amount > 0
   580→                         AND (el.start_year < :year OR (el.start_year = :year AND el.start_month <= :month))
   581→                         AND el.id NOT IN (

   583→                             WHERE lem.month = :month AND lem.year = :year
   584→                         )",

   589→                    }
   590→
   591→                    // Total deductions (now includes loan_emi)
   592→                    $totalDeductions = round($pfEmployee + $esiEmployee + $pt + $lwfEmployee + $salaryAdvance + $officeDeduction + $loanEmi, 2);
   593→
   594→                    // Net pay (always round off)
   595→                    $netPay = round($grossWithOT - $totalDeductions); // Round to nearest ₹.00
   596→
   597→                    // Employer contributions
   598→                    $employerContribution = round($pfEmployer + $epsEmployer + $edlisEmployer + $epfAdmin + $esiEmployer + $lwfEmployer, 2);
   599→
   600→                    // Bonus provision (8.33% of basic_da, max ₹7000)
   601→                    $bonusProvision = 0;
   602→                    if ($emp['bonus_applicable'] ?? 0) {
   603→                        $bonusBase = min($basicDa, 7000);
   604→                        $bonusProvision = round($bonusBase * 8.33 / 100, 2);
   605→                    }
   606→
   607→                    // Gratuity provision (4.81% of basic_da)
   608→                    $gratuityProvision = 0;
   609→                    if ($emp['gratuity_applicable'] ?? 0) {
   610→                        $gratuityProvision = round($basicDa * 4.81 / 100, 2);
   611→                    }
   612→
   613→                    // CTC
   614→                    $ctc = round($grossWithOT + $employerContribution + $bonusProvision + $gratuityProvision, 2);
   615→
   616→                    // Minimum wage check (Audit Issue #14)
   617→                    $minWage = $this->db->fetch(
   618→                        "SELECT mw.minimum_wage FROM minimum_wages mw
   619→                         WHERE (mw.state = :state OR mw.state = 'All')
   620→                         AND mw.designation LIKE CONCAT('%', :cat, '%')
   621→                         AND mw.effective_from <= :dt
   622→                         ORDER BY mw.effective_from DESC LIMIT 1",

   624→                    );

   628→                            'employee_name' => $emp['full_name'],
   629→                            'type' => 'Below Minimum Wage',
   630→                            'message' => 'Gross ₹' . number_format($grossEarnings, 2) . ' < min wage ₹' . number_format($minWage['minimum_wage'], 2) . ' for ' . ($emp['worker_category'] ?? 'N/A')
   631→                        ];
   632→                    }
   633→
   634→                    // Insert or update payroll record - matching database schema with basic_da column
   635→                    $this->db->query(
   636→                        "INSERT INTO payroll (
   637→                            payroll_period_id, employee_id, unit_id,

   640→                            overtime_amount, extra_days_amount, gross_earnings,
   641→                            pf_employee, esi_employee, professional_tax, lwf_employee,
   642→                            salary_advance, loan_emi, office_deduction, total_deductions,
   643→                            pf_employer, eps_employer, edlis_employer, epf_admin_charges, esi_employer,
   644→                            bonus_provision, gratuity_provision,
   645→                            total_employer_contribution,

   807→    /**
   808→     * Hold salary for specific employees
   809→     * 
   810→     * @param int $periodId The payroll period ID
   811→     * @param array $employeeCodes Array of employee codes
   812→     * @param string $reason Reason for holding salary
   813→     * @return array Result with success status
   814→     */
   815→    public function holdSalary($periodId, $employeeCodes, $reason = '') {
   816→        // Check if period is frozen
   817→        $period = $this->db->fetch(
   818→            "SELECT status FROM payroll_periods WHERE id = :id",
   819→            ['id' => $periodId]
   820→        );
   821→
   822→        if (!$period || in_array($period['status'], ['Frozen', 'Locked', 'Paid'])) {
   823→            return ['success' => false, 'message' => 'Cannot hold salary for this period.'];
   824→        }
   825→
   826→        $placeholders = implode(',', array_fill(0, count($employeeCodes), '?'));
   827→        $params = array_merge([$periodId], $employeeCodes);
   828→
   829→        $this->db->query(
   830→            "UPDATE payroll SET 
   831→                salary_hold = 1, 
   832→                hold_reason = ?, 
   833→                hold_date = CURDATE(),
   834→                status = 'Hold'
   835→            WHERE payroll_period_id = ? AND employee_id IN ($placeholders)",
   836→            array_merge([$reason], $params)
   837→        );
   838→
   839→        // Update hold count in period
   840→        $holdCount = $this->db->fetch(
   841→            "SELECT COUNT(*) as count FROM payroll WHERE payroll_period_id = ? AND salary_hold = 1",
   842→            [$periodId]
   843→        );
   844→
   845→        $this->db->update('payroll_periods', [
   846→            'hold_count' => $holdCount['count']
   847→        ], SQL_WHERE_ID, ['id' => $periodId]);
   848→
   849→        return ['success' => true, 'message' => 'Salary hold applied successfully.'];
   850→    }

   966→     * 
   967→     * @param int $periodId The payroll period ID
   968→     * @param string $paymentDate The payment date
   969→     * @param array $excludeHeld Whether to exclude held salaries
   970→     * @return array Result with success status
   971→     */
   972→    public function markAsPaid($periodId, $paymentDate = null, $excludeHeld = true) {
   973→        if (!$paymentDate) {
   974→            $paymentDate = date('Y-m-d');
   975→        }
   976→
   977→        // Check if period is frozen
   978→        $period = $this->db->fetch(
   979→            "SELECT status FROM payroll_periods WHERE id = :id",
   980→            ['id' => $periodId]
   981→        );
   982→
   983→        if (!$period) {
   984→            return ['success' => false, 'message' => 'Payroll period not found.'];
   985→        }
   986→
   987→        if ($period['status'] === 'Frozen' || $period['status'] === 'Locked') {
   988→            return ['success' => false, 'message' => 'Cannot modify frozen payroll period.'];
   989→        }
   990→
   991→        // Update period
   992→        $this->db->update('payroll_periods', [
   993→            'status' => 'Paid',
   994→            'payment_date' => $paymentDate
   995→        ], SQL_WHERE_ID, ['id' => $periodId]);

   997→        // Update payroll records (exclude held if requested)
   998→        $whereClause = $excludeHeld 
   999→            ? "payroll_period_id = :period_id AND salary_hold = 0"
  1000→            : "payroll_period_id = :period_id";
  1001→
  1002→        $this->db->query(
  1003→            "UPDATE payroll SET status = 'Paid', payment_status = 'Paid' WHERE $whereClause",
  1004→            ['period_id' => $periodId]
  1005→        );
  1006→
  1007→        return ['success' => true, 'message' => 'Payroll marked as paid.'];

  1257→    /**
  1258→     * Calculate Professional Tax (State-wise slabs)
  1259→     * Default slabs - can be overridden based on employee's work state
  1260→     * Common PT slabs across India (monthly)
  1261→     */
  1262→    private function calculatePT($gross, $state = 'MH') {
  1263→        // Maharashtra slabs (most common)
  1264→        if ($state === 'MH' || $state === 'Maharashtra') {
  1265→            if ($gross > 10000) {
  1266→                return 200;  // Fixed ₹200 for salary above ₹10,000
  1267→            }
  1268→            return 0;
  1269→        }
  1270→        
  1271→        // Karnataka slabs
  1272→        if ($state === 'KA' || $state === 'Karnataka') {
  1273→            if ($gross > 15000) {
  1274→                return 200;
  1275→            } elseif ($gross > 10000) {
  1276→                return 150;
  1277→            }
  1278→            return 0;
  1279→        }
  1280→        
  1281→        // Tamil Nadu slabs
  1282→        if ($state === 'TN' || $state === 'Tamil Nadu') {
  1283→            if ($gross > 75000) { // Half yearly threshold
  1284→                return 1250 / 6; // Approx monthly
  1285→            } elseif ($gross > 50000) {
  1286→                return 833 / 6;
  1287→            }
  1288→            return 0;
  1289→        }
  1290→        
  1291→        // Delhi slabs (no PT for most)
  1292→        if ($state === 'DL' || $state === 'Delhi') {
  1293→            if ($gross > 25000) {
  1294→                return 200;
  1295→            }
  1296→            return 0;
  1297→        }
  1298→        
  1299→        // Gujarat slabs
  1300→        if ($state === 'GJ' || $state === 'Gujarat') {
  1301→            if ($gross > 12000) {
  1302→                return 200;
  1303→            }
  1304→            return 0;
  1305→        }
  1306→        
  1307→        // Default: Maharashtra-style slab
  1308→        if ($gross > 10000) {
  1309→            return 200;
  1310→        }
  1311→        return 0;
  1312→    }
