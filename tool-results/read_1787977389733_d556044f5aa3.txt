   633→function recalcTemplate() {
   634→    var net = parseFloat(document.getElementById('tplNetSalary').value) || 0;
   635→    if (net <= 0) { clearCalcResults(); return; }
   636→
   637→    // Get selected worker category (single radio)
   638→    var catEl = document.querySelector('[name="worker_cat"]:checked');
   639→    var workerCategory = catEl ? catEl.value : '';
   640→    if (!workerCategory) {
   641→        clearCalcResults();
   642→        var w = document.getElementById('calcWarnings');
   643→        w.classList.remove('d-none');
   644→        w.innerHTML = '<div class="alert alert-info py-2 small mb-0"><i class="bi bi-info-circle me-1"></i>Please select a Worker Category to look up minimum wage.</div>';
   645→        return;
   646→    }
   647→
   648→    var params = {
   649→        action: 'reverse_calc',
   650→        net_salary: net,
   651→        unit_id: UNIT_ID,
   652→        worker_category: workerCategory,
   653→        // Send explicit overrides for the 4 statutory flags (Q1=A: defaults from unit, override allowed)
   654→        pf:  document.getElementById('stat_pf').checked  ? '1' : '0',
   655→        esi: document.getElementById('stat_esi').checked ? '1' : '0',
   656→        pt:  document.getElementById('stat_pt').checked  ? '1' : '0',
   657→        lwf: document.getElementById('stat_lwf').checked ? '1' : '0'
   658→        // No bonus_applicable — bonus is always auto (skipped only if it would overshoot)
   659→    };
   660→
   661→    fetch('index.php?page=api/salary-calc', {
   662→        method: 'POST',
   663→        headers: { 'Content-Type': 'application/json', 'X-CSRF-Token': CSRF_TOKEN },
   664→        body: JSON.stringify(params)
   665→    })
   666→    .then(function(r) { return r.json(); })
   667→    .then(function(data) {
   668→        // Handle hard error (e.g. target below min wage, target not achievable)
   669→        if (!data.success) {
   670→            clearCalcResults();
   671→            var w = document.getElementById('calcWarnings');
   672→            w.classList.remove('d-none');
   673→            var errType = (data.data && data.data.error) ? data.data.error : 'ERROR';
   674→            var alertClass = 'alert-danger';
   675→            if (errType === 'TARGET_BELOW_MIN_WAGE') alertClass = 'alert-warning';
   676→            w.innerHTML = '<div class="alert ' + alertClass + ' py-2 small mb-0"><i class="bi bi-exclamation-octagon me-1"></i><strong>Cannot calculate:</strong> ' + (data.error || 'Unknown error') + '</div>';
   677→            return;
   678→        }
   679→        var d = data.data;
   680→        document.getElementById('calcBasicDa').textContent = fmt(d.basic_da);
   681→        document.getElementById('calcLeave').textContent   = fmt(d.leave_encashment);
   682→        document.getElementById('calcBonus').textContent   = fmt(d.bonus_encashment);
   683→        document.getElementById('calcHra').textContent     = fmt(d.hra);
   684→        document.getElementById('calcGross').textContent   = fmt(d.gross_salary);
   685→        document.getElementById('calcPf').textContent      = fmt(d.deductions.pf);
   686→        document.getElementById('calcEsi').textContent     = fmt(d.deductions.esi);
   687→        document.getElementById('calcPt').textContent      = fmt(d.deductions.pt);
   688→        document.getElementById('calcLwf').textContent     = fmt(d.deductions.lwf);
   689→        document.getElementById('calcTotalDed').textContent = fmt(d.total_deductions);
   690→        document.getElementById('calcNet').textContent     = fmt(d.calculated_net);
   691→        document.getElementById('calcBasicPct').textContent = d.basic_percent + '% of Gross';
   692→
   693→        // Min wage + escalation level
   694→        document.getElementById('calcMinWage').textContent = d.min_wage > 0 ? fmt(d.min_wage) : 'Not found';
   695→
   696→        // Min wage diagnostic (explains why it's 0 / Not found)
   697→        var dbgEl = document.getElementById('calcMinWageDebug');
   698→        if (dbgEl) {
   699→            var dbg = d.min_wage_debug;
   700→            if (d.min_wage > 0 || !dbg) {
   701→                dbgEl.style.display = 'none';
   702→                dbgEl.textContent = '';
   703→            } else {
   704→                var msg = dbg.reason || 'Min wage not found.';
   705→                var extra = '';
   706→                if (dbg.categories_for_state && dbg.categories_for_state.length) {
   707→                    extra += ' | Categories in DB: ' + dbg.categories_for_state.join(', ');
   708→                }
   709→                if (dbg.zones_for_state && dbg.zones_for_state.length) {
   710→                    extra += ' | Zones: ' + dbg.zones_for_state.join(', ');
   711→                }
   712→                dbgEl.textContent = msg + extra;
   713→                dbgEl.style.display = 'block';
   714→            }
   715→        }

   717→        var levelClass = 'bg-info';

   719→        else if (d.level_reached === 3) levelClass = 'bg-warning text-dark';

   735→        if (d.fifty_fifty_ok) {
   736→            ff.innerHTML = ' <i class="bi bi-check-circle-fill text-success"></i> 50-50 OK';

   750→        // Warnings (escalation notes + min wage + non-convergence)

   773→        w.innerHTML = '<div class="alert alert-danger py-2 small mb-0"><i class="bi bi-x-circle me-1"></i>Request failed: ' + err.message + '</div>';
   774→    });
   775→}
