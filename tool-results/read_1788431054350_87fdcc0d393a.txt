    46→function _ensureTables(mysqli $conn): void
    47→{
    48→    // Main visits table
    49→    $conn->query("
    50→        CREATE TABLE IF NOT EXISTS ess_unit_visits (
    51→            id INT AUTO_INCREMENT PRIMARY KEY,
    52→            employee_id VARCHAR(50) NOT NULL,
    53→            unit_id INT NOT NULL,
    54→            visit_number TINYINT NOT NULL COMMENT '1=first visit, 2=second visit',
    55→            visit_month INT NOT NULL COMMENT '1-12',
    56→            visit_year INT NOT NULL,
    57→            document_url VARCHAR(500) DEFAULT '',
    58→            document_type VARCHAR(20) DEFAULT 'image' COMMENT 'image or pdf',
    59→            notes TEXT DEFAULT NULL,
    60→            status ENUM('submitted','approved','rejected') DEFAULT 'submitted',
    61→            rejection_reason TEXT DEFAULT NULL,
    62→            approved_by VARCHAR(50) DEFAULT NULL,
    63→            approved_at DATETIME DEFAULT NULL,
    64→            total_score DECIMAL(8,2) DEFAULT 0,
    65→            max_score DECIMAL(8,2) DEFAULT 0,
    66→            score_percent DECIMAL(5,2) DEFAULT 0,
    67→            created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    68→            updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    69→            INDEX idx_employee_month (employee_id, visit_month, visit_year),
    70→            INDEX idx_unit_month (unit_id, visit_month, visit_year),
    71→            INDEX idx_status (status)
    72→        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci
    73→    ");
    74→
    75→    // Add columns that may not exist in old schema
    76→    $cols = $conn->query("SHOW COLUMNS FROM ess_unit_visits LIKE 'rejection_reason'")->num_rows;
    77→    if ($cols == 0) {
    78→        $conn->query("ALTER TABLE ess_unit_visits ADD COLUMN rejection_reason TEXT DEFAULT NULL AFTER status");
    79→        $conn->query("ALTER TABLE ess_unit_visits ADD COLUMN approved_by VARCHAR(50) DEFAULT NULL AFTER rejection_reason");
    80→        $conn->query("ALTER TABLE ess_unit_visits ADD COLUMN approved_at DATETIME DEFAULT NULL AFTER approved_by");
    81→        $conn->query("ALTER TABLE ess_unit_visits ADD COLUMN total_score DECIMAL(8,2) DEFAULT 0 AFTER approved_at");
    82→        $conn->query("ALTER TABLE ess_unit_visits ADD COLUMN max_score DECIMAL(8,2) DEFAULT 0 AFTER total_score");
    83→        $conn->query("ALTER TABLE ess_unit_visits ADD COLUMN score_percent DECIMAL(5,2) DEFAULT 0 AFTER max_score");
    84→        $conn->query("ALTER TABLE ess_unit_visits ADD INDEX idx_status (status)");
    85→    }
    86→
    87→    // Drop old unique key if exists and recreate
    88→    try {
    89→        $conn->query("ALTER TABLE ess_unit_visits DROP INDEX uk_visit");
    90→    } catch (\Throwable $e) { /* index may not exist */ }
    91→    try {
    92→        $conn->query("ALTER TABLE ess_unit_visits ADD UNIQUE KEY uk_visit (employee_id, unit_id, visit_month, visit_year, visit_number)");
    93→    } catch (\Throwable $e) { /* may already exist */ }
    94→
    95→    // Checklist items table
    96→    $conn->query("
    97→        CREATE TABLE IF NOT EXISTS ess_visit_checklist_items (
    98→            id INT AUTO_INCREMENT PRIMARY KEY,
    99→            visit_id INT NOT NULL,
   100→            checklist_item_id INT NOT NULL,
   101→            category_id INT NOT NULL,
   102→            status ENUM('yes','no','na') DEFAULT 'yes',
   103→            remarks TEXT DEFAULT NULL,
   104→            photo_url VARCHAR(500) DEFAULT NULL,
   105→            created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
   106→            INDEX idx_visit (visit_id)
   107→        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci
   108→    ");
   109→
   110→    // Audit log table
   111→    $conn->query("
   112→        CREATE TABLE IF NOT EXISTS ess_visit_audit_log (
   113→            id INT AUTO_INCREMENT PRIMARY KEY,
   114→            visit_id INT NOT NULL,
   115→            action VARCHAR(50) NOT NULL,
   116→            performed_by VARCHAR(50) NOT NULL,
   117→            details TEXT DEFAULT NULL,
   118→            created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
   119→            INDEX idx_visit (visit_id),
   120→            INDEX idx_action (action)
   121→        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci
   122→    ");
   123→}

   129→function _calculateScore(mysqli $conn, int $visitId): array
   130→{
   131→    $stmt = $conn->prepare('
   132→        SELECT vci.status, ci.weight
   133→        FROM ess_visit_checklist_items vci
   134→        JOIN ess_checklist_items ci ON ci.id = vci.checklist_item_id
   135→        WHERE vci.visit_id = ?
   136→    ');
   137→    $stmt->bind_param('i', $visitId);
   138→    $stmt->execute();
   139→    $result = $stmt->get_result();
   140→
   141→    $total = 0;
   142→    $max = 0;
   143→    while ($row = $result->fetch_assoc()) {
   144→        $w = (float)($row['weight'] ?? 1);
   145→        if ($row['status'] !== 'na') {
   146→            $max += $w;
   147→            if ($row['status'] === 'yes') {
   148→                $total += $w;
   149→            }
   150→        }
   151→    }
   152→    $stmt->close();
   153→
   154→    $percent = $max > 0 ? round(($total / $max) * 100, 2) : 0;
   155→
   156→    // Update visit record
   157→    $upd = $conn->prepare('UPDATE ess_unit_visits SET total_score = ?, max_score = ?, score_percent = ? WHERE id = ?');
   158→    $upd->bind_param('dddi', $total, $max, $percent, $visitId);
   159→    $upd->execute();
   160→    $upd->close();
   161→
   162→    return ['total' => $total, 'max' => $max, 'percent' => $percent];
   163→}

   196→function _handleGet(): void
   197→{
   198→    $authId = requireAuth();
   199→    $conn = getDbConnection();
   200→    _ensureTables($conn);
   201→
   202→    $view = $_GET['view'] ?? '';
   203→
   204→    if ($view === 'detail') {
   205→        _handleGetDetail($conn, $authId);
   206→    } elseif ($view === 'dashboard') {
   207→        _handleGetDashboard($conn, $authId);
   208→    } else {
   209→        _handleGetList($conn, $authId);
   210→    }

   213→function _handleGetList(mysqli $conn, string $authId): void
   214→{
   215→    $employeeId = $_GET['employee_id'] ?? $authId;
   216→    $month = (int)($_GET['month'] ?? 0);
   217→    $year = (int)($_GET['year'] ?? 0);
   218→    $unitId = (int)($_GET['unit_id'] ?? 0);
   219→    $status = $_GET['status'] ?? '';
   220→    $includeChecklist = ($_GET['include_checklist'] ?? '') === '1';
   221→    [$page, $limit, $offset] = getPaginationParams();
   222→
   223→    // Build where
   224→    $where = 'WHERE v.employee_id = ?';
   225→    $types = 's';
   226→    $params = [$employeeId];
   227→
   228→    if ($month > 0) { $where .= ' AND v.visit_month = ?'; $types .= 'i'; $params[] = $month; }
   229→    if ($year > 0) { $where .= ' AND v.visit_year = ?'; $types .= 'i'; $params[] = $year; }
   230→    if ($unitId > 0) { $where .= ' AND v.unit_id = ?'; $types .= 'i'; $params[] = $unitId; }
   231→    if ($status) { $where .= ' AND v.status = ?'; $types .= 's'; $params[] = $status; }
   232→
   233→    // Count
   234→    $countStmt = $conn->prepare("SELECT COUNT(*) AS total FROM ess_unit_visits v {$where}");
   235→    bindDynamicParams($countStmt, $types, $params);
   236→    $countStmt->execute();
   237→    $total = (int)$countStmt->get_result()->fetch_assoc()['total'];
   238→    $countStmt->close();
   239→
   240→    // Fetch
   241→    $dataQuery = "
   242→        SELECT v.id, v.employee_id, v.unit_id, v.visit_number, v.visit_month, v.visit_year,
   243→               v.document_url, v.document_type, v.notes, v.status, v.rejection_reason,
   244→               v.approved_by, v.approved_at, v.total_score, v.max_score, v.score_percent,
   245→               v.created_at, v.updated_at,
   246→               u.name AS unit_name, c.name AS client_name,
   247→               e.full_name AS employee_name, e.employee_code
   248→        FROM ess_unit_visits v
   249→        LEFT JOIN units u ON u.id = v.unit_id
   250→        LEFT JOIN clients c ON c.id = u.client_id
   251→        LEFT JOIN employees e ON e.id = v.employee_id
   252→        {$where}
   253→        ORDER BY v.visit_year DESC, v.visit_month DESC, v.created_at DESC
   254→        LIMIT ? OFFSET ?
   255→    ";
   256→    $dataTypes = $types . 'ii';
   257→    $dataParams = [...$params, $limit, $offset];
   258→
   259→    $stmt = $conn->prepare($dataQuery);
   260→    bindDynamicParams($stmt, $dataTypes, $dataParams);
   261→    $stmt->execute();
   262→    $result = $stmt->get_result();
   263→
   264→    $visits = [];
   265→    while ($row = $result->fetch_assoc()) {
   266→        $visit = [
   267→            'id' => (int)$row['id'],
   268→            'employee_id' => (int)$row['employee_id'],
   269→            'employee_name' => $row['employee_name'] ?? '',
   270→            'employee_code' => $row['employee_code'] ?? '',
   271→            'unit_id' => (int)$row['unit_id'],
   272→            'unit_name' => $row['unit_name'] ?? '',
   273→            'client_name' => $row['client_name'] ?? '',
   274→            'visit_number' => (int)$row['visit_number'],
   275→            'visit_month' => (int)$row['visit_month'],
   276→            'visit_year' => (int)$row['visit_year'],
   277→            'document_url' => $row['document_url'] ?? '',
   278→            'document_type' => $row['document_type'] ?? 'image',
   279→            'notes' => $row['notes'] ?? '',
   280→            'status' => $row['status'] ?? 'submitted',
   281→            'rejection_reason' => $row['rejection_reason'] ?? '',
   282→            'approved_by' => $row['approved_by'] ? (int)$row['approved_by'] : null,
   283→            'approved_at' => $row['approved_at'] ?? null,
   284→            'total_score' => (float)($row['total_score'] ?? 0),
   285→            'max_score' => (float)($row['max_score'] ?? 0),
   286→            'score_percent' => (float)($row['score_percent'] ?? 0),
   287→            'created_at' => $row['created_at'] ?? '',
   288→            'updated_at' => $row['updated_at'] ?? '',
   289→        ];
   290→
   291→        if ($includeChecklist) {
   292→            $visit['checklist_items'] = _fetchChecklistItems($conn, (int)$row['id']);
   293→        }
   294→
   295→        $visits[] = $visit;
   296→    }
   297→    $stmt->close();
   298→    $conn->close();
   299→
   300→    jsonOutput([
   301→        'success' => true,
   302→        'data' => [
   303→            'items' => $visits,
   304→            ...buildPagination($total, $page, $limit)
   305→        ]
   306→    ]);
   307→}

   309→function _handleGetDetail(mysqli $conn, string $authId): void
   310→{
   311→    $visitId = (int)($_GET['id'] ?? 0);
   312→    if ($visitId <= 0) {
   313→        jsonOutput(['success' => false, 'error' => 'Visit ID is required'], 400);
   314→    }
   315→
   316→    $stmt = $conn->prepare("
   317→        SELECT v.*, u.name AS unit_name, c.name AS client_name,
   318→               e.full_name AS employee_name, e.employee_code, e.mobile_number AS employee_mobile
   319→        FROM ess_unit_visits v
   320→        LEFT JOIN units u ON u.id = v.unit_id
   321→        LEFT JOIN clients c ON c.id = u.client_id
   322→        LEFT JOIN employees e ON e.id = v.employee_id
   323→        WHERE v.id = ?
   324→    ");
   325→    $stmt->bind_param('i', $visitId);
   326→    $stmt->execute();
   327→    $row = $stmt->get_result()->fetch_assoc();
   328→    $stmt->close();
   329→
   330→    if (!$row) {
   331→        $conn->close();
   332→        jsonOutput(['success' => false, 'error' => 'Visit not found'], 404);
   333→    }
   334→
   335→    // SECURITY (ownership): previously any authenticated user could read ANY
   336→    // visit detail (photos, mobile, audit log) by passing ?id=N. Now: only
   337→    // the visit submitter or a supervisor+ role may view it.
   338→    requireOwnershipOrRole($authId, (string)($row['employee_id'] ?? ''), ESS_GUARD_ROLES_SUPERVISOR, $conn);
   339→
   340→    // Recalculate score to ensure it's current
   341→    $scores = _calculateScore($conn, $visitId);
   342→
   343→    $visit = [
   344→        'id' => (int)$row['id'],
   345→        'employee_id' => (int)$row['employee_id'],
   346→        'employee_name' => $row['employee_name'] ?? '',
   347→        'employee_code' => $row['employee_code'] ?? '',
   348→        'employee_mobile' => $row['employee_mobile'] ?? '',
   349→        'unit_id' => (int)$row['unit_id'],
   350→        'unit_name' => $row['unit_name'] ?? '',
   351→        'client_name' => $row['client_name'] ?? '',
   352→        'visit_number' => (int)$row['visit_number'],
   353→        'visit_month' => (int)$row['visit_month'],
   354→        'visit_year' => (int)$row['visit_year'],
   355→        'document_url' => $row['document_url'] ?? '',
   356→        'document_type' => $row['document_type'] ?? 'image',
   357→        'notes' => $row['notes'] ?? '',
   358→        'status' => $row['status'] ?? 'submitted',
   359→        'rejection_reason' => $row['rejection_reason'] ?? '',
   360→        'approved_by' => $row['approved_by'] ? (int)$row['approved_by'] : null,
   361→        'approved_at' => $row['approved_at'] ?? null,
   362→        'total_score' => $scores['total'],
   363→        'max_score' => $scores['max'],
   364→        'score_percent' => $scores['percent'],
   365→        'checklist_items' => _fetchChecklistItems($conn, $visitId),
   366→        'created_at' => $row['created_at'] ?? '',
   367→        'updated_at' => $row['updated_at'] ?? '',
   368→    ];
   369→
   370→    // Fetch audit log
   371→    $auditStmt = $conn->prepare('SELECT * FROM ess_visit_audit_log WHERE visit_id = ? ORDER BY created_at ASC');
   372→    $auditStmt->bind_param('i', $visitId);
   373→    $auditStmt->execute();
   374→    $auditResult = $auditStmt->get_result();
   375→    $auditLog = [];
   376→    while ($aRow = $auditResult->fetch_assoc()) {
   377→        $auditLog[] = [
   378→            'id' => (int)$aRow['id'],
   379→            'action' => $aRow['action'],
   380→            'performed_by' => $aRow['performed_by'],
   381→            'details' => $aRow['details'] ?? '',
   382→            'created_at' => $aRow['created_at'],
   383→        ];
   384→    }
   385→    $auditStmt->close();
   386→    $visit['audit_log'] = $auditLog;
   387→
   388→    $conn->close();
   389→    jsonOutput(['success' => true, 'data' => $visit]);
   390→}

   392→function _handleGetDashboard(mysqli $conn, string $authId): void
   393→{
   394→    $employeeId = (int)($_GET['employee_id'] ?? $authId);
   395→    $now = new \DateTime('now', new \DateTimeZone('Asia/Kolkata'));
   396→    $currentMonth = (int)$now->format('n');
   397→    $currentYear = (int)$now->format('Y');
   398→
   399→    // Basic counts
   400→    $counts = $conn->query("
   401→        SELECT
   402→            COUNT(*) AS total_visits,
   403→            SUM(CASE WHEN status = 'submitted' THEN 1 ELSE 0 END) AS pending_visits,
   404→            SUM(CASE WHEN status = 'approved' THEN 1 ELSE 0 END) AS approved_visits,
   405→            SUM(CASE WHEN status = 'rejected' THEN 1 ELSE 0 END) AS rejected_visits,
   406→            SUM(CASE WHEN visit_month = {$currentMonth} AND visit_year = {$currentYear} THEN 1 ELSE 0 END) AS this_month_visits
   407→        FROM ess_unit_visits WHERE employee_id = {$employeeId}
   408→    ")->fetch_assoc();
   409→
   410→    // This month avg score
   411→    $monthAvg = $conn->query("
   412→        SELECT AVG(score_percent) AS avg_score
   413→        FROM ess_unit_visits
   414→        WHERE employee_id = {$employeeId} AND visit_month = {$currentMonth} AND visit_year = {$currentYear}
   415→            AND status IN ('approved','submitted') AND max_score > 0
   416→    ")->fetch_assoc();
   417→
   418→    // Score trend - last 6 months
   419→    $trend = [];
   420→    for ($i = 5; $i >= 0; $i--) {
   421→        $d = clone $now;
   422→        $d->modify("-{$i} months");
   423→        $m = (int)$d->format('n');
   424→        $y = (int)$d->format('Y');
   425→        $label = $d->format('M Y');
   426→        $avgRow = $conn->query("
   427→            SELECT AVG(score_percent) AS avg_score
   428→            FROM ess_unit_visits
   429→            WHERE employee_id = {$employeeId} AND visit_month = {$m} AND visit_year = {$y}
   430→                AND status IN ('approved','submitted') AND max_score > 0
   431→        ")->fetch_assoc();
   432→        $trend[] = ['month' => $label, 'avg_score' => round((float)($avgRow['avg_score'] ?? 0), 1)];
   433→    }
   434→
   435→    // Category scores
   436→    $catScores = $conn->query("
   437→        SELECT cc.name AS category, AVG(
   438→            CASE WHEN vci.status = 'yes' THEN ci.weight ELSE 0 END
   439→        ) / NULLIF(AVG(CASE WHEN vci.status != 'na' THEN ci.weight ELSE NULL END), 0) * 100 AS avg_score
   440→        FROM ess_visit_checklist_items vci
   441→        JOIN ess_checklist_items ci ON ci.id = vci.checklist_item_id
   442→        JOIN ess_checklist_categories cc ON cc.id = ci.category_id
   443→        JOIN ess_unit_visits v ON v.id = vci.visit_id
   444→        WHERE v.employee_id = {$employeeId} AND v.status IN ('approved','submitted')
   445→        GROUP BY cc.id, cc.name
   446→        ORDER BY cc.display_order
   447→    ");
   448→    $categoryScores = [];
   449→    while ($cr = $catScores->fetch_assoc()) {
   450→        $categoryScores[] = [
   451→            'category' => $cr['category'] ?? '',
   452→            'avg_score' => round((float)($cr['avg_score'] ?? 0), 1),
   453→        ];
   454→    }
   455→
   456→    // Unit compliance
   457→    $unitComp = $conn->query("
   458→        SELECT u.name AS unit_name, AVG(v.score_percent) AS score, COUNT(*) AS visits
   459→        FROM ess_unit_visits v
   460→        JOIN units u ON u.id = v.unit_id
   461→        WHERE v.employee_id = {$employeeId} AND v.status IN ('approved','submitted') AND v.max_score > 0
   462→        GROUP BY v.unit_id, u.name
   463→        ORDER BY score DESC
   464→        LIMIT 10
   465→    ");
   466→    $unitCompliance = [];
   467→    while ($ur = $unitComp->fetch_assoc()) {
   468→        $unitCompliance[] = [
   469→            'unit_name' => $ur['unit_name'] ?? '',
   470→            'score' => round((float)($ur['score'] ?? 0), 1),
   471→            'visits' => (int)$ur['visits'],
   472→        ];
   473→    }
   474→
   475→    $conn->close();
   476→
   477→    jsonOutput([
   478→        'success' => true,
   479→        'data' => [
   480→            'total_visits' => (int)($counts['total_visits'] ?? 0),
   481→            'pending_visits' => (int)($counts['pending_visits'] ?? 0),
   482→            'approved_visits' => (int)($counts['approved_visits'] ?? 0),
   483→            'rejected_visits' => (int)($counts['rejected_visits'] ?? 0),
   484→            'this_month_visits' => (int)($counts['this_month_visits'] ?? 0),
   485→            'this_month_score_avg' => round((float)($monthAvg['avg_score'] ?? 0), 1),
   486→            'score_trend' => $trend,
   487→            'category_scores' => $categoryScores,
   488→            'unit_compliance' => $unitCompliance,
   489→        ]
   490→    ]);

   533→// POST: Create Visit
   534→// ══════════════════════════════════════════════════════════════════════════
   535→
   536→function _handlePost(): void
   537→{
   538→    $employeeId = requireAuth();
   539→    $input = getInput();
   540→    $conn = getDbConnection();
   541→    _ensureTables($conn);
   542→
   543→    // Email action — send visit report email
   544→    if (($input['action'] ?? '') === 'send_email') {
   545→        $visitId = (int)($input['visit_id'] ?? 0);
   546→        if ($visitId <= 0) {
   547→            jsonOutput(['success' => false, 'error' => 'Visit ID required'], 400);
   548→        }
   549→        _logAudit($conn, $visitId, 'email_sent', $employeeId, 'Report email requested');
   550→
   551→        // Attempt to send email (best-effort)
   552→        try {
   553→            $emailSent = _sendVisitEmailInline($conn, $visitId);
   554→        } catch (\Throwable $e) {
   555→            $emailSent = false;
   556→        }
   557→        $conn->close();
   558→
   559→        jsonOutput(['success' => true, 'message' => $emailSent ? 'Report email sent successfully' : 'Email will be sent shortly']);
   560→        return;
   561→    }
   562→
   563→    // Validate
   564→    $unitId = (int)($input['unit_id'] ?? 0);
   565→    $visitMonth = (int)($input['visit_month'] ?? 0);
   566→    $visitYear = (int)($input['visit_year'] ?? 0);
   567→    $visitNumber = (int)($input['visit_number'] ?? 0);
   568→    $documentUrl = trim($input['document_url'] ?? '');
   569→    $documentType = trim($input['document_type'] ?? 'image');
   570→    $notes = trim($input['notes'] ?? '');
   571→    $checklistItems = $input['checklist_items'] ?? [];
   572→
   573→    if ($unitId <= 0) jsonOutput(['success' => false, 'error' => 'Unit is required'], 400);
   574→    if ($visitMonth < 1 || $visitMonth > 12) jsonOutput(['success' => false, 'error' => 'Invalid month (1-12 required)'], 400);
   575→    if ($visitYear < 2020 || $visitYear > 2099) jsonOutput(['success' => false, 'error' => 'Invalid year'], 400);
   576→    if ($visitNumber < 1 || $visitNumber > 2) jsonOutput(['success' => false, 'error' => 'Visit number must be 1 or 2'], 400);
   577→    if (empty($documentUrl)) jsonOutput(['success' => false, 'error' => 'Document (JPG/PDF) is required'], 400);
   578→
   579→    // Duplicate check
   580→    $dupStmt = $conn->prepare('
   581→        SELECT id FROM ess_unit_visits
   582→        WHERE employee_id = ? AND unit_id = ? AND visit_month = ? AND visit_year = ? AND visit_number = ?
   583→    ');
   584→    $dupStmt->bind_param('siiii', $employeeId, $unitId, $visitMonth, $visitYear, $visitNumber);
   585→    $dupStmt->execute();
   586→    $existing = $dupStmt->get_result()->fetch_assoc();
   587→    $dupStmt->close();
   588→
   589→    if ($existing) {
   590→        jsonOutput([
   591→            'success' => false,
   592→            'error' => 'A ' . _visitLabel($visitNumber) . ' for this unit in ' . _monthName($visitMonth) . ' ' . $visitYear . ' already exists',
   593→            'existing_id' => (int)$existing['id']
   594→        ], 409);
   595→    }
   596→
   597→    // Insert visit
   598→    $stmt = $conn->prepare('
   599→        INSERT INTO ess_unit_visits (employee_id, unit_id, visit_number, visit_month, visit_year, document_url, document_type, notes, status)
   600→        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
   601→    ');
   602→    bindDynamicParams($stmt, 'siiiissss', [
   603→        $employeeId, $unitId, $visitNumber, $visitMonth, $visitYear,
   604→        $documentUrl, $documentType, $notes, 'submitted'
   605→    ]);
   606→    $stmt->execute();
   607→    $newId = $stmt->insert_id;
   608→    $stmt->close();
   609→
   610→    // Insert checklist items
   611→    if (is_array($checklistItems) && count($checklistItems) > 0) {
   612→        $insStmt = $conn->prepare('
   613→            INSERT INTO ess_visit_checklist_items (visit_id, checklist_item_id, category_id, status, remarks, photo_url)
   614→            VALUES (?, ?, ?, ?, ?, ?)
   615→        ');
   616→        foreach ($checklistItems as $item) {
   617→            $itemId = (int)($item['checklist_item_id'] ?? 0);
   618→            $catId = (int)($item['category_id'] ?? 0);
   619→            $status = in_array($item['status'] ?? '', ['yes', 'no', 'na']) ? $item['status'] : 'yes';
   620→            $remarks = trim($item['remarks'] ?? '');
   621→            $photoUrl = trim($item['photo_url'] ?? '');
   622→            $insStmt->bind_param('iiisss', $newId, $itemId, $catId, $status, $remarks, $photoUrl);
   623→            $insStmt->execute();
   624→        }
   625→        $insStmt->close();
   626→    }
   627→
   628→    // Calculate score
   629→    $scores = _calculateScore($conn, $newId);
   630→
   631→    // Audit log
   632→    _logAudit($conn, $newId, 'created', $employeeId, "Visit submitted with {$scores['percent']}% score");
   633→
   634→    // Fetch unit info for response
   635→    $unitStmt = $conn->prepare('SELECT u.name AS unit_name, c.name AS client_name FROM units u LEFT JOIN clients c ON c.id = u.client_id WHERE u.id = ?');
   636→    $unitStmt->bind_param('i', $unitId);
   637→    $unitStmt->execute();
   638→    $unitInfo = $unitStmt->get_result()->fetch_assoc();
   639→    $unitStmt->close();
   640→
   641→    // Fetch employee info
   642→    $empStmt = $conn->prepare('SELECT full_name, employee_code FROM employees WHERE id = ?');
   643→    $empStmt->bind_param('s', $employeeId);
   644→    $empStmt->execute();
   645→    $empInfo = $empStmt->get_result()->fetch_assoc();
   646→    $empStmt->close();
   647→    $conn->close();
   648→
   649→    jsonOutput([
   650→        'success' => true,
   651→        'data' => [
   652→            'id' => $newId,
   653→            'employee_id' => (int)$employeeId,
   654→            'employee_name' => $empInfo['full_name'] ?? '',
   655→            'employee_code' => $empInfo['employee_code'] ?? '',
   656→            'unit_id' => $unitId,
   657→            'unit_name' => $unitInfo['unit_name'] ?? '',
   658→            'client_name' => $unitInfo['client_name'] ?? '',
   659→            'visit_number' => $visitNumber,
   660→            'visit_month' => $visitMonth,
   661→            'visit_year' => $visitYear,
   662→            'document_url' => $documentUrl,
   663→            'document_type' => $documentType,
   664→            'notes' => $notes,
   665→            'status' => 'submitted',
   666→            'total_score' => $scores['total'],
   667→            'max_score' => $scores['max'],
   668→            'score_percent' => $scores['percent'],
   669→            'message' => 'Unit visit ' . _visitLabel($visitNumber) . ' submitted successfully'
   670→        ]
   671→    ]);

   675→// PUT: Approve / Reject
   676→// ══════════════════════════════════════════════════════════════════════════
   677→
   678→function _handlePut(): void
   679→{
   680→    $employeeId = requireAuth();
   681→    $input = getInput();
   682→    $conn = getDbConnection();
   683→    _ensureTables($conn);
   684→
   685→    // SECURITY: approving/rejecting a unit visit is a manager+ action.
   686→    // Previously any authenticated employee could approve/reject ANY visit.
   687→    requireRole(ESS_GUARD_ROLES_MANAGER, $conn);
   688→
   689→    $visitId = (int)($input['id'] ?? 0);
   690→    $action = $input['action'] ?? '';
   691→
   692→    if ($visitId <= 0) jsonOutput(['success' => false, 'error' => 'Visit ID is required'], 400);
   693→    if (!in_array($action, ['approve', 'reject'])) jsonOutput(['success' => false, 'error' => 'Action must be approve or reject'], 400);
   694→
   695→    // Verify visit exists
   696→    $checkStmt = $conn->prepare('SELECT id, status, employee_id FROM ess_unit_visits WHERE id = ?');
   697→    $checkStmt->bind_param('i', $visitId);
   698→    $checkStmt->execute();
   699→    $visit = $checkStmt->get_result()->fetch_assoc();
   700→    $checkStmt->close();
   701→
   702→    if (!$visit) { $conn->close(); jsonOutput(['success' => false, 'error' => 'Visit not found'], 404); }
   703→    if ($visit['status'] !== 'submitted') { $conn->close(); jsonOutput(['success' => false, 'error' => 'Only submitted visits can be ' . $action . 'd'], 400); }
   704→
   705→    if ($action === 'approve') {
   706→        $stmt = $conn->prepare('UPDATE ess_unit_visits SET status = ?, approved_by = ?, approved_at = NOW() WHERE id = ?');
   707→        $stmt->bind_param('ssi', 'approved', $employeeId, $visitId);
   708→        $stmt->execute();
   709→        $stmt->close();
   710→        _logAudit($conn, $visitId, 'approved', $employeeId);
   711→    } else {
   712→        $rejectionReason = trim($input['rejection_reason'] ?? '');
   713→        $stmt = $conn->prepare('UPDATE ess_unit_visits SET status = ?, rejection_reason = ?, approved_by = ?, approved_at = NOW() WHERE id = ?');
   714→        $stmt->bind_param('sssi', 'rejected', $rejectionReason, $employeeId, $visitId);
   715→        $stmt->execute();
   716→        $stmt->close();
   717→        _logAudit($conn, $visitId, 'rejected', $employeeId, $rejectionReason);
   718→    }
   719→
   720→    // Fetch updated visit
   721→    $fetchStmt = $conn->prepare("
   722→        SELECT v.*, u.name AS unit_name, c.name AS client_name,
   723→               e.full_name AS employee_name, e.employee_code
   724→        FROM ess_unit_visits v
   725→        LEFT JOIN units u ON u.id = v.unit_id
   726→        LEFT JOIN clients c ON c.id = u.client_id
   727→        LEFT JOIN employees e ON e.id = v.employee_id
   728→        WHERE v.id = ?
   729→    ");
   730→    $fetchStmt->bind_param('i', $visitId);
   731→    $fetchStmt->execute();
   732→    $row = $fetchStmt->get_result()->fetch_assoc();
   733→    $fetchStmt->close();
   734→    $conn->close();
   735→
   736→    jsonOutput([
   737→        'success' => true,
   738→        'data' => [
   739→            'id' => (int)$row['id'],
   740→            'status' => $row['status'],
   741→            'message' => 'Visit ' . $action . 'd successfully'
   742→        ]
   743→    ]);
   744→}

   787→// Email Helper (inline)
   788→// ══════════════════════════════════════════════════════════════════════════
   789→
   790→function _sendVisitEmailInline(mysqli $conn, int $visitId): bool
   791→{
   792→    // Fetch visit with employee email
   793→    $stmt = $conn->prepare("
   794→        SELECT v.*, u.name AS unit_name, c.name AS client_name,
   795→               e.full_name AS employee_name, e.employee_code, e.email AS employee_email
   796→        FROM ess_unit_visits v
   797→        LEFT JOIN units u ON u.id = v.unit_id
   798→        LEFT JOIN clients c ON c.id = u.client_id
   799→        LEFT JOIN employees e ON e.id = v.employee_id
   800→        WHERE v.id = ?
   801→    ");
   802→    $stmt->bind_param('i', $visitId);
   803→    $stmt->execute();
   804→    $visit = $stmt->get_result()->fetch_assoc();
   805→    $stmt->close();
   806→
   807→    if (!$visit || empty($visit['employee_email'])) return false;
   808→
   809→    // Fetch checklist items
   810→    $itemsStmt = $conn->prepare('
   811→        SELECT ci.status, ci.remarks,
   812→               cmi.name AS item_name, cmc.name AS category_name
   813→        FROM ess_visit_checklist_items ci
   814→        LEFT JOIN ess_checklist_items cmi ON cmi.id = ci.checklist_item_id
   815→        LEFT JOIN ess_checklist_categories cmc ON cmc.id = ci.category_id
   816→        WHERE ci.visit_id = ?
   817→        ORDER BY cmc.display_order ASC, cmi.display_order ASC
   818→    ');
   819→    $itemsStmt->bind_param('i', $visitId);
   820→    $itemsStmt->execute();
   821→    $itemsResult = $itemsStmt->get_result();
   822→    $items = [];
   823→    while ($row = $itemsResult->fetch_assoc()) {
   824→        $items[] = $row;
   825→    }
   826→    $itemsResult->free();
   827→    $itemsStmt->close();
   828→
   829→    $score = (int)($visit['score_percent'] ?? 0);
   830→    $subject = sprintf('Unit Visit Report: %s - %s %s (%d%%)',
   831→        $visit['unit_name'] ?? 'Unknown',
   832→        _monthName((int)$visit['visit_month']),
   833→        $visit['visit_year'],
   834→        $score
   835→    );
   836→
   837→    // Build plain text email
   838→    $text = "UNIT VISIT CHECKLIST REPORT\nRCS TRUE FACILITIES PVT LTD\n" . str_repeat('=', 50) . "\n\n";
   839→    $text .= "Employee: " . ($visit['employee_name'] ?? '') . "\n";
   840→    $text .= "Client/Unit: " . ($visit['client_name'] ?? '') . " / " . ($visit['unit_name'] ?? '') . "\n";
   841→    $text .= "Visit: " . ((int)$visit['visit_number'] === 1 ? 'First' : 'Second') . " Visit - " . _monthName((int)$visit['visit_month']) . " " . $visit['visit_year'] . "\n";
   842→    $text .= "Score: {$score}% (" . ($visit['total_score'] ?? 0) . "/" . ($visit['max_score'] ?? 0) . " points)\n\n";
   843→
   844→    $grouped = [];
   845→    foreach ($items as $item) {
   846→        $cat = $item['category_name'] ?? 'Unknown';
   847→        if (!isset($grouped[$cat])) $grouped[$cat] = [];
   848→        $status = $item['status'] === 'na' ? 'N/A' : ($item['status'] === 'yes' ? '[YES]' : '[NO]');
   849→        $line = "  {$status} " . ($item['item_name'] ?? '');
   850→        if (!empty($item['remarks'])) $line .= " - " . $item['remarks'];
   851→        $grouped[$cat][] = $line;
   852→    }
   853→    foreach ($grouped as $cat => $lines) {
   854→        $text .= "\n--- " . strtoupper($cat) . " ---\n" . implode("\n", $lines) . "\n";
   855→    }
   856→
   857→    if (!empty($visit['notes'])) $text .= "\nNOTES: " . $visit['notes'] . "\n";
   858→
   859→    $headers = "From: RCS ESS Portal <noreply@rcsfacility.com>\r\nReply-To: support@rcsfacility.com\r\nX-Mailer: RCS-ESS/2.0";
   860→    return mail($visit['employee_email'], $subject, $text, $headers, '-fnoreply@rcsfacility.com');
   861→}
