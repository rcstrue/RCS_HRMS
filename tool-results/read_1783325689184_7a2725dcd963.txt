   537→   537→function updateOTAmounts() {
   538→   538→    let totalOT = 0;
   539→   539→    let totalEstAmount = 0;
   540→   540→    
   541→   541→    document.querySelectorAll('.ot-input').forEach(input => {
   542→   542→        const empId = input.dataset.empId;
   543→   543→        const otRate = parseFloat(input.dataset.otRate) || 0;
   544→   544→        const otHours = parseFloat(input.value) || 0;
   545→   545→        const estAmount = otRate * otHours;
   546→   546→        
   547→   547→        totalOT += otHours;
   548→   548→        totalEstAmount += estAmount;
   549→   549→        
   550→   550→        const amountEl = document.getElementById('ot_amount_' + empId);
   551→   551→        if (amountEl) {
   552→   552→            amountEl.textContent = '₹' + estAmount.toLocaleString('en-IN', {minimumFractionDigits: 0, maximumFractionDigits: 0});
   553→   553→        }
   554→   554→    });
   555→   555→

   557→   557→    document.getElementById('new_ot_total').textContent = totalOT.toFixed(1);
   558→   558→    document.getElementById('est_ot_amount_total').textContent = fmt(totalEstAmount);
