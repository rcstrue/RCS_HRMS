   912→// Auto-populate Worker Category from the selected Designation
   913→// (the option carries a data-category attribute sourced from the
   914→//  designations.worker_category column)
   915→function autoFillWorkerCategory(designationSelect) {
   916→    const opt = designationSelect.options[designationSelect.selectedIndex];
   917→    const cat = opt ? (opt.getAttribute('data-category') || '') : '';
   918→    const wcSelect = document.getElementById('worker_category');
   919→    if (!wcSelect || !cat) return;
   920→
   921→    // Try exact match first
   922→    let matched = false;
   923→    for (const o of wcSelect.options) {
   924→        if (o.value === cat) { o.selected = true; matched = true; break; }
   925→    }
   926→    // Fallback: case-insensitive match (handles "Semi-skilled" vs "Semi-Skilled")
   927→    if (!matched) {
   928→        const lower = cat.toLowerCase();
   929→        for (const o of wcSelect.options) {
   930→            if (o.value.toLowerCase() === lower) { o.selected = true; matched = true; break; }
   931→        }
   932→    }
   933→    if (matched) {
   934→        wcSelect.classList.add('is-valid');
   935→        setTimeout(() => wcSelect.classList.remove('is-valid'), 1200);
   936→    }
   937→}
