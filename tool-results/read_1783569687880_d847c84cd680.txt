   790→// Calculate row values
   791→function calculateRow(empId) {
   792→    // Get earnings
   793→    const basicDa = parseFloat(document.querySelector(`[name="basic_da[${empId}]"]`)?.value) || 0;
   794→    const hra = parseFloat(document.querySelector(`[name="hra[${empId}]"]`)?.value) || 0;
   795→    const leaveEncashment = parseFloat(document.querySelector(`[name="leave_encashment[${empId}]"]`)?.value) || 0;
   796→    const bonusEncashment = parseFloat(document.querySelector(`[name="bonus_encashment[${empId}]"]`)?.value) || 0;
   797→    const washingAllowance = parseFloat(document.querySelector(`[name="washing_allowance[${empId}]"]`)?.value) || 0;
   798→    
   799→    const gross = basicDa + hra + leaveEncashment + bonusEncashment + washingAllowance;
   800→    
   801→    // Get statutory checkboxes
   802→    const pfApplicable = document.querySelector(`[name="pf_applicable[${empId}]"]`)?.checked || false;
   803→    const esiApplicable = document.querySelector(`[name="esi_applicable[${empId}]"]`)?.checked || false;
   804→    const ptApplicable = document.querySelector(`[name="pt_applicable[${empId}]"]`)?.checked || false;
   805→    
   806→    // Calculate PF (12% of Basic+DA, max on 15000)
   807→    const pfBase = Math.min(basicDa, 15000);
   808→    const pf = pfApplicable ? Math.round(pfBase * 0.12 * 100) / 100 : 0;
   809→    
   810→    // Calculate ESI (0.75% of Gross if gross <= 21000)
   811→    const esi = esiApplicable && gross <= 21000 ? Math.round(gross * 0.0075 * 100) / 100 : 0;
   812→    
   813→    // PT (simplified)
   814→    const pt = ptApplicable ? (gross > 15000 ? 200 : 0) : 0;
   815→    
   816→    const totalDed = pf + esi + pt;
   817→    const netSalary = gross - totalDed;
   818→    
   819→    // Update display
   820→    const grossEl = document.querySelector(`.gross-${empId}`);
   821→    const pfEl = document.querySelector(`.pf-${empId}`);
   822→    const esiEl = document.querySelector(`.esi-${empId}`);
   823→    const totalDedEl = document.querySelector(`.totalded-${empId}`);
   824→    const netEl = document.querySelector(`.netsalary-${empId}`);
   825→    
   826→    if (grossEl) grossEl.textContent = gross.toFixed(2);
   827→    if (pfEl) pfEl.textContent = pf.toFixed(2);
   828→    if (esiEl) esiEl.textContent = esi.toFixed(2);
   829→    if (totalDedEl) totalDedEl.textContent = totalDed.toFixed(2);
   830→    if (netEl) netEl.textContent = netSalary.toFixed(2);
   831→    
   832→    // Update totals
   833→    calculateTotals();
   834→}
