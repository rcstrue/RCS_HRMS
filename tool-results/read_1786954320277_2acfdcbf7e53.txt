     1→<?php

     3→
     4→/**
     5→ * RCS HRMS Pro — Employee Change Requests Approval Page
     6→ *
     7→ * Allows HR/Admin to view, approve, or reject change requests
     8→ * submitted by employees via the ESS mobile app.
     9→ *
    10→ * Approve: updates the employee record with the new value.
    11→ * Reject: stores rejection reason; employee sees it in ESS.
    12→ * Both actions send email + WhatsApp notification to the employee.
    13→ */
    14→
    15→$pageTitle = 'Change Request Approvals';
    16→
    17→// ─── POST Actions ────────────────────────────────────────────────────────────
    18→
    19→if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
    20→    // CSRF check
    21→    if (!validateCSRFToken($_POST['csrf_token'] ?? '')) {
    22→        setFlash('error', 'Invalid request. Please refresh the page and try again.');
    23→        redirect($_SERVER['REQUEST_URI'] ?? 'index.php?page=employee/change-requests');
    24→    }
    25→    $action = sanitize($_POST['action']);
    26→
    27→    // ── Whitelist of fields that can be updated on approval ──
    28→    $APPROVAL_FIELDS = [
    29→        'full_name', 'father_name', 'date_of_birth', 'gender',
    30→        'designation', 'department',
    31→    ];
    32→
    33→    // ── 1. Approve ─────────────────────────────────────────────
    34→    if ($action === 'approve') {
    35→        $id = (int) ($_POST['id'] ?? 0);
    36→        if ($id > 0) {
    37→            $request = $db->fetch(
    38→                "SELECT r.*, e.full_name AS emp_name, e.email, e.mobile_number,
    39→                        e.employee_code, e.designation
    40→                 FROM employee_change_requests r
    41→                 JOIN employees e ON e.id = r.employee_id
    42→                 WHERE r.id = :id AND r.status = 'pending'
    43→                 LIMIT 1",
    44→                ['id' => $id]
    45→            );
    46→
    47→            if ($request) {
    48→                $field = $request['field_name'];
    49→                $newValue = $request['new_value'];
    50→
    51→                // Apply the change to employee record if it's a whitelisted field
    52→                if (in_array($field, $APPROVAL_FIELDS, true)) {
    53→                    try {
    54→                        $db->update(
    55→                            'employees',
    56→                            [$field => $newValue],
    57→                            'id = :id',
    58→                            ['id' => $request['employee_id']]
    59→                        );
    60→                    } catch (Exception $e) {
    61→                        setFlash('error', 'Failed to update employee record: ' . $e->getMessage());
    62→                        redirect('index.php?page=employee/change-requests');
    63→                    }
    64→                }
    65→
    66→                // Mark request as approved
    67→                $db->update(
    68→                    'employee_change_requests',
    69→                    [
    70→                        'status'      => 'approved',
    71→                        'reviewed_at' => date('Y-m-d H:i:s'),
    72→                        'reviewed_by' => $_SESSION['user_id'] ?? null,
    73→                    ],
    74→                    'id = :id',
    75→                    ['id' => $id]
    76→                );

    89→                            <div style='background:#10b981;padding:20px;text-align:center;'>
    90→                                <h2 style='color:#fff;margin:0;'>RCS True Facilities Pvt Ltd</h2>
    91→                            </div>
    92→                            <div style='padding:25px;background:#f9fafb;border:1px solid #e5e7eb;'>
    93→                                <p>Dear {$request['emp_name']},</p>
    94→                                <p>Your change request has been <strong style='color:#10b981;'>APPROVED</strong>.</p>
    95→                                <table style='width:100%;border-collapse:collapse;margin:15px 0;'>
    96→                                    <tr><td style='padding:8px;border:1px solid #e5e7eb;background:#f3f4f6;font-weight:bold;'>Field</td>
    97→                                        <td style='padding:8px;border:1px solid #e5e7eb;'>{$fieldLabel}</td></tr>
    98→                                    <tr><td style='padding:8px;border:1px solid #e5e7eb;background:#f3f4f6;font-weight:bold;'>Old Value</td>
    99→                                        <td style='padding:8px;border:1px solid #e5e7eb;'>" . htmlspecialchars($request['old_value']) . "</td></tr>
   100→                                    <tr><td style='padding:8px;border:1px solid #e5e7eb;background:#f3f4f6;font-weight:bold;'>New Value</td>
   101→                                        <td style='padding:8px;border:1px solid #e5e7eb;'>" . htmlspecialchars($newValue) . "</td></tr>
   102→                                </table>
   103→                                <p>If you did not request this change, please contact HR immediately.</p>
   104→                            </div>
   105→                            <p style='text-align:center;color:#9ca3af;font-size:12px;'>This is an automated message from RCS HRMS Pro.</p>
   106→                        </div>";

   108→                    }
   109→
   110→                    // WhatsApp

   116→                    }

   126→                }

   129→            } else {
   130→                setFlash('error', 'Request not found or already processed.');
   131→            }
   132→        }
   133→        redirect('index.php?page=employee/change-requests');
   134→    }
   135→

   142→            $request = $db->fetch(
   143→                "SELECT r.*, e.full_name AS emp_name, e.email, e.mobile_number,
   144→                        e.employee_code, e.designation
   145→                 FROM employee_change_requests r
   146→                 JOIN employees e ON e.id = r.employee_id
   147→                 WHERE r.id = :id AND r.status = 'pending'
   148→                 LIMIT 1",
   149→                ['id' => $id]
   150→            );
   151→
   152→            if ($request) {
   153→                $field = $request['field_name'];

   155→                // Mark request as rejected

   157→                    'employee_change_requests',

   166→                );

   172→
   173→                    $empEmail = $request['email'];
   174→                    if (!empty($empEmail)) {
   175→                        $fieldLabel = ucfirst(str_replace('_', ' ', $field));
   176→                        $htmlBody = "
   177→                        <div style='font-family:Arial,sans-serif;max-width:600px;margin:0 auto;padding:20px;'>
   178→                            <div style='background:#ef4444;padding:20px;text-align:center;'>
   179→                                <h2 style='color:#fff;margin:0;'>RCS True Facilities Pvt Ltd</h2>
   180→                            </div>
   181→                            <div style='padding:25px;background:#f9fafb;border:1px solid #e5e7eb;'>
   182→                                <p>Dear {$request['emp_name']},</p>
   183→                                <p>Your change request has been <strong style='color:#ef4444;'>REJECTED</strong>.</p>
   184→                                <table style='width:100%;border-collapse:collapse;margin:15px 0;'>
   185→                                    <tr><td style='padding:8px;border:1px solid #e5e7eb;background:#f3f4f6;font-weight:bold;'>Field</td>
   186→                                        <td style='padding:8px;border:1px solid #e5e7eb;'>{$fieldLabel}</td></tr>
   187→                                    <tr><td style='padding:8px;border:1px solid #e5e7eb;background:#f3f4f6;font-weight:bold;'>Requested Value</td>
   188→                                        <td style='padding:8px;border:1px solid #e5e7eb;'>" . htmlspecialchars($request['new_value']) . "</td></tr>
   189→                                    <tr><td style='padding:8px;border:1px solid #e5e7eb;background:#f3f4f6;font-weight:bold;'>Rejection Reason</td>
   190→                                        <td style='padding:8px;border:1px solid #e5e7eb;'>" . htmlspecialchars($reason) . "</td></tr>
   191→                                </table>
   192→                                <p>If you believe this was an error, please contact HR.</p>
   193→                            </div>
   194→                            <p style='text-align:center;color:#9ca3af;font-size:12px;'>This is an automated message from RCS HRMS Pro.</p>
   195→                        </div>";
   196→                        $notif->sendEmail($empEmail, "Change Request Rejected — {$fieldLabel}", $htmlBody);
   197→                    }
   198→
   199→                    // WhatsApp
   200→                    $empMobile = $request['mobile_number'];
   201→                    if (!empty($empMobile)) {
   202→                        $fieldLabel = ucfirst(str_replace('_', ' ', $field));
   203→                        $waMsg = "Hello {$request['emp_name']},\n\nYour change request has been REJECTED.\n\nField: {$fieldLabel}\nReason: {$reason}\n\n- RCS HRMS Pro";
   204→                        $notif->sendWhatsApp($empMobile, $waMsg);
   205→                    }
   206→                } catch (Exception $e) {
   207→                    error_log('[Change Request Rejection] Notification error: ' . $e->getMessage());
   208→                }
   209→
   210→                if (function_exists('logActivity')) {
   211→                    logActivity('reject_change_request', 'employee', $request['employee_id'],
   212→                        "Rejected change request #{$id}: {$field} — " . substr($reason, 0, 50));
   213→                }
   214→
   215→                setFlash('success', 'Change request rejected. Employee has been notified.');
   216→            } else {
   217→                setFlash('error', 'Request not found or already processed.');
   218→            }
   219→        } elseif ($id > 0 && $reason === '') {
   220→            setFlash('danger', 'Rejection reason is required.');
   221→        }
   222→        redirect('index.php?page=employee/change-requests');
   223→    }
   224→
   225→    // ── 3. Bulk approve ───────────────────────────────────────
   226→    if ($action === 'bulk_approve') {
   227→        $ids = $_POST['selected_ids'] ?? [];
   228→        if (!empty($ids) && is_array($ids)) {
   229→            $APPROVAL_FIELDS = [
   230→                'full_name', 'father_name', 'date_of_birth', 'gender',
   231→                'designation', 'department',
   232→            ];
   233→            $count = 0;
   234→            $errors = 0;
   235→
   236→            foreach ($ids as $id) {
   237→                $id = (int) $id;
   238→                if ($id <= 0) continue;
   239→
   240→                $request = $db->fetch(
   241→                    "SELECT r.* FROM employee_change_requests r
   242→                     WHERE r.id = :id AND r.status = 'pending'
   243→                     LIMIT 1",
   244→                    ['id' => $id]
   245→                );
   246→
   247→                if (!$request) continue;
   248→
   249→                $field = $request['field_name'];
   250→                $newValue = $request['new_value'];
   251→
   252→                if (in_array($field, $APPROVAL_FIELDS, true)) {
   253→                    try {
   254→                        $db->update('employees', [$field => $newValue], 'id = :id', ['id' => $request['employee_id']]);
   255→                    } catch (Exception $e) {
   256→                        $errors++;
   257→                        continue;
   258→                    }
   259→                }
   260→
   261→                $db->update(
   262→                    'employee_change_requests',
   263→                    [
   264→                        'status'      => 'approved',
   265→                        'reviewed_at' => date('Y-m-d H:i:s'),
   266→                        'reviewed_by' => $_SESSION['user_id'] ?? null,
   267→                    ],
   268→                    'id = :id',
   269→                    ['id' => $id]
   270→                );
   271→                $count++;
   272→            }
   273→
   274→            $msg = "{$count} change request(s) approved.";
   275→            if ($errors > 0) $msg .= " ({$errors} error(s))";
   276→            setFlash('success', $msg);
   277→        } else {
   278→            setFlash('warning', 'No requests selected.');
   279→        }
   280→        redirect('index.php?page=employee/change-requests');
   281→    }
   282→}

   285→
   286→$fStatus = sanitize($_GET['status'] ?? 'all');
   287→$fSearch = sanitize(trim($_GET['search'] ?? ''));
   288→
   289→// ─── Build WHERE clause ─────────────────────────────────────────────────────
   290→
   291→$where  = ['1=1'];
   292→$params = [];
   293→
   294→if ($fStatus !== 'all' && in_array($fStatus, ['pending', 'approved', 'rejected'], true)) {
   295→    $where[]  = 'r.status = :status';
   296→    $params['status'] = $fStatus;
   297→}
   298→
   299→if ($fSearch !== '') {
   300→    $where[] = '(e.full_name LIKE :search1 OR e.employee_code LIKE :search2 OR r.field_name LIKE :search3)';
   301→    $params['search1'] = "%{$fSearch}%";
   302→    $params['search2'] = "%{$fSearch}%";
   303→    $params['search3'] = "%{$fSearch}%";
   304→}
   305→
   306→$whereSql = implode(' AND ', $where);
   307→
   308→// ─── Fetch requests ─────────────────────────────────────────────────────────
   309→
   310→$requests = $db->fetchAll(
   311→    "SELECT r.*,
   312→            e.full_name AS emp_name, e.employee_code, e.mobile_number, e.email,
   313→            e.designation AS emp_designation, e.client_id,
   314→            c.name AS client_name, u.name AS unit_name,
   315→            reviewer.full_name AS reviewed_by_name
   316→     FROM employee_change_requests r
   317→     JOIN employees e ON e.id = r.employee_id
   318→     LEFT JOIN clients c ON c.id = e.client_id
   319→     LEFT JOIN units u   ON u.id = e.unit_id
   320→     LEFT JOIN employees reviewer ON reviewer.id = r.reviewed_by
   321→     WHERE {$whereSql}
   322→     ORDER BY
   323→        CASE WHEN r.status = 'pending' THEN 0 ELSE 1 END,
   324→        r.created_at DESC",
   325→    $params
   326→);
   327→
   328→// ─── Summary counts ──────────────────────────────────────────────────────────
   329→
   330→$pendingCount = 0;
   331→$approvedCount = 0;
   332→$rejectedCount = 0;
   333→foreach ($requests as $r) {
   334→    switch ($r['status']) {
   335→        case 'pending':  $pendingCount++;  break;
   336→        case 'approved': $approvedCount++; break;
   337→        case 'rejected': $rejectedCount++; break;
   338→    }
   339→}

   348→    'designation'   => 'Designation',
   349→    'department'    => 'Department',
   350→];

   401→                    </div>

   575→<script>
   576→(function() {
   577→    // Select all checkbox
   578→    const selectAll = document.getElementById('selectAll');
   579→    if (selectAll) {
   580→        selectAll.addEventListener('change', function() {
   581→            document.querySelectorAll('.row-checkbox').forEach(cb => cb.checked = this.checked);
   582→            updateSelectedCount();
   583→        });
   584→    }
   585→
   586→    // Individual checkboxes
   587→    document.querySelectorAll('.row-checkbox').forEach(cb => {
   588→        cb.addEventListener('change', updateSelectedCount);
   589→    });
   590→
   591→    function updateSelectedCount() {
   592→        const count = document.querySelectorAll('.row-checkbox:checked').length;
   593→        const el = document.getElementById('selectedCount');
   594→        if (el) el.textContent = count > 0 ? count + ' selected' : '';
   595→    }
   596→
   597→    // Approve single request
   598→    window.approveRequest = function(id) {
   599→        if (!confirm('Are you sure you want to approve this change request? The employee record will be updated.')) return;
   600→
   601→        const form = document.createElement('form');
   602→        form.method = 'POST';
   603→        form.action = '?page=employee/change-requests<?= $fStatus !== 'all' ? '&status=' . $fStatus : '' ?>';
   604→
   605→        form.appendChild(createInput('action', 'approve'));
   606→        form.appendChild(createInput('id', id));
   607→        form.appendChild(createInput('csrf_token', '<?= htmlspecialchars(getCSRFToken()) ?>'));
   608→
   609→        document.body.appendChild(form);
   610→        form.submit();
   611→    };
   612→
   613→    // Show reject modal
   614→    window.showRejectModal = function(id) {
   615→        document.getElementById('rejectId').value = id;
   616→        document.getElementById('rejection_reason').value = '';
   617→        new bootstrap.Modal(document.getElementById('rejectModal')).show();
   618→    };
   619→
   620→    // Bulk approve
   621→    window.bulkApprove = function() {
   622→        const checked = document.querySelectorAll('.row-checkbox:checked');
   623→        if (checked.length === 0) {
   624→            alert('Please select at least one request.');
   625→            return;
   626→        }
   627→        if (!confirm('Approve ' + checked.length + ' selected change request(s)? Employee records will be updated.')) return;
   628→        document.getElementById('bulkForm').submit();
   629→    };
   630→
   631→    function createInput(name, value) {
   632→        const input = document.createElement('input');
   633→        input.type = 'hidden';
   634→        input.name = name;
   635→        input.value = value;
   636→        return input;
   637→    }
   638→})();
   639→</script>
