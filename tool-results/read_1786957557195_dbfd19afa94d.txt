     1→<?php
     2→if (!isset($db) || !is_object($db)) { header("Location: index.php"); exit; }
     3→
     4→/**
     5→ * RCS HRMS Pro — Employee Change Requests Approval Page
     6→ *
     7→ * Allows HR/Admin to view, approve, or reject change requests
     8→ * submitted by employees via the ESS mobile app.
     9→ *
    10→ * Approve: updates the employee record with the new value.
    11→ * Reject: stores rejection reason; employee sees it in ESS.
    12→ * Both actions send email + WhatsApp notification to the employee.
    13→ */
    14→
    15→$pageTitle = 'Change Request Approvals';
    16→
    17→// ─── Ensure table exists ──────────────────────────────────────────────────────
    18→try {
    19→    $db->exec("CREATE TABLE IF NOT EXISTS employee_change_requests (
    20→        id INT AUTO_INCREMENT PRIMARY KEY,
    21→        employee_id INT NOT NULL,
    22→        field_name VARCHAR(100) NOT NULL,
    23→        old_value TEXT,
    24→        new_value TEXT NOT NULL,
    25→        reason TEXT,
    26→        status ENUM('pending','approved','rejected') NOT NULL DEFAULT 'pending',
    27→        created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    28→        reviewed_at DATETIME NULL,
    29→        reviewed_by INT NULL,
    30→        rejection_reason TEXT NULL,
    31→        INDEX idx_employee_status (employee_id, status),
    32→        INDEX idx_field_pending (employee_id, field_name, status)
    33→    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4");
    34→} catch (Exception $e) {
    35→    error_log('[change-requests] Table create error: ' . $e->getMessage());
    36→}
    37→
    38→// ─── POST Actions ────────────────────────────────────────────────────────────
    39→
    40→if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
    41→    // CSRF check
    42→    if (!validateCSRFToken($_POST['csrf_token'] ?? '')) {
    43→        setFlash('error', 'Invalid request. Please refresh the page and try again.');
    44→        redirect($_SERVER['REQUEST_URI'] ?? 'index.php?page=employee/change-requests');
    45→    }
    46→    $action = sanitize($_POST['action']);
    47→
    48→    // ── Whitelist of fields that can be updated on approval ──
    49→    $APPROVAL_FIELDS = [
    50→        'full_name', 'father_name', 'date_of_birth', 'gender',
    51→        'designation', 'department', 'profile_pic_url',
    52→    ];
    53→
    54→    // ── 1. Approve ─────────────────────────────────────────────
    55→    if ($action === 'approve') {
    56→        $id = (int) ($_POST['id'] ?? 0);
    57→        if ($id > 0) {
    58→            $request = $db->fetch(
    59→                "SELECT r.*, e.full_name AS emp_name, e.email, e.mobile_number,
    60→                        e.employee_code, e.designation
    61→                 FROM employee_change_requests r
    62→                 JOIN employees e ON e.id = r.employee_id
    63→                 WHERE r.id = :id AND r.status = 'pending'
    64→                 LIMIT 1",
    65→                ['id' => $id]
    66→            );
    67→
    68→            if ($request) {
    69→                $field = $request['field_name'];
    70→                $newValue = $request['new_value'];
    71→
    72→                // Apply the change to employee record if it's a whitelisted field
    73→                if (in_array($field, $APPROVAL_FIELDS, true)) {
    74→                    try {
    75→                        $db->update(
    76→                            'employees',
    77→                            [$field => $newValue],
    78→                            'id = :id',
    79→                            ['id' => $request['employee_id']]
    80→                        );
    81→                    } catch (Exception $e) {
    82→                        setFlash('error', 'Failed to update employee record: ' . $e->getMessage());
    83→                        redirect('index.php?page=employee/change-requests');
    84→                    }
    85→                }
    86→
    87→                // Mark request as approved
    88→                $db->update(
    89→                    'employee_change_requests',
    90→                    [
    91→                        'status'      => 'approved',
    92→                        'reviewed_at' => date('Y-m-d H:i:s'),
    93→                        'reviewed_by' => $_SESSION['user_id'] ?? null,
    94→                    ],
    95→                    'id = :id',
    96→                    ['id' => $id]
    97→                );
    98→
    99→                // ── Notification to employee ──
   100→                try {
   101→                    require_once __DIR__ . '/../../includes/class.notification.php';
   102→                    $notif = new Notification();
   103→
   104→                    // Email
   105→                    $empEmail = $request['email'];
   106→                    if (!empty($empEmail)) {
   107→                        $fieldLabel = ucfirst(str_replace('_', ' ', $field));
   108→                        $htmlBody = "
   109→                        <div style='font-family:Arial,sans-serif;max-width:600px;margin:0 auto;padding:20px;'>
   110→                            <div style='background:#10b981;padding:20px;text-align:center;'>
   111→                                <h2 style='color:#fff;margin:0;'>RCS True Facilities Pvt Ltd</h2>
   112→                            </div>
   113→                            <div style='padding:25px;background:#f9fafb;border:1px solid #e5e7eb;'>
   114→                                <p>Dear {$request['emp_name']},</p>
   115→                                <p>Your change request has been <strong style='color:#10b981;'>APPROVED</strong>.</p>
   116→                                <table style='width:100%;border-collapse:collapse;margin:15px 0;'>
   117→                                    <tr><td style='padding:8px;border:1px solid #e5e7eb;background:#f3f4f6;font-weight:bold;'>Field</td>
   118→                                        <td style='padding:8px;border:1px solid #e5e7eb;'>{$fieldLabel}</td></tr>
   119→                                    <tr><td style='padding:8px;border:1px solid #e5e7eb;background:#f3f4f6;font-weight:bold;'>Old Value</td>
   120→                                        <td style='padding:8px;border:1px solid #e5e7eb;'>" . htmlspecialchars($request['old_value']) . "</td></tr>
   121→                                    <tr><td style='padding:8px;border:1px solid #e5e7eb;background:#f3f4f6;font-weight:bold;'>New Value</td>
   122→                                        <td style='padding:8px;border:1px solid #e5e7eb;'>" . htmlspecialchars($newValue) . "</td></tr>
   123→                                </table>
   124→                                <p>If you did not request this change, please contact HR immediately.</p>
   125→                            </div>
   126→                            <p style='text-align:center;color:#9ca3af;font-size:12px;'>This is an automated message from RCS HRMS Pro.</p>

   129→                    }

   131→                    // WhatsApp
   132→                    $empMobile = $request['mobile_number'];
   133→                    if (!empty($empMobile)) {
   134→                        $fieldLabel = ucfirst(str_replace('_', ' ', $field));
   135→                        $waMsg = "Hello {$request['emp_name']},\n\nYour change request has been APPROVED.\n\nField: {$fieldLabel}\nNew Value: {$newValue}\n\n- RCS HRMS Pro";
   136→                        $notif->sendWhatsApp($empMobile, $waMsg);
   137→                    }
   138→                } catch (Exception $e) {
   139→                    // Log but don't fail the approval
   140→                    error_log('[Change Request Approval] Notification error: ' . $e->getMessage());
   141→                }
   142→
   143→                // Audit log
   144→                if (function_exists('logActivity')) {
   145→                    logActivity('approve_change_request', 'employee', $request['employee_id'],
   146→                        "Approved change request #{$id}: {$field} = " . substr($newValue, 0, 50));
   147→                }
   148→
   149→                setFlash('success', 'Change request approved. Employee has been notified.');
   150→            } else {
   151→                setFlash('error', 'Request not found or already processed.');
   152→            }
   153→        }
   154→        redirect('index.php?page=employee/change-requests');
   155→    }
   156→
   157→    // ── 2. Reject ──────────────────────────────────────────────
   158→    if ($action === 'reject') {
   159→        $id     = (int) ($_POST['id'] ?? 0);
   160→        $reason = sanitize(trim($_POST['rejection_reason'] ?? ''));
   161→
   162→        if ($id > 0 && $reason !== '') {
   163→            $request = $db->fetch(
   164→                "SELECT r.*, e.full_name AS emp_name, e.email, e.mobile_number,
   165→                        e.employee_code, e.designation
   166→                 FROM employee_change_requests r
   167→                 JOIN employees e ON e.id = r.employee_id
   168→                 WHERE r.id = :id AND r.status = 'pending'
   169→                 LIMIT 1",
   170→                ['id' => $id]
   171→            );
   172→
   173→            if ($request) {
   174→                $field = $request['field_name'];
   175→
   176→                // Mark request as rejected
   177→                $db->update(
   178→                    'employee_change_requests',
   179→                    [
   180→                        'status'           => 'rejected',
   181→                        'reviewed_at'      => date('Y-m-d H:i:s'),
   182→                        'reviewed_by'      => $_SESSION['user_id'] ?? null,
   183→                        'rejection_reason' => $reason,
   184→                    ],
   185→                    'id = :id',
   186→                    ['id' => $id]
   187→                );
   188→
   189→                // ── Notification to employee ──
   190→                try {
   191→                    require_once __DIR__ . '/../../includes/class.notification.php';
   192→                    $notif = new Notification();
   193→
   194→                    $empEmail = $request['email'];
   195→                    if (!empty($empEmail)) {
   196→                        $fieldLabel = ucfirst(str_replace('_', ' ', $field));
   197→                        $htmlBody = "
   198→                        <div style='font-family:Arial,sans-serif;max-width:600px;margin:0 auto;padding:20px;'>
   199→                            <div style='background:#ef4444;padding:20px;text-align:center;'>
   200→                                <h2 style='color:#fff;margin:0;'>RCS True Facilities Pvt Ltd</h2>
   201→                            </div>
   202→                            <div style='padding:25px;background:#f9fafb;border:1px solid #e5e7eb;'>
   203→                                <p>Dear {$request['emp_name']},</p>
   204→                                <p>Your change request has been <strong style='color:#ef4444;'>REJECTED</strong>.</p>
   205→                                <table style='width:100%;border-collapse:collapse;margin:15px 0;'>
   206→                                    <tr><td style='padding:8px;border:1px solid #e5e7eb;background:#f3f4f6;font-weight:bold;'>Field</td>
   207→                                        <td style='padding:8px;border:1px solid #e5e7eb;'>{$fieldLabel}</td></tr>
   208→                                    <tr><td style='padding:8px;border:1px solid #e5e7eb;background:#f3f4f6;font-weight:bold;'>Requested Value</td>
   209→                                        <td style='padding:8px;border:1px solid #e5e7eb;'>" . htmlspecialchars($request['new_value']) . "</td></tr>
   210→                                    <tr><td style='padding:8px;border:1px solid #e5e7eb;background:#f3f4f6;font-weight:bold;'>Rejection Reason</td>
   211→                                        <td style='padding:8px;border:1px solid #e5e7eb;'>" . htmlspecialchars($reason) . "</td></tr>
   212→                                </table>
   213→                                <p>If you believe this was an error, please contact HR.</p>
   214→                            </div>
   215→                            <p style='text-align:center;color:#9ca3af;font-size:12px;'>This is an automated message from RCS HRMS Pro.</p>
   216→                        </div>";
   217→                        $notif->sendEmail($empEmail, "Change Request Rejected — {$fieldLabel}", $htmlBody);
   218→                    }
   219→
   220→                    // WhatsApp
   221→                    $empMobile = $request['mobile_number'];
   222→                    if (!empty($empMobile)) {
   223→                        $fieldLabel = ucfirst(str_replace('_', ' ', $field));
   224→                        $waMsg = "Hello {$request['emp_name']},\n\nYour change request has been REJECTED.\n\nField: {$fieldLabel}\nReason: {$reason}\n\n- RCS HRMS Pro";
   225→                        $notif->sendWhatsApp($empMobile, $waMsg);
   226→                    }
   227→                } catch (Exception $e) {
   228→                    error_log('[Change Request Rejection] Notification error: ' . $e->getMessage());
   229→                }
   230→
   231→                if (function_exists('logActivity')) {
   232→                    logActivity('reject_change_request', 'employee', $request['employee_id'],
   233→                        "Rejected change request #{$id}: {$field} — " . substr($reason, 0, 50));
   234→                }
   235→
   236→                setFlash('success', 'Change request rejected. Employee has been notified.');
   237→            } else {
   238→                setFlash('error', 'Request not found or already processed.');
   239→            }
   240→        } elseif ($id > 0 && $reason === '') {
   241→            setFlash('danger', 'Rejection reason is required.');
   242→        }
   243→        redirect('index.php?page=employee/change-requests');
   244→    }
   245→
   246→    // ── 3. Bulk approve ───────────────────────────────────────
   247→    if ($action === 'bulk_approve') {
   248→        $ids = $_POST['selected_ids'] ?? [];
   249→        if (!empty($ids) && is_array($ids)) {
   250→            $APPROVAL_FIELDS = [
   251→                'full_name', 'father_name', 'date_of_birth', 'gender',
   252→                'designation', 'department', 'profile_pic_url',
   253→            ];
   254→            $count = 0;
   255→            $errors = 0;
   256→
   257→            foreach ($ids as $id) {
   258→                $id = (int) $id;
   259→                if ($id <= 0) continue;
   260→
   261→                $request = $db->fetch(
   262→                    "SELECT r.* FROM employee_change_requests r
   263→                     WHERE r.id = :id AND r.status = 'pending'
   264→                     LIMIT 1",
   265→                    ['id' => $id]
   266→                );
   267→
   268→                if (!$request) continue;
   269→
   270→                $field = $request['field_name'];
   271→                $newValue = $request['new_value'];
   272→
   273→                if (in_array($field, $APPROVAL_FIELDS, true)) {
   274→                    try {
   275→                        $db->update('employees', [$field => $newValue], 'id = :id', ['id' => $request['employee_id']]);
   276→                    } catch (Exception $e) {
   277→                        $errors++;
   278→                        continue;
   279→                    }
   280→                }
   281→
   282→                $db->update(
   283→                    'employee_change_requests',
   284→                    [
   285→                        'status'      => 'approved',
   286→                        'reviewed_at' => date('Y-m-d H:i:s'),
   287→                        'reviewed_by' => $_SESSION['user_id'] ?? null,
   288→                    ],
   289→                    'id = :id',
   290→                    ['id' => $id]
   291→                );
   292→                $count++;
   293→            }
   294→
   295→            $msg = "{$count} change request(s) approved.";
   296→            if ($errors > 0) $msg .= " ({$errors} error(s))";
   297→            setFlash('success', $msg);
   298→        } else {
   299→            setFlash('warning', 'No requests selected.');
   300→        }
   301→        redirect('index.php?page=employee/change-requests');
   302→    }
   303→}
   304→
   305→// ─── GET Filters ────────────────────────────────────────────────────────────
   306→
   307→$fStatus = sanitize($_GET['status'] ?? 'all');
   308→$fSearch = sanitize(trim($_GET['search'] ?? ''));
   309→
   310→// ─── Build WHERE clause ─────────────────────────────────────────────────────
   311→
   312→$where  = ['1=1'];
   313→$params = [];
   314→
   315→if ($fStatus !== 'all' && in_array($fStatus, ['pending', 'approved', 'rejected'], true)) {
   316→    $where[]  = 'r.status = :status';
   317→    $params['status'] = $fStatus;
   318→}
   319→
   320→if ($fSearch !== '') {
   321→    $where[] = '(e.full_name LIKE :search1 OR e.employee_code LIKE :search2 OR r.field_name LIKE :search3)';
   322→    $params['search1'] = "%{$fSearch}%";
   323→    $params['search2'] = "%{$fSearch}%";
   324→    $params['search3'] = "%{$fSearch}%";
   325→}
   326→
   327→$whereSql = implode(' AND ', $where);
   328→
   329→// ─── Fetch requests ─────────────────────────────────────────────────────────
   330→
   331→// Fetch requests — try full query with JOINs, fallback to simple query
   332→$requests = [];
   333→try {
   334→    $requests = $db->fetchAll(
   335→        "SELECT r.*,
   336→                e.full_name AS emp_name, e.employee_code, e.mobile_number, e.email,
   337→                e.designation AS emp_designation, e.client_id,
   338→                c.name AS client_name, u.name AS unit_name,
   339→                reviewer.full_name AS reviewed_by_name
   340→         FROM employee_change_requests r
   341→         JOIN employees e ON e.id = r.employee_id
   342→         LEFT JOIN clients c ON c.id = e.client_id
   343→         LEFT JOIN units u   ON u.id = e.unit_id
   344→         LEFT JOIN employees reviewer ON reviewer.id = r.reviewed_by
   345→         WHERE {$whereSql}
   346→         ORDER BY
   347→            CASE WHEN r.status = 'pending' THEN 0 ELSE 1 END,
   348→            r.created_at DESC",
   349→        $params
   350→    );
   351→} catch (Exception $e) {
   352→    error_log('[change-requests] Full query failed: ' . $e->getMessage());
   353→    // Fallback: query without JOINs to clients/units tables (they may not exist)
   354→    try {
   355→        $requests = $db->fetchAll(
   356→            "SELECT r.*,
   357→                    e.full_name AS emp_name, e.employee_code, e.mobile_number, e.email,
   358→                    e.designation AS emp_designation, e.client_id,
   359→                    NULL AS client_name, NULL AS unit_name,
   360→                    reviewer.full_name AS reviewed_by_name
   361→             FROM employee_change_requests r
   362→             JOIN employees e ON e.id = r.employee_id
   363→             LEFT JOIN employees reviewer ON reviewer.id = r.reviewed_by
   364→             WHERE {$whereSql}
   365→             ORDER BY
   366→                CASE WHEN r.status = 'pending' THEN 0 ELSE 1 END,
   367→                r.created_at DESC",
   368→            $params
   369→        );
   370→    } catch (Exception $e2) {
   371→        error_log('[change-requests] Fallback query also failed: ' . $e2->getMessage());
   372→        $requests = [];
   373→    }
   374→}

   377→
   378→$pendingCount = 0;

   381→foreach ($requests as $r) {
   382→    switch ($r['status']) {
   383→        case 'pending':  $pendingCount++;  break;
   384→        case 'approved': $approvedCount++; break;
   385→        case 'rejected': $rejectedCount++; break;
   386→    }
   387→}

   634→<script>
   635→(function() {
   636→    // Select all checkbox
   637→    const selectAll = document.getElementById('selectAll');
   638→    if (selectAll) {
   639→        selectAll.addEventListener('change', function() {
   640→            document.querySelectorAll('.row-checkbox').forEach(cb => cb.checked = this.checked);
   641→            updateSelectedCount();
   642→        });
   643→    }
   644→
   645→    // Individual checkboxes
   646→    document.querySelectorAll('.row-checkbox').forEach(cb => {
   647→        cb.addEventListener('change', updateSelectedCount);
   648→    });
   649→
   650→    function updateSelectedCount() {
   651→        const count = document.querySelectorAll('.row-checkbox:checked').length;
   652→        const el = document.getElementById('selectedCount');
   653→        if (el) el.textContent = count > 0 ? count + ' selected' : '';
   654→    }
   655→
   656→    // Approve single request
   657→    window.approveRequest = function(id) {
   658→        if (!confirm('Are you sure you want to approve this change request? The employee record will be updated.')) return;
   659→
   660→        const form = document.createElement('form');
   661→        form.method = 'POST';
   662→        form.action = '?page=employee/change-requests<?= $fStatus !== 'all' ? '&status=' . $fStatus : '' ?>';
   663→
   664→        form.appendChild(createInput('action', 'approve'));
   665→        form.appendChild(createInput('id', id));
   666→        form.appendChild(createInput('csrf_token', '<?= htmlspecialchars(getCSRFToken()) ?>'));
   667→
   668→        document.body.appendChild(form);
   669→        form.submit();
   670→    };
   671→
   672→    // Show reject modal
   673→    window.showRejectModal = function(id) {
   674→        document.getElementById('rejectId').value = id;
   675→        document.getElementById('rejection_reason').value = '';
   676→        new bootstrap.Modal(document.getElementById('rejectModal')).show();
   677→    };
   678→
   679→    // Bulk approve
   680→    window.bulkApprove = function() {
   681→        const checked = document.querySelectorAll('.row-checkbox:checked');
   682→        if (checked.length === 0) {
   683→            alert('Please select at least one request.');
   684→            return;
   685→        }
   686→        if (!confirm('Approve ' + checked.length + ' selected change request(s)? Employee records will be updated.')) return;
   687→        document.getElementById('bulkForm').submit();
   688→    };
   689→
   690→    function createInput(name, value) {
   691→        const input = document.createElement('input');
   692→        input.type = 'hidden';
   693→        input.name = name;
   694→        input.value = value;
   695→        return input;
   696→    }
   697→})();
   698→</script>
