   289→    // Get compliance calendar
   290→    public function getComplianceCalendar($month = null, $year = null) {
   291→        $sql = "SELECT cc.*, s.state_name,
   292→                CASE
   293→                    WHEN cc.due_date < CURDATE() THEN 'Overdue'
   294→                    WHEN cc.due_date <= DATE_ADD(CURDATE(), INTERVAL 7 DAY) THEN 'Due Soon'
   295→                    ELSE 'Upcoming'
   296→                END as urgency
   297→                FROM compliance_calendar cc
   298→                LEFT JOIN states s ON cc.state_id = s.id
   299→                WHERE cc.is_active = 1";
   300→        $params = [];
   301→
   302→        if ($month && $year) {
   303→            $sql .= " AND MONTH(cc.due_date) = :month AND YEAR(cc.due_date) = :year";
   304→            $params['month'] = $month;
   305→            $params['year'] = $year;
   306→        } elseif ($year) {
   307→            $sql .= " AND YEAR(cc.due_date) = :year";
   308→            $params['year'] = $year;
   309→        }
   310→
   311→        $sql .= " ORDER BY cc.due_date ASC";
   312→
   313→        return $this->db->fetchAll($sql, $params);

   419→    // Calculate PT for given salary and state
   420→    public function calculatePT($grossSalary, $stateId, $gender = 'All') {
   421→        $rate = $this->db->fetch(
   422→            "SELECT pt_amount FROM professional_tax_rates
   423→             WHERE state_id = :state_id
   424→             AND is_active = 1
   425→             AND effective_from <= CURDATE()
   426→             AND salary_from <= :gross
   427→             AND (salary_to IS NULL OR salary_to >= :gross)
   428→             AND (gender_specific = 'All' OR gender_specific = :gender)
   429→             ORDER BY effective_from DESC, salary_from DESC
   430→             LIMIT 1",
   431→            ['state_id' => $stateId, 'gross' => $grossSalary, 'gender' => $gender]
   432→        );
   433→
   434→        return $rate ? (float)$rate['pt_amount'] : 0;
   435→    }

   497→    // Get notifications
   498→    public function getNotifications($limit = 10) {
   499→        $notifications = [];
   500→
   501→        try {
   502→            // Check for pending filings
   503→            $pending = $this->db->fetchAll(
   504→                "SELECT * FROM compliance_filings
   505→                 WHERE status = 'Pending'
   506→                 ORDER BY due_date ASC
   507→                 LIMIT :limit",
   508→                ['limit' => $limit]
   509→            );
   510→
   511→            foreach ($pending as $p) {
   512→                $isOverdue = strtotime($p['due_date']) < time();
   513→                $notifications[] = [
   514→                    'type' => $isOverdue ? 'danger' : 'warning',
   515→                    'title' => $p['compliance_type'] . ' Filing ' . ($isOverdue ? 'Overdue' : 'Pending'),
   516→                    'description' => sprintf(
   517→                        '%s return for %s/%s is %s',
   518→                        $p['compliance_type'],
   519→                        str_pad($p['filing_period_month'], 2, '0', STR_PAD_LEFT),
   520→                        $p['filing_period_year'],
   521→                        $isOverdue ? 'overdue' : 'pending'
   522→                    ),
   523→                    'created_at' => $p['due_date']
   524→                ];
   525→            }
   526→
   527→            // Check for wage updates
   528→            $wageUpdates = $this->db->fetchAll(
   529→                "SELECT s.state_name, MAX(mw.effective_from) as last_update
   530→                 FROM minimum_wages mw
   531→                 JOIN states s ON mw.state_id = s.id
   532→                 WHERE mw.is_active = 1
   533→                 GROUP BY s.id
   534→                 HAVING last_update < DATE_SUB(CURDATE(), INTERVAL 6 MONTH)
   535→                 LIMIT 5"
   536→            );
   537→
   538→            foreach ($wageUpdates as $w) {
   539→                $notifications[] = [
   540→                    'type' => 'info',
   541→                    'title' => 'Minimum Wage Update Needed',
   542→                    'description' => $w['state_name'] . ' minimum wages may need updating. Last update: ' . $w['last_update'],
   543→                    'created_at' => date('Y-m-d')
   544→                ];
   545→            }
   546→
   547→        } catch (Exception $e) {
   548→            error_log('Compliance getNotifications error: ' . $e->getMessage());
   549→        }
   550→
   551→        return array_slice($notifications, 0, $limit);
   552→    }

   554→    // Get monthly summary
   555→    public function getMonthlySummary($month, $year) {
   556→        $summary = [
   557→            'pf_employee' => 0,
   558→            'pf_employer' => 0,
   559→            'eps_employer' => 0,
   560→            'esi_employee' => 0,
   561→            'esi_employer' => 0,
   562→            'professional_tax' => 0,
   563→            'lwf_employee' => 0,
   564→            'lwf_employer' => 0
   565→        ];
   566→
   567→        try {
   568→            $period = $this->db->fetch(
   569→                SQL_GET_PAYROLL_PERIOD,
   570→                ['month' => $month, 'year' => $year]
   571→            );
   572→
   573→            if ($period) {
   574→                $totals = $this->db->fetch(
   575→                    "SELECT
   576→                        SUM(pf_employee) as pf_employee,
   577→                        SUM(pf_employer) as pf_employer,
   578→                        SUM(eps_employer) as eps_employer,
   579→                        SUM(esi_employee) as esi_employee,
   580→                        SUM(esi_employer) as esi_employer,
   581→                        SUM(professional_tax) as professional_tax,
   582→                        SUM(lwf_employee) as lwf_employee,
   583→                        SUM(lwf_employer) as lwf_employer
   584→                     FROM payroll
   585→                     WHERE payroll_period_id = :period_id",
   586→                    ['period_id' => $period['id']]
   587→                );
   588→
   589→                if ($totals) {
   590→                    $summary = array_merge($summary, $totals);

   656→    public function generateFormXVI($unitId, $month, $year) {
   657→        // Try to get attendance data
   658→        try {
   659→            return $this->db->fetchAll(
   660→                "SELECT e.employee_code, e.full_name, e.father_name, e.designation, e.worker_category,
   661→                        COALESCE(a.present_days, 0) as present_days,
   662→                        COALESCE(a.absent_days, 0) as absent_days,
   663→                        COALESCE(a.weekly_offs, 0) as weekly_offs,
   664→                        COALESCE(a.holidays, 0) as holidays,
   665→                        COALESCE(a.total_working_days, 0) as total_working_days,
   666→                        COALESCE(a.overtime_hours, 0) as total_overtime_hours
   667→                 FROM employees e
   668→                 LEFT JOIN attendance a ON e.id = a.employee_id AND MONTH(a.attendance_date) = :month AND YEAR(a.attendance_date) = :year
   669→                 WHERE e.unit_id = :unit_id
   670→                 AND e.status = 'approved'
   671→                 ORDER BY e.employee_code",
   672→                ['unit_id' => $unitId, 'month' => $month, 'year' => $year]
   673→            );
   674→        } catch (Exception $e) {
   675→            // Return just employees if attendance table doesn't exist
   676→            return $this->db->fetchAll(
   677→                "SELECT e.employee_code, e.full_name, e.father_name, e.designation, e.worker_category,
   678→                        0 as present_days, 0 as absent_days, 0 as weekly_offs, 0 as holidays, 0 as total_working_days, 0 as total_overtime_hours
   679→                 FROM employees e
   680→                 WHERE e.unit_id = :unit_id
   681→                 AND e.status = 'approved'
   682→                 ORDER BY e.employee_code",
   683→                ['unit_id' => $unitId]
   684→            );
   685→        }
   686→    }
