   211→function categoryBadge($cat): string

   254→function filterUrl($baseUrl, $override = []): string
   255→{
   256→    $params = $_GET;
   257→    foreach ($override as $k => $v) {
   258→        if ($v === '' || $v === 'all') {
   259→            unset($params[$k]);

   261→            $params[$k] = $v;
   262→        }
   263→    }
   264→    return $baseUrl . (empty($params) ? '' : '&' . http_build_query($params));

   617→document.addEventListener('DOMContentLoaded', function () {
   618→
   619→    const allCheckboxes    = document.querySelectorAll('.row-checkbox');
   620→    const selectAllHeader  = document.getElementById('selectAllHeader');
   621→    const selectAllBelow   = document.getElementById('selectAll');
   622→    const selectedCountEl  = document.getElementById('selectedCount');
   623→    const bulkApproveBtn   = document.getElementById('bulkApproveBtn');
   624→    const selectedIdsInput = document.getElementById('selectedIdsInput');
   625→    const bulkApproveForm  = document.getElementById('bulkApproveForm');
   626→
   627→    function updateSelectedCount() {
   628→        const checked = document.querySelectorAll('.row-checkbox:checked');
   629→        const count   = checked.length;
   630→        selectedCountEl.textContent = count + ' selected';
   631→        bulkApproveBtn.disabled = (count === 0);
   632→        const ids = Array.from(checked).map(cb => cb.value);
   633→        selectedIdsInput.value = ids.join(',');
   634→    }
   635→
   636→    function syncSelectAll() {
   637→        const total  = allCheckboxes.length;
   638→        const checked = document.querySelectorAll('.row-checkbox:checked').length;
   639→        const state  = (checked === total && total > 0);
   640→        if (selectAllHeader) selectAllHeader.checked = state;
   641→        if (selectAllBelow)  selectAllBelow.checked  = state;
   642→    }
   643→
   644→    if (selectAllHeader) {
   645→        selectAllHeader.addEventListener('change', function () {
   646→            allCheckboxes.forEach(cb => { cb.checked = this.checked; });
   647→            syncSelectAll();
   648→            updateSelectedCount();
   649→        });
   650→    }
   651→
   652→    if (selectAllBelow) {
   653→        selectAllBelow.addEventListener('change', function () {
   654→            allCheckboxes.forEach(cb => { cb.checked = this.checked; });
   655→            syncSelectAll();
   656→            updateSelectedCount();
   657→        });
   658→    }
   659→
   660→    allCheckboxes.forEach(cb => {
   661→        cb.addEventListener('change', function () {
   662→            syncSelectAll();
   663→            updateSelectedCount();
   664→        });
   665→    });
   666→
   667→    if (bulkApproveBtn) {
   668→        bulkApproveBtn.addEventListener('click', function () {
   669→            const count = document.querySelectorAll('.row-checkbox:checked').length;
   670→            if (count === 0) return;
   671→            if (!confirm('Approve ' + count + ' selected expense(s)?')) return;
   672→            const existing = bulkApproveForm.querySelectorAll('input[name="selected_ids[]"]');
   673→            existing.forEach(el => el.remove());
   674→            document.querySelectorAll('.row-checkbox:checked').forEach(cb => {
   675→                const input = document.createElement('input');
   676→                input.type  = 'hidden';
   677→                input.name  = 'selected_ids[]';
   678→                input.value = cb.value;
   679→                bulkApproveForm.appendChild(input);
   680→            });
   681→            selectedIdsInput.remove();
   682→            bulkApproveForm.submit();
   683→        });

   694→function quickApprove(id) {
