   183→   383→function _handleGetDashboard(mysqli $conn, string $authId): void
   184→   384→{
   185→   385→    $employeeId = (int)($_GET['employee_id'] ?? $authId);
   186→   386→    $now = new \DateTime('now', new \DateTimeZone('Asia/Kolkata'));
   187→   387→    $currentMonth = (int)$now->format('n');
   188→   388→    $currentYear = (int)$now->format('Y');
   189→   389→
   190→   390→    // Basic counts
   191→   391→    $counts = $conn->query("
   192→   392→        SELECT
   193→   393→            COUNT(*) AS total_visits,
   194→   394→            SUM(CASE WHEN status = 'submitted' THEN 1 ELSE 0 END) AS pending_visits,
   195→   395→            SUM(CASE WHEN status = 'approved' THEN 1 ELSE 0 END) AS approved_visits,
   196→   396→            SUM(CASE WHEN status = 'rejected' THEN 1 ELSE 0 END) AS rejected_visits,
   197→   397→            SUM(CASE WHEN visit_month = {$currentMonth} AND visit_year = {$currentYear} THEN 1 ELSE 0 END) AS this_month_visits
   198→   398→        FROM ess_unit_visits WHERE employee_id = {$employeeId}
   199→   399→    ")->fetch_assoc();
   200→   400→
   201→   401→    // This month avg score
   202→   402→    $monthAvg = $conn->query("
   203→   403→        SELECT AVG(score_percent) AS avg_score
   204→   404→        FROM ess_unit_visits
   205→   405→        WHERE employee_id = {$employeeId} AND visit_month = {$currentMonth} AND visit_year = {$currentYear}
   206→   406→            AND status IN ('approved','submitted') AND max_score > 0
   207→   407→    ")->fetch_assoc();
   208→   408→
   209→   409→    // Score trend - last 6 months
   210→   410→    $trend = [];
   211→   411→    for ($i = 5; $i >= 0; $i--) {
   212→   412→        $d = clone $now;
   213→   413→        $d->modify("-{$i} months");
   214→   414→        $m = (int)$d->format('n');
   215→   415→        $y = (int)$d->format('Y');
   216→   416→        $label = $d->format('M Y');
   217→   417→        $avgRow = $conn->query("
   218→   418→            SELECT AVG(score_percent) AS avg_score
   219→   419→            FROM ess_unit_visits
   220→   420→            WHERE employee_id = {$employeeId} AND visit_month = {$m} AND visit_year = {$y}
   221→   421→                AND status IN ('approved','submitted') AND max_score > 0
   222→   422→        ")->fetch_assoc();
   223→   423→        $trend[] = ['month' => $label, 'avg_score' => round((float)($avgRow['avg_score'] ?? 0), 1)];
   224→   424→    }
   225→   425→
   226→   426→    // Category scores
   227→   427→    $catScores = $conn->query("
   228→   428→        SELECT cc.name AS category, AVG(
   229→   429→            CASE WHEN vci.status = 'yes' THEN ci.weight ELSE 0 END
   230→   430→        ) / NULLIF(AVG(CASE WHEN vci.status != 'na' THEN ci.weight ELSE NULL END), 0) * 100 AS avg_score
   231→   431→        FROM ess_visit_checklist_items vci
   232→   432→        JOIN ess_checklist_items ci ON ci.id = vci.checklist_item_id
   233→   433→        JOIN ess_checklist_categories cc ON cc.id = ci.category_id
   234→   434→        JOIN ess_unit_visits v ON v.id = vci.visit_id
   235→   435→        WHERE v.employee_id = {$employeeId} AND v.status IN ('approved','submitted')
   236→   436→        GROUP BY cc.id, cc.name
   237→   437→        ORDER BY cc.display_order
   238→   438→    ");
   239→   439→    $categoryScores = [];
   240→   440→    while ($cr = $catScores->fetch_assoc()) {
   241→   441→        $categoryScores[] = [
   242→   442→            'category' => $cr['category'] ?? '',
   243→   443→            'avg_score' => round((float)($cr['avg_score'] ?? 0), 1),
   244→   444→        ];
   245→   445→    }
   246→   446→
   247→   447→    // Unit compliance
   248→   448→    $unitComp = $conn->query("
   249→   449→        SELECT u.name AS unit_name, AVG(v.score_percent) AS score, COUNT(*) AS visits
   250→   450→        FROM ess_unit_visits v
   251→   451→        JOIN units u ON u.id = v.unit_id
   252→   452→        WHERE v.employee_id = {$employeeId} AND v.status IN ('approved','submitted') AND v.max_score > 0
   253→   453→        GROUP BY v.unit_id, u.name
   254→   454→        ORDER BY score DESC
   255→   455→        LIMIT 10
   256→   456→    ");
   257→   457→    $unitCompliance = [];
   258→   458→    while ($ur = $unitComp->fetch_assoc()) {
   259→   459→        $unitCompliance[] = [
   260→   460→            'unit_name' => $ur['unit_name'] ?? '',
   261→   461→            'score' => round((float)($ur['score'] ?? 0), 1),
   262→   462→            'visits' => (int)$ur['visits'],
   263→   463→        ];
   264→   464→    }
   265→   465→
   266→   466→    $conn->close();
   267→   467→
   268→   468→    jsonOutput([
   269→   469→        'success' => true,
   270→   470→        'data' => [
   271→   471→            'total_visits' => (int)($counts['total_visits'] ?? 0),
   272→   472→            'pending_visits' => (int)($counts['pending_visits'] ?? 0),
   273→   473→            'approved_visits' => (int)($counts['approved_visits'] ?? 0),
   274→   474→            'rejected_visits' => (int)($counts['rejected_visits'] ?? 0),

   468→   668→

   470→   670→{

   473→   673→    $conn = getDbConnection();
   474→   674→    _ensureTables($conn);
   475→   675→

   479→   679→    if ($visitId <= 0) jsonOutput(['success' => false, 'error' => 'Visit ID is required'], 400);

   481→   681→
   482→   682→    // Verify visit exists

   485→   685→    $checkStmt->execute();
   486→   686→    $visit = $checkStmt->get_result()->fetch_assoc();

   493→   693→        $stmt = $conn->prepare('UPDATE ess_unit_visits SET status = ?, approved_by = ?, approved_at = NOW() WHERE id = ?');

   495→   695→        $stmt->execute();
   496→   696→        $stmt->close();
   497→   697→        _logAudit($conn, $visitId, 'approved', $employeeId);
   498→   698→    } else {
   499→   699→        $rejectionReason = trim($input['rejection_reason'] ?? '');
   500→   700→        $stmt = $conn->prepare('UPDATE ess_unit_visits SET status = ?, rejection_reason = ?, approved_by = ?, approved_at = NOW() WHERE id = ?');
   501→   701→        $stmt->bind_param('sssi', 'rejected', $rejectionReason, $employeeId, $visitId);
   502→   702→        $stmt->execute();
   503→   703→        $stmt->close();
   504→   704→        _logAudit($conn, $visitId, 'rejected', $employeeId, $rejectionReason);
   505→   705→    }
   506→   706→
   507→   707→    // Fetch updated visit
   508→   708→    $fetchStmt = $conn->prepare("
   509→   709→        SELECT v.*, u.name AS unit_name, c.name AS client_name,
   510→   710→               e.full_name AS employee_name, e.employee_code
   511→   711→        FROM ess_unit_visits v
   512→   712→        LEFT JOIN units u ON u.id = v.unit_id
   513→   713→        LEFT JOIN clients c ON c.id = u.client_id
   514→   714→        LEFT JOIN employees e ON e.id = v.employee_id
   515→   715→        WHERE v.id = ?
   516→   716→    ");
   517→   717→    $fetchStmt->bind_param('i', $visitId);
   518→   718→    $fetchStmt->execute();
   519→   719→    $row = $fetchStmt->get_result()->fetch_assoc();
   520→   720→    $fetchStmt->close();
   521→   721→    $conn->close();
   522→   722→
   523→   723→    jsonOutput([
   524→   724→        'success' => true,
   525→   725→        'data' => [
   526→   726→            'id' => (int)$row['id'],
   527→   727→            'status' => $row['status'],
   528→   728→            'message' => 'Visit ' . $action . 'd successfully'

   609→   809→    $items = [];

   612→   812→    }
   613→   813→    $itemsResult->free();
   614→   814→    $itemsStmt->close();
   615→   815→
   616→   816→    $score = (int)($visit['score_percent'] ?? 0);
   617→   817→    $subject = sprintf('Unit Visit Report: %s - %s %s (%d%%)',
   618→   818→        $visit['unit_name'] ?? 'Unknown',

   620→   820→        $visit['visit_year'],
