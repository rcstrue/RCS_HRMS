    79→    79→    public function getAll($filters = [], $page = 1, $perPage = DEFAULT_PAGE_SIZE) {
    80→    80→        // Check if employee_salary_structures table exists
    81→    81→        $useSalaryTable = $this->checkSalaryTableExists();
    82→    82→        $baseColumns = $this->getBaseColumns();
    83→    83→

    85→    85→            $sql = "SELECT $baseColumns,
    86→    86→                           ess.basic_da, ess.hra, ess.leave_encashment, ess.bonus_encashment,
    87→    87→                           ess.washing_allowance, ess.gross_salary,
    88→    88→                           ess.pf_applicable, ess.esi_applicable, ess.pt_applicable,
    89→    89→                           c.name as client_name_display,
    90→    90→                           u.name as unit_name_display

    93→    93→                        SELECT employee_id, basic_da, hra, leave_encashment, bonus_encashment,
    94→    94→                               washing_allowance, gross_salary, pf_applicable, esi_applicable, pt_applicable
    95→    95→                        FROM employee_salary_structures
    96→    96→                        WHERE effective_to IS NULL OR effective_to >= CURDATE()
    97→    97→                        GROUP BY employee_id
    98→    98→                    ) ess ON e.id = ess.employee_id
    99→    99→                    LEFT JOIN clients c ON e.client_id = c.id
   100→   100→                    LEFT JOIN units u ON e.unit_id = u.id
   101→   101→                    WHERE 1=1";
   102→   102→        } else {
   103→   103→            $sql = "SELECT $baseColumns,
   104→   104→                           c.name as client_name_display,
   105→   105→                           u.name as unit_name_display
   106→   106→                    FROM employees e
   107→   107→                    LEFT JOIN clients c ON e.client_id = c.id
   108→   108→                    LEFT JOIN units u ON e.unit_id = u.id
   109→   109→                    WHERE 1=1";
   110→   110→        }
   111→   111→        $params = [];
   112→   112→
   113→   113→        // Status filter - use strtolower for case-insensitive comparison
   114→   114→        if (!empty($filters['status'])) {
   115→   115→            $statusLower = strtolower($filters['status']);
   116→   116→            if ($statusLower === 'active' || $statusLower === 'approved') {
   117→   117→                $sql .= " AND e.status = :status";
   118→   118→                $params['status'] = STATUS_APPROVED;
   119→   119→            } elseif ($statusLower === 'pending') {
   120→   120→                $sql .= " AND e.status LIKE :status";
   121→   121→                $params['status'] = STATUS_PENDING . '%';
   122→   122→            } else {
   123→   123→                $sql .= " AND e.status = :status";
   124→   124→                $params['status'] = $filters['status'];
   125→   125→            }
   126→   126→        }
   127→   127→
   128→   128→        // Client filter
   129→   129→        if (!empty($filters['client_id'])) {
   130→   130→            $sql .= " AND e.client_id = :client_id";
   131→   131→            $params['client_id'] = $filters['client_id'];
   132→   132→        }
   133→   133→
   134→   134→        // Unit filter
   135→   135→        if (!empty($filters['unit_id'])) {
   136→   136→            $sql .= " AND e.unit_id = :unit_id";
   137→   137→            $params['unit_id'] = $filters['unit_id'];
   138→   138→        }
   139→   139→
   140→   140→        // Worker category filter
   141→   141→        if (!empty($filters['worker_category'])) {
   142→   142→            $sql .= " AND e.worker_category = :worker_category";
   143→   143→            $params['worker_category'] = $filters['worker_category'];
   144→   144→        }
   145→   145→
   146→   146→        // Search filter - use unique parameter names for PDO with emulated prepares disabled
   147→   147→        if (!empty($filters['search'])) {
   148→   148→            $sql .= " AND (e.employee_code LIKE :search1
   149→   149→                      OR e.full_name LIKE :search2
   150→   150→                      OR e.mobile_number LIKE :search3
   151→   151→                      OR e.aadhaar_number LIKE :search4)";
   152→   152→            $searchTerm = '%' . $filters['search'] . '%';
   153→   153→            $params['search1'] = $searchTerm;
   154→   154→            $params['search2'] = $searchTerm;
   155→   155→            $params['search3'] = $searchTerm;
   156→   156→            $params['search4'] = $searchTerm;
   157→   157→        }
   158→   158→
   159→   159→        // Count total
   160→   160→        $countSql = "SELECT COUNT(*) as total FROM ($sql) as t";
   161→   161→        $total = $this->db->fetchColumn($countSql, $params);
   162→   162→
   163→   163→        // Pagination
   164→   164→        $offset = ($page - 1) * $perPage;
   165→   165→        $sql .= " ORDER BY e.created_at DESC LIMIT $perPage OFFSET $offset";
   166→   166→
   167→   167→        $employees = $this->db->fetchAll($sql, $params);
   168→   168→
   169→   169→        return [
   170→   170→            'data' => $employees,
   171→   171→            'total' => $total ?: 0,
   172→   172→            'page' => $page,
   173→   173→            'per_page' => $perPage,
   174→   174→            'total_pages' => ceil(($total ?: 0) / $perPage)
   175→   175→        ];
   176→   176→    }

   303→   303→    // Create new employee
   304→   304→    public function create($data) {
   305→   305→        // Generate employee code if not provided
   306→   306→        if (empty($data['employee_code'])) {
   307→   307→            $data['employee_code'] = $this->generateEmployeeCode();
   308→   308→        }
   309→   309→        
   310→   310→        // Set default status
   311→   311→        $data['status'] = $data['status'] ?? STATUS_PENDING_HR;
   312→   312→        // Set default app_role to 'employee' if not provided
   313→   313→        $data['app_role'] = $data['app_role'] ?? 'employee';
   314→   314→        $data['created_at'] = date(DATETIME_FORMAT_DB);
   315→   315→
   316→   316→        // Map form fields to database columns
   317→   317→        $dbData = $this->mapFormDataToDb($data);
   318→   318→
   319→   319→        // Check if employee_salary_structures table exists
   320→   320→        $useSalaryTable = $this->checkSalaryTableExists();
   321→   321→
   322→   322→        try {
   323→   323→            $this->db->beginTransaction();
   324→   324→            
   325→   325→            // Insert employee
   326→   326→            $this->db->insert('employees', $dbData);
   327→   327→            
   328→   328→            // Get the newly created employee ID
   329→   329→            $employeeId = $this->db->lastInsertId();
   330→   330→            
   331→   331→            // Insert salary structure if table exists and any salary/statutory data provided
   332→   332→            $hasSalaryData = !empty($data['basic_da']) || !empty($data['basic_salary']) ||
   333→   333→                isset($data['pf_applicable']) || isset($data['esi_applicable']) ||
   334→   334→                isset($data['pt_applicable']) || isset($data['lwf_applicable']) ||
   335→   335→                isset($data['bonus_applicable']) || isset($data['gratuity_applicable']) ||
   336→   336→                isset($data['overtime_applicable']);
   337→   337→            
   338→   338→            if ($useSalaryTable && $hasSalaryData && $employeeId) {
   339→   339→                $salaryData = $this->buildSalaryData($data, $employeeId);
   340→   340→                $this->db->insert('employee_salary_structures', $salaryData);
   341→   341→            }
   342→   342→            
   343→   343→            $this->db->commit();
   344→   344→            
   345→   345→            // Send notification for pending employee registration
   346→   346→            if (strpos($data['status'], 'pending') !== false) {
   347→   347→                $notification = new Notification();
   348→   348→                $notification->notifyNewEmployeeRegistration([
   349→   349→                    'id' => $employeeId,
   350→   350→                    'full_name' => $data['full_name'] ?? '',
   351→   351→                    'employee_code' => $data['employee_code']
   352→   352→                ]);
   353→   353→            }
   354→   354→            
   355→   355→            return ['success' => true, 'message' => MSG_RECORD_CREATED, 'employee_id' => $employeeId];
   356→   356→        } catch (Exception $e) {
   357→   357→            $this->db->rollBack();
   358→   358→            return ['success' => false, 'message' => 'Failed to create employee: ' . $e->getMessage()];
   359→   359→        }
   360→   360→    }

   362→   362→    // Map form data to database columns
   363→   363→    private function mapFormDataToDb($data) {
   364→   364→        $mapped = [];
   365→   365→        
   366→   366→        // Direct mapping - form fields match database columns (removed client_name/unit_name as they don't exist)
   367→   367→        $directFields = [
   368→   368→            'full_name', 'father_name', 'employee_code', 'mobile_number', 'alternate_mobile',
   369→   369→            'email', 'gender', 'date_of_birth', 'marital_status', 'blood_group',
   370→   370→            'aadhaar_number', 'uan_number', 'esic_number',
   371→   371→            'address', 'pin_code', 'state', 'district',
   372→   372→            'bank_name', 'account_number', 'ifsc_code', 'account_holder_name',
   373→   373→            'client_id', 'unit_id', 'app_role',
   374→   374→            'designation', 'department',
   375→   375→            'worker_category', 'employment_type', 'date_of_joining', 'probation_period',
   376→   376→            'nominee_name', 'nominee_relationship', 'nominee_dob', 'nominee_contact',
   377→   377→            'emergency_contact_name', 'emergency_contact_relation',
   378→   378→            'status', 'profile_pic_url', 'profile_pic_cropped_url',
   379→   379→            'aadhaar_front_url', 'aadhaar_back_url', 'bank_document_url'
   380→   380→        ];
   381→   381→        
   382→   382→        foreach ($directFields as $field) {
   383→   383→            if (isset($data[$field])) {
   384→   384→                $mapped[$field] = $data[$field];
   385→   385→            }
   386→   386→        }
   387→   387→        
   388→   388→        // Handle middle_name -> father_name mapping (for backward compatibility)
   389→   389→        if (isset($data['middle_name']) && !isset($data['father_name'])) {
   390→   390→            $mapped['father_name'] = $data['middle_name'];
   391→   391→        }
   392→   392→        
   393→   393→        // Look up client_id from client_name if not provided (for import compatibility)
   394→   394→        if (!empty($data['client_name']) && empty($data['client_id'])) {
   395→   395→            $client = $this->db->fetch(
   396→   396→                "SELECT id FROM clients WHERE name = :name LIMIT 1",
   397→   397→                ['name' => $data['client_name']]
   398→   398→            );
   399→   399→            if ($client) {
   400→   400→                $mapped['client_id'] = $client['id'];
   401→   401→            }
   402→   402→        }
   403→   403→        
   404→   404→        // Look up unit_id from unit_name if not provided (for import compatibility)
   405→   405→        if (!empty($data['unit_name']) && empty($data['unit_id'])) {
   406→   406→            $unit = $this->db->fetch(
   407→   407→                "SELECT id FROM units WHERE name = :name LIMIT 1",
   408→   408→                ['name' => $data['unit_name']]
   409→   409→            );
   410→   410→            if ($unit) {
   411→   411→                $mapped['unit_id'] = $unit['id'];
   412→   412→            }
   413→   413→        }
   414→   414→        
   415→   415→        return $mapped;
   416→   416→    }

   462→   462→        try {
   463→   463→            $this->db->beginTransaction();
   464→   464→            
   465→   465→            // Update employee
   466→   466→            $this->db->update('employees', $dbData, SQL_WHERE_ID, ['id' => $id]);
   467→   467→            
   468→   468→            // Update salary structure if provided and table exists
   469→   469→            if (!empty($salaryData)) {
   470→   470→                // Check if salary structure exists
   471→   471→                $existing = $this->db->fetch(
   472→   472→                    "SELECT id FROM employee_salary_structures WHERE employee_id = :id AND (effective_to IS NULL OR effective_to >= CURDATE())",

   480→   480→                    $salaryData['effective_from'] = date(DATE_FORMAT_DB);
   481→   481→                    $this->db->insert('employee_salary_structures', $salaryData);
   482→   482→                }
   483→   483→            }
   484→   484→            
   485→   485→            $this->db->commit();

   487→   487→        } catch (Exception $e) {

   647→   647→        $errors = [];
   648→   648→        $duplicates = [];
   649→   649→

   654→   654→                    $exists = $this->db->fetch(
   655→   655→                        "SELECT id FROM employees WHERE mobile_number = :mobile",
   656→   656→                        ['mobile' => $row['mobile_number']]
   657→   657→                    );
   658→   658→                    if ($exists && $skipDuplicates) {
   659→   659→                        $duplicates[] = $row['mobile_number'];
   660→   660→                        continue;
   661→   661→                    }
   662→   662→                }
   663→   663→
   664→   664→                $result = $this->create($row);
   665→   665→                if ($result['success']) {
   666→   666→                    $imported++;
   667→   667→                } else {
   668→   668→                    $errors[] = $result['message'];
   669→   669→                }
   670→   670→            } catch (Exception $e) {
   671→   671→                $errors[] = "Error importing row: " . $e->getMessage();
   672→   672→            }
   673→   673→        }
   674→   674→
   675→   675→        return [
   676→   676→            'success' => true,
   677→   677→            'imported' => $imported,
   678→   678→            'duplicates' => count($duplicates),
   679→   679→            'errors' => $errors
