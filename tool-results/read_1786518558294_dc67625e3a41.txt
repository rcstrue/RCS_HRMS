    19→

    23→$clientId = getSessionFilter('client_id', 0);
    24→$unitId   = getSessionFilter('unit_id', 0);
    25→$month    = getSessionFilter('month', prev_month_num());
    26→$year     = getSessionFilter('year', prev_month_year());
    27→
    28→$calendarDays = cal_days_in_month(CAL_GREGORIAN, $month, $year);
    29→
    30→// ── Fetch unit's pay_days_type from salary formula ──────────────
    31→$unitFormula = null;
    32→if ($unitId > 0) {
    33→    $unitFormula = $db->fetch(
    34→        "SELECT * FROM unit_salary_formulas
    35→         WHERE unit_id = ? AND is_active = 1
    36→         ORDER BY effective_from DESC LIMIT 1",
    37→        [$unitId]
    38→    );
    39→}
    40→$payDaysType = $unitFormula['pay_days_type'] ?? 'actual';
    41→
    42→// Calculate totalDays based on pay_days_type
    43→$totalDays = $calendarDays;
    44→switch ($payDaysType) {
    45→    case 'fixed_30':
    46→        $totalDays = 30;
    47→        break;
    48→    case 'previous_month':
    49→        $prevMonth = $month - 1;
    50→        $prevYear  = $year;
    51→        if ($prevMonth < 1) { $prevMonth = 12; $prevYear--; }
    52→        $totalDays = (int)cal_days_in_month(CAL_GREGORIAN, $prevMonth, $prevYear);
    53→        break;
    54→    case 'calendar_minus_sundays':
    55→        // Count all Sundays in the selected month
    56→        $sundays = 0;
    57→        for ($d = 1; $d <= $calendarDays; $d++) {
    58→            $dow = (int)date('N', strtotime("$year-$month-$d"));
    59→            if ($dow === 7) $sundays++;
    60→        }
    61→        $totalDays = $calendarDays - $sundays;
    62→        break;
    63→    default: // 'actual'
    64→        $totalDays = $calendarDays;
    65→        break;
    66→}
    67→
    68→// Also get OT settings from unit formula
    69→$unitOtCalcType = $unitFormula['ot_calculation_type'] ?? 'double_pay';
    70→$unitOtCalcOn   = $unitFormula['ot_calculation_on'] ?? 'basic_da';
    71→$unitOtHrsPerDay = (float)($unitFormula['ot_hours_per_day'] ?? 8);
    72→
    73→// ── Pay days calculation is handled above via unit_salary_formulas ──
    74→
    75→// ── Fetch PF/ESI rates from DB (match process.php class.payroll.php) ──
    76→$pfEmployeeShare = 12.00;
    77→$pfWageCeiling = 15000;
    78→$pfEmployerShare = 3.67;
    79→$pfEmployerEps = 8.33;
    80→$pfEmployerEdlis = 0.50;
    81→$pfEpfAdmin = 0.50;
    82→$esiEmployeeShare = 0.75;
    83→$esiEmployerShare = 3.25;
    84→$esiWageCeiling = 21000;
    85→try {

    89→        $pfWageCeiling  = (float)$pfRate['wage_ceiling'];
    90→        $pfEmployerShare = (float)$pfRate['employer_share_pf'];
    91→        $pfEmployerEps  = (float)$pfRate['employer_share_eps'];
    92→        $pfEmployerEdlis = (float)$pfRate['employer_share_edlis'];
    93→        $pfEpfAdmin     = (float)$pfRate['epf_admin_charges'];

    97→    $esiRate = $db->fetch("SELECT * FROM esi_rates WHERE is_active = 1 ORDER BY effective_from DESC LIMIT 1");
    98→    if ($esiRate) {
    99→        $esiEmployeeShare = (float)$esiRate['employee_share'];
   100→        $esiEmployerShare = (float)$esiRate['employer_share'];
   101→        $esiWageCeiling   = (float)$esiRate['wage_ceiling'];
   102→    }
   103→} catch (Exception $e) {}

   105→// ── Fetch unit state + PT + LWF configuration ────────────────

   107→$unitStatePtApplicable = false;
   108→$ptSlabs = [];
   109→$lwfAmount = 0; // Default 0, will be set per-state below
   110→if ($unitId > 0) {
   111→    $unitInfo = $db->fetch(
   112→        "SELECT u.state, s.pt_applicable as state_pt_applicable, s.id as state_id
   113→         FROM units u
   114→         LEFT JOIN states s ON (u.state = s.state_code OR u.state = s.state_name)
   115→         WHERE u.id = ?",
   116→        [$unitId]
   117→    );
   118→    if ($unitInfo) {
   119→        $unitState = $unitInfo['state'] ?? '';
   120→        $unitStatePtApplicable = !empty($unitInfo['state_pt_applicable']);
   121→    }
   122→    // Fetch PT slabs for this state
   123→    if ($unitStatePtApplicable && $unitState) {
   124→        try {
   125→            $ptSlabs = $db->fetchAll(
   126→                "SELECT min_gross, max_gross, pt_amount
   127→                 FROM professional_tax_slabs
   128→                 WHERE (state_name = ? OR state_name = ?) AND is_active = 1
   129→                 ORDER BY min_gross",
   130→                [$unitState, ucfirst(strtolower($unitState))]
   131→            );
   132→        } catch (Exception $e) {}
   133→    }
   134→    // Fetch LWF rate for this state (from lwf_rates → lwf_state_rates fallback)
   135→    if ($unitState) {
   136→        try {
   137→            $lwfRate = $db->fetch(
   138→                "SELECT lr.employee_share, lr.contribution_months
   139→                 FROM lwf_rates lr
   140→                 JOIN states s ON lr.state_id = s.id
   141→                 WHERE (s.state_code = ? OR s.state_name = ?)
   142→                 AND lr.is_active = 1 AND lr.effective_from <= CURDATE()
   143→                 ORDER BY lr.effective_from DESC LIMIT 1",
   144→                [$unitState, $unitState]
   145→            );
   146→            if (!$lwfRate) {
   147→                $lwfRate = $db->fetch(
   148→                    "SELECT employee_share, contribution_months
   149→                     FROM lwf_state_rates
   150→                     WHERE (state_name = ? OR state_name = ?) AND is_active = 1
   151→                     AND effective_from <= CURDATE()
   152→                     ORDER BY effective_from DESC LIMIT 1",
   153→                    [$unitState, ucfirst(strtolower($unitState))]
   154→                );
   155→            }
   156→            if ($lwfRate) {
   157→                $contribMonths = array_filter(array_map('intval', explode(',', $lwfRate['contribution_months'] ?? '')));
   158→                if (in_array($month, $contribMonths)) {
   159→                    $lwfAmount = floatval($lwfRate['employee_share']);
   160→                }
   161→            }
   162→        } catch (Exception $e) {}
   163→    }

   167→$clients = $db->fetchAll("SELECT id, name FROM clients WHERE is_active = 1 ORDER BY name");
   168→
   169→// Units filtered by selected client (same pattern as attendance/add)
   170→$units = [];
   171→if ($clientId > 0) {
   172→    $units = $db->fetchAll(
   173→        "SELECT id, name, unit_code FROM units WHERE client_id = ? AND is_active = 1 ORDER BY name",
   174→        [$clientId]
   175→    );
   176→}
   177→
   178→// ── Handle Load Employees via POST (AJAX or form) ───────────────
   179→$employeeData = [];
   180→$loaded = false;
   181→
   182→if (isset($_GET['load']) && (int)$_GET['load'] === 1 && $unitId > 0) {
   183→    $loaded = true;
   184→
   185→    // Ensure loan tables exist before querying them in the loop
   186→    try {
   187→        $db->exec("CREATE TABLE IF NOT EXISTS `employee_loans` (
   188→            `id` int(11) NOT NULL AUTO_INCREMENT,
   189→            `employee_id` int(11) NOT NULL,
   190→            `unit_id` int(11) DEFAULT NULL,
   191→            `loan_type` varchar(50) DEFAULT 'Personal',
   192→            `amount` decimal(12,2) NOT NULL,
   193→            `interest_rate` decimal(5,2) DEFAULT 0.00,
   194→            `tenure_months` int(11) NOT NULL,
   195→            `emi_amount` decimal(12,2) NOT NULL,
   196→            `total_interest` decimal(12,2) DEFAULT 0.00,
   197→            `total_repayable` decimal(12,2) NOT NULL,
   198→            `balance_amount` decimal(12,2) NOT NULL,
   199→            `emi_deducted` int(11) DEFAULT 0,
   200→            `start_month` int(2) NOT NULL,
   201→            `start_year` int(4) NOT NULL,
   202→            `status` enum('Active','Closed','Settled','Written Off') DEFAULT 'Active',
   203→            `remarks` text DEFAULT NULL,
   204→            `created_at` timestamp DEFAULT CURRENT_TIMESTAMP,
   205→            `updated_at` timestamp DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
   206→            PRIMARY KEY (`id`),
   207→            KEY `idx_employee` (`employee_id`),
   208→            KEY `idx_unit` (`unit_id`),
   209→            KEY `idx_status` (`status`)
   210→        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4");
   211→        $db->exec("CREATE TABLE IF NOT EXISTS `loan_emi_log` (
   212→            `id` int(11) NOT NULL AUTO_INCREMENT,
   213→            `loan_id` int(11) NOT NULL,
   214→            `employee_id` int(11) NOT NULL,
   215→            `month` int(2) NOT NULL,
   216→            `year` int(4) NOT NULL,
   217→            `emi_amount` decimal(12,2) NOT NULL,
   218→            `principal_component` decimal(12,2) DEFAULT 0.00,
   219→            `interest_component` decimal(12,2) DEFAULT 0.00,
   220→            `balance_after` decimal(12,2) NOT NULL,
   221→            `deducted_via_payroll` tinyint(1) DEFAULT 1,
   222→            `created_at` timestamp DEFAULT CURRENT_TIMESTAMP,
   223→            PRIMARY KEY (`id`),
   224→            UNIQUE KEY `uniq_loan_month_year` (`loan_id`, `month`, `year`),
   225→            KEY `idx_employee_month` (`employee_id`, `month`, `year`)
   226→        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4");
   227→    } catch (Exception $e) {}
   228→
   229→    $employees = $db->fetchAll(
   230→        "SELECT id, employee_code, full_name, designation
   231→         FROM employees
   232→         WHERE unit_id = ? AND status = 'approved'
   233→         ORDER BY employee_code",
   234→        [$unitId]
   235→    );
   236→
   237→    // Use last day of SELECTED month — not CURDATE() — so editing a past month
   238→    // loads the salary structure that was actually active during that month,
   239→    // not the latest one created for a future month.
   240→    $effDateStr = sprintf('%04d-%02d-%02d', $year, $month, cal_days_in_month(CAL_GREGORIAN, $month, $year));
   241→
   242→    foreach ($employees as $emp) {
   243→        $empId  = (int)$emp['id'];
   244→        $empCode = $emp['employee_code'];
   245→
   246→        // Salary structure (latest effective for selected month)
   247→        $salary = $db->fetch(
   248→            "SELECT * FROM employee_salary_structures
   249→             WHERE employee_id = ? AND effective_from <= ?
   250→             ORDER BY effective_from DESC LIMIT 1",
   251→            [$empId, $effDateStr]
   252→        );
   253→
   254→        // Attendance
   255→        $att = [];
   256→        try {
   257→            $att = $db->fetch(
   258→                "SELECT * FROM attendance_summary
   259→                 WHERE employee_id = ? AND month = ? AND year = ?",

   261→            );
   262→        } catch (Exception $e) {
   263→            $att = [];
   264→        }
   265→
   266→        // Advances
   267→        $adv = [];
   268→        try {
   269→            $adv = $db->fetch(
   270→                "SELECT COALESCE(adv1,0)+COALESCE(adv2,0)+COALESCE(office_advance,0)+COALESCE(dress_advance,0) as total_advance,
   271→                        COALESCE(adv1,0) as adv1,
   272→                        COALESCE(adv2,0) as adv2,
   273→                        COALESCE(office_advance,0) as office_advance,
   274→                        COALESCE(dress_advance,0) as dress_advance
   275→                 FROM employee_advances
   276→                 WHERE employee_id = ? AND month = ? AND year = ?",
   277→                [$empId, $month, $year]
   278→            );
   279→        } catch (Exception $e) {
   280→            $adv = [];
   281→        }
   282→
   283→        // Existing payroll record (if already saved for this month/year)
   284→        $existing = [];
   285→        try {
   286→            // Try direct month/year lookup first (new schema)
   287→            $existing = $db->fetch(
   288→                "SELECT * FROM payroll
   289→                 WHERE employee_id = ? AND month = ? AND year = ?",
   290→                [$empCode, $month, $year]
   291→            );
   292→        } catch (Exception $e) {
   293→            $existing = [];
   294→        }
   295→
   296→        // Active loan EMI for this employee (with balance > 0, not already deducted this month)
   297→        $loanEmi = 0;
   298→        try {
   299→            // Sum EMI from ALL active loans
   300→            $loans = $db->fetchAll(
   301→                "SELECT el.emi_amount, el.id as loan_id
   302→                 FROM employee_loans el
   303→                 WHERE el.employee_id = ? AND el.status = 'Active'
   304→                 AND el.balance_amount > 0
   305→                 AND (el.start_year < ? OR (el.start_year = ? AND el.start_month <= ?))
   306→                 AND el.id NOT IN (
   307→                     SELECT lem.loan_id FROM loan_emi_log lem
   308→                     WHERE lem.month = ? AND lem.year = ?
   309→                 )",
   310→                [$empId, $year, $year, $month, $month, $year]
   311→            );
   312→            foreach ($loans as $loan) {
   313→                $loanEmi += floatval($loan['emi_amount']);
   314→            }
   315→        } catch (Exception $e) {
   316→            // Fallback: try without emi_log check (in case table has different structure)
   317→            try {
   318→                $loans2 = $db->fetchAll(
   319→                    "SELECT el.emi_amount, el.id as loan_id
   320→                     FROM employee_loans el
   321→                     WHERE el.employee_id = ? AND el.status = 'Active'
   322→                     AND el.balance_amount > 0
   323→                     AND (el.start_year < ? OR (el.start_year = ? AND el.start_month <= ?))",
   324→                    [$empId, $year, $year, $month]
   325→                );
   326→                foreach ($loans2 as $loan2) {
   327→                    $loanEmi += floatval($loan2['emi_amount']);
   328→                }
   329→            } catch (Exception $e2) {
   330→                $loanEmi = 0;
   331→            }
   332→        }
   333→
   334→        $employeeData[] = [
   335→            'emp'      => $emp,
   336→            'salary'   => $salary ?: [],
   337→            'att'      => $att ?: [],
   338→            'adv'      => $adv ?: [],
   339→            'existing' => $existing ?: [],
   340→            'loan_emi' => $loanEmi,
   341→        ];
   342→    }
   343→}

   346→$monthNames = [1=>'January','February','March','April','May','June',
   347→               7=>'July','August','September','October','November','December'];
   348→$monthLabel = $monthNames[$month] ?? '';

  1431→    function previewModalCalc() {
  1432→        var totalDays = TOTAL_DAYS;

  1434→        var attWO      = parseFloat($('#modalAttWO').val()) || 0;
  1435→        var attExtra   = parseFloat($('#modalAttExtra').val()) || 0;
  1436→        var otHours    = parseFloat($('#modalOtHours').val()) || 0;
  1437→        var paidDays   = attPresent + attWO + attExtra;
  1438→
  1439→        var wBasicDa   = parseFloat($('#modalWBasicDa').val()) || 0;
  1440→        var wHra       = parseFloat($('#modalWHra').val()) || 0;
  1441→        var wLeaveEnc  = parseFloat($('#modalWLeaveEnc').val()) || 0;
  1442→        var wBonusEnc  = parseFloat($('#modalWBonusEnc').val()) || 0;
  1443→        var wWash      = parseFloat($('#modalWWash').val()) || 0;
  1444→        var wOtRate    = parseFloat($('#modalWOtRate').val()) || 0;
  1445→
  1446→        var advance    = parseFloat($('#modalAdvance').val()) || 0;
  1447→        var officeDed  = parseFloat($('#modalOfficeDed').val()) || 0;
  1448→        var loanEmiM   = parseFloat($('#modalLoanEmi').val()) || 0;
  1449→
  1450→        if (totalDays === 0 || paidDays === 0) {
  1451→            $('#modalPreviewGross').text('₹0.00');
  1452→            $('#modalPreviewDed').text('₹0.00');
  1453→            $('#modalPreviewNet').text('₹0.00');

  1455→        }
  1456→
  1457→        var pfApp   = $('#modalPfApp').is(':checked');
  1458→        var esiApp  = $('#modalEsiApp').is(':checked');
  1459→        var ptApp   = $('#modalPtApp').is(':checked');
  1460→        var lwfApp  = $('#modalLwfApp').is(':checked');
  1461→
  1462→        var otMult  = modalCurrentRow ? (parseFloat(modalCurrentRow.data('ot-multiplier')) || 1) : 1;
  1463→        var lwfAmt  = modalCurrentRow ? (parseFloat(modalCurrentRow.data('lwf-amount')) || 0) : 0;
  1464→        var existGr = modalCurrentRow ? (parseFloat(modalCurrentRow.data('existing-gross')) || 0) : 0;
  1465→
  1466→        var basicDa  = round2(wBasicDa * paidDays / totalDays);
  1467→        var hra      = round2(wHra * paidDays / totalDays);

  1469→        var bonusEnc = round2(wBonusEnc * paidDays / totalDays);
  1470→        var wash     = round2(wWash * paidDays / totalDays);
  1471→        var otAmt    = round2(wOtRate * otHours * otMult);
  1472→
  1473→        var gross = round2(basicDa + hra + leaveEnc + bonusEnc + wash + otAmt);
  1474→        var pf    = pfApp ? round2(Math.min(basicDa, PF_WAGE_CEILING) * PF_EMPLOYEE_SHARE / 100) : 0;
  1475→        var esiG  = existGr > 0 ? existGr : 0; // full monthly gross
  1476→        var esi   = (esiApp && esiG > 0 && esiG <= ESI_WAGE_CEILING) ? round2(gross * ESI_EMPLOYEE_SHARE / 100) : 0;
