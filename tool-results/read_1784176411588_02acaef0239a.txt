   127→   127→function _calculateScore(mysqli $conn, int $visitId): array
   128→   128→{
   129→   129→    $stmt = $conn->prepare('
   130→   130→        SELECT vci.status, ci.weight
   131→   131→        FROM ess_visit_checklist_items vci
   132→   132→        JOIN ess_checklist_items ci ON ci.id = vci.checklist_item_id
   133→   133→        WHERE vci.visit_id = ?
   134→   134→    ');
   135→   135→    $stmt->bind_param('i', $visitId);
   136→   136→    $stmt->execute();
   137→   137→    $result = $stmt->get_result();
   138→   138→
   139→   139→    $total = 0;
   140→   140→    $max = 0;
   141→   141→    while ($row = $result->fetch_assoc()) {
   142→   142→        $w = (float)($row['weight'] ?? 1);
   143→   143→        if ($row['status'] !== 'na') {
   144→   144→            $max += $w;
   145→   145→            if ($row['status'] === 'yes') {
   146→   146→                $total += $w;
   147→   147→            }
   148→   148→        }
   149→   149→    }
   150→   150→    $stmt->close();
   151→   151→
   152→   152→    $percent = $max > 0 ? round(($total / $max) * 100, 2) : 0;
   153→   153→
   154→   154→    // Update visit record
   155→   155→    $upd = $conn->prepare('UPDATE ess_unit_visits SET total_score = ?, max_score = ?, score_percent = ? WHERE id = ?');
   156→   156→    $upd->bind_param('dddi', $total, $max, $percent, $visitId);
   157→   157→    $upd->execute();
   158→   158→    $upd->close();
   159→   159→
   160→   160→    return ['total' => $total, 'max' => $max, 'percent' => $percent];
   161→   161→}

   383→   383→}

   385→   385→function _handleGetDashboard(mysqli $conn, string $authId): void
   386→   386→{
   387→   387→    $employeeId = (int)($_GET['employee_id'] ?? $authId);
   388→   388→    $now = new \DateTime('now', new \DateTimeZone('Asia/Kolkata'));
   389→   389→    $currentMonth = (int)$now->format('n');
   390→   390→    $currentYear = (int)$now->format('Y');
   391→   391→
   392→   392→    // Basic counts
   393→   393→    $counts = $conn->query("
   394→   394→        SELECT
   395→   395→            COUNT(*) AS total_visits,
   396→   396→            SUM(CASE WHEN status = 'submitted' THEN 1 ELSE 0 END) AS pending_visits,
   397→   397→            SUM(CASE WHEN status = 'approved' THEN 1 ELSE 0 END) AS approved_visits,
   398→   398→            SUM(CASE WHEN status = 'rejected' THEN 1 ELSE 0 END) AS rejected_visits,
   399→   399→            SUM(CASE WHEN visit_month = {$currentMonth} AND visit_year = {$currentYear} THEN 1 ELSE 0 END) AS this_month_visits
   400→   400→        FROM ess_unit_visits WHERE employee_id = {$employeeId}
   401→   401→    ")->fetch_assoc();
   402→   402→
   403→   403→    // This month avg score
   404→   404→    $monthAvg = $conn->query("
   405→   405→        SELECT AVG(score_percent) AS avg_score
   406→   406→        FROM ess_unit_visits
   407→   407→        WHERE employee_id = {$employeeId} AND visit_month = {$currentMonth} AND visit_year = {$currentYear}
   408→   408→            AND status IN ('approved','submitted') AND max_score > 0
   409→   409→    ")->fetch_assoc();
   410→   410→
   411→   411→    // Score trend - last 6 months
   412→   412→    $trend = [];
   413→   413→    for ($i = 5; $i >= 0; $i--) {
   414→   414→        $d = clone $now;
   415→   415→        $d->modify("-{$i} months");
   416→   416→        $m = (int)$d->format('n');
   417→   417→        $y = (int)$d->format('Y');
   418→   418→        $label = $d->format('M Y');
   419→   419→        $avgRow = $conn->query("
