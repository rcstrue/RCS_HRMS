   141→function formatDate(val: string | null | undefined): string {
   142→  if (!val) return '—'
   143→  try {
   144→    return new Date(val).toLocaleDateString('en-IN', {
   145→      day: '2-digit', month: 'short', year: 'numeric',
   146→    })
   147→  } catch {
   148→    return val
   149→  }
   150→}
   151→
   152→function formatCurrency(val: number): string {
   153→  if (!val) return '₹0'
   154→  return new Intl.NumberFormat('en-IN', {
   155→    style: 'currency', currency: 'INR', maximumFractionDigits: 0,
   156→  }).format(val)
   157→}

   159→// ── CSV Export ───────────────────────────────────────────────────────────
   160→
   161→function exportCSV(records: ContractorRecord[]) {
   162→  if (!records.length) return
   163→  const headers = [
   164→    'Contractor Name', 'Address', 'Registration No', 'Date of Registration', 'PAN',
   165→    'Establishment Name', 'Establishment Address', 'Nature of Work', 'Max Workmen',
   166→    'Licensing Authority', 'License No', 'License Issue Date', 'License Valid From',
   167→    'License Valid To', 'License Fee', 'GSTIN', 'PF Reg No',
   168→    'Contact Person', 'Contact Designation', 'Contact Mobile', 'Contact Email',
   169→    'Bank Name', 'Bank Account No', 'Bank IFSC', 'ESI Reg No',
   170→    'Contract Start', 'Contract End', 'Contract Value', 'Security Deposit',
   171→    'Work Location', 'Remarks',
   172→  ]
   173→  const keys: (keyof ContractorRecord)[] = [
   174→    'contractorName', 'contractorAddress', 'registrationNumber', 'dateOfRegistration', 'contractorPan',
   175→    'establishmentName', 'establishmentAddress', 'natureOfWork', 'maxWorkmen',
   176→    'licensingAuthority', 'licenseNumber', 'licenseIssueDate', 'licenseValidFrom',
   177→    'licenseValidTo', 'licenseFee', 'contractorGstin', 'pfRegistrationNumber',
   178→    'contactPersonName', 'contactPersonDesignation', 'contactPersonMobile', 'contactPersonEmail',
   179→    'bankName', 'bankAccountNumber', 'bankIfscCode', 'esiRegistrationNumber',
   180→    'contractStartDate', 'contractEndDate', 'contractValue', 'securityDeposit',
   181→    'contractWorkLocation', 'remarks',
   182→  ]
   183→  const rows = records.map(r =>
   184→    keys.map(k => {
   185→      const v = r[k]
   186→      if (v === null || v === undefined) return ''
   187→      const s = String(v)
   188→      return s.includes(',') || s.includes('"') || s.includes('\n')
   189→        ? `"${s.replace(/"/g, '""')}"`
   190→        : s
   191→    }).join(',')
   192→  )
   193→  const csv = [headers.join(','), ...rows].join('\n')
   194→  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' })
   195→  const link = document.createElement('a')
   196→  link.href = URL.createObjectURL(blob)
   197→  link.download = `Form_A_Labour_Contractor_Register_${new Date().toISOString().slice(0, 10)}.csv`
   198→  link.click()
   199→  URL.revokeObjectURL(link.href)
   200→}
