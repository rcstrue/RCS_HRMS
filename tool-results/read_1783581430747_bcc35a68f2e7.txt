   383→   383→    ];

   385→   385→    // Employer contribution columns (matching process.php)
   386→   386→    $payData['pf_employer']            = round2($payroll['pf_employer'] ?? 0);
   387→   387→    $payData['eps_employer']           = round2($payroll['eps_employer'] ?? 0);
   388→   388→    $payData['edlis_employer']         = round2($payroll['edlis_employer'] ?? 0);
   389→   389→    $payData['epf_admin_charges']      = round2($payroll['epf_admin_charges'] ?? 0);
   390→   390→    $payData['esi_employer']           = round2($payroll['esi_employer'] ?? 0);
   391→   391→    $payData['total_employer_contribution'] = round2($payroll['total_employer_contribution'] ?? 0);
   392→   392→
   393→   393→    // Include loan_emi from the grid input
   394→   394→    $payData['loan_emi'] = round2($advances['loan_emi'] ?? 0);
   395→   395→
   396→   396→    if ($existingPay) {
   397→   397→        $db->update('payroll', $payData, 'id = :id', ['id' => $existingPay['id']]);
   398→   398→    } else {
   399→   399→        $payData['created_at'] = date('Y-m-d H:i:s');
   400→   400→        $db->insert('payroll', $payData);
   401→   401→    }
   402→   402→
   403→   403→    // ── Update employee statutory flags in salary structure ──────
   404→   404→    try {
   405→   405→        $db->update('employee_salary_structures', [
   406→   406→            'pf_applicable'  => $pfApplicable,
   407→   407→            'esi_applicable' => $esiApplicable,
   408→   408→            'pt_applicable'  => $ptApplicable,
   409→   409→            'lwf_applicable' => $lwfApplicable,
   410→   410→        ], 'employee_id = :eid AND effective_to IS NULL', ['eid' => $employeeId]);
   411→   411→    } catch (Exception $e) {
   412→   412→        // Column may not exist — ignore
   413→   413→    }
   414→   414→
   415→   415→    // ═══════════════════════════════════════════════════════════════
   416→   416→    // 5. LOAN EMI LOG & BALANCE UPDATE
   417→   417→    // (Payroll record already saved with loan_emi from grid — don't double-count)
   418→   418→    // Only update loan_emi_log and employee_loans balance.
   419→   419→    // ═══════════════════════════════════════════════════════════════
   420→   420→    $loanDeductionTotal = 0;
   421→   421→    try {
   422→   422→        // Ensure loan tables exist
   423→   423→        $db->exec("CREATE TABLE IF NOT EXISTS `employee_loans` (
   424→   424→            `id` int(11) NOT NULL AUTO_INCREMENT,
   425→   425→            `employee_id` int(11) NOT NULL,
   426→   426→            `unit_id` int(11) DEFAULT NULL,
   427→   427→            `loan_type` varchar(50) DEFAULT 'Personal',
   428→   428→            `amount` decimal(12,2) NOT NULL,
   429→   429→            `interest_rate` decimal(5,2) DEFAULT 0.00,
   430→   430→            `tenure_months` int(11) NOT NULL,
   431→   431→            `emi_amount` decimal(12,2) NOT NULL,
   432→   432→            `total_interest` decimal(12,2) DEFAULT 0.00,
   433→   433→            `total_repayable` decimal(12,2) NOT NULL,
   434→   434→            `balance_amount` decimal(12,2) NOT NULL,
   435→   435→            `emi_deducted` int(11) DEFAULT 0,
   436→   436→            `start_month` int(2) NOT NULL,
   437→   437→            `start_year` int(4) NOT NULL,
   438→   438→            `status` enum('Active','Closed','Settled','Written Off') DEFAULT 'Active',
   439→   439→            `remarks` text DEFAULT NULL,
   440→   440→            `created_at` timestamp DEFAULT CURRENT_TIMESTAMP,
   441→   441→            `updated_at` timestamp DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
   442→   442→            PRIMARY KEY (`id`),
   443→   443→            KEY `idx_employee` (`employee_id`),
   444→   444→            KEY `idx_status` (`status`)
   445→   445→        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4");
   446→   446→
   447→   447→        $db->exec("CREATE TABLE IF NOT EXISTS `loan_emi_log` (
   448→   448→            `id` int(11) NOT NULL AUTO_INCREMENT,
   449→   449→            `loan_id` int(11) NOT NULL,
   450→   450→            `employee_id` int(11) NOT NULL,
   451→   451→            `month` int(2) NOT NULL,
   452→   452→            `year` int(4) NOT NULL,
   453→   453→            `emi_amount` decimal(12,2) NOT NULL,
   454→   454→            `principal_component` decimal(12,2) DEFAULT 0.00,
   455→   455→            `interest_component` decimal(12,2) DEFAULT 0.00,
   456→   456→            `balance_after` decimal(12,2) NOT NULL,
   457→   457→            `payroll_id` int(11) DEFAULT NULL,
   458→   458→            `deducted_via_payroll` tinyint(1) DEFAULT 1,
   459→   459→            `created_at` timestamp DEFAULT CURRENT_TIMESTAMP,
   460→   460→            PRIMARY KEY (`id`),
   461→   461→            UNIQUE KEY `uniq_loan_month_year` (`loan_id`, `month`, `year`),
   462→   462→            KEY `idx_employee_month` (`employee_id`, `month`, `year`)
   463→   463→        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4");
   464→   464→
   465→   465→        // Only proceed if the grid had a loan_emi value > 0
   466→   466→        $gridLoanEmi = (float)($advances['loan_emi'] ?? 0);
   467→   467→        if ($gridLoanEmi <= 0) {
   468→   468→            // No EMI to process — but clean up any stale emi_log entries
   469→   469→            // (from previous failed saves that partially committed)
   470→   470→            try {
   471→   471→                $db->query("DELETE FROM loan_emi_log WHERE employee_id = ? AND month = ? AND year = ? AND loan_id IN (SELECT id FROM employee_loans WHERE employee_id = ?)",
   472→   472→                    [$employeeId, $month, $year, $employeeId]);
   473→   473→            } catch (Exception $e) {}
   474→   474→            goto loan_done;
   475→   475→        }
   476→   476→
   477→   477→        // Get active loans for this employee that have started by this month
   478→   478→        $empLoans = $db->fetchAll(
   479→   479→            "SELECT * FROM employee_loans
   480→   480→             WHERE employee_id = ? AND status = 'Active'
   481→   481→             AND balance_amount > 0
   482→   482→             AND (start_year < ? OR (start_year = ? AND start_month <= ?))",
   483→   483→            [$employeeId, $year, $year, $month]
   484→   484→        );
   485→   485→
   486→   486→        foreach ($empLoans as $empLoan) {
   487→   487→            // Check if EMI already deducted for this month
   488→   488→            $alreadyDeducted = $db->fetch(
   489→   489→                "SELECT id FROM loan_emi_log WHERE loan_id = ? AND month = ? AND year = ?",
   490→   490→                [$empLoan['id'], $month, $year]
   491→   491→            );
   492→   492→
   493→   493→            if ($alreadyDeducted) {
   494→   494→                continue;
   495→   495→            }
   496→   496→
   497→   497→            $emiAmount = (float)$empLoan['emi_amount'];
   498→   498→            $balanceAmount = (float)$empLoan['balance_amount'];
   499→   499→
   500→   500→            // Calculate interest/principal split
   501→   501→            if ((float)$empLoan['interest_rate'] > 0) {
   502→   502→                $monthlyRate = (float)$empLoan['interest_rate'] / 12 / 100;
   503→   503→                $interestComponent = round($balanceAmount * $monthlyRate, 2);
   504→   504→                $principalComponent = round($emiAmount - $interestComponent, 2);
   505→   505→            } else {
   506→   506→                $interestComponent = 0;
   507→   507→                $principalComponent = $emiAmount;
   508→   508→            }
   509→   509→
   510→   510→            // Handle last EMI: adjust to remaining balance
   511→   511→            if ($emiAmount > $balanceAmount) {
   512→   512→                $emiAmount = $balanceAmount;
   513→   513→                $principalComponent = $emiAmount - $interestComponent;
   514→   514→                if ($principalComponent < 0) {
   515→   515→                    $interestComponent = $emiAmount;
   516→   516→                    $principalComponent = 0;
   517→   517→                }
   518→   518→            }
   519→   519→
   520→   520→            $newBalance = round($balanceAmount - $emiAmount, 2);
   521→   521→            if ($newBalance < 0) $newBalance = 0;
   522→   522→
   523→   523→            // Record EMI log (linked to payroll row — Audit Issue #7)
   524→   524→            $db->insert('loan_emi_log', [
   525→   525→                'loan_id' => $empLoan['id'],
   526→   526→                'employee_id' => $employeeId,
   527→   527→                'month' => $month,
   528→   528→                'year' => $year,
   529→   529→                'emi_amount' => $emiAmount,
   530→   530→                'principal_component' => $principalComponent,
   531→   531→                'interest_component' => $interestComponent,
   532→   532→                'balance_after' => $newBalance,
   533→   533→                'deducted_via_payroll' => 1
   534→   534→            ]);
   535→   535→
   536→   536→            // Try to link to payroll row by month/year
   537→   537→            try {
   538→   538→                $linkedPayrollId = null;
   539→   539→                $pRow = $db->fetch(
   540→   540→                    "SELECT id FROM payroll WHERE employee_id = ? AND month = ? AND year = ?",
   541→   541→                    [$employeeCode, $month, $year]
   542→   542→                );
   543→   543→                $linkedPayrollId = $pRow ? (int)$pRow['id'] : null;
   544→   544→                if ($linkedPayrollId) {
   545→   545→                    $db->query(
   546→   546→                        "UPDATE loan_emi_log SET payroll_id = ? WHERE loan_id = ? AND month = ? AND year = ?",
   547→   547→                        [$linkedPayrollId, $empLoan['id'], $month, $year]
   548→   548→                    );
   549→   549→                }
   550→   550→            } catch (Exception $e) {}
   551→   551→
   552→   552→            // Update loan balance and status
   553→   553→            $newStatus = ($newBalance <= 0) ? 'Closed' : 'Active';
   554→   554→            $db->update('employee_loans', [
   555→   555→                'balance_amount' => $newBalance,
   556→   556→                'emi_deducted' => (int)$empLoan['emi_deducted'] + 1,
   557→   557→                'status' => $newStatus
   558→   558→            ], 'id = :id', ['id' => $empLoan['id']]);
   559→   559→
   560→   560→            $loanDeductionTotal += $emiAmount;
   561→   561→        }
   562→   562→
   563→   563→        // NOTE: We do NOT update the payroll record here.
   564→   564→        // The loan_emi was already included in the grid input and saved in step 4.
   565→   565→        // Updating again would double-count the deduction.
   566→   566→
   567→   567→    } catch (Exception $loanEx) {
   568→   568→        // Log but don't fail payroll save if loan deduction fails
   569→   569→        error_log('Loan EMI auto-deduction failed: ' . $loanEx->getMessage());
   570→   570→    }
   571→   571→    loan_done:
   572→   572→
   573→   573→    $db->exec("COMMIT");
   574→   574→
   575→   575→    echo json_encode([
   576→   576→        'success' => true,
   577→   577→        'message' => $otWarning . 'Payroll saved for ' . $employeeCode . ($loanDeductionTotal > 0 ? ' (Loan EMI ₹' . number_format($loanDeductionTotal, 2) . ' auto-deducted)' : ''),
   578→   578→        'employee_code' => $employeeCode,
   579→   579→        'net_pay' => $payData['net_pay'],
   580→   580→        'gross_salary' => $payData['gross_salary'],
   581→   581→        'loan_deduction' => $loanDeductionTotal,
   582→   582→    ]);
   583→   583→
   584→   584→} catch (Exception $e) {
   585→   585→    // Rollback on error
   586→   586→    try { $db->exec("ROLLBACK"); } catch (Exception $re) {}
   587→   587→
   588→   588→    error_log("Payroll Save Row Error [{$employeeCode}]: " . $e->getMessage());
   589→   589→
   590→   590→    echo json_encode([
   591→   591→        'success' => false,
   592→   592→        'message' => 'Error saving payroll: ' . $e->getMessage(),
   593→   593→        'employee_code' => $employeeCode,
   594→   594→    ]);
   595→   595→}
