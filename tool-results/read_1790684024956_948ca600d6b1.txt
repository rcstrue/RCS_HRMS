    47→    47→// EditProfilePage — Clean edit form, single Save Changes button
    48→    48→// ══════════════════════════════════════════════════════════════
    49→    49→//
    50→    50→// Design:
    51→    51→//   - All fields render as simple input fields (no per-field buttons)
    52→    52→//   - A small badge next to each label shows "Saves directly" (green)
    53→    53→//     or "Needs approval" (amber) based on resolveEffectiveRule()
    54→    54→//   - One "Save Changes" button at the bottom handles everything:
    55→    55→//     → Free fields are saved directly via onSaveFreeFields
    56→    56→//     → Approval-needed fields are submitted as change requests
    57→    57→//     → A summary toast tells the user what happened
    58→    58→//   - Profile photo: upload button (handled by Save for approval)
    59→    59→
    60→    60→interface EditProfilePageProps {
    61→    61→  employee: Employee;
    62→    62→  pendingChangeRequests?: ChangeRequest[];
    63→    63→  onSaveFreeFields: (fields: Record<string, string | null>) => Promise<{ success: boolean; error?: string }>;
    64→    64→  onSubmitChangeRequest: (data: { field_name: string; old_value: string; new_value: string; reason?: string }) => Promise<{ success: boolean; error?: string }>;
    65→    65→  onBack: () => void;
    66→    66→}

    79→    79→export default function EditProfilePage({
    80→    80→  employee,
    81→    81→  pendingChangeRequests = [],
    82→    82→  onSaveFreeFields,
    83→    83→  onSubmitChangeRequest,
    84→    84→  onBack,
    85→    85→}: EditProfilePageProps) {
    86→    86→  const [saving, setSaving] = useState(false);
    87→    87→  const [formValues, setFormValues] = useState<Record<string, string>>({});
    88→    88→  const [uploadingPhoto, setUploadingPhoto] = useState(false);
    89→    89→  const [pendingPhotoUrl, setPendingPhotoUrl] = useState<string | null>(null);
    90→    90→  const fileInputRef = useRef<HTMLInputElement>(null);
    91→    91→
    92→    92→  // ── KYC Document image uploads ──
    93→    93→  // Track pending uploaded URLs for image fields (aadhaar_front_url, aadhaar_back_url, bank_document_url)
    94→    94→  const [pendingImageUrls, setPendingImageUrls] = useState<Record<string, string>>({});
    95→    95→  const [uploadingImageKey, setUploadingImageKey] = useState<string | null>(null);
    96→    96→  const imageInputRefs = useRef<Record<string, HTMLInputElement | null>>({});
    97→    97→
    98→    98→  const pendingFieldNames = useMemo(() => {
    99→    99→    const set = new Set<string>();
   100→   100→    for (const r of pendingChangeRequests) {
   101→   101→      if (r.status === 'pending') set.add(r.field_name);
   102→   102→    }
   103→   103→    return set;
   104→   104→  }, [pendingChangeRequests]);
   105→   105→
   106→   106→  const getValue = (key: string): string => {
   107→   107→    const val = (employee as unknown as Record<string, unknown>)[key];
   108→   108→    return (val as string) || '';
   109→   109→  };
   110→   110→
   111→   111→  // Initialize form with current values for all editable fields
   112→   112→  useEffect(() => {
   113→   113→    const initial: Record<string, string> = {};
   114→   114→    for (const f of FIELD_RULES) {
   115→   115→      if (f.rule !== 'readonly') {
   116→   116→        initial[f.key] = getValue(f.key);
   117→   117→      }
   118→   118→    }
   119→   119→    setFormValues(initial);
   120→   120→  }, [employee]); // eslint-disable-line react-hooks/exhaustive-deps
   121→   121→
   122→   122→  // Only show sections that have at least one editable field
   123→   123→  const sectionsWithFields = useMemo(() => {
   124→   124→    return FIELD_SECTIONS.map(sectionDef => {
   125→   125→      const fields = FIELD_RULES.filter(f => f.section === sectionDef.key && f.rule !== 'readonly');
   126→   126→      return { ...sectionDef, fields };
   127→   127→    }).filter(s => s.fields.length > 0);
   128→   128→  }, []);
   129→   129→
   130→   130→  const updateField = (key: string, value: string) => {
   131→   131→    setFormValues(prev => ({ ...prev, [key]: value }));
   132→   132→  };
   133→   133→
   134→   134→  // Count changes for the save button badge
   135→   135→  const changedCount = useMemo(() => {
   136→   136→    let count = 0;
   137→   137→    for (const f of FIELD_RULES) {
   138→   138→      if (f.rule === 'readonly') continue;
   139→   139→      if (f.inputType === 'photo') continue; // handled separately below
   140→   140→      if (f.inputType === 'image') continue; // handled separately below
   141→   141→      const current = getValue(f.key);
   142→   142→      const newVal = formValues[f.key] ?? '';
   143→   143→      if (newVal !== current) count++;
   144→   144→    }
   145→   145→    // Photo change
   146→   146→    const currentPhoto = getValue('profile_pic_url');
   147→   147→    if (pendingPhotoUrl && pendingPhotoUrl !== currentPhoto) count++;
   148→   148→    // Document image changes
   149→   149→    for (const f of FIELD_RULES) {
   150→   150→      if (f.inputType !== 'image') continue;
   151→   151→      const current = getValue(f.key);
   152→   152→      const pending = pendingImageUrls[f.key];
   153→   153→      if (pending && pending !== current) count++;
   154→   154→    }
   155→   155→    return count;
   156→   156→  }, [formValues, employee, pendingPhotoUrl, pendingImageUrls]); // eslint-disable-line react-hooks/exhaustive-deps
   157→   157→
   158→   158→  // ── Single Save Changes handler ──
   159→   159→  // Auto-routes: free fields → direct save, approval fields → change requests
   160→   160→  const handleSave = async () => {
   161→   161→    setSaving(true);
   162→   162→    try {
   163→   163→      const freeChanged: Record<string, string | null> = {};
   164→   164→      const approvalFields: { field: FieldRule; newVal: string }[] = [];
   165→   165→
   166→   166→      for (const f of FIELD_RULES) {
   167→   167→        if (f.rule === 'readonly') continue;
   168→   168→        if (f.inputType === 'photo') continue; // handle photo separately
   169→   169→        if (f.inputType === 'image') continue; // handle image uploads separately
   170→   170→
   171→   171→        const current = getValue(f.key);
   172→   172→        const newVal = formValues[f.key] ?? '';
   173→   173→        if (newVal === current) continue; // no change
   174→   174→
   175→   175→        const effectiveRule = resolveEffectiveRule(f.rule, current);
   176→   176→
   177→   177→        if (effectiveRule === 'free') {
   178→   178→          freeChanged[f.key] = newVal || null;
   179→   179→        } else if (effectiveRule === 'admin_approval') {
   180→   180→          if (!newVal) continue; // skip empty values for approval fields
   181→   181→          approvalFields.push({ field: f, newVal });
   182→   182→        }
   183→   183→      }
   184→   184→
   185→   185→      // Handle photo (always needs approval)
   186→   186→      const currentPhoto = getValue('profile_pic_url');
   187→   187→      if (pendingPhotoUrl && pendingPhotoUrl !== currentPhoto) {
   188→   188→        approvalFields.push({
   189→   189→          field: FIELD_RULES.find(f => f.key === 'profile_pic_url')!,
   190→   190→          newVal: pendingPhotoUrl,
   191→   191→        });
   192→   192→      }
   193→   193→
   194→   194→      // Handle KYC document image uploads (always need approval)
   195→   195→      for (const f of FIELD_RULES) {
   196→   196→        if (f.inputType !== 'image') continue;
   197→   197→        const current = getValue(f.key);
   198→   198→        const pending = pendingImageUrls[f.key];
   199→   199→        if (pending && pending !== current) {
   200→   200→          approvalFields.push({ field: f, newVal: pending });
   201→   201→        }
   202→   202→      }
   203→   203→
   204→   204→      const totalChanges = Object.keys(freeChanged).length + approvalFields.length;
   205→   205→      if (totalChanges === 0) {
   206→   206→        toast.info('No changes to save.');
   207→   207→        setSaving(false);
   208→   208→        return;
   209→   209→      }
   210→   210→
   211→   211→      // 1. Save free fields directly
   212→   212→      let savedCount = 0;
   213→   213→      if (Object.keys(freeChanged).length > 0) {
   214→   214→        const result = await onSaveFreeFields(freeChanged);
   215→   215→        if (result.success) {
   216→   216→          savedCount = Object.keys(freeChanged).length;
   217→   217→        } else {
   218→   218→          toast.error(result.error || 'Failed to save some fields');
   219→   219→          setSaving(false);
   220→   220→          return;
   221→   221→        }
   222→   222→      }
   223→   223→
   224→   224→      // 2. Submit change requests for approval-needed fields
   225→   225→      let requestCount = 0;
   226→   226→      let requestErrors = 0;
   227→   227→      for (const { field, newVal } of approvalFields) {
   228→   228→        if (pendingFieldNames.has(field.key)) continue; // already pending
   229→   229→        const result = await onSubmitChangeRequest({
   230→   230→          field_name: field.key,
   231→   231→          old_value: getValue(field.key),
   232→   232→          new_value: newVal,
   233→   233→        });
   234→   234→        if (result.success) {
   235→   235→          requestCount++;
   236→   236→        } else {
   237→   237→          requestErrors++;
   238→   238→        }
   239→   239→      }
   240→   240→
   241→   241→      // 3. Summary toast
   242→   242→      if (requestErrors > 0) {
   243→   243→        toast.error(`Some change requests failed. ${savedCount} field${savedCount !== 1 ? 's' : ''} saved, ${requestCount} request${requestCount !== 1 ? 's' : ''} submitted.`);
   244→   244→      } else if (savedCount > 0 && requestCount > 0) {
   245→   245→        toast.success(`${savedCount} field${savedCount !== 1 ? 's' : ''} saved directly, ${requestCount} change request${requestCount !== 1 ? 's' : ''} submitted for approval.`);
   246→   246→      } else if (savedCount > 0) {
   247→   247→        toast.success('Profile updated successfully');
   248→   248→      } else if (requestCount > 0) {
   249→   249→        toast.success(`${requestCount} change request${requestCount !== 1 ? 's' : ''} submitted for HR approval.`);
   250→   250→      }
   251→   251→
   252→   252→      onBack();
   253→   253→    } catch {
   254→   254→      toast.error('An unexpected error occurred');
   255→   255→    } finally {
   256→   256→      setSaving(false);
   257→   257→    }
   258→   258→  };
   259→   259→
   260→   260→  // ── Profile Photo Upload ──
   261→   261→  const handlePhotoSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   262→   262→    const file = e.target.files?.[0];
   263→   263→    if (!file) return;
   264→   264→
   265→   265→    if (!file.type.startsWith('image/')) {
   266→   266→      toast.error('Please select an image file');
   267→   267→      return;
   268→   268→    }
   269→   269→    if (file.size > 5 * 1024 * 1024) {
   270→   270→      toast.error('Image must be less than 5MB');
   271→   271→      return;
   272→   272→    }
   273→   273→
   274→   274→    setUploadingPhoto(true);
   275→   275→    try {
   276→   276→      const formData = new FormData();
   277→   277→      formData.append('photo', file);
   278→   278→      formData.append('employee_id', String(employee.id));
   279→   279→
   280→   280→      const API_BASE = (import.meta as Record<string, Record<string, string>>).env?.VITE_API_BASE_URL ?? '';
   281→   281→      const API_KEY = (import.meta as Record<string, Record<string, string>>).env?.VITE_API_KEY ?? '';
   282→   282→      const token = typeof window !== 'undefined' ? localStorage.getItem('ess_token') || '' : '';
   283→   283→
   284→   284→      const resp = await fetch(`${API_BASE}/api/ess/upload.php`, {
   285→   285→        method: 'POST',
   286→   286→        headers: { 'X-API-KEY': API_KEY, 'Authorization': `Bearer ${token}` },
   287→   287→        body: formData,
   288→   288→      });
   289→   289→
   290→   290→      const json = await resp.json();
   291→   291→      if (json.success && json.data?.url) {
   292→   292→        setPendingPhotoUrl(json.data.url);
   293→   293→        setFormValues(prev => ({ ...prev, profile_pic_url: json.data.url }));
   294→   294→        toast.success('Photo uploaded. Will be submitted for approval when you Save.');
   295→   295→      } else {
   296→   296→        toast.error(json.error || 'Upload failed');
   297→   297→      }
   298→   298→    } catch {
   299→   299→      toast.error('Upload failed. Please try again.');
   300→   300→    } finally {
   301→   301→      setUploadingPhoto(false);
   302→   302→      if (fileInputRef.current) fileInputRef.current.value = '';
   303→   303→    }
   304→   304→  };
   305→   305→
   306→   306→  // ── KYC Document Image Upload ──
   307→   307→  const handleImageSelect = async (fieldKey: string, folder: string, filename: string, e: React.ChangeEvent<HTMLInputElement>) => {
   308→   308→    const file = e.target.files?.[0];
   309→   309→    if (!file) return;
   310→   310→
   311→   311→    if (!file.type.startsWith('image/')) {
   312→   312→      toast.error('Please select an image file');
   313→   313→      return;
   314→   314→    }
   315→   315→    if (file.size > 5 * 1024 * 1024) {
   316→   316→      toast.error('Image must be less than 5MB');
   317→   317→      return;
   318→   318→    }
   319→   319→
   320→   320→    setUploadingImageKey(fieldKey);
   321→   321→    try {
   322→   322→      // Convert file to base64 for upload-base64 endpoint
   323→   323→      const base64Data = await new Promise<string>((resolve, reject) => {
   324→   324→        const reader = new FileReader();
   325→   325→        reader.onload = () => resolve(reader.result as string);
   326→   326→        reader.onerror = reject;
   327→   327→        reader.readAsDataURL(file);
   328→   328→      });
   329→   329→
   330→   330→      const { url, error } = await uploadBase64Image(base64Data, filename, folder);
   331→   331→      if (url) {
   332→   332→        setPendingImageUrls(prev => ({ ...prev, [fieldKey]: url }));
   333→   333→        setFormValues(prev => ({ ...prev, [fieldKey]: url }));
   334→   334→        toast.success('Document uploaded. Will be submitted for approval when you Save.');
   335→   335→      } else {
   336→   336→        toast.error(error || 'Upload failed');
   337→   337→      }
   338→   338→    } catch {
   339→   339→      toast.error('Upload failed. Please try again.');
   340→   340→    } finally {
   341→   341→      setUploadingImageKey(null);
   342→   342→      // Reset file input
   343→   343→      const inputRef = imageInputRefs.current[fieldKey];
   344→   344→      if (inputRef) inputRef.value = '';
   345→   345→    }
   346→   346→  };
   347→   347→
   348→   348→  // Render input based on field type
   349→   349→  const renderInput = (
   350→   350→    field: FieldRule,
   351→   351→    value: string,
   352→   352→    onChange: (v: string) => void,
   353→   353→    disabled: boolean,
   354→   354→  ) => {
   355→   355→    const inputId = `field-${field.key}`;
   356→   356→
   357→   357→    if (field.inputType === 'select' && field.options) {
   358→   358→      const validValue = (value && field.options.includes(value)) ? value : '';
   359→   359→      return (
   360→   360→        <Select value={validValue} onValueChange={onChange} disabled={disabled}>
   361→   361→          <SelectTrigger id={inputId}>
   362→   362→            <SelectValue placeholder={`Select ${field.label.toLowerCase()}`} />
   363→   363→          </SelectTrigger>
   364→   364→          <SelectContent>
   365→   365→            {field.options.map(opt => (
   366→   366→              <SelectItem key={opt} value={opt}>{opt}</SelectItem>
   367→   367→            ))}
   368→   368→          </SelectContent>
   369→   369→        </Select>
   370→   370→      );
   371→   371→    }
   372→   372→
   373→   373→    if (field.inputType === 'textarea') {
   374→   374→      return (
   375→   375→        <Textarea
   376→   376→          id={inputId}
   377→   377→          placeholder={`Enter ${field.label.toLowerCase()}`}
   378→   378→          value={value}
   379→   379→          onChange={e => onChange(e.target.value)}
   380→   380→          rows={3}
   381→   381→          disabled={disabled}
   382→   382→        />
   383→   383→      );
   384→   384→    }
   385→   385→
   386→   386→    if (field.inputType === 'date') {
   387→   387→      return (
   388→   388→        <Input
   389→   389→          id={inputId}
   390→   390→          type="date"
   391→   391→          value={value}
   392→   392→          onChange={e => onChange(e.target.value)}
   393→   393→          disabled={disabled}
   394→   394→        />
   395→   395→      );
   396→   396→    }
   397→   397→
   398→   398→    if (field.inputType === 'tel') {
   399→   399→      return (
   400→   400→        <Input
   401→   401→          id={inputId}
   402→   402→          type="tel"
   403→   403→          placeholder={`Enter ${field.label.toLowerCase()}`}
   404→   404→          value={value}
   405→   405→          onChange={e => onChange(e.target.value)}
   406→   406→          disabled={disabled}
   407→   407→        />
   408→   408→      );
   409→   409→    }
   410→   410→
   411→   411→    if (field.inputType === 'email') {
   412→   412→      return (
   413→   413→        <Input
   414→   414→          id={inputId}
   415→   415→          type="email"
   416→   416→          placeholder={`Enter ${field.label.toLowerCase()}`}
   417→   417→          value={value}
   418→   418→          onChange={e => onChange(e.target.value)}
   419→   419→          disabled={disabled}
   420→   420→        />
   421→   421→      );
   422→   422→    }
   423→   423→
   424→   424→    return (
   425→   425→      <Input
   426→   426→        id={inputId}
   427→   427→        type="text"
   428→   428→        placeholder={`Enter ${field.label.toLowerCase()}`}
   429→   429→        value={value}
   430→   430→        onChange={e => onChange(e.target.value)}
   431→   431→        disabled={disabled}
   432→   432→      />
   433→   433→    );
   434→   434→  };
   435→   435→
   436→   436→  // Render a single field row — clean, no per-field buttons
   437→   437→  const renderField = (field: FieldRule) => {
   438→   438→    if (field.rule === 'readonly') return null;
   439→   439→
   440→   440→    const hasPending = pendingFieldNames.has(field.key);
   441→   441→    const effectiveRule = resolveEffectiveRule(field.rule, getValue(field.key));
   442→   442→    const current = getValue(field.key);
   443→   443→    const newVal = formValues[field.key] ?? '';
   444→   444→    const isChanged = newVal !== current;
   445→   445→
   446→   446→    // ── Special: Profile Photo ──
   447→   447→    if (field.inputType === 'photo') {
   448→   448→      const currentPhoto = current;
   449→   449→      const displayPhoto = pendingPhotoUrl || currentPhoto;
   450→   450→      const photoChanged = !!pendingPhotoUrl && pendingPhotoUrl !== currentPhoto;
   451→   451→
   452→   452→      return (
   453→   453→        <div key={field.key} className="space-y-2">
   454→   454→          <div className="flex items-center justify-between">
   455→   455→            <Label className="text-sm font-medium">{field.label}</Label>
   456→   456→            {hasPending ? (
   457→   457→              <Badge variant="outline" className="bg-amber-50 text-amber-600 border-amber-200 text-[10px] px-1.5 py-0">
   458→   458→                <Clock className="w-3 h-3 mr-0.5" /> Pending
   459→   459→              </Badge>
   460→   460→            ) : (
   461→   461→              <Badge variant="outline" className="bg-amber-50 text-amber-600 border-amber-200 text-[10px] px-1.5 py-0">
   462→   462→                Needs approval
   463→   463→              </Badge>
   464→   464→            )}
   465→   465→          </div>
   466→   466→          <div className="flex items-center gap-4">
   467→   467→            <div className="w-16 h-16 rounded-full bg-gray-100 border-2 border-gray-200 overflow-hidden flex-shrink-0">
   468→   468→              {displayPhoto ? (
   469→   469→                <img
   470→   470→                  src={getFileUrl(displayPhoto)}
   471→   471→                  alt="Profile"
   472→   472→                  className="w-full h-full object-cover"
   473→   473→                  onError={e => { (e.target as HTMLImageElement).style.display = 'none'; }}
