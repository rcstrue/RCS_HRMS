   290→   290→    // Get compliance calendar
   291→   291→    public function getComplianceCalendar($month = null, $year = null) {
   292→   292→        $sql = "SELECT cc.*, s.state_name,
   293→   293→                CASE
   294→   294→                    WHEN cc.due_date < CURDATE() THEN 'Overdue'
   295→   295→                    WHEN cc.due_date <= DATE_ADD(CURDATE(), INTERVAL 7 DAY) THEN 'Due Soon'
   296→   296→                    ELSE 'Upcoming'
   297→   297→                END as urgency
   298→   298→                FROM compliance_calendar cc
   299→   299→                LEFT JOIN states s ON cc.state_id = s.id
   300→   300→                WHERE cc.is_active = 1";
   301→   301→        $params = [];
   302→   302→
   303→   303→        if ($month && $year) {
   304→   304→            $sql .= " AND MONTH(cc.due_date) = :month AND YEAR(cc.due_date) = :year";
   305→   305→            $params['month'] = $month;
   306→   306→            $params['year'] = $year;
   307→   307→        } elseif ($year) {
   308→   308→            $sql .= " AND YEAR(cc.due_date) = :year";
   309→   309→            $params['year'] = $year;
   310→   310→        }
   311→   311→
   312→   312→        $sql .= " ORDER BY cc.due_date ASC";

   533→   533→                 WHERE mw.is_active = 1

   538→   538→
   539→   539→            foreach ($wageUpdates as $w) {
   540→   540→                $notifications[] = [
