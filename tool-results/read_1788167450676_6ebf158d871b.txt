    35→ * Handle file uploads to web root
    36→ * @param array $file The $_FILES array element
    37→ * @param string $uploadDir Directory to upload to (relative to web root)
    38→ * @return array|null Result with 'path' or 'error' key, or null if no file
    39→ */
    40→function handleFileUpload($file, $uploadDir = 'uploads/profile/') {
    41→    $response = null;
    42→    $maxSize = 5 * 1024 * 1024; // 5MB
    43→
    44→    // Extension whitelist — NEVER trust the client-supplied filename extension
    45→    $allowedImageExtensions = ['jpg', 'jpeg', 'png', 'gif'];
    46→    $allowedPdfExtension    = ['pdf'];
    47→    $allowedExtensions      = array_merge($allowedImageExtensions, $allowedPdfExtension);
    48→
    49→    // Map of real MIME type (from finfo) → safe forced extension
    50→    $mimeToExtension = [
    51→        'image/jpeg' => 'jpg',
    52→        'image/png'  => 'png',
    53→        'image/gif'  => 'gif',
    54→        'application/pdf' => 'pdf',
    55→    ];
    56→
    57→    // Check if file was uploaded via PHP
    58→    if (!isset($file['tmp_name']) || empty($file['tmp_name'])) {
    59→        return $response;
    60→    }
    61→
    62→    // Check for upload errors
    63→    if ($file['error'] !== UPLOAD_ERR_OK) {
    64→        return $response;
    65→    }
    66→
    67→    // Validate file size
    68→    if ($file['size'] > $maxSize) {
    69→        $response = ['error' => 'File size must be less than 5MB.'];
    70→        return $response;
    71→    }
    72→
    73→    // ─── Validate REAL file content with finfo (magic bytes), not client Content-Type ───
    74→    $finfo = new finfo(FILEINFO_MIME_TYPE);
    75→    $detectedMime = $finfo->file($file['tmp_name']);
    76→
    77→    if (!isset($mimeToExtension[$detectedMime])) {
    78→        $response = ['error' => 'Invalid file type. Only JPG, PNG, GIF, PDF allowed.'];
    79→        return $response;
    80→    }
    81→
    82→    // Force the extension from the detected MIME type — ignore the uploaded filename entirely
    83→    $safeExtension = $mimeToExtension[$detectedMime];
    84→
    85→    // For images, also validate with getimagesize() to confirm it's a real image
    86→    if (in_array($safeExtension, $allowedImageExtensions)) {
    87→        $imageInfo = @getimagesize($file['tmp_name']);
    88→        if ($imageInfo === false) {
    89→            $response = ['error' => 'Invalid image file. The content is not a valid image.'];
    90→            return $response;
    91→        }
    92→        // Re-encode images server-side to strip any embedded payloads
    93→        $destImage = null;
    94→        switch ($safeExtension) {
    95→            case 'jpg':
    96→            case 'jpeg':
    97→                $src = @imagecreatefromjpeg($file['tmp_name']);
    98→                if ($src) { $destImage = $src; }
    99→                break;
   100→            case 'png':
   101→                $src = @imagecreatefrompng($file['tmp_name']);
   102→                if ($src) { $destImage = $src; }
   103→                break;
   104→            case 'gif':
   105→                $src = @imagecreatefromgif($file['tmp_name']);
   106→                if ($src) { $destImage = $src; }
   107→                break;
   108→        }
   109→        if (!$destImage) {
   110→            $response = ['error' => 'Failed to process image. Please upload a valid image file.'];
   111→            return $response;
   112→        }
   113→    }
   114→
   115→    // Get web root (parent of APP_ROOT which is hrms folder)
   116→    $webRoot = dirname(APP_ROOT);
   117→
   118→    // Create upload directory if not exists
   119→    $fullPath = $webRoot . '/' . $uploadDir;
   120→    if (!is_dir($fullPath)) {
   121→        mkdir($fullPath, 0755, true);
   122→    }
   123→
   124→    // Generate unique filename using ONLY the safe, forced extension
   125→    $filename = uniqid() . '_' . time() . '.' . $safeExtension;
   126→
   127→    // Database path starts with / for web root
   128→    $dbPath = '/' . $uploadDir . $filename;
   129→
   130→    // Save file: re-encode images via GD to strip payloads; copy PDFs directly
   131→    $targetPath = $fullPath . $filename;
   132→    if (in_array($safeExtension, $allowedImageExtensions)) {
   133→        // Re-encode and save via GD (strips EXIF, comments, embedded scripts)
   134→        $saveOk = false;
   135→        switch ($safeExtension) {
   136→            case 'jpg':
   137→            case 'jpeg':
   138→                $saveOk = imagejpeg($destImage, $targetPath, 90);
   139→                break;
   140→            case 'png':
   141→                $saveOk = imagepng($destImage, $targetPath);
   142→                break;
   143→            case 'gif':
   144→                $saveOk = imagegif($destImage, $targetPath);
   145→                break;
   146→        }
   147→        imagedestroy($destImage);
   148→        if (!$saveOk) {
   149→            $response = ['error' => 'Failed to save processed image.'];
   150→            return $response;
   151→        }
   152→    } else {
   153→        // PDF: just move the validated file (finfo already confirmed it's a PDF)
   154→        if (!move_uploaded_file($file['tmp_name'], $targetPath)) {
   155→            $response = ['error' => 'Failed to upload file.'];
   156→            return $response;
   157→        }
   158→    }
   159→
   160→    $response = ['path' => $dbPath];
   161→    return $response;
   162→}

   909→// Auto-populate Worker Category from the selected Designation
   910→// (the option carries a data-category attribute sourced from the
   911→//  designations.worker_category column)
   912→function autoFillWorkerCategory(designationSelect) {
   913→    const opt = designationSelect.options[designationSelect.selectedIndex];
   914→    const cat = opt ? (opt.getAttribute('data-category') || '') : '';
   915→    const wcSelect = document.getElementById('worker_category');
   916→    if (!wcSelect || !cat) return;
   917→
   918→    // Try exact match first
   919→    let matched = false;
   920→    for (const o of wcSelect.options) {
   921→        if (o.value === cat) { o.selected = true; matched = true; break; }
   922→    }
   923→    // Fallback: case-insensitive match (handles "Semi-skilled" vs "Semi-Skilled")
   924→    if (!matched) {
   925→        const lower = cat.toLowerCase();
   926→        for (const o of wcSelect.options) {
   927→            if (o.value.toLowerCase() === lower) { o.selected = true; matched = true; break; }
   928→        }
   929→    }
   930→    if (matched) {
   931→        wcSelect.classList.add('is-valid');
   932→        setTimeout(() => wcSelect.classList.remove('is-valid'), 1200);
   933→    }
   934→}
