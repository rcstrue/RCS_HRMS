    79→if (!function_exists('menuPermissionsTableExists')) {
    80→    function menuPermissionsTableExists($db) {
    81→        try {

    90→/**
    91→ * Sync role permissions to role_menu_permissions table for sidebar visibility.
    92→ * Maps each permission module to its corresponding sidebar menu key.
    93→ */
    94→if (!function_exists('syncMenuPermissions')) {
    95→    function syncMenuPermissions($db, $roleId, $permissions) {
    96→        if (!menuPermissionsTableExists($db)) {
    97→            return false;
    98→        }
    99→        
   100→        // Map permission module keys to sidebar menu keys
   101→        $menuMap = [
   102→            'dashboard'  => 'dashboard',
   103→            'employee'   => 'employee',
   104→            'attendance' => 'attendance',
   105→            'payroll'    => 'payroll',
   106→            'client'     => 'client',
   107→            'unit'       => 'unit',
   108→            'forms'      => 'forms',
   109→            'leave'      => 'leave',
   110→            'report'     => 'report',
   111→            'settings'   => 'settings',
   112→        ];
   113→        
   114→        // Get all menus from auth
   115→        global $auth;
   116→        $menus = $auth ? $auth->getAllMenus() : [];
   117→        
   118→        if (empty($menus)) {
   119→            return false;
   120→        }
   121→        
   122→        try {
   123→            // Clear existing permissions for this role
   124→            $db->query("DELETE FROM role_menu_permissions WHERE role_id = :role_id", ['role_id' => $roleId]);
   125→            
   126→            foreach ($menus as $menuKey => $menuInfo) {
   127→                $isVisible = 0;
   128→                
   129→                // Check if user has view permission for this menu's module
   130→                foreach ($menuMap as $permModule => $menuMatch) {
   131→                    if (strpos($menuKey, $menuMatch) !== false || $menuKey === $menuMatch) {
   132→                        if (isset($permissions[$permModule]['view']) && $permissions[$permModule]['view']) {
   133→                            $isVisible = 1;
   134→                        }
   135→                        break;
   136→                    }
   137→                }
   138→                
   139→                $db->insert('role_menu_permissions', [
   140→                    'role_id'     => $roleId,
   141→                    'menu_key'    => $menuKey,
   142→                    'submenu_key' => null,
   143→                    'is_visible'  => $isVisible,
   144→                    'can_view'    => $isVisible,
   145→                    'can_add'     => isset($permissions[$menuKey]['add']) ? 1 : 0,
   146→                    'can_edit'    => isset($permissions[$menuKey]['edit']) ? 1 : 0,
   147→                    'can_delete'  => isset($permissions[$menuKey]['delete']) ? 1 : 0,
   148→                ]);
   149→                
   150→                // Handle submenus
   151→                if (!empty($menuInfo['submenus'])) {
   152→                    foreach ($menuInfo['submenus'] as $submenuKey => $submenuInfo) {
   153→                        $db->insert('role_menu_permissions', [
   154→                            'role_id'     => $roleId,
   155→                            'menu_key'    => $menuKey,
   156→                            'submenu_key' => $submenuKey,
   157→                            'is_visible'  => $isVisible,
   158→                            'can_view'    => $isVisible,
   159→                            'can_add'     => 0,
   160→                            'can_edit'    => 0,
   161→                            'can_delete'  => 0,
   162→                        ]);
   163→                    }
   164→                }
   165→            }
   166→            return true;
   167→        } catch (Exception $e) {
   168→            error_log('syncMenuPermissions error: ' . $e->getMessage());
   169→            return false;
   170→        }
   171→    }
   172→}

   221→    }

   245→            'level'       => $level,
   246→            'is_active'   => $isActive,
   247→        ], 'id = :id', ['id' => $roleId]);

   249→        syncMenuPermissions($db, $roleId, $permissions);
   250→        logActivity('update', 'roles', $roleId, "Updated role: $roleName");
   251→        setFlash('success', 'Role updated successfully!');
   252→        redirect('index.php?page=settings/roles');
   253→    }
   254→    
   255→    if ($action === 'delete' && isset($_POST['role_id'])) {
   256→        $roleId = (int)$_POST['role_id'];
   257→        $userCount = $db->fetch("SELECT COUNT(*) as count FROM users WHERE role_id = :id", ['id' => $roleId]);
   258→        
   259→        if ($userCount['count'] > 0) {
   260→            setFlash('error', 'Cannot delete role. It is assigned to ' . $userCount['count'] . ' user(s).');
   261→        } else {
   262→            try {
   263→                if (menuPermissionsTableExists($db)) {
   264→                    $db->query("DELETE FROM role_menu_permissions WHERE role_id = :id", ['id' => $roleId]);
   265→                }
   266→            } catch (Exception $e) {}
   267→            
   268→            $db->delete('roles', 'id = :id', ['id' => $roleId]);
   269→            logActivity('delete', 'roles', $roleId, "Deleted role");
   270→            setFlash('success', 'Role deleted successfully!');
   271→        }
   272→        redirect('index.php?page=settings/roles');
   273→    }
   274→    
   275→    if ($action === 'toggle' && isset($_POST['role_id'])) {
   276→        $roleId = (int)$_POST['role_id'];
   277→        $role = $db->fetch("SELECT * FROM roles WHERE id = :id", ['id' => $roleId]);
   278→        
   279→        if ($role) {
   280→            $newStatus = $role['is_active'] ? 0 : 1;
   281→            $db->update('roles', ['is_active' => $newStatus], 'id = :id', ['id' => $roleId]);
   282→            setFlash('success', 'Role status updated!');
   283→        }
   284→        redirect('index.php?page=settings/roles');
   285→    }

   575→function editRole(role) {
   576→    document.getElementById('edit_role_id').value = role.id;
   577→    document.getElementById('edit_role_name').value = role.role_name;
   578→    document.getElementById('edit_role_code').value = role.role_code;
   579→    document.getElementById('edit_description').value = role.description || '';
   580→    document.getElementById('edit_level').value = role.level || 1;
   581→    document.getElementById('edit_is_active').checked = role.is_active == 1;
   582→    
   583→    var perms = JSON.parse(role.permissions || '{}');
   584→    document.querySelectorAll('[id^="edit_perm_"]').forEach(function(cb) { cb.checked = false; });
   585→    
   586→    for (var mod in perms) {
   587→        for (var action in perms[mod]) {
   588→            var cb = document.getElementById('edit_perm_' + mod + '_' + action);
   589→            if (cb) cb.checked = perms[mod][action];
   590→        }
   591→    }
   592→    
   593→    new bootstrap.Modal(document.getElementById('editRoleModal')).show();
   594→}
   595→
   596→function deleteRole(id, name) {
   597→    document.getElementById('delete_role_id').value = id;
   598→    document.getElementById('delete_role_name').textContent = name;
   599→    new bootstrap.Modal(document.getElementById('deleteRoleModal')).show();
   600→}

   602→function toggleModulePerms(prefix, module) {
   603→    var boxes = document.querySelectorAll('.' + prefix + '-' + module);
   604→    var allChecked = Array.from(boxes).every(function(cb) { return cb.checked; });
   605→    boxes.forEach(function(cb) { cb.checked = !allChecked; });
   606→}
