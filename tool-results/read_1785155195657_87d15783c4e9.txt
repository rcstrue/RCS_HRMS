   552→    // ── EMI Preview Calculator ──
   553→    function calculateEmiPreview() {
   554→        var amount = parseFloat($('#loanAmount').val()) || 0;
   555→        var rate = parseFloat($('#loanInterestRate').val()) || 0;
   556→        var tenure = parseInt($('#loanTenure').val()) || 0;
   557→
   558→        if (amount > 0 && tenure > 0) {
   559→            var emi, totalInterest, totalRepayable;
   560→            if (rate <= 0) {
   561→                emi = amount / tenure;
   562→                totalInterest = 0;
   563→                totalRepayable = amount;
   564→            } else {
   565→                var monthlyRate = rate / 12 / 100;
   566→                var onePlusR = 1 + monthlyRate;
   567→                var power = Math.pow(onePlusR, tenure);
   568→                emi = amount * monthlyRate * power / (power - 1);
   569→                totalRepayable = emi * tenure;
   570→                totalInterest = totalRepayable - amount;
   571→            }
   572→            $('#previewEmi').text('₹' + emi.toFixed(2));
   573→            $('#previewInterest').text('₹' + totalInterest.toFixed(2));
   574→            $('#previewTotal').text('₹' + totalRepayable.toFixed(2));
   575→            $('#emiPreview').slideDown();
   576→        } else {
   577→            $('#emiPreview').slideUp();
   578→        }
   579→    }
   580→
   581→    $('#loanAmount, #loanInterestRate, #loanTenure').on('input', calculateEmiPreview);
