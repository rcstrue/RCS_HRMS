   587→// Auto-calculate gross and totals
   588→document.querySelectorAll('.salary-input').forEach(input => {
   589→    input.addEventListener('input', calculateRowGross);
   590→});
   591→
   592→function calculateRowGross(e) {
   593→    const empId = e?.target?.dataset?.empId;
   594→    if (empId) {
   595→        const form = document.getElementById('salaryForm');
   596→        const basic = parseFloat(form.querySelector(`[name="basic_da[${empId}]"]`)?.value) || 0;
   597→        const hra = parseFloat(form.querySelector(`[name="hra[${empId}]"]`)?.value) || 0;
   598→        const leaveEnc = parseFloat(form.querySelector(`[name="leave_encashment[${empId}]"]`)?.value) || 0;
   599→        const bonusEnc = parseFloat(form.querySelector(`[name="bonus_encashment[${empId}]"]`)?.value) || 0;
   600→        const washing = parseFloat(form.querySelector(`[name="washing_allowance[${empId}]"]`)?.value) || 0;
   601→        const gross = basic + hra + leaveEnc + bonusEnc + washing;
   602→        
   603→        const grossEl = document.getElementById('gross_' + empId);
   604→        if (grossEl) grossEl.textContent = '₹' + gross.toLocaleString('en-IN');
   605→    }
   606→    calculateTotals();
   607→}
   608→
   609→function calculateTotals() {
   610→    let totalBasic = 0, totalHra = 0, totalLeaveEnc = 0, totalBonusEnc = 0, totalWashing = 0, totalGross = 0;
   611→    
   612→    document.querySelectorAll('[name^="basic_da["]').forEach(input => {
   613→        totalBasic += parseFloat(input.value) || 0;
   614→    });
   615→    document.querySelectorAll('[name^="hra["]').forEach(input => {
   616→        totalHra += parseFloat(input.value) || 0;
   617→    });
   618→    document.querySelectorAll('[name^="leave_encashment["]').forEach(input => {
   619→        totalLeaveEnc += parseFloat(input.value) || 0;
   620→    });
   621→    document.querySelectorAll('[name^="bonus_encashment["]').forEach(input => {
   622→        totalBonusEnc += parseFloat(input.value) || 0;
   623→    });
   624→    document.querySelectorAll('[name^="washing_allowance["]').forEach(input => {
   625→        totalWashing += parseFloat(input.value) || 0;
   626→    });
   627→    
   628→    totalGross = totalBasic + totalHra + totalLeaveEnc + totalBonusEnc + totalWashing;
   629→    
   630→    const fmt = (v) => '₹' + v.toLocaleString('en-IN');
   631→    document.getElementById('total_basic_da').textContent = fmt(totalBasic);
   632→    document.getElementById('total_hra').textContent = fmt(totalHra);
   633→    document.getElementById('total_leave_enc').textContent = fmt(totalLeaveEnc);
   634→    document.getElementById('total_bonus_enc').textContent = fmt(totalBonusEnc);
   635→    document.getElementById('total_washing').textContent = fmt(totalWashing);
   636→    document.getElementById('total_gross_col').textContent = fmt(totalGross);
   637→}
