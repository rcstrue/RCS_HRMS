   227→   227→    /**
   228→   228→     * Process payroll for period with selective filters
   229→   229→     * 
   230→   230→     * @param int $periodId The payroll period ID
   231→   231→     * @param array $filters Optional filters for selective processing
   232→   232→     *                       - client_id: Process only this client's employees
   233→   233→     *                       - unit_id: Process only this unit's employees
   234→   234→     *                       - employee_codes: Array of specific employee codes
   235→   235→     *                       - recalculate_dirty: Only recalculate dirty records
   236→   236→     * @return array Result with success status and processed count
   237→   237→     */
   238→   238→    public function processPayroll($periodId, $filters = []) {
   239→   239→        // Get period info
   240→   240→        $period = $this->db->fetch(
   241→   241→            "SELECT * FROM payroll_periods WHERE id = :id",
   242→   242→            ['id' => $periodId]
   243→   243→        );
   244→   244→
   245→   245→        if (!$period) {
   246→   246→            return ['success' => false, 'message' => 'Payroll period not found.'];
   247→   247→        }
   248→   248→
   249→   249→        // Check if period is frozen
   250→   250→        if ($period['status'] === 'Frozen' || $period['status'] === 'Locked') {
   251→   251→            return ['success' => false, 'message' => 'Payroll period is frozen. Cannot process.'];
   252→   252→        }
   253→   253→
   254→   254→        // Ensure payroll table has extra_days_amount column
   255→   255→        try {
   256→   256→            $checkColumn = $this->db->fetch("SHOW COLUMNS FROM payroll LIKE 'extra_days_amount'");
   257→   257→            if (!$checkColumn) {
   258→   258→                $this->db->exec("ALTER TABLE payroll ADD COLUMN extra_days_amount DECIMAL(10,2) DEFAULT 0.00 AFTER overtime_amount");
   259→   259→            }
   260→   260→        } catch (Exception $e) {
   261→   261→            // Column might already exist or table might not have records
   262→   262→        }
   263→   263→
   264→   264→        // Build employee query with filters
   265→   265→        $sql = "SELECT e.id, e.employee_code, e.full_name, e.date_of_joining,
   266→   266→                       e.date_of_leaving, e.client_id, e.unit_id, e.worker_category,
   267→   267→                       e.bank_name, e.account_number, e.ifsc_code,
   268→   268→                       c.name as client_name, u.name as unit_name,
   269→   269→                       ess.basic_da, ess.hra, ess.leave_encashment, ess.bonus_encashment, ess.washing_allowance,
   270→   270→                       ess.gross_salary, ess.pf_applicable, ess.esi_applicable,
   271→   271→                       ess.pt_applicable, ess.lwf_applicable, ess.overtime_applicable,
   272→   272→                       ess.bonus_applicable, ess.gratuity_applicable,
   273→   273→                       usf.pay_days_type, usf.ot_calculation_type, usf.ot_calculation_on, usf.ot_hours_per_day,
   274→   274→                       u.state as unit_state
   275→   275→                FROM employees e
   276→   276→                INNER JOIN employee_salary_structures ess ON e.id = ess.employee_id
   277→   277→                    AND (ess.effective_to IS NULL OR ess.effective_to >= CURDATE())
   278→   278→                LEFT JOIN clients c ON e.client_id = c.id
   279→   279→                LEFT JOIN units u ON e.unit_id = u.id
   280→   280→                LEFT JOIN unit_salary_formulas usf ON usf.unit_id = e.unit_id
   281→   281→                    AND usf.is_active = 1
   282→   282→                    AND (usf.effective_to IS NULL OR usf.effective_to >= CURDATE())
   283→   283→                WHERE e.status = 'approved'";
   284→   284→
   285→   285→        $params = [];
   286→   286→
   287→   287→        // Apply filters for selective processing
   288→   288→        if (!empty($filters['client_id'])) {
   289→   289→            $sql .= " AND e.client_id = :client_id";
   290→   290→            $params['client_id'] = $filters['client_id'];
   291→   291→        }
   292→   292→
   293→   293→        if (!empty($filters['unit_id'])) {
   294→   294→            $sql .= " AND e.unit_id = :unit_id";
   295→   295→            $params['unit_id'] = $filters['unit_id'];
   296→   296→        }
   297→   297→
   298→   298→        // Employee codes filter - use named params to avoid mixing with positional
   299→   299→        if (!empty($filters['employee_codes']) && is_array($filters['employee_codes'])) {
   300→   300→            $empPlaceholders = [];
   301→   301→            $idx = 1;
   302→   302→            foreach ($filters['employee_codes'] as $code) {
   303→   303→                $key = 'emp_code_' . $idx++;
   304→   304→                $empPlaceholders[] = ':' . $key;
   305→   305→                $params[$key] = $code;
   306→   306→            }
   307→   307→            $sql .= " AND e.employee_code IN (" . implode(',', $empPlaceholders) . ")";
   308→   308→        }
   309→   309→
   310→   310→        // Only recalculate dirty records if specified
   311→   311→        if (!empty($filters['recalculate_dirty'])) {
   312→   312→            $sql .= " AND EXISTS (
   313→   313→                SELECT 1 FROM payroll p 
   314→   314→                WHERE p.employee_id = e.employee_code 
   315→   315→                AND p.payroll_period_id = :period_id_check
   316→   316→                AND p.payroll_dirty = 1
   317→   317→            )";
   318→   318→            $params['period_id_check'] = $periodId;
   319→   319→        }
   320→   320→
   321→   321→        $employees = $this->db->fetchAll($sql, $params);
   322→   322→
   323→   323→        $processed = 0;
   324→   324→        $errors = [];
   325→   325→        $exceptions = [];
   326→   326→        $totalGross = 0;
   327→   327→        $totalNet = 0;
   328→   328→        $totalPF = 0;
   329→   329→        $totalESI = 0;
   330→   330→
   331→   331→        $this->db->beginTransaction();
   332→   332→
   333→   333→        try {
   334→   334→            foreach ($employees as $emp) {
   335→   335→                try {
   336→   336→                    // Check for exceptions
   337→   337→                    $exceptionType = null;
   338→   338→                    
   339→   339→                    // Check if employee joined after period end
   340→   340→                    $periodEndDate = $period['end_date'];
   341→   341→                    if ($emp['date_of_joining'] && strtotime($emp['date_of_joining']) > strtotime($periodEndDate)) {
   342→   342→                        continue;
   343→   343→                    }
   344→   344→
   345→   345→                    // Check for missing salary structure
   346→   346→                    if (empty($emp['basic_da']) && empty($emp['gross_salary'])) {
   347→   347→                        $exceptionType = 'Undefined Salary';
   348→   348→                        $exceptions[] = [
   349→   349→                            'employee_id' => $emp['employee_code'],
   350→   350→                            'employee_name' => $emp['full_name'],
   351→   351→                            'type' => 'Undefined Salary',
   352→   352→                            'message' => 'No salary structure defined'
   353→   353→                        ];
   354→   354→                    }
   355→   355→
   356→   356→                    // Check for missing bank details
   357→   357→                    if (empty($emp['bank_name']) || empty($emp['account_number']) || empty($emp['ifsc_code'])) {
   358→   358→                        if (!$exceptionType) {
   359→   359→                            $exceptionType = 'Missing Bank Details';
   360→   360→                        }
   361→   361→                        $exceptions[] = [
   362→   362→                            'employee_id' => $emp['employee_code'],
   363→   363→                            'employee_name' => $emp['full_name'],
   364→   364→                            'type' => 'Missing Bank Details',
   365→   365→                            'message' => 'Bank account details incomplete'
   366→   366→                        ];
   367→   367→                    }
   368→   368→
   369→   369→                    // Get attendance summary — prefer unit_id match, fallback to any unit (backward compatible)
   370→   370→                    $attendance = $this->db->fetch(
   371→   371→                        "SELECT
   372→   372→                            total_present as present_days,
   373→   373→                            total_extra,
   374→   374→                            overtime_hours,
   375→   375→                            total_wo as weekly_offs,
   376→   376→                            total_paid_days
   377→   377→                        FROM attendance_summary
   378→   378→                        WHERE employee_id = :emp_id
   379→   379→                        AND month = :month
   380→   380→                        AND year = :year
   381→   381→                        ORDER BY unit_id = :unit_id DESC
   382→   382→                        LIMIT 1",
   383→   383→                        ['emp_id' => $emp['id'], 'unit_id' => $emp['unit_id'], 'month' => $period['month'], 'year' => $period['year']]
   384→   384→                    );
   385→   385→
   386→   386→                    // Calculate total days based on unit's pay_days_type setting
   387→   387→                    $payDaysType = $emp['pay_days_type'] ?? 'actual';
   388→   388→                    switch ($payDaysType) {
   389→   389→                        case 'fixed_30':
   390→   390→                            $totalDays = 30;
   391→   391→                            break;
   392→   392→                        case 'previous_month':
   393→   393→                            // For 26-25 cycle: use previous month's days
   394→   394→                            $prevMonth = $period['month'] - 1;
   395→   395→                            $prevYear = $period['year'];
   396→   396→                            if ($prevMonth < 1) { $prevMonth = 12; $prevYear--; }
   397→   397→                            $totalDays = (int)date('t', strtotime("$prevYear-$prevMonth-01"));
   398→   398→                            break;
   399→   399→                        case 'calendar_minus_sundays':
   400→   400→                            // Month days minus all Sundays
   401→   401→                            $calDays = (int)cal_days_in_month(CAL_GREGORIAN, $period['month'], $period['year']);
   402→   402→                            $sundays = 0;
   403→   403→                            for ($d = 1; $d <= $calDays; $d++) {
   404→   404→                                $dow = (int)date('N', strtotime(sprintf('%04d-%02d-%02d', $period['year'], $period['month'], $d)));
   405→   405→                                if ($dow === 7) $sundays++;
   406→   406→                            }
   407→   407→                            $totalDays = $calDays - $sundays;
   408→   408→                            break;
   409→   409→                        default: // 'actual'
   410→   410→                            $totalDays = $period['pay_days'] ?? 30;
   411→   411→                            break;
   412→   412→                    }
   413→   413→                    
   414→   414→                    // Use total_paid_days from attendance_summary if available, otherwise calculate
   415→   415→                    $preCalcPaidDays = floatval($attendance['total_paid_days'] ?? 0);
   416→   416→                    if ($preCalcPaidDays > 0) {
   417→   417→                        $paidDays = $preCalcPaidDays;
   418→   418→                    } else {
   419→   419→                        // Fallback: calculate from individual components
   420→   420→                        $paidDays = floatval($attendance['total_present'] ?? 0);
   421→   421→                        $paidDays += floatval($attendance['weekly_offs'] ?? 0);
   422→   422→                        $paidDays += floatval($attendance['total_extra'] ?? 0);
   423→   423→                    }
   424→   424→                    
   425→   425→                    // Check if attendance record exists (must check is_array first to avoid warnings)
   426→   426→                    // NOTE: SQL aliases total_present → present_days, total_wo → weekly_offs
   427→   427→                    $hasAttendance = is_array($attendance) && array_key_exists('present_days', $attendance) && $attendance['present_days'] !== null;
   428→   428→
   429→   429→                    // Check for missing attendance — SKIP employee, do NOT assume full month
   430→   430→                    if (!$hasAttendance) {
   431→   431→                        $exceptions[] = [
   432→   432→                            'employee_id' => $emp['employee_code'],
   433→   433→                            'employee_name' => $emp['full_name'],
   434→   434→                            'type' => 'Missing Attendance',
   435→   435→                            'message' => 'No attendance data found — employee skipped. Upload attendance first.'
   436→   436→                        ];
   437→   437→                        continue; // Skip this employee entirely
   438→   438→                    }
   439→   439→
   440→   440→                    $unpaidDays = max(0, $totalDays - $paidDays);
   441→   441→                    $overtimeHours = floatval($attendance['overtime_hours'] ?? 0);
   442→   442→
   443→   443→                    // Calculate earnings (pro-rated) - using basic_da as combined column
   444→   444→                    $basicDa = round(($emp['basic_da'] ?? 0) * $paidDays / $totalDays, 2);
   445→   445→                    $hra = round(($emp['hra'] ?? 0) * $paidDays / $totalDays, 2);
   446→   446→                    $leaveEncashment = round(($emp['leave_encashment'] ?? 0) * $paidDays / $totalDays, 2);
   447→   447→                    $bonusEncashment = round(($emp['bonus_encashment'] ?? 0) * $paidDays / $totalDays, 2);
   448→   448→                    $washingAllowance = round(($emp['washing_allowance'] ?? 0) * $paidDays / $totalDays, 2);
   449→   449→
   450→   450→                    $grossEarnings = round($basicDa + $hra + $leaveEncashment + $bonusEncashment + $washingAllowance, 2);
   451→   451→
   452→   452→                    // Extra days are now included in paid_days, so no separate calculation needed
   453→   453→                    $extraDaysAmount = 0;
   454→   454→
   455→   455→                    // Calculate overtime amount based on unit salary formula settings
   456→   456→                    $overtimeAmount = 0;

   459→   459→                        $otCalcOn = $emp['ot_calculation_on'] ?? 'basic_da';      // basic, basic_da, basic_hra, gross

   461→   461→
   462→   462→                        // Determine OT base amount

   464→   464→                            case 'gross':

   467→   467→                            case 'basic_hra':

   474→   474→                            default: // basic_da

  1257→  1257→    /**
  1258→  1258→     * Calculate Professional Tax (State-wise slabs)
  1259→  1259→     * Default slabs - can be overridden based on employee's work state
  1260→  1260→     * Common PT slabs across India (monthly)
  1261→  1261→     */
  1262→  1262→    private function calculatePT($gross, $state = 'MH') {
  1263→  1263→        // Maharashtra slabs (most common)
  1264→  1264→        if ($state === 'MH' || $state === 'Maharashtra') {
  1265→  1265→            if ($gross > 10000) {
  1266→  1266→                return 200;  // Fixed ₹200 for salary above ₹10,000
  1267→  1267→            }
  1268→  1268→            return 0;
  1269→  1269→        }
  1270→  1270→        
  1271→  1271→        // Karnataka slabs
  1272→  1272→        if ($state === 'KA' || $state === 'Karnataka') {
  1273→  1273→            if ($gross > 15000) {
  1274→  1274→                return 200;
  1275→  1275→            } elseif ($gross > 10000) {
  1276→  1276→                return 150;
  1277→  1277→            }
  1278→  1278→            return 0;
  1279→  1279→        }
  1280→  1280→        
  1281→  1281→        // Tamil Nadu slabs
  1282→  1282→        if ($state === 'TN' || $state === 'Tamil Nadu') {
  1283→  1283→            if ($gross > 75000) { // Half yearly threshold
  1284→  1284→                return 1250 / 6; // Approx monthly
  1285→  1285→            } elseif ($gross > 50000) {
  1286→  1286→                return 833 / 6;
  1287→  1287→            }
  1288→  1288→            return 0;
  1289→  1289→        }
  1290→  1290→        
  1291→  1291→        // Delhi slabs (no PT for most)
  1292→  1292→        if ($state === 'DL' || $state === 'Delhi') {
  1293→  1293→            if ($gross > 25000) {
  1294→  1294→                return 200;
  1295→  1295→            }
  1296→  1296→            return 0;
  1297→  1297→        }
  1298→  1298→        
  1299→  1299→        // Gujarat slabs
  1300→  1300→        if ($state === 'GJ' || $state === 'Gujarat') {
  1301→  1301→            if ($gross > 12000) {
  1302→  1302→                return 200;
  1303→  1303→            }
  1304→  1304→            return 0;
  1305→  1305→        }
  1306→  1306→        
  1307→  1307→        // Default: Maharashtra-style slab
  1308→  1308→        if ($gross > 10000) {
  1309→  1309→            return 200;
  1310→  1310→        }
  1311→  1311→        return 0;
  1312→  1312→    }
