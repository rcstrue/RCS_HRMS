   670→
   671→// Load minimum-wage zones for a state into a <select> from api/mw-zones.
   672→// selectSel   = selector for the zone <select>
   673→// currentVal  = value to pre-select (used in edit modal)
   674→// helpSel     = selector for the small help text under the select
   675→window.loadMwZones = function(state, selectSel, currentVal, helpSel) {
   676→    var $sel = $(selectSel);
   677→    var $help = $(helpSel);
   678→    // reset to just the "state-wide" option
   679→    $sel.html('<option value="">— State-wide rate (no zone) —</option>');
   680→    if (!state) {
   681→        if ($help) $help.text('Select the state first to load available zones from the minimum wage table.');
   682→        return;
   683→    }
   684→    $sel.append('<option disabled>Loading zones…</option>');
   685→    if ($help) $help.text('Loading zones for ' + state + '…');
   686→    $.getJSON('index.php?page=api/mw-zones', { state: state })
   687→    .done(function(resp) {
   688→        $sel.find('option:disabled').remove();
   689→        if (!resp || !resp.success) {
   690→            if ($help) $help.text('Could not load zones — you can still save with state-wide rate.');
   691→            return;
   692→        }
   693→        var zones = resp.zones || [];
   694→        if (zones.length === 0) {
   695→            if ($help) $help.text(resp.state_found
   696→                ? 'No zone-specific rates for ' + state + ' — state-wide rate applies.'
   697→                : 'State not found in minimum wage table. Run the minimum-wage sync first.');
   698→            return;
   699→        }
   700→        zones.forEach(function(z) {
   701→            $sel.append('<option value="' + $('<div>').text(z).html() + '">' + $('<div>').text(z).html() + '</option>');
   702→        });
   703→        if (currentVal) {
   704→            $sel.val(currentVal);
   705→            if ($sel.val() !== currentVal) {
   706→                // value isn't in the list — add it so it isn't lost
   707→                $sel.append('<option value="' + $('<div>').text(currentVal).html() + '">' + $('<div>').text(currentVal).html() + ' (saved)</option>');
   708→                $sel.val(currentVal);
   709→            }
   710→        }
   711→        if ($help) $help.text(zones.length + ' zone(s) available for ' + state + '. Leave blank for state-wide rate.');
   712→    })
   713→    .fail(function() {
   714→        $sel.find('option:disabled').remove();
   715→        if ($help) $help.text('Could not load zones (network error). You can still save with state-wide rate.');
   716→    });
   717→};
