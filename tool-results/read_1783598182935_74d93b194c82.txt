    22→// ── Session Filters ──────────────────────────────────────────────

    30→if (isset($_GET['unit_id']))   $unitId   = (int)$_GET['unit_id'];
    31→if (isset($_GET['month']))     $month    = (int)$_GET['month'];
    32→if (isset($_GET['year']))      $year     = (int)$_GET['year'];
    33→// Save back to session for persistence
    34→$_SESSION['filter_client_id'] = $clientId;
    35→$_SESSION['filter_unit_id']   = $unitId;
    36→$_SESSION['filter_month']     = $month;
    37→$_SESSION['filter_year']      = $year;
    38→
    39→$calendarDays = cal_days_in_month(CAL_GREGORIAN, $month, $year);
    40→
    41→// ── Fetch unit's pay_days_type from salary formula ──────────────
    42→$unitFormula = null;
    43→if ($unitId > 0) {
    44→    $unitFormula = $db->fetch(
    45→        "SELECT * FROM unit_salary_formulas
    46→         WHERE unit_id = ? AND is_active = 1
    47→         ORDER BY effective_from DESC LIMIT 1",
    48→        [$unitId]
    49→    );
    50→}
    51→$payDaysType = $unitFormula['pay_days_type'] ?? 'actual';
    52→
    53→// Calculate totalDays based on pay_days_type
    54→$totalDays = $calendarDays;
    55→switch ($payDaysType) {
    56→    case 'fixed_30':
    57→        $totalDays = 30;
    58→        break;
    59→    case 'previous_month':
    60→        $prevMonth = $month - 1;
    61→        $prevYear  = $year;
    62→        if ($prevMonth < 1) { $prevMonth = 12; $prevYear--; }
    63→        $totalDays = (int)cal_days_in_month(CAL_GREGORIAN, $prevMonth, $prevYear);
    64→        break;
    65→    case 'calendar_minus_sundays':
    66→        // Count all Sundays in the selected month
    67→        $sundays = 0;
    68→        for ($d = 1; $d <= $calendarDays; $d++) {
    69→            $dow = (int)date('N', strtotime("$year-$month-$d"));
    70→            if ($dow === 7) $sundays++;
    71→        }
    72→        $totalDays = $calendarDays - $sundays;
    73→        break;
    74→    default: // 'actual'
    75→        $totalDays = $calendarDays;
    76→        break;
    77→}
    78→
    79→// Also get OT settings from unit formula
    80→$unitOtCalcType = $unitFormula['ot_calculation_type'] ?? 'double_pay';
    81→$unitOtCalcOn   = $unitFormula['ot_calculation_on'] ?? 'basic_da';
    82→$unitOtHrsPerDay = (float)($unitFormula['ot_hours_per_day'] ?? 8);
    83→
    84→// ── Pay days calculation is handled above via unit_salary_formulas ──

    87→$pfEmployeeShare = 12.00;
    88→$pfWageCeiling = 15000;

    91→$pfEmployerEdlis = 0.50;

    94→$esiEmployerShare = 3.25;

   100→        $pfWageCeiling  = (float)$pfRate['wage_ceiling'];
   101→        $pfEmployerShare = (float)$pfRate['employer_share_pf'];
   102→        $pfEmployerEps  = (float)$pfRate['employer_share_eps'];
   103→        $pfEmployerEdlis = (float)$pfRate['employer_share_edlis'];

   111→        $esiEmployerShare = (float)$esiRate['employer_share'];

   113→    }
   114→} catch (Exception $e) {}

   116→// ── Fetch unit state + PT + LWF configuration ────────────────

   118→$unitStatePtApplicable = false;
   119→$ptSlabs = [];
   120→$lwfAmount = 0; // Default 0, will be set per-state below
   121→if ($unitId > 0) {
   122→    $unitInfo = $db->fetch(
   123→        "SELECT u.state, s.pt_applicable as state_pt_applicable, s.id as state_id
   124→         FROM units u
   125→         LEFT JOIN states s ON (u.state = s.state_code OR u.state = s.state_name)
   126→         WHERE u.id = ?",
   127→        [$unitId]
   128→    );
   129→    if ($unitInfo) {
   130→        $unitState = $unitInfo['state'] ?? '';
   131→        $unitStatePtApplicable = !empty($unitInfo['state_pt_applicable']);
   132→    }
   133→    // Fetch PT slabs for this state
   134→    if ($unitStatePtApplicable && $unitState) {
   135→        try {
   136→            $ptSlabs = $db->fetchAll(
   137→                "SELECT min_gross, max_gross, pt_amount
   138→                 FROM professional_tax_slabs
   139→                 WHERE (state_name = ? OR state_name = ?) AND is_active = 1
   140→                 ORDER BY min_gross",
   141→                [$unitState, ucfirst(strtolower($unitState))]
   142→            );
   143→        } catch (Exception $e) {}
   144→    }
   145→    // Fetch LWF rate for this state (from lwf_rates → lwf_state_rates fallback)
   146→    if ($unitState) {
   147→        try {
   148→            $lwfRate = $db->fetch(
   149→                "SELECT lr.employee_share, lr.contribution_months
   150→                 FROM lwf_rates lr
   151→                 JOIN states s ON lr.state_id = s.id
   152→                 WHERE (s.state_code = ? OR s.state_name = ?)
   153→                 AND lr.is_active = 1 AND lr.effective_from <= CURDATE()
   154→                 ORDER BY lr.effective_from DESC LIMIT 1",
   155→                [$unitState, $unitState]
   156→            );
   157→            if (!$lwfRate) {
   158→                $lwfRate = $db->fetch(
   159→                    "SELECT employee_share, contribution_months
   160→                     FROM lwf_state_rates
   161→                     WHERE (state_name = ? OR state_name = ?) AND is_active = 1
   162→                     AND effective_from <= CURDATE()
   163→                     ORDER BY effective_from DESC LIMIT 1",
   164→                    [$unitState, ucfirst(strtolower($unitState))]
   165→                );
   166→            }
   167→            if ($lwfRate) {
   168→                $contribMonths = array_filter(array_map('intval', explode(',', $lwfRate['contribution_months'] ?? '')));
   169→                if (in_array($month, $contribMonths)) {
   170→                    $lwfAmount = floatval($lwfRate['employee_share']);
   171→                }
   172→            }
   173→        } catch (Exception $e) {}
   174→    }

   178→$clients = $db->fetchAll("SELECT id, name FROM clients WHERE is_active = 1 ORDER BY name");
   179→
   180→// Units filtered by selected client (same pattern as attendance/add)
   181→$units = [];
   182→if ($clientId > 0) {
   183→    $units = $db->fetchAll(
   184→        "SELECT id, name, unit_code FROM units WHERE client_id = ? AND is_active = 1 ORDER BY name",
   185→        [$clientId]

   189→// ── Handle Load Employees via POST (AJAX or form) ───────────────
   190→$employeeData = [];

   193→if (isset($_GET['load']) && (int)$_GET['load'] === 1 && $unitId > 0) {
   194→    $loaded = true;
   195→
   196→    // Ensure loan tables exist before querying them in the loop
   197→    try {
   198→        $db->exec("CREATE TABLE IF NOT EXISTS `employee_loans` (
   199→            `id` int(11) NOT NULL AUTO_INCREMENT,
   200→            `employee_id` int(11) NOT NULL,
   201→            `unit_id` int(11) DEFAULT NULL,
   202→            `loan_type` varchar(50) DEFAULT 'Personal',
   203→            `amount` decimal(12,2) NOT NULL,
   204→            `interest_rate` decimal(5,2) DEFAULT 0.00,
   205→            `tenure_months` int(11) NOT NULL,
   206→            `emi_amount` decimal(12,2) NOT NULL,
   207→            `total_interest` decimal(12,2) DEFAULT 0.00,
   208→            `total_repayable` decimal(12,2) NOT NULL,
   209→            `balance_amount` decimal(12,2) NOT NULL,
   210→            `emi_deducted` int(11) DEFAULT 0,
   211→            `start_month` int(2) NOT NULL,
   212→            `start_year` int(4) NOT NULL,
   213→            `status` enum('Active','Closed','Settled','Written Off') DEFAULT 'Active',
   214→            `remarks` text DEFAULT NULL,
   215→            `created_at` timestamp DEFAULT CURRENT_TIMESTAMP,
   216→            `updated_at` timestamp DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
   217→            PRIMARY KEY (`id`),
   218→            KEY `idx_employee` (`employee_id`),
   219→            KEY `idx_unit` (`unit_id`),
   220→            KEY `idx_status` (`status`)
   221→        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4");
   222→        $db->exec("CREATE TABLE IF NOT EXISTS `loan_emi_log` (
   223→            `id` int(11) NOT NULL AUTO_INCREMENT,
   224→            `loan_id` int(11) NOT NULL,
   225→            `employee_id` int(11) NOT NULL,
   226→            `month` int(2) NOT NULL,
   227→            `year` int(4) NOT NULL,
   228→            `emi_amount` decimal(12,2) NOT NULL,
   229→            `principal_component` decimal(12,2) DEFAULT 0.00,
   230→            `interest_component` decimal(12,2) DEFAULT 0.00,
   231→            `balance_after` decimal(12,2) NOT NULL,
   232→            `deducted_via_payroll` tinyint(1) DEFAULT 1,
   233→            `created_at` timestamp DEFAULT CURRENT_TIMESTAMP,
   234→            PRIMARY KEY (`id`),
   235→            UNIQUE KEY `uniq_loan_month_year` (`loan_id`, `month`, `year`),
   236→            KEY `idx_employee_month` (`employee_id`, `month`, `year`)
   237→        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4");
   238→    } catch (Exception $e) {}
   239→
   240→    $employees = $db->fetchAll(
   241→        "SELECT id, employee_code, full_name, designation
   242→         FROM employees
   243→         WHERE unit_id = ? AND status IN ('approved', 'active')
   244→         ORDER BY employee_code",

   246→    );
   247→

   249→    // loads the salary structure that was actually active during that month,
   250→    // not the latest one created for a future month.
   251→    $effDateStr = sprintf('%04d-%02d-%02d', $year, $month, cal_days_in_month(CAL_GREGORIAN, $month, $year));
   252→
   253→    foreach ($employees as $emp) {
   254→        $empId  = (int)$emp['id'];
   255→        $empCode = $emp['employee_code'];
   256→
   257→        // Salary structure (latest effective for selected month)
   258→        $salary = $db->fetch(
   259→            "SELECT * FROM employee_salary_structures
   260→             WHERE employee_id = ? AND effective_from <= ?
   261→             ORDER BY effective_from DESC LIMIT 1",
   262→            [$empId, $effDateStr]
   263→        );
   264→
   265→        // Attendance
   266→        $att = [];
   267→        try {
   268→            $att = $db->fetch(
   269→                "SELECT * FROM attendance_summary
   270→                 WHERE employee_id = ? AND month = ? AND year = ?",
   271→                [$empId, $month, $year]
   272→            );
   273→        } catch (Exception $e) {
   274→            $att = [];
   275→        }
   276→
   277→        // Advances
   278→        $adv = [];
   279→        try {
   280→            $adv = $db->fetch(
   281→                "SELECT COALESCE(adv1,0)+COALESCE(adv2,0)+COALESCE(office_advance,0)+COALESCE(dress_advance,0) as total_advance,
   282→                        COALESCE(adv1,0) as adv1,
   283→                        COALESCE(adv2,0) as adv2,
   284→                        COALESCE(office_advance,0) as office_advance,
   285→                        COALESCE(dress_advance,0) as dress_advance
   286→                 FROM employee_advances
   287→                 WHERE employee_id = ? AND month = ? AND year = ?",
   288→                [$empId, $month, $year]
   289→            );
   290→        } catch (Exception $e) {
   291→            $adv = [];
   292→        }
   293→
   294→        // Existing payroll record (if already saved for this month/year)
   295→        $existing = [];
   296→        try {
   297→            // Try direct month/year lookup first (new schema)
   298→            $existing = $db->fetch(
   299→                "SELECT * FROM payroll
   300→                 WHERE employee_id = ? AND month = ? AND year = ?",
   301→                [$empCode, $month, $year]
   302→            );
   303→        } catch (Exception $e) {
   304→            $existing = [];
   305→        }
   306→
   307→        // Active loan EMI for this employee (with balance > 0, not already deducted this month)
   308→        $loanEmi = 0;
   309→        try {
   310→            // Sum EMI from ALL active loans
   311→            $loans = $db->fetchAll(
   312→                "SELECT el.emi_amount, el.id as loan_id
   313→                 FROM employee_loans el
   314→                 WHERE el.employee_id = ? AND el.status = 'Active'
   315→                 AND el.balance_amount > 0
   316→                 AND (el.start_year < ? OR (el.start_year = ? AND el.start_month <= ?))
   317→                 AND el.id NOT IN (
   318→                     SELECT lem.loan_id FROM loan_emi_log lem
   319→                     WHERE lem.month = ? AND lem.year = ?
   320→                 )",
   321→                [$empId, $year, $year, $month, $month, $year]
   322→            );
   323→            foreach ($loans as $loan) {
   324→                $loanEmi += floatval($loan['emi_amount']);
   325→            }
   326→        } catch (Exception $e) {
   327→            // Fallback: try without emi_log check (in case table has different structure)
   328→            try {
   329→                $loans2 = $db->fetchAll(
   330→                    "SELECT el.emi_amount, el.id as loan_id
   331→                     FROM employee_loans el
   332→                     WHERE el.employee_id = ? AND el.status = 'Active'
   333→                     AND el.balance_amount > 0
   334→                     AND (el.start_year < ? OR (el.start_year = ? AND el.start_month <= ?))",
   335→                    [$empId, $year, $year, $month]
   336→                );
   337→                foreach ($loans2 as $loan2) {
   338→                    $loanEmi += floatval($loan2['emi_amount']);
   339→                }
   340→            } catch (Exception $e2) {
   341→                $loanEmi = 0;
   342→            }
   343→        }
   344→
   345→        $employeeData[] = [
   346→            'emp'      => $emp,
   347→            'salary'   => $salary ?: [],
   348→            'att'      => $att ?: [],
   349→            'adv'      => $adv ?: [],
   350→            'existing' => $existing ?: [],
   351→            'loan_emi' => $loanEmi,
   352→        ];
   353→    }
   354→}
