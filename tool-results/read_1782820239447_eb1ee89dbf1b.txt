    34→ * @param string $uploadDir Directory to upload to (relative to web root)
    35→ * @return array|null Result with 'path' or 'error' key, or null if no file
    36→ */
    37→function handleFileUpload($file, $uploadDir = 'uploads/profile/') {
    38→    $response = null;
    39→    $maxSize = 5 * 1024 * 1024; // 5MB
    40→
    41→    // Extension whitelist — NEVER trust the client-supplied filename extension
    42→    $allowedImageExtensions = ['jpg', 'jpeg', 'png', 'gif'];
    43→    $allowedPdfExtension    = ['pdf'];
    44→    $allowedExtensions      = array_merge($allowedImageExtensions, $allowedPdfExtension);
    45→
    46→    // Map of real MIME type (from finfo) → safe forced extension
    47→    $mimeToExtension = [
    48→        'image/jpeg' => 'jpg',
    49→        'image/png'  => 'png',
    50→        'image/gif'  => 'gif',
    51→        'application/pdf' => 'pdf',
    52→    ];
    53→
    54→    // Check if file was uploaded via PHP
    55→    if (!isset($file['tmp_name']) || empty($file['tmp_name'])) {
    56→        return $response;
    57→    }
    58→
    59→    // Check for upload errors
    60→    if ($file['error'] !== UPLOAD_ERR_OK) {
    61→        return $response;
    62→    }
    63→
    64→    // Validate file size
    65→    if ($file['size'] > $maxSize) {
    66→        $response = ['error' => 'File size must be less than 5MB.'];
    67→        return $response;
    68→    }
    69→
    70→    // ─── Validate REAL file content with finfo (magic bytes), not client Content-Type ───
    71→    $finfo = new finfo(FILEINFO_MIME_TYPE);
    72→    $detectedMime = $finfo->file($file['tmp_name']);
    73→
    74→    if (!isset($mimeToExtension[$detectedMime])) {
    75→        $response = ['error' => 'Invalid file type. Only JPG, PNG, GIF, PDF allowed.'];
    76→        return $response;
    77→    }
    78→
    79→    // Force the extension from the detected MIME type — ignore the uploaded filename entirely
    80→    $safeExtension = $mimeToExtension[$detectedMime];
    81→
    82→    // For images, also validate with getimagesize() to confirm it's a real image
    83→    if (in_array($safeExtension, $allowedImageExtensions)) {
    84→        $imageInfo = @getimagesize($file['tmp_name']);
    85→        if ($imageInfo === false) {
    86→            $response = ['error' => 'Invalid image file. The content is not a valid image.'];
    87→            return $response;
    88→        }
    89→        // Re-encode images server-side to strip any embedded payloads
    90→        $destImage = null;
    91→        switch ($safeExtension) {
    92→            case 'jpg':
    93→            case 'jpeg':
    94→                $src = @imagecreatefromjpeg($file['tmp_name']);
    95→                if ($src) { $destImage = $src; }
    96→                break;
    97→            case 'png':
    98→                $src = @imagecreatefrompng($file['tmp_name']);
    99→                if ($src) { $destImage = $src; }
   100→                break;
   101→            case 'gif':
   102→                $src = @imagecreatefromgif($file['tmp_name']);
   103→                if ($src) { $destImage = $src; }
   104→                break;
   105→        }
   106→        if (!$destImage) {
   107→            $response = ['error' => 'Failed to process image. Please upload a valid image file.'];
   108→            return $response;
   109→        }
   110→    }
   111→
   112→    // Get web root (parent of APP_ROOT which is hrms folder)
   113→    $webRoot = dirname(APP_ROOT);
   114→
   115→    // Create upload directory if not exists
   116→    $fullPath = $webRoot . '/' . $uploadDir;
   117→    if (!is_dir($fullPath)) {
   118→        mkdir($fullPath, 0755, true);
   119→    }
   120→
   121→    // Generate unique filename using ONLY the safe, forced extension
   122→    $filename = uniqid() . '_' . time() . '.' . $safeExtension;
   123→
   124→    // Database path starts with / for web root
   125→    $dbPath = '/' . $uploadDir . $filename;
   126→
   127→    // Save file: re-encode images via GD to strip payloads; copy PDFs directly
   128→    $targetPath = $fullPath . $filename;
   129→    if (in_array($safeExtension, $allowedImageExtensions)) {
   130→        // Re-encode and save via GD (strips EXIF, comments, embedded scripts)
   131→        $saveOk = false;
   132→        switch ($safeExtension) {
   133→            case 'jpg':
   134→            case 'jpeg':
   135→                $saveOk = imagejpeg($destImage, $targetPath, 90);
   136→                break;
   137→            case 'png':
   138→                $saveOk = imagepng($destImage, $targetPath);
   139→                break;
   140→            case 'gif':
   141→                $saveOk = imagegif($destImage, $targetPath);
   142→                break;
   143→        }
   144→        imagedestroy($destImage);
   145→        if (!$saveOk) {
   146→            $response = ['error' => 'Failed to save processed image.'];
   147→            return $response;
   148→        }
   149→    } else {
   150→        // PDF: just move the validated file (finfo already confirmed it's a PDF)
   151→        if (!move_uploaded_file($file['tmp_name'], $targetPath)) {
   152→            $response = ['error' => 'Failed to upload file.'];
   153→            return $response;
   154→        }
   155→    }
   156→
   157→    $response = ['path' => $dbPath];
   158→    return $response;
   159→}
