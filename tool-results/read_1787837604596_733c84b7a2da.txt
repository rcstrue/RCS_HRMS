   305→// Helper functions
   306→function parseCSV($filePath) {
   307→    $data = [];
   308→    if (($handle = fopen($filePath, "r")) !== FALSE) {
   309→        while (($row = fgetcsv($handle, 1000, ",")) !== FALSE) {
   310→            $data[] = $row;
   311→        }
   312→        fclose($handle);
   313→    }
   314→    return $data;
   315→}

   345→function downloadCurrentSalary($db, $clientId, $unitId) {
   346→    // Note: employee_salary_structures has basic_da column
   347→    $sql = "SELECT e.employee_code, e.full_name, c.name as client_name, u.name as unit_name,
   348→            COALESCE(ess.basic_da,0) as basic_da,
   349→            COALESCE(ess.hra,0) as hra,
   350→            COALESCE(ess.leave_encashment,0) as leave_encashment,
   351→            COALESCE(ess.bonus_encashment,0) as bonus_encashment,
   352→            COALESCE(ess.washing_allowance,0) as washing,
   353→
   354→            COALESCE(ess.gross_salary,0) as gross_salary
   355→            FROM employees e
   356→            LEFT JOIN employee_salary_structures ess ON e.id = ess.employee_id 
   357→                AND (ess.effective_to IS NULL OR ess.effective_to >= CURDATE())
   358→            LEFT JOIN clients c ON e.client_id = c.id
   359→            LEFT JOIN units u ON e.unit_id = u.id
   360→            WHERE e.status = 'approved'";
   361→    
   362→    $params = [];
   363→    if ($clientId) {
   364→        $sql .= " AND e.client_id = :client_id";
   365→        $params['client_id'] = $clientId;
   366→    }
   367→    if ($unitId) {
   368→        $sql .= " AND e.unit_id = :unit_id";
   369→        $params['unit_id'] = $unitId;
   370→    }
   371→    
   372→    $sql .= " ORDER BY c.name, u.name, e.employee_code";
   373→    
   374→    $employees = $db->fetchAll($sql, $params);
   375→    
   376→    $filename = 'Current_Salary_' . date('Ymd') . '.csv';
   377→    
   378→    header('Content-Type: text/csv');
   379→    header('Content-Disposition: attachment; filename="' . $filename . '"');
   380→    header('Pragma: no-cache');
   381→    header('Expires: 0');
   382→    
   383→    $output = fopen('php://output', 'w');
   384→    fputcsv($output, ['Emp Code', 'Name', 'Client', 'Unit', 'Basic+DA', 'HRA', 'Leave Encashment', 'Bonus Encashment', 'Washing', 'Gross'], ',', '"', '\\');
   385→    
   386→    foreach ($employees as $emp) {
   387→        fputcsv($output, [
   388→            $emp['employee_code'],
   389→            $emp['full_name'],
   390→            $emp['client_name'],
   391→            $emp['unit_name'],
   392→            $emp['basic_da'],
   393→            $emp['hra'],
   394→            $emp['leave_encashment'],
   395→            $emp['bonus_encashment'],
   396→            $emp['washing'],
   397→
   398→            $emp['gross_salary']
   399→        ], ',', '"', '\\');
   400→    }
   401→    
   402→    fclose($output);
   403→    exit;
   404→}
