    95→// ── Helpers ────────────────────────────────────────────
    96→function getInitials(name: string): string {
    97→  return name
    98→    .split(' ')
    99→    .map((w) => w[0])
   100→    .filter(Boolean)
   101→    .slice(0, 2)
   102→    .join('')
   103→    .toUpperCase();
   104→}
   105→
   106→function maskMobile(mobile: string): string {
   107→  // Show full mobile number — no masking
   108→  return mobile || '';
   109→}
   110→
   111→function maskSensitive(value: string | number | undefined): string {
   112→  const str = value != null ? String(value) : '';
   113→  if (!str) return '****';
   114→  if (str.length <= 4) return '****';
   115→  const first = str.slice(0, 2);
   116→  const last = str.slice(-2);
   117→  const middleLen = str.length - 4;
   118→  return first + '*'.repeat(middleLen) + last;
   119→}
   120→
   121→function formatDate(dateStr?: string): string {
   122→  if (!dateStr) return '—';
   123→  const d = new Date(dateStr);
   124→  return d.toLocaleDateString('en-IN', {
   125→    day: 'numeric',
   126→    month: 'short',
   127→    year: 'numeric',
   128→  });
   129→}
   130→
   131→function isActive(employee: Employee): boolean {
   132→  return !employee.date_of_leaving && employee.status !== 'inactive' && employee.status !== 'resigned';
   133→}
