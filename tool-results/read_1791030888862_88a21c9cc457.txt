   999→    if (!imgEl) return;
  1000→    var empId = <?php echo (int)$employeeId; ?>;
  1001→    var currentUrl = imgEl.getAttribute('src');
  1002→    if (!currentUrl) return;
  1003→
  1004→    openLiteEditor(currentUrl, function(base64DataUrl) {
  1005→        var formData = new FormData();
  1006→        formData.append('action', 'crop_save_doc');
  1007→        formData.append('employee_id', empId);
  1008→        formData.append('field', field);
  1009→        formData.append('image_data', base64DataUrl);
  1010→        formData.append('original_url', currentUrl);
  1011→
  1012→        fetch(window.location.pathname + '?page=api/crop-save', {
  1013→            method: 'POST',
  1014→            body: formData
  1015→        })
  1016→        .then(function(r) { return r.json(); })
  1017→        .then(function(res) {
  1018→            if (res.ok) {
  1019→                imgEl.src = res.url + '?t=' + Date.now();
  1020→                var link = imgEl.closest('a');
  1021→                if (link) link.href = res.url;
  1022→                // Also update the sidebar profile photo if it was the profile pic
  1023→                if (field.indexOf('profile_pic') !== -1) {
  1024→                    var sidePhoto = document.querySelector('.card-body.text-center img.rounded-circle');
  1025→                    if (sidePhoto) sidePhoto.src = res.url + '?t=' + Date.now();
  1026→                }
  1027→                alert('Image saved successfully!');
  1028→            } else {
  1029→                alert('Save failed: ' + (res.msg || 'Unknown error'));
  1030→            }
  1031→        })
  1032→        .catch(function(err) { alert('Error: ' + err.message); });
  1033→    });
