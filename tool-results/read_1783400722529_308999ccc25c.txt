    40→    40→
    41→    41→    // Extension whitelist — NEVER trust the client-supplied filename extension
    42→    42→    $allowedImageExtensions = ['jpg', 'jpeg', 'png', 'gif'];

    44→    44→    $allowedExtensions      = array_merge($allowedImageExtensions, $allowedPdfExtension);
    45→    45→
    46→    46→    // Map of real MIME type (from finfo) → safe forced extension
    47→    47→    $mimeToExtension = [
    48→    48→        'image/jpeg' => 'jpg',
    49→    49→        'image/png'  => 'png',
    50→    50→        'image/gif'  => 'gif',
    51→    51→        'application/pdf' => 'pdf',
    52→    52→    ];
    53→    53→
    54→    54→    // Check if file was uploaded via PHP
    55→    55→    if (!isset($file['tmp_name']) || empty($file['tmp_name'])) {
    56→    56→        return $response;
    57→    57→    }
    58→    58→
    59→    59→    // Check for upload errors

    62→    62→    }

    64→    64→    // Validate file size
    65→    65→    if ($file['size'] > $maxSize) {

    67→    67→        return $response;
    68→    68→    }
    69→    69→
    70→    70→    // ─── Validate REAL file content with finfo (magic bytes), not client Content-Type ───
    71→    71→    $finfo = new finfo(FILEINFO_MIME_TYPE);
    72→    72→    $detectedMime = $finfo->file($file['tmp_name']);

    74→    74→    if (!isset($mimeToExtension[$detectedMime])) {
    75→    75→        $response = ['error' => 'Invalid file type. Only JPG, PNG, GIF, PDF allowed.'];
    76→    76→        return $response;
    77→    77→    }
    78→    78→
    79→    79→    // Force the extension from the detected MIME type — ignore the uploaded filename entirely

    86→    86→            $response = ['error' => 'Invalid image file. The content is not a valid image.'];
    87→    87→            return $response;
    88→    88→        }
    89→    89→        // Re-encode images server-side to strip any embedded payloads
    90→    90→        $destImage = null;
    91→    91→        switch ($safeExtension) {
    92→    92→            case 'jpg':
    93→    93→            case 'jpeg':
    94→    94→                $src = @imagecreatefromjpeg($file['tmp_name']);
    95→    95→                if ($src) { $destImage = $src; }
    96→    96→                break;
    97→    97→            case 'png':
    98→    98→                $src = @imagecreatefrompng($file['tmp_name']);
    99→    99→                if ($src) { $destImage = $src; }
   100→   100→                break;
   101→   101→            case 'gif':
   102→   102→                $src = @imagecreatefromgif($file['tmp_name']);
   103→   103→                if ($src) { $destImage = $src; }
   104→   104→                break;
   105→   105→        }
   106→   106→        if (!$destImage) {
   107→   107→            $response = ['error' => 'Failed to process image. Please upload a valid image file.'];
   108→   108→            return $response;
   109→   109→        }
   110→   110→    }
   111→   111→
   112→   112→    // Get web root (parent of APP_ROOT which is hrms folder)
   113→   113→    $webRoot = dirname(APP_ROOT);
   114→   114→
   115→   115→    // Create upload directory if not exists
   116→   116→    $fullPath = $webRoot . '/' . $uploadDir;

   118→   118→        mkdir($fullPath, 0755, true);
   119→   119→    }

   135→   135→                $saveOk = imagejpeg($destImage, $targetPath, 90);
   136→   136→                break;

   138→   138→                $saveOk = imagepng($destImage, $targetPath);
   139→   139→                break;
   140→   140→            case 'gif':
   141→   141→                $saveOk = imagegif($destImage, $targetPath);
   142→   142→                break;
   143→   143→        }
   144→   144→        imagedestroy($destImage);
   145→   145→        if (!$saveOk) {
   146→   146→            $response = ['error' => 'Failed to save processed image.'];
   147→   147→            return $response;
   148→   148→        }
   149→   149→    } else {
   150→   150→        // PDF: just move the validated file (finfo already confirmed it's a PDF)
   151→   151→        if (!move_uploaded_file($file['tmp_name'], $targetPath)) {
   152→   152→            $response = ['error' => 'Failed to upload file.'];
   153→   153→            return $response;
   154→   154→        }
   155→   155→    }
