     1→<?php
     2→/**
     3→ * RCS HRMS Pro - Payroll Save Row API
     4→ * Version: 1.0.0
     5→ *
     6→ * AJAX endpoint for saving a single employee's payroll row.
     7→ * Route: index.php?page=api/payroll-save-row
     8→ *
     9→ * Receives JSON POST with:
    10→ *   - employee_id (int, employees.id)
    11→ *   - employee_code (string, employees.employee_code)
    12→ *   - month, year, unit_id
    13→ *   - pf_applicable, esi_applicable, pt_applicable, lwf_applicable (bool)
    14→ *   - wage_details { basic_da, hra, leave_encashment, bonus_encashment, washing_allowance, gross_salary }
    15→ *   - attendance { total_present, total_wo, total_extra, overtime_hours, total_paid_days }
    16→ *   - advances { advance, office_deduction }
    17→ *   - payroll { all calculated values }
    18→ *
    19→ * Handles:
    20→ *   1. Upsert attendance_summary (employee_id = employees.id)
    21→ *   2. Upsert employee_salary_structures (effective_from = YYYY-MM-01)
    22→ *   3. Upsert employee_advances (employee_id = employees.id)
    23→ *   4. Upsert payroll (employee_id = employee_code string)
    24→ */
    25→
    26→header('Content-Type: application/json');
    27→
    28→// ── Authentication check ────────────────────────────────────────
    29→if (!isset($_SESSION['user_id'])) {
    30→    http_response_code(401);
    31→    echo json_encode(['success' => false, 'message' => 'Not authenticated']);
    32→    exit;
    33→}
    34→
    35→// Role check — payroll row save is admin/HR/manager only
    36→$roleCode = $_SESSION['role_code'] ?? '';
    37→if (!in_array($roleCode, ['admin', 'hr_executive', 'hr', 'manager'])) {
    38→    http_response_code(403);
    39→    echo json_encode(['success' => false, 'message' => 'Access denied. Insufficient permissions for payroll save.']);
    40→    exit;
    41→}
    42→
    43→// CSRF check — state-changing POST must carry a valid token (Round 4)
    44→// Note: this endpoint reads JSON body, so the token may come via X-CSRF-Token header
    45→// or in the JSON body as csrf_token. We read the body ONCE and reuse it below.
    46→$rawInput = file_get_contents('php://input');
    47→$csrfInput = json_decode($rawInput, true);
    48→$csrfToken = $_SERVER['HTTP_X_CSRF_TOKEN'] ?? ($csrfInput['csrf_token'] ?? '') ?? '';
    49→if (!validateCSRFToken($csrfToken)) {
    50→    http_response_code(403);
    51→    echo json_encode(['success' => false, 'message' => 'Invalid or missing CSRF token. Please refresh the page and try again.']);
    52→    exit;
    53→}
    54→
    55→// ── Only POST ───────────────────────────────────────────────────
    56→if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    57→    http_response_code(405);
    58→    echo json_encode(['success' => false, 'message' => 'Method not allowed']);
    59→    exit;
    60→}
    61→
    62→// ── Parse JSON input (reuse the body already read for CSRF) ─────
    63→$data = $csrfInput;
    64→
    65→if (json_last_error() !== JSON_ERROR_NONE) {
    66→    echo json_encode(['success' => false, 'message' => 'Invalid JSON: ' . json_last_error_msg()]);
    67→    exit;
    68→}
    69→
    70→// ── Validate required fields ────────────────────────────────────
    71→$employeeId   = (int)($data['employee_id'] ?? 0);
    72→$employeeCode = trim($data['employee_code'] ?? '');
    73→$month        = (int)($data['month'] ?? 0);
    74→$year         = (int)($data['year'] ?? 0);
    75→$unitId       = (int)($data['unit_id'] ?? 0);
    76→if ($employeeId <= 0 || empty($employeeCode) || $month < 1 || $month > 12 || $year < 2020) {
    77→    echo json_encode(['success' => false, 'message' => 'Missing required fields (employee_id, employee_code, month, year)']);
    78→    exit;
    79→}
    80→
    81→$wageDetails = $data['wage_details'] ?? [];
    82→$attendance  = $data['attendance'] ?? [];
    83→$advances    = $data['advances'] ?? [];
    84→$payroll     = $data['payroll'] ?? [];
    85→
    86→// ── Server-side attendance validation (Audit Issue #5) ──
    87→$attPresent  = (float)($attendance['total_present'] ?? 0);
    88→$attWO       = (float)($attendance['total_wo'] ?? 0);
    89→$attExtra    = (float)($attendance['total_extra'] ?? 0);
    90→$attOtHours  = (float)($attendance['overtime_hours'] ?? 0);
    91→$attPaidDays = (float)($attendance['total_paid_days'] ?? 0);
    92→
    93→$validationErrors = [];
    94→if ($attPresent < 0) $validationErrors[] = 'Present days cannot be negative';
    95→if ($attWO < 0)      $validationErrors[] = 'Weekly off days cannot be negative';
    96→if ($attExtra < 0)   $validationErrors[] = 'Extra days cannot be negative';
    97→if ($attOtHours < 0) $validationErrors[] = 'OT hours cannot be negative';
    98→
    99→// Get total_days for validation from unit_salary_formulas (same logic as process-edit.php)
   100→$calDays = cal_days_in_month(CAL_GREGORIAN, $month, $year);
   101→$vTotalDays = $calDays;
   102→if ($unitId > 0) {
   103→    $vUnitFormula = $db->fetch(
   104→        "SELECT * FROM unit_salary_formulas WHERE unit_id = ? AND is_active = 1 ORDER BY effective_from DESC LIMIT 1",
   105→        [$unitId]
   106→    );
   107→    if ($vUnitFormula) {
   108→        $vPayDaysType = $vUnitFormula['pay_days_type'] ?? 'actual';
   109→        switch ($vPayDaysType) {
   110→            case 'fixed_30': $vTotalDays = 30; break;
   111→            case 'previous_month':
   112→                $pm = $month - 1; $py = $year;
   113→                if ($pm < 1) { $pm = 12; $py--; }
   114→                $vTotalDays = (int)cal_days_in_month(CAL_GREGORIAN, $pm, $py);
   115→                break;
   116→            case 'calendar_minus_sundays':
   117→                $s = 0;
   118→                for ($d = 1; $d <= $calDays; $d++) {
   119→                    if ((int)date('N', strtotime("$year-$month-$d")) === 7) $s++;
   120→                }
   121→                $vTotalDays = $calDays - $s;
   122→                break;
   123→            default: $vTotalDays = $calDays; break;
   124→        }
   125→    }
   126→}
   127→$computedPaidDays = $attPresent + $attWO + $attExtra;
   128→$paidDaysWarning = '';
   129→if ($computedPaidDays > $vTotalDays) {
   130→    $paidDaysWarning = "⚠ Paid days ($computedPaidDays) exceeds total days ($vTotalDays) — saved with override. ";
   131→}
   132→// OT monthly limit: ot_hours_per_day × total_days from unit config
   133→// (overtime_hours is a MONTHLY total, not per-day)
   134→$otMonthlyCap = 0;
   135→if ($unitId > 0) {
   136→    $unitFormula = $db->fetch(
   137→        "SELECT ot_hours_per_day FROM unit_salary_formulas
   138→         WHERE unit_id = ? AND is_active = 1
   139→         ORDER BY effective_from DESC LIMIT 1",
   140→        [$unitId]
   141→    );
   142→    if ($unitFormula) {
   143→        $otHrsPerDay = (float)($unitFormula['ot_hours_per_day'] ?? 8);
   144→        $otMonthlyCap = round($otHrsPerDay * $vTotalDays);
   145→    }
   146→}
   147→// Only warn if OT exceeds monthly cap (not a hard block — allow override)
   148→$otWarning = '';
   149→if ($otMonthlyCap > 0 && $attOtHours > $otMonthlyCap) {
   150→    $otWarning = "OT hours ($attOtHours) exceeds monthly cap ($otMonthlyCap) — saved with override. ";
   151→}
   152→
   153→if (!empty($validationErrors)) {
   154→    echo json_encode([
   155→        'success' => false,
   156→        'message' => 'Validation: ' . implode('; ', $validationErrors)
   157→    ]);
   158→    exit;
   159→}
   160→
   161→// ══ AUTO-MIGRATION: Must run OUTSIDE transaction (DDL causes implicit COMMIT in InnoDB) ══
   162→// Step 1: Detect which columns exist in payroll table
   163→$payrollColNames = [];
   164→try {
   165→    $colRows = $db->fetchAll("SHOW COLUMNS FROM payroll");
   166→    $payrollColNames = array_column($colRows, 'Field');
   167→} catch (Exception $e) {
   168→    error_log('Payroll column detection failed: ' . $e->getMessage());
   169→}
   170→$payrollColSet = array_flip($payrollColNames); // for O(1) lookup
   171→
   172→// Step 2: Add missing columns via ALTER TABLE
   173→if (!isset($payrollColSet['loan_emi'])) {
   174→    try {
   175→        $db->exec("ALTER TABLE payroll ADD COLUMN loan_emi DECIMAL(10,2) DEFAULT 0.00");
   176→        $payrollColSet['loan_emi'] = true;
   177→        error_log('Auto-migrate: Added loan_emi column to payroll table');
   178→    } catch (Exception $e) {
   179→        error_log('Auto-migrate loan_emi FAILED: ' . $e->getMessage());
   180→    }
   181→}
   182→$monthYearMissing = !isset($payrollColSet['month']) || !isset($payrollColSet['year']);
   183→if ($monthYearMissing) {
   184→    try {
   185→        if (!isset($payrollColSet['month'])) {
   186→            $db->exec("ALTER TABLE payroll ADD COLUMN month INT NOT NULL DEFAULT 0");
   187→            error_log('Auto-migrate: Added month column to payroll table');
   188→        }
   189→        if (!isset($payrollColSet['year'])) {
   190→            $db->exec("ALTER TABLE payroll ADD COLUMN year INT NOT NULL DEFAULT 0");
   191→            error_log('Auto-migrate: Added year column to payroll table');
   192→        }
   193→        $db->exec("ALTER TABLE payroll ADD INDEX idx_month_year (month, year)");
   194→        // Backfill no longer needed — payroll_period_id column removed
   195→        $monthYearMissing = false;
   196→    } catch (Exception $e) {
   197→        error_log('Auto-migrate month/year FAILED: ' . $e->getMessage());
   198→    }
   199→}
   200→
   201→try {
   202→    $db->exec("START TRANSACTION");
   203→
   204→    // ═══════════════════════════════════════════════════════════════
   205→    // 1. ATTENDANCE_SUMMARY — Upsert on employee_id + month + year
   206→    // ═══════════════════════════════════════════════════════════════
   207→    $existingAtt = $db->fetch(
   208→        "SELECT id FROM attendance_summary WHERE employee_id = ? AND month = ? AND year = ?",
   209→        [$employeeId, $month, $year]
   210→    );
   211→
   212→    $attData = [
   213→        'employee_id'    => $employeeId,
   214→        'month'          => $month,
   215→        'year'           => $year,
   216→        'total_present'  => (float)($attendance['total_present'] ?? 0),
   217→        'total_wo'       => (float)($attendance['total_wo'] ?? 0),
   218→        'total_extra'    => (float)($attendance['total_extra'] ?? 0),
   219→        'overtime_hours' => (float)($attendance['overtime_hours'] ?? 0),
   220→        'total_paid_days'=> (float)($attendance['total_paid_days'] ?? 0),
   221→        'updated_at'     => date('Y-m-d H:i:s'),
   222→    ];
   223→
   224→    if ($existingAtt) {
   225→        $db->update('attendance_summary', $attData, 'id = :id', ['id' => $existingAtt['id']]);
   226→    } else {
   227→        $attData['created_at'] = date('Y-m-d H:i:s');
   228→        $db->insert('attendance_summary', $attData);
   229→    }
   230→
   231→    // ═══════════════════════════════════════════════════════════════
   232→    // 2. EMPLOYEE_SALARY_STRUCTURES — Upsert for this month
   233→    // ═══════════════════════════════════════════════════════════════
   234→    $effFrom = sprintf('%04d-%02d-01', $year, $month);
   235→
   236→    $existingSal = $db->fetch(
   237→        "SELECT id FROM employee_salary_structures 
   238→         WHERE employee_id = ? AND effective_from = ?",
   239→        [$employeeId, $effFrom]
   240→    );
   241→
   242→    $grossSalary = (float)($wageDetails['gross_salary'] ?? 0);
   243→    if ($grossSalary == 0) {
   244→        $grossSalary = (float)($wageDetails['basic_da'] ?? 0)
   245→                     + (float)($wageDetails['hra'] ?? 0)
   246→                     + (float)($wageDetails['leave_encashment'] ?? 0)
   247→                     + (float)($wageDetails['bonus_encashment'] ?? 0)
   248→                     + (float)($wageDetails['washing_allowance'] ?? 0);
   249→    }
   250→
   251→    $pfApplicable  = !empty($data['pf_applicable']) ? 1 : 0;
   252→    $esiApplicable = !empty($data['esi_applicable']) ? 1 : 0;
   253→    $ptApplicable  = !empty($data['pt_applicable']) ? 1 : 0;
   254→    $lwfApplicable = !empty($data['lwf_applicable']) ? 1 : 0;
   255→
   256→    $salData = [
   257→        'employee_id'       => $employeeId,
   258→        'effective_from'    => $effFrom,
   259→        'effective_to'      => null,  // Current (no end date)
   260→        'basic_da'          => (float)($wageDetails['basic_da'] ?? 0),
   261→        'hra'               => (float)($wageDetails['hra'] ?? 0),
   262→        'leave_encashment'  => (float)($wageDetails['leave_encashment'] ?? 0),
   263→        'bonus_encashment'  => (float)($wageDetails['bonus_encashment'] ?? 0),
   264→        'washing_allowance' => (float)($wageDetails['washing_allowance'] ?? 0),
   265→        'gross_salary'      => $grossSalary,
   266→        'pf_applicable'     => $pfApplicable,
   267→        'esi_applicable'    => $esiApplicable,
   268→        'pt_applicable'     => $ptApplicable,
   269→        'lwf_applicable'    => $lwfApplicable,
   270→        'overtime_applicable'=> 1,
   271→        'updated_at'        => date('Y-m-d H:i:s'),
   272→    ];
   273→
   274→    if ($existingSal) {
   275→        $db->update('employee_salary_structures', $salData, 'id = :id', ['id' => $existingSal['id']]);
   276→    } else {
   277→        $salData['created_at'] = date('Y-m-d H:i:s');
   278→        $db->insert('employee_salary_structures', $salData);
   279→    }
   280→
   281→    // Close any previous salary structure that was open-ended
   282→    // (effective_to IS NULL and effective_from < current effFrom)
   283→    $db->query(
   284→        "UPDATE employee_salary_structures 
   285→         SET effective_to = DATE_SUB(?, INTERVAL 1 DAY), updated_at = NOW() 
   286→         WHERE employee_id = ? AND effective_from < ? AND effective_to IS NULL AND id != ?",
   287→        [$effFrom, $employeeId, $effFrom, $existingSal ? $existingSal['id'] : 0]
   288→    );
   289→
   290→    // ═══════════════════════════════════════════════════════════════
   291→    // 3. EMPLOYEE_ADVANCES — Upsert on employee_id + month + year
   292→    // ═══════════════════════════════════════════════════════════════
   293→    $existingAdv = $db->fetch(
   294→        "SELECT id FROM employee_advances WHERE employee_id = ? AND month = ? AND year = ?",
   295→        [$employeeId, $month, $year]
   296→    );
   297→
   298→    $totalAdvance = (float)($advances['advance'] ?? 0);
   299→    $officeDeduction = (float)($advances['office_deduction'] ?? 0);
   300→
   301→    $advData = [
   302→        'employee_id'    => $employeeId,
   303→        'month'          => $month,
   304→        'year'           => $year,
   305→        'adv1'           => $totalAdvance,   // Total advance stored in adv1
   306→        'adv2'           => 0,
   307→        'office_advance' => $officeDeduction, // Office deduction stored in office_advance
   308→        'dress_advance'  => 0,
   309→        'updated_at'     => date('Y-m-d H:i:s'),
   310→    ];
   311→
   312→    // Fix: exclude office_deduction from advance total to avoid double-counting
   313→    // process.php queries advance separately from office_deduction.
   314→    // We store them in separate columns so total_advance = adv1 only (no office_advance in it).
   315→
   316→    if ($existingAdv) {
   317→        $db->update('employee_advances', $advData, 'id = :id', ['id' => $existingAdv['id']]);
   318→    } else {
   319→        $advData['created_at'] = date('Y-m-d H:i:s');
   320→        $db->insert('employee_advances', $advData);
   321→    }
   322→
   323→    // ═══════════════════════════════════════════════════════════════
   324→    // 4. PAYROLL — Upsert on employee_id(=code) + month + year
   325→    // ═══════════════════════════════════════════════════════════════
   326→    $existingPay = null;
   327→    try {
   328→        $existingPay = $db->fetch(
   329→            "SELECT id FROM payroll WHERE employee_id = ? AND month = ? AND year = ?",
   330→            [$employeeCode, $month, $year]
   331→        );
   332→    } catch (Exception $e) {}
   333→
   334→    $totalDays = (int)($payroll['total_days'] ?? 0) ?: cal_days_in_month(CAL_GREGORIAN, $month, $year);
   335→
   336→    // Read PF/ESI rates from DB (for employer contribution calculation if not sent from JS)
   337→    $pfEmployerShare = 3.67; $pfEmployerEps = 8.33; $pfEmployerEdlis = 0.50; $pfEpfAdmin = 0.50; $esiEmployerShare = 3.25;
   338→    try {
   339→        $pfRate = $db->fetch("SELECT * FROM pf_rates WHERE is_active = 1 ORDER BY effective_from DESC LIMIT 1");
   340→        if ($pfRate) {
   341→            $pfEmployerShare = (float)$pfRate['employer_share_pf'];
   342→            $pfEmployerEps  = (float)$pfRate['employer_share_eps'];
   343→            $pfEmployerEdlis = (float)$pfRate['employer_share_edlis'];
   344→            $pfEpfAdmin     = (float)$pfRate['epf_admin_charges'];
   345→        }
   346→    } catch (Exception $e) {}
   347→    try {
   348→        $esiRate = $db->fetch("SELECT * FROM esi_rates WHERE is_active = 1 ORDER BY effective_from DESC LIMIT 1");
   349→        if ($esiRate) {
   350→            $esiEmployerShare = (float)$esiRate['employer_share'];
   351→        }
   352→    } catch (Exception $e) {}
   353→
   354→    // CTC: gross + employer contributions (matching class.payroll.php)
   355→    $ctcFromPayload = round2($payroll['ctc'] ?? 0);
   356→    if ($ctcFromPayload > 0) {
   357→        $ctc = $ctcFromPayload;
   358→    } else {
   359→        // Fallback: calculate from saved values
   360→        $grossSalary = round2($payroll['gross_salary'] ?? 0);
   361→        $pfEmployeeVal = round2($payroll['pf_employee'] ?? 0);
   362→        $esiEmployeeVal = round2($payroll['esi_employee'] ?? 0);
   363→        $pfBase = min($pfEmployeeVal, round2($payroll['basic_da'] ?? 0));
   364→        $employerContrib = round2($pfBase * ($pfEmployerShare + $pfEmployerEps + $pfEmployerEdlis + $pfEpfAdmin) / 100);
   365→        $ctc = round2($grossSalary + $employerContrib);
   366→    }
   367→
   368→    // Round all monetary values to 2 decimal places
   369→    // NOTE: gross_earnings from JS already INCLUDES overtime_amount — do NOT add it again
   370→    $calcGross  = round2($payroll['gross_earnings'] ?? 0);
   371→    $calcDed    = round2($payroll['total_deductions'] ?? 0);
   372→    $calcNetPay = round($calcGross - $calcDed); // Round to nearest ₹.00
   373→
   374→    $payData = [
   375→        'employee_id'      => $employeeCode,
   376→        'unit_id'          => $unitId,
   377→        'month'            => $month,
   378→        'year'             => $year,
   379→        'basic_da'         => round2($payroll['basic_da'] ?? 0),
   380→        'hra'              => round2($payroll['hra'] ?? 0),
   381→        'leave_encashment' => round2($payroll['leave_encashment'] ?? 0),
   382→        'bonus_encashment' => round2($payroll['bonus_encashment'] ?? 0),
   383→        'washing_allowance'=> round2($payroll['washing_allowance'] ?? 0),
   384→        'overtime_amount'  => round2($payroll['overtime_amount'] ?? 0),
   385→        'overtime_hours'   => round2($payroll['overtime_hours'] ?? 0),
   386→        'gross_earnings'   => $calcGross,
   387→        'gross_salary'     => round2($payroll['gross_salary'] ?? 0),
   388→        'pf_employee'      => round2($payroll['pf_employee'] ?? 0),
   389→        'esi_employee'     => round2($payroll['esi_employee'] ?? 0),
   390→        'professional_tax' => round2($payroll['professional_tax'] ?? 0),
   391→        'lwf_employee'     => round2($payroll['lwf_employee'] ?? 0),
   392→        'salary_advance'   => round2($payroll['salary_advance'] ?? 0),
   393→        'office_deduction' => round2($payroll['office_deduction'] ?? 0),
   394→        'total_deductions' => $calcDed,
   395→        'net_pay'          => $calcNetPay,
   396→        'paid_days'        => (int)($payroll['paid_days'] ?? 0),
   397→        'total_days'       => $totalDays,
   398→        'ctc'              => $ctc,
   399→        'status'           => 'Processed',
   400→        'salary_hold'      => 0,
   401→        'payroll_dirty'    => 0,
   402→        'last_calculated_at' => date('Y-m-d H:i:s'),
   403→        'calculated_by'    => (int)($_SESSION['user_id'] ?? 0) ?: null,
   404→        'updated_at'       => date('Y-m-d H:i:s'),
   405→    ];
   406→
   407→    // Employer contribution columns (matching process.php)
   408→    $payData['pf_employer']            = round2($payroll['pf_employer'] ?? 0);
   409→    $payData['eps_employer']           = round2($payroll['eps_employer'] ?? 0);
   410→    $payData['edlis_employer']         = round2($payroll['edlis_employer'] ?? 0);
   411→    $payData['epf_admin_charges']      = round2($payroll['epf_admin_charges'] ?? 0);
   412→    $payData['esi_employer']           = round2($payroll['esi_employer'] ?? 0);
   413→    $payData['total_employer_contribution'] = round2($payroll['total_employer_contribution'] ?? 0);
   414→
   415→    // Include loan_emi from the grid input
   416→    $payData['loan_emi'] = round2($advances['loan_emi'] ?? 0);
   417→
   418→    // Filter $payData to only include columns that actually exist in the payroll table
   419→    // This prevents SQL errors if auto-migration failed to add month/year
   420→    $safePayData = array_intersect_key($payData, $payrollColSet);
   421→    if (empty($safePayData)) {
   422→        throw new Exception('Payroll table column detection failed — cannot save');
   423→    }
   424→
   425→    if ($existingPay) {
   426→        $db->update('payroll', $safePayData, 'id = :id', ['id' => $existingPay['id']]);
   427→    } else {
   428→        $safePayData['created_at'] = date('Y-m-d H:i:s');
   429→        $db->insert('payroll', $safePayData);
   430→    }
   431→
   432→    // ── Update employee statutory flags in salary structure ──────
   433→    try {
   434→        $db->update('employee_salary_structures', [
   435→            'pf_applicable'  => $pfApplicable,
   436→            'esi_applicable' => $esiApplicable,
   437→            'pt_applicable'  => $ptApplicable,
   438→            'lwf_applicable' => $lwfApplicable,
   439→        ], 'employee_id = :eid AND effective_to IS NULL', ['eid' => $employeeId]);
   440→    } catch (Exception $e) {
   441→        // Column may not exist — ignore
   442→    }
   443→
   444→    // ═══════════════════════════════════════════════════════════════
   445→    // 5. LOAN EMI LOG & BALANCE UPDATE
   446→    // (Payroll record already saved with loan_emi from grid — don't double-count)
   447→    // Only update loan_emi_log and employee_loans balance.
   448→    // ═══════════════════════════════════════════════════════════════
   449→    $loanDeductionTotal = 0;
   450→    try {
   451→        // Ensure loan tables exist
   452→        $db->exec("CREATE TABLE IF NOT EXISTS `employee_loans` (
   453→            `id` int(11) NOT NULL AUTO_INCREMENT,
   454→            `employee_id` int(11) NOT NULL,
   455→            `unit_id` int(11) DEFAULT NULL,
   456→            `loan_type` varchar(50) DEFAULT 'Personal',
   457→            `amount` decimal(12,2) NOT NULL,
   458→            `interest_rate` decimal(5,2) DEFAULT 0.00,
   459→            `tenure_months` int(11) NOT NULL,
   460→            `emi_amount` decimal(12,2) NOT NULL,
   461→            `total_interest` decimal(12,2) DEFAULT 0.00,
   462→            `total_repayable` decimal(12,2) NOT NULL,
   463→            `balance_amount` decimal(12,2) NOT NULL,
   464→            `emi_deducted` int(11) DEFAULT 0,
   465→            `start_month` int(2) NOT NULL,
   466→            `start_year` int(4) NOT NULL,
   467→            `status` enum('Active','Closed','Settled','Written Off') DEFAULT 'Active',
   468→            `remarks` text DEFAULT NULL,
   469→            `created_at` timestamp DEFAULT CURRENT_TIMESTAMP,
   470→            `updated_at` timestamp DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
   471→            PRIMARY KEY (`id`),
   472→            KEY `idx_employee` (`employee_id`),
   473→            KEY `idx_status` (`status`)
   474→        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4");
   475→
   476→        $db->exec("CREATE TABLE IF NOT EXISTS `loan_emi_log` (
   477→            `id` int(11) NOT NULL AUTO_INCREMENT,
   478→            `loan_id` int(11) NOT NULL,
   479→            `employee_id` int(11) NOT NULL,
   480→            `month` int(2) NOT NULL,
   481→            `year` int(4) NOT NULL,
   482→            `emi_amount` decimal(12,2) NOT NULL,
   483→            `principal_component` decimal(12,2) DEFAULT 0.00,
   484→            `interest_component` decimal(12,2) DEFAULT 0.00,
   485→            `balance_after` decimal(12,2) NOT NULL,
   486→            `payroll_id` int(11) DEFAULT NULL,
   487→            `deducted_via_payroll` tinyint(1) DEFAULT 1,
   488→            `created_at` timestamp DEFAULT CURRENT_TIMESTAMP,
   489→            PRIMARY KEY (`id`),
   490→            UNIQUE KEY `uniq_loan_month_year` (`loan_id`, `month`, `year`),
   491→            KEY `idx_employee_month` (`employee_id`, `month`, `year`)
   492→        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4");
   493→
   494→        // Only proceed if the grid had a loan_emi value > 0
   495→        $gridLoanEmi = (float)($advances['loan_emi'] ?? 0);
   496→        if ($gridLoanEmi <= 0) {
   497→            // No EMI to process — but clean up any stale emi_log entries
   498→            // (from previous failed saves that partially committed)
   499→            try {
   500→                $db->query("DELETE FROM loan_emi_log WHERE employee_id = ? AND month = ? AND year = ? AND loan_id IN (SELECT id FROM employee_loans WHERE employee_id = ?)",
   501→                    [$employeeId, $month, $year, $employeeId]);
   502→            } catch (Exception $e) {}
   503→            goto loan_done;
   504→        }
   505→
   506→        // Get active loans for this employee that have started by this month
   507→        $empLoans = $db->fetchAll(
   508→            "SELECT * FROM employee_loans
   509→             WHERE employee_id = ? AND status = 'Active'
   510→             AND balance_amount > 0
   511→             AND (start_year < ? OR (start_year = ? AND start_month <= ?))",
   512→            [$employeeId, $year, $year, $month]
   513→        );
   514→
   515→        foreach ($empLoans as $empLoan) {
   516→            // Check if EMI already deducted for this month
   517→            $alreadyDeducted = $db->fetch(
   518→                "SELECT id FROM loan_emi_log WHERE loan_id = ? AND month = ? AND year = ?",
   519→                [$empLoan['id'], $month, $year]
   520→            );
   521→
   522→            if ($alreadyDeducted) {
   523→                continue;
   524→            }
   525→
   526→            $emiAmount = (float)$empLoan['emi_amount'];
   527→            $balanceAmount = (float)$empLoan['balance_amount'];
   528→
   529→            // Calculate interest/principal split
   530→            if ((float)$empLoan['interest_rate'] > 0) {
   531→                $monthlyRate = (float)$empLoan['interest_rate'] / 12 / 100;
   532→                $interestComponent = round($balanceAmount * $monthlyRate, 2);
   533→                $principalComponent = round($emiAmount - $interestComponent, 2);
   534→            } else {
   535→                $interestComponent = 0;
   536→                $principalComponent = $emiAmount;
   537→            }
   538→
   539→            // Handle last EMI: adjust to remaining balance
   540→            if ($emiAmount > $balanceAmount) {
   541→                $emiAmount = $balanceAmount;
   542→                $principalComponent = $emiAmount - $interestComponent;
   543→                if ($principalComponent < 0) {
   544→                    $interestComponent = $emiAmount;
   545→                    $principalComponent = 0;
   546→                }
   547→            }
   548→
   549→            $newBalance = round($balanceAmount - $emiAmount, 2);
   550→            if ($newBalance < 0) $newBalance = 0;
   551→
   552→            // Record EMI log (linked to payroll row — Audit Issue #7)
   553→            $db->insert('loan_emi_log', [
   554→                'loan_id' => $empLoan['id'],
   555→                'employee_id' => $employeeId,
   556→                'month' => $month,
   557→                'year' => $year,
   558→                'emi_amount' => $emiAmount,
   559→                'principal_component' => $principalComponent,
   560→                'interest_component' => $interestComponent,
   561→                'balance_after' => $newBalance,
   562→                'deducted_via_payroll' => 1
   563→            ]);
   564→
   565→            // Try to link to payroll row by month/year
   566→            try {
   567→                $linkedPayrollId = null;
   568→                $pRow = $db->fetch(
   569→                    "SELECT id FROM payroll WHERE employee_id = ? AND month = ? AND year = ?",
   570→                    [$employeeCode, $month, $year]
   571→                );
   572→                $linkedPayrollId = $pRow ? (int)$pRow['id'] : null;
   573→                if ($linkedPayrollId) {
   574→                    $db->query(
   575→                        "UPDATE loan_emi_log SET payroll_id = ? WHERE loan_id = ? AND month = ? AND year = ?",
   576→                        [$linkedPayrollId, $empLoan['id'], $month, $year]
   577→                    );
   578→                }
   579→            } catch (Exception $e) {}
   580→
   581→            // Update loan balance and status
   582→            $newStatus = ($newBalance <= 0) ? 'Closed' : 'Active';
   583→            $db->update('employee_loans', [
   584→                'balance_amount' => $newBalance,
   585→                'emi_deducted' => (int)$empLoan['emi_deducted'] + 1,
   586→                'status' => $newStatus
   587→            ], 'id = :id', ['id' => $empLoan['id']]);
   588→
   589→            $loanDeductionTotal += $emiAmount;
   590→        }
   591→
   592→        // NOTE: We do NOT update the payroll record here.
   593→        // The loan_emi was already included in the grid input and saved in step 4.
   594→        // Updating again would double-count the deduction.
   595→
   596→    } catch (Exception $loanEx) {
   597→        // Log but don't fail payroll save if loan deduction fails
   598→        error_log('Loan EMI auto-deduction failed: ' . $loanEx->getMessage());
   599→    }
   600→    loan_done:
   601→
   602→    $db->exec("COMMIT");
   603→
   604→    echo json_encode([
   605→        'success' => true,
   606→        'message' => $paidDaysWarning . $otWarning . 'Payroll saved for ' . $employeeCode . ($loanDeductionTotal > 0 ? ' (Loan EMI ₹' . number_format($loanDeductionTotal, 2) . ' auto-deducted)' : ''),
   607→        'employee_code' => $employeeCode,
   608→        'net_pay' => $payData['net_pay'],
   609→        'gross_salary' => $payData['gross_salary'],
   610→        'loan_deduction' => $loanDeductionTotal,
   611→    ]);
   612→
   613→} catch (Exception $e) {
   614→    // Rollback on error
   615→    try { $db->exec("ROLLBACK"); } catch (Exception $re) {}
   616→
   617→    error_log("Payroll Save Row Error [{$employeeCode}]: " . $e->getMessage());
   618→
   619→    echo json_encode([
   620→        'success' => false,
   621→        'message' => 'Error saving payroll: ' . $e->getMessage(),
   622→        'employee_code' => $employeeCode,
   623→    ]);
   624→}

   627→ * Helper: round to 2 decimal places
   628→ */
   629→function round2($val) {
   630→    return round((float)$val, 2);
   631→}
