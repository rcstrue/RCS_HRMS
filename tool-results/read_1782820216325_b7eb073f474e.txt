    28→    // Check existing session
    29→    private function checkSession() {
    30→        if (isset($_SESSION['user_id'])) {
    31→            $user = $this->db->fetch(
    32→                "SELECT u.id, u.username, u.email, u.role_id, u.first_name, u.last_name, u.is_active,
    33→                        r.role_name, r.role_code
    34→                 FROM users u 
    35→                 LEFT JOIN roles r ON u.role_id = r.id
    36→                 WHERE u.id = :id AND u.is_active = 1",
    37→                ['id' => $_SESSION['user_id']]
    38→            );
    39→            
    40→            if ($user) {
    41→                $this->user = $user;
    42→            }
    43→        }
    44→    }

    46→    // Login user
    47→    public function login($username, $password, $remember = false) {
    48→        // ── Rate limiting / lockout check ──
    49→        $ip = $_SERVER['REMOTE_ADDR'] ?? '0.0.0.0';
    50→        $lockout = $this->_checkLoginLockout($username, $ip);
    51→        if ($lockout) {
    52→            return ['success' => false, 'error' => $lockout];
    53→        }
    54→
    55→        // Get user with role info
    56→        $user = $this->db->fetch(
    57→            "SELECT u.id, u.username, u.email, u.password, u.role_id, u.first_name, u.last_name, u.is_active,
    58→                    r.role_name, r.role_code
    59→             FROM users u 
    60→             LEFT JOIN roles r ON u.role_id = r.id 
    61→             WHERE (u.username = :username OR u.email = :email)",
    62→            ['username' => $username, 'email' => $username]
    63→        );
    64→        
    65→        // Check if user exists
    66→        if (!$user) {
    67→            $this->_recordFailedLogin($username, $ip);
    68→            return ['success' => false, 'error' => 'Invalid username or password.'];
    69→        }
    70→        
    71→        // Check if active
    72→        if (empty($user['is_active'])) {
    73→            return ['success' => false, 'error' => 'Account is inactive.'];
    74→        }
    75→        
    76→        // Verify password
    77→        if (!password_verify($password, $user['password'])) {
    78→            $this->_recordFailedLogin($username, $ip);
    79→            return ['success' => false, 'error' => 'Invalid username or password.'];
    80→        }
    81→
    82→        // Clear failed attempts on successful login
    83→        $this->_clearFailedLogins($username, $ip);
    84→        
    85→        // Set session variables
    86→        $_SESSION['user_id'] = $user['id'];
    87→        $_SESSION['username'] = $user['username'];
    88→        $_SESSION['email'] = $user['email'];
    89→        $_SESSION['role_id'] = $user['role_id'];
    90→        $_SESSION['role_code'] = $user['role_code'] ?? 'worker';
    91→        $_SESSION['role_name'] = $user['role_name'] ?? 'User';
    92→        $_SESSION['first_name'] = $user['first_name'] ?? '';
    93→        $_SESSION['last_name'] = $user['last_name'] ?? '';
    94→        
    95→        // Update last login
    96→        try {
    97→            $this->db->update('users', [
    98→                'last_login' => date('Y-m-d H:i:s')
    99→            ], SQL_WHERE_USER_ID, ['id' => $user['id']]);
   100→        } catch (Exception $e) {
   101→            // Ignore if last_login column doesn't exist
   102→        }
   103→        
   104→        $this->user = $user;
   105→        
   106→        return [
   107→            'success' => true,
   108→            'message' => 'Login successful.',
   109→            'user' => $user
   110→        ];
   111→    }

   145→    // Check if user has role level or higher
   146→    public function hasRoleLevel($role) {
   147→        if (!isset($_SESSION['role_code'])) {
   148→            return false;
   149→        }
   150→        
   151→        $userLevel = $this->roleHierarchy[$_SESSION['role_code']] ?? 0;
   152→        $requiredLevel = $this->roleHierarchy[$role] ?? 0;
   153→        
   154→        return $userLevel >= $requiredLevel;
   155→    }

   157→    // Change password
   158→    public function changePassword($userId, $currentPassword, $newPassword) {
   159→        $user = $this->db->fetch(
   160→            "SELECT password FROM users WHERE id = :id",
   161→            ['id' => $userId]
   162→        );
   163→        
   164→        if (!$user || !password_verify($currentPassword, $user['password'])) {
   165→            return ['success' => false, 'message' => 'Current password is incorrect.'];
   166→        }
   167→        
   168→        $hashedPassword = password_hash($newPassword, PASSWORD_BCRYPT, ['cost' => 12]);
   169→        
   170→        $this->db->update('users', [
   171→            'password' => $hashedPassword
   172→        ], SQL_WHERE_USER_ID, ['id' => $userId]);
   173→        
   174→        return ['success' => true, 'message' => 'Password changed successfully.'];
   175→    }

   177→    // Create new user
   178→    public function createUser($data) {
   179→        $exists = $this->db->fetch(
   180→            "SELECT id FROM users WHERE username = :username OR email = :email",
   181→            ['username' => $data['username'], 'email' => $data['email']]
   182→        );
   183→        
   184→        if ($exists) {
   185→            return ['success' => false, 'message' => 'Username or email already exists.'];
   186→        }
   187→        
   188→        $hashedPassword = password_hash($data['password'], PASSWORD_BCRYPT, ['cost' => 12]);
   189→        
   190→        $userId = $this->db->insert('users', [
   191→            'username' => $data['username'],
   192→            'email' => $data['email'],
   193→            'password' => $hashedPassword,
   194→            'role_id' => $data['role_id'] ?? 5,
   195→            'first_name' => $data['first_name'] ?? '',
   196→            'last_name' => $data['last_name'] ?? '',

   200→        if ($userId) {
   201→            return ['success' => true, 'message' => 'User created successfully.', 'user_id' => $userId];

   222→    // Get all users (admin/hr)
   223→    public function getUsers($filters = []) {
   224→        $sql = "SELECT u.id, u.username, u.email, u.role_id, u.is_active, u.last_login, 
   225→                       u.created_at, u.first_name, u.last_name, r.role_name, r.role_code
   226→                FROM users u
   227→                LEFT JOIN roles r ON u.role_id = r.id
   228→                WHERE 1=1";
   229→        $params = [];
   230→        
   231→        if (!empty($filters['role_id'])) {
   232→            $sql .= " AND u.role_id = :role_id";
   233→            $params['role_id'] = $filters['role_id'];
   234→        }
   235→        
   236→        if (isset($filters['is_active'])) {
   237→            $sql .= " AND u.is_active = :is_active";
   238→            $params['is_active'] = $filters['is_active'];
   239→        }
   240→        
   241→        $sql .= " ORDER BY u.created_at DESC";
   242→        
   243→        return $this->db->fetchAll($sql, $params);
   244→    }

   267→    // Update user
   268→    public function updateUser($userId, $data) {
   269→        $allowedFields = ['email', 'role_id', 'first_name', 'last_name', 'phone', 'is_active'];
   270→        $updateData = [];
   271→        
   272→        foreach ($allowedFields as $field) {
   273→            if (isset($data[$field])) {
   274→                $updateData[$field] = $data[$field];
   275→            }
   276→        }
   277→        
   278→        if (empty($updateData)) {
   279→            return ['success' => false, 'message' => 'No data to update.'];
   280→        }
   281→        
   282→        $result = $this->db->update('users', $updateData, SQL_WHERE_USER_ID, ['id' => $userId]);
   283→        
   284→        return ['success' => true, 'message' => 'User updated successfully.'];
   285→    }

   290→        if (isset($_SESSION['user_id']) && $userId == $_SESSION['user_id']) {
   291→            return ['success' => false, 'message' => 'Cannot delete your own account.'];
   292→        }
   293→        
   294→        $result = $this->db->delete('users', SQL_WHERE_USER_ID, ['id' => $userId]);
   295→        
   296→        return ['success' => true, 'message' => 'User deleted successfully.'];
   297→    }

   440→    /**
   441→     * Create menu permissions table if not exists (enhanced version with submenus and actions)
   442→     */
   443→    public function ensureMenuPermissionsTable() {
   444→        try {
   445→            // Check if table exists with new structure
   446→            $checkColumn = $this->db->fetch("SHOW COLUMNS FROM role_menu_permissions LIKE 'submenu_key'");
   447→            
   448→            if (!$checkColumn) {
   449→                // Table exists but needs migration - add new columns
   450→                try {
   451→                    $this->db->query("ALTER TABLE role_menu_permissions 
   452→                        ADD COLUMN submenu_key VARCHAR(100) DEFAULT NULL AFTER menu_key,
   453→                        ADD COLUMN can_view TINYINT(1) DEFAULT 1 AFTER is_visible,
   454→                        ADD COLUMN can_add TINYINT(1) DEFAULT 0 AFTER can_view,
   455→                        ADD COLUMN can_edit TINYINT(1) DEFAULT 0 AFTER can_add,
   456→                        ADD COLUMN can_delete TINYINT(1) DEFAULT 0 AFTER can_edit,
   457→                        ADD COLUMN can_export TINYINT(1) DEFAULT 0 AFTER can_delete,
   458→                        ADD COLUMN can_import TINYINT(1) DEFAULT 0 AFTER can_export,
   459→                        ADD COLUMN can_print TINYINT(1) DEFAULT 0 AFTER can_import,
   460→                        DROP INDEX uniq_role_menu,
   461→                        ADD UNIQUE KEY uniq_role_menu_submenu (role_id, menu_key, submenu_key),
   462→                        ADD INDEX idx_role_menu (role_id, menu_key),
   463→                        ADD INDEX idx_role_submenu (role_id, submenu_key)");
   464→                } catch (Exception $e) {
   465→                    // Columns might already exist
   466→                }
   467→            }
   468→        } catch (Exception $e) {

   933→    private function _ensureLoginAttemptsTable(): void
   934→    {
   935→        static $created = false;
   936→        if ($created) return;
   937→        $this->db->query("
   938→            CREATE TABLE IF NOT EXISTS login_attempts (
   939→                id          INT AUTO_INCREMENT PRIMARY KEY,
   940→                username    VARCHAR(255) NOT NULL,
   941→                ip          VARCHAR(45)  NOT NULL,
   942→                attempts    INT          NOT NULL DEFAULT 0,
   943→                last_attempt DATETIME    NOT NULL,
   944→                locked_until DATETIME    NULL,
   945→                created_at  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
   946→                UNIQUE KEY uk_user_ip (username, ip)
   947→            ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4
   948→        ");
   949→        $created = true;

   952→    /**
   953→     * Check if the username+IP is currently locked out.
   954→     * Returns an error message string if locked, or null if allowed.
   955→     */
   956→    private function _checkLoginLockout($username, $ip): ?string
   957→    {
   958→        $this->_ensureLoginAttemptsTable();
   959→
   960→        $row = $this->db->fetch(
   961→            "SELECT attempts, locked_until FROM login_attempts WHERE username = :u AND ip = :i",
   962→            ['u' => $username, 'i' => $ip]
   963→        );
   964→
   965→        if (!$row) {
   966→            return null;
   967→        }
   968→
   969→        // If there is a lockout window, check if it has expired
   970→        if ($row['locked_until'] !== null) {
   971→            $now  = new \DateTime('now');
   972→            $lock = new \DateTime($row['locked_until']);
   973→            if ($now < $lock) {
   974→                $diff  = $now->diff($lock);
   975→                $mins  = $diff->i + ($diff->h * 60) + ($diff->d * 1440);
   976→                $hours = (int)floor($mins / 60);
   977→                if ($hours >= 1) {
