   118→            $status = 'Pending';

   120→            $status = 'Current';
   121→        } else {
   122→            $status = 'Future';
   123→        }

   125→        // Skip future months if loan is closed
   126→        if ($loan['status'] !== 'Active' && $status === 'Future') {
   127→            break;
   128→        }
   129→

   131→        if ($loan['interest_rate'] > 0 && $runningBalance > 0) {
   132→            $monthlyRate = (float)$loan['interest_rate'] / 12 / 100;
   133→            $interestComp = round($runningBalance * $monthlyRate, 2);
   134→            $principalComp = round($emiAmount - $interestComp, 2);
   135→        } else {
   136→            $interestComp = 0;
   137→            $principalComp = $emiAmount;
   138→        }
   139→
   140→        // Adjust last EMI
   141→        if ($emiAmount > $runningBalance && $runningBalance > 0) {
   142→            $emiAmount = $runningBalance;
   143→            $principalComp = $emiAmount;
   144→        }

   148→            'year' => $schedYear,
   149→            'emi' => $status === 'Future' ? $emiAmount : 0,
   150→            'principal' => $principalComp,
   151→            'interest' => $interestComp,

   153→            'status' => $status,
   154→            'paid_via_payroll' => false,
   155→            'paid_date' => null
   156→        ];

   159→            $runningBalance = max(0, $runningBalance - $emiAmount);
   160→        }
   161→    }

   164→        break;
