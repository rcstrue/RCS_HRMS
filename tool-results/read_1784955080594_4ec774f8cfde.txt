     9→ * =================================
    10→ * 1. Always use client_id and unit_id for filtering (NOT client_name/unit_name)

    83→    // Get payroll periods
    84→    public function getPeriods($status = null) {
    85→        $sql = "SELECT pp.*, 
    86→                       COUNT(p.id) as employee_count,
    87→                       SUM(CASE WHEN p.salary_hold = 1 THEN 1 ELSE 0 END) as hold_count
    88→                FROM payroll_periods pp
    89→                LEFT JOIN payroll p ON pp.id = p.payroll_period_id
    90→                WHERE 1=1";
    91→        $params = [];
    92→
    93→        if ($status) {
    94→            $sql .= " AND pp.status = :status";
    95→            $params['status'] = $status;
    96→        }
    97→
    98→        $sql .= " GROUP BY pp.id ORDER BY pp.year DESC, pp.month DESC";
    99→
   100→        return $this->db->fetchAll($sql, $params);
   101→    }

   156→    // Get payroll for period with filters
   157→    public function getPayrollReport($periodId, $filters = []) {
   158→        $sql = "SELECT p.*,
   159→                       e.employee_code, e.full_name, e.designation, e.department,
   160→                       e.bank_name, e.account_number, e.ifsc_code,
   161→                       c.name as client_name, u.name as unit_name
   162→                FROM payroll p
   163→                JOIN employees e ON p.employee_id = e.employee_code
   164→                LEFT JOIN clients c ON e.client_id = c.id
   165→                LEFT JOIN units u ON e.unit_id = u.id
   166→                WHERE p.payroll_period_id = :period_id";
   167→        $params = ['period_id' => $periodId];
   168→
   169→        // Apply filters
   170→        if (!empty($filters['unit_id'])) {
   171→            $sql .= " AND e.unit_id = :unit_id";
   172→            $params['unit_id'] = $filters['unit_id'];
   173→        }
   174→
   175→        if (!empty($filters['client_id'])) {
   176→            $sql .= " AND e.client_id = :client_id";
   177→            $params['client_id'] = $filters['client_id'];
   178→        }
   179→
   180→        if (!empty($filters['status'])) {
   181→            $sql .= " AND p.status = :pstatus";
   182→            $params['pstatus'] = $filters['status'];
   183→        }
   184→
   185→        if (!empty($filters['salary_hold'])) {
   186→            $sql .= " AND p.salary_hold = :salary_hold";
   187→            $params['salary_hold'] = $filters['salary_hold'];
   188→        }
   189→
   190→        if (!empty($filters['search'])) {
   191→            $sql .= " AND (e.full_name LIKE :search1 OR e.employee_code LIKE :search2)";
   192→            $params['search1'] = '%' . $filters['search'] . '%';
   193→            $params['search2'] = '%' . $filters['search'] . '%';
   194→        }
   195→
   196→        $sql .= " ORDER BY c.name, u.name, e.employee_code";
   197→
   198→        return $this->db->fetchAll($sql, $params);

   226→    /**
   227→     * Process payroll for period with selective filters
   228→     * 
   229→     * @param int $periodId The payroll period ID
   230→     * @param array $filters Optional filters for selective processing
   231→     *                       - client_id: Process only this client's employees
   232→     *                       - unit_id: Process only this unit's employees
   233→     *                       - employee_codes: Array of specific employee codes
   234→     *                       - recalculate_dirty: Only recalculate dirty records
   235→     * @return array Result with success status and processed count
   236→     */
   237→    public function processPayroll($periodId, $filters = []) {
   238→        // Get period info
   239→        $period = $this->db->fetch(
   240→            "SELECT * FROM payroll_periods WHERE id = :id",
   241→            ['id' => $periodId]
   242→        );
   243→
   244→        if (!$period) {
   245→            return ['success' => false, 'message' => 'Payroll period not found.'];
   246→        }
   247→
   248→        // Check if period is frozen
   249→        if ($period['status'] === 'Frozen' || $period['status'] === 'Locked') {
   250→            return ['success' => false, 'message' => 'Payroll period is frozen. Cannot process.'];
   251→        }
   252→
   253→        // Ensure payroll table has extra_days_amount column
   254→        try {
   255→            $checkColumn = $this->db->fetch("SHOW COLUMNS FROM payroll LIKE 'extra_days_amount'");
   256→            if (!$checkColumn) {
   257→                $this->db->exec("ALTER TABLE payroll ADD COLUMN extra_days_amount DECIMAL(10,2) DEFAULT 0.00 AFTER overtime_amount");
   258→            }
   259→        } catch (Exception $e) {
   260→            // Column might already exist or table might not have records
   261→        }
   262→
   263→        // Build employee query with filters
   264→        $sql = "SELECT e.id, e.employee_code, e.full_name, e.date_of_joining,
   265→                       e.date_of_leaving, e.client_id, e.unit_id, e.worker_category,
   266→                       e.bank_name, e.account_number, e.ifsc_code,
   267→                       c.name as client_name, u.name as unit_name,
   268→                       ess.basic_da, ess.hra, ess.leave_encashment, ess.bonus_encashment, ess.washing_allowance,
   269→                       ess.gross_salary, ess.pf_applicable, ess.esi_applicable,
   270→                       ess.pt_applicable, ess.lwf_applicable, ess.overtime_applicable,
   271→                       ess.bonus_applicable, ess.gratuity_applicable,
   272→                       usf.pay_days_type, usf.ot_calculation_type, usf.ot_calculation_on, usf.ot_hours_per_day,
   273→                       u.state as unit_state
   274→                FROM employees e
   275→                INNER JOIN employee_salary_structures ess ON e.id = ess.employee_id
   276→                    AND (ess.effective_to IS NULL OR ess.effective_to >= CURDATE())
   277→                LEFT JOIN clients c ON e.client_id = c.id
   278→                LEFT JOIN units u ON e.unit_id = u.id
   279→                LEFT JOIN unit_salary_formulas usf ON usf.unit_id = e.unit_id
   280→                    AND usf.is_active = 1
   281→                    AND (usf.effective_to IS NULL OR usf.effective_to >= CURDATE())
   282→                WHERE e.status IN ('approved', 'active')";
   283→
   284→        $params = [];
   285→
   286→        // Apply filters for selective processing
   287→        if (!empty($filters['client_id'])) {
   288→            $sql .= " AND e.client_id = :client_id";
   289→            $params['client_id'] = $filters['client_id'];
   290→        }
   291→
   292→        if (!empty($filters['unit_id'])) {
   293→            $sql .= " AND e.unit_id = :unit_id";
   294→            $params['unit_id'] = $filters['unit_id'];
   295→        }
   296→
   297→        // Employee codes filter - use named params to avoid mixing with positional
   298→        if (!empty($filters['employee_codes']) && is_array($filters['employee_codes'])) {
   299→            $empPlaceholders = [];
   300→            $idx = 1;
   301→            foreach ($filters['employee_codes'] as $code) {
   302→                $key = 'emp_code_' . $idx++;
   303→                $empPlaceholders[] = ':' . $key;
   304→                $params[$key] = $code;
   305→            }
   306→            $sql .= " AND e.employee_code IN (" . implode(',', $empPlaceholders) . ")";
   307→        }
   308→
   309→        // Only recalculate dirty records if specified
   310→        if (!empty($filters['recalculate_dirty'])) {
   311→            $sql .= " AND EXISTS (
   312→                SELECT 1 FROM payroll p 
   313→                WHERE p.employee_id = e.employee_code 
   314→                AND p.payroll_period_id = :period_id_check
   315→                AND p.payroll_dirty = 1
   316→            )";
   317→            $params['period_id_check'] = $periodId;
   318→        }
   319→
   320→        $employees = $this->db->fetchAll($sql, $params);
   321→
   322→        $processed = 0;
   323→        $errors = [];
   324→        $exceptions = [];
   325→        $totalGross = 0;
   326→        $totalNet = 0;
   327→        $totalPF = 0;
   328→        $totalESI = 0;
   329→
   330→        $this->db->beginTransaction();
   331→
   332→        try {
   333→            foreach ($employees as $emp) {
   334→                try {
   335→                    // Check for exceptions
   336→                    $exceptionType = null;
   337→                    
   338→                    // Check if employee joined after period end
   339→                    $periodEndDate = $period['end_date'];
   340→                    if ($emp['date_of_joining'] && strtotime($emp['date_of_joining']) > strtotime($periodEndDate)) {
   341→                        continue;
   342→                    }
   343→
   344→                    // Check for missing salary structure
   345→                    if (empty($emp['basic_da']) && empty($emp['gross_salary'])) {
   346→                        $exceptionType = 'Undefined Salary';
   347→                        $exceptions[] = [
   348→                            'employee_id' => $emp['employee_code'],
   349→                            'employee_name' => $emp['full_name'],
   350→                            'type' => 'Undefined Salary',
   351→                            'message' => 'No salary structure defined'
   352→                        ];
   353→                    }
   354→
   355→                    // Check for missing bank details
   356→                    if (empty($emp['bank_name']) || empty($emp['account_number']) || empty($emp['ifsc_code'])) {
   357→                        if (!$exceptionType) {
   358→                            $exceptionType = 'Missing Bank Details';
   359→                        }
   360→                        $exceptions[] = [
   361→                            'employee_id' => $emp['employee_code'],
   362→                            'employee_name' => $emp['full_name'],
   363→                            'type' => 'Missing Bank Details',
   364→                            'message' => 'Bank account details incomplete'
   365→                        ];
   366→                    }
   367→
   368→                    // Get attendance summary — prefer unit_id match, fallback to any unit (backward compatible)
   369→                    $attendance = $this->db->fetch(
   370→                        "SELECT
   371→                            total_present as present_days,
   372→                            total_extra,
   373→                            overtime_hours,
   374→                            total_wo as weekly_offs,
   375→                            total_paid_days
   376→                        FROM attendance_summary
   377→                        WHERE employee_id = :emp_id
   378→                        AND month = :month
   379→                        AND year = :year
   380→                        ORDER BY unit_id = :unit_id DESC
   381→                        LIMIT 1",
   382→                        ['emp_id' => $emp['id'], 'unit_id' => $emp['unit_id'], 'month' => $period['month'], 'year' => $period['year']]
   383→                    );
   384→
   385→                    // Calculate total days based on unit's pay_days_type setting
   386→                    $payDaysType = $emp['pay_days_type'] ?? 'actual';
   387→                    switch ($payDaysType) {
   388→                        case 'fixed_30':
   389→                            $totalDays = 30;
   390→                            break;
   391→                        case 'previous_month':
   392→                            // For 26-25 cycle: use previous month's days
   393→                            $prevMonth = $period['month'] - 1;
   394→                            $prevYear = $period['year'];
   395→                            if ($prevMonth < 1) { $prevMonth = 12; $prevYear--; }
   396→                            $totalDays = (int)date('t', strtotime("$prevYear-$prevMonth-01"));
   397→                            break;
   398→                        case 'calendar_minus_sundays':
   399→                            // Month days minus all Sundays
   400→                            $calDays = (int)cal_days_in_month(CAL_GREGORIAN, $period['month'], $period['year']);
   401→                            $sundays = 0;
   402→                            for ($d = 1; $d <= $calDays; $d++) {
   403→                                $dow = (int)date('N', strtotime(sprintf('%04d-%02d-%02d', $period['year'], $period['month'], $d)));
   404→                                if ($dow === 7) $sundays++;
   405→                            }
   406→                            $totalDays = $calDays - $sundays;
   407→                            break;
   408→                        default: // 'actual'
   409→                            $totalDays = $period['pay_days'] ?? 30;
   410→                            break;
   411→                    }
   412→                    
   413→                    // Use total_paid_days from attendance_summary if available, otherwise calculate
   414→                    $preCalcPaidDays = floatval($attendance['total_paid_days'] ?? 0);
   415→                    if ($preCalcPaidDays > 0) {
   416→                        $paidDays = $preCalcPaidDays;
   417→                    } else {
   418→                        // Fallback: calculate from individual components
   419→                        $paidDays = floatval($attendance['total_present'] ?? 0);
   420→                        $paidDays += floatval($attendance['weekly_offs'] ?? 0);
   421→                        $paidDays += floatval($attendance['total_extra'] ?? 0);
   422→                    }
   423→                    
   424→                    // Check if attendance record exists (must check is_array first to avoid warnings)
   425→                    // NOTE: SQL aliases total_present → present_days, total_wo → weekly_offs
   426→                    $hasAttendance = is_array($attendance) && array_key_exists('present_days', $attendance) && $attendance['present_days'] !== null;
   427→
   428→                    // Check for missing attendance — SKIP employee, do NOT assume full month
   429→                    if (!$hasAttendance) {
   430→                        $exceptions[] = [
   431→                            'employee_id' => $emp['employee_code'],
   432→                            'employee_name' => $emp['full_name'],
   433→                            'type' => 'Missing Attendance',
   434→                            'message' => 'No attendance data found — employee skipped. Upload attendance first.'
   435→                        ];
   436→                        continue; // Skip this employee entirely
   437→                    }
   438→
   439→                    $unpaidDays = max(0, $totalDays - $paidDays);
   440→                    $overtimeHours = floatval($attendance['overtime_hours'] ?? 0);
   441→
   442→                    // Calculate earnings (pro-rated) - using basic_da as combined column
   443→                    $basicDa = round(($emp['basic_da'] ?? 0) * $paidDays / $totalDays, 2);
   444→                    $hra = round(($emp['hra'] ?? 0) * $paidDays / $totalDays, 2);
   445→                    $leaveEncashment = round(($emp['leave_encashment'] ?? 0) * $paidDays / $totalDays, 2);
   446→                    $bonusEncashment = round(($emp['bonus_encashment'] ?? 0) * $paidDays / $totalDays, 2);
   447→                    $washingAllowance = round(($emp['washing_allowance'] ?? 0) * $paidDays / $totalDays, 2);
   448→
   449→                    $grossEarnings = round($basicDa + $hra + $leaveEncashment + $bonusEncashment + $washingAllowance, 2);
   450→
   451→                    // Extra days are now included in paid_days, so no separate calculation needed
   452→                    $extraDaysAmount = 0;
   453→
   454→                    // Calculate overtime amount based on unit salary formula settings
   455→                    $overtimeAmount = 0;
   456→                    if ($overtimeHours > 0 && ($emp['overtime_applicable'] ?? 0)) {
   457→                        $otCalcType = $emp['ot_calculation_type'] ?? 'single_pay'; // single_pay or double_pay
   458→                        $otCalcOn = $emp['ot_calculation_on'] ?? 'basic_da';      // basic, basic_da, basic_hra, gross
   459→                        $otHoursPerDay = floatval($emp['ot_hours_per_day'] ?? 8);
   460→
   461→                        // Determine OT base amount
   462→                        switch ($otCalcOn) {
   463→                            case 'gross':
   464→                                $otBase = $emp['gross_salary'] ?? 0;
   465→                                break;
   466→                            case 'basic_hra':
   467→                                $otBase = ($emp['basic_da'] ?? 0) + ($emp['hra'] ?? 0);
   468→                                break;
   469→                            case 'basic':
   470→                                // Basic only (without DA) — approximate from basic_da
   471→                                $otBase = ($emp['basic_da'] ?? 0) * 0.5; // rough split
   472→                                break;
   473→                            default: // basic_da
   474→                                $otBase = $emp['basic_da'] ?? 0;
   475→                                break;
   476→                        }
   477→
   478→                        $hourlyRate = $otBase / $totalDays / $otHoursPerDay;
   479→                        $otMultiplier = ($otCalcType === 'double_pay') ? 2 : 1;
   480→                        $overtimeAmount = round($hourlyRate * $overtimeHours * $otMultiplier, 2);
   481→                    }
   482→
   483→                    $grossWithOT = round($grossEarnings + $overtimeAmount, 2);
   484→
   485→                    // Calculate PF
   486→                    $pfEmployee = 0;
   487→                    $pfEmployer = 0;
   488→                    $epsEmployer = 0;
   489→                    $edlisEmployer = 0;
   490→                    $epfAdmin = 0;
   491→
   492→                    if ($emp['pf_applicable'] ?? 0) {
   493→                        $pfBase = min($basicDa, $this->pfRates['wage_ceiling']);
   494→                        $pfEmployee = round($pfBase * $this->pfRates['employee_share'] / 100, 2);
   495→                        $pfEmployer = round($pfBase * $this->pfRates['employer_pf'] / 100, 2);
   496→                        $epsEmployer = round($pfBase * $this->pfRates['employer_eps'] / 100, 2);
   497→                        $edlisEmployer = round($pfBase * $this->pfRates['employer_edlis'] / 100, 2);
   498→                        $epfAdmin = round($pfBase * $this->pfRates['epf_admin'] / 100, 2);
   499→                    }
   500→
   501→                    // Calculate ESI - Check against actual gross salary (not pro-rated)
   502→                    $esiEmployee = 0;
   503→                    $esiEmployer = 0;
   504→                    $actualGrossSalary = floatval($emp['gross_salary'] ?? 0);
   505→
   506→                    if (($emp['esi_applicable'] ?? 0) && $actualGrossSalary <= $this->esiRates['wage_ceiling']) {
   507→                        $esiEmployee = round($grossWithOT * $this->esiRates['employee_share'] / 100, 2);
   508→                        $esiEmployer = round($grossWithOT * $this->esiRates['employer_share'] / 100, 2);
   509→                    }
   510→
   511→                    // Calculate Professional Tax
   512→                    $pt = 0;
   513→                    if ($emp['pt_applicable'] ?? 0) {
   514→                        $pt = $this->calculatePT($grossWithOT, $emp['unit_state'] ?? '');
   515→                    }
   516→
   517→                    // Calculate LWF (Labour Welfare Fund) — state-specific from lwf_rates table
   518→                    $lwfEmployee = 0;
   519→                    $lwfEmployer = 0;
   520→                    if (($emp['lwf_applicable'] ?? 0) && !empty($emp['unit_state'])) {
   521→                        $stateCode = $emp['unit_state'];
   522→                        $lwfRate = $this->db->fetch(
   523→                            "SELECT lr.employee_share, lr.employer_share, lr.contribution_months
   524→                             FROM lwf_rates lr
   525→                             JOIN states s ON lr.state_id = s.id
   526→                             WHERE (s.state_code = :st OR s.state_name = :st)
   527→                             AND lr.is_active = 1
   528→                             AND lr.effective_from <= CURDATE()
   529→                             ORDER BY lr.effective_from DESC LIMIT 1",
   530→                            ['st' => $stateCode]
   531→                        );
   532→                        if (!$lwfRate) {
   533→                            // Fallback: try lwf_state_rates by state_name
   534→                            $lwfRate = $this->db->fetch(
   535→                                "SELECT employee_share, employer_share, contribution_months
   536→                                 FROM lwf_state_rates
   537→                                 WHERE state_name = :st AND is_active = 1
   538→                                 AND effective_from <= CURDATE()
   539→                                 ORDER BY effective_from DESC LIMIT 1",
   540→                                ['st' => $stateCode]
   541→                            );
   542→                        }
   543→                        if ($lwfRate) {
   544→                            // Check if current month is a contribution month
   545→                            $contribMonths = array_filter(array_map('intval', explode(',', $lwfRate['contribution_months'] ?? '')));
   546→                            if (in_array((int)$period['month'], $contribMonths)) {
   547→                                $lwfEmployee = floatval($lwfRate['employee_share']);
   548→                                $lwfEmployer = floatval($lwfRate['employer_share']);
   549→                            }
   550→                        }
   551→                    }
   552→
   553→                    // Get advances for the period (scoped to unit_id to avoid double-deduction)
   554→                    $advance = $this->db->fetch(
   555→                        "SELECT COALESCE(SUM(adv1 + adv2 + dress_advance), 0) as total_advance
   556→                        FROM employee_advances
   557→                        WHERE employee_id = :emp_id AND unit_id = :unit_id AND month = :month AND year = :year",
   558→                        ['emp_id' => $emp['id'], 'unit_id' => $emp['unit_id'], 'month' => $period['month'], 'year' => $period['year']]
   559→                    );
   560→                    $salaryAdvance = $advance['total_advance'] ?? 0;
   561→
   562→                    // Get office deduction for the period (scoped to unit_id)
   563→                    $officeDeduction = 0;
   564→                    $officeAdv = $this->db->fetch(
   565→                        "SELECT COALESCE(office_advance, 0) as office_ded
   566→                         FROM employee_advances
   567→                         WHERE employee_id = :emp_id AND unit_id = :unit_id AND month = :month AND year = :year",

   569→                    );
   570→                    $officeDeduction = floatval($officeAdv['office_ded'] ?? 0);
   571→
   572→                    // Get active loan EMI for this employee (Audit Issue #8 — include in total_deductions)
   573→                    $loanEmi = 0;
   574→                    $activeLoans = $this->db->fetchAll(
   575→                        "SELECT el.emi_amount
   576→                         FROM employee_loans el
   577→                         WHERE el.employee_id = :emp_id AND el.status = 'Active'
   578→                         AND el.balance_amount > 0
   579→                         AND (el.start_year < :year OR (el.start_year = :year AND el.start_month <= :month))
   580→                         AND el.id NOT IN (
   581→                             SELECT lem.loan_id FROM loan_emi_log lem
   582→                             WHERE lem.month = :month AND lem.year = :year
   583→                         )",
   584→                        ['emp_id' => $emp['id'], 'year' => $period['year'], 'month' => $period['month']]
   585→                    );
   586→                    foreach ($activeLoans as $aLoan) {
   587→                        $loanEmi += floatval($aLoan['emi_amount']);
   588→                    }
   589→
   590→                    // Total deductions (now includes loan_emi)
   591→                    $totalDeductions = round($pfEmployee + $esiEmployee + $pt + $lwfEmployee + $salaryAdvance + $officeDeduction + $loanEmi, 2);
   592→
   593→                    // Net pay (always round off)
   594→                    $netPay = round($grossWithOT - $totalDeductions); // Round to nearest ₹.00
   595→
   596→                    // Employer contributions
   597→                    $employerContribution = round($pfEmployer + $epsEmployer + $edlisEmployer + $epfAdmin + $esiEmployer + $lwfEmployer, 2);
   598→
   599→                    // Bonus provision (8.33% of basic_da, max ₹7000)
   600→                    $bonusProvision = 0;
   601→                    if ($emp['bonus_applicable'] ?? 0) {
   602→                        $bonusBase = min($basicDa, 7000);
   603→                        $bonusProvision = round($bonusBase * 8.33 / 100, 2);
   604→                    }
   605→
   606→                    // Gratuity provision (4.81% of basic_da)
   607→                    $gratuityProvision = 0;
   608→                    if ($emp['gratuity_applicable'] ?? 0) {
   609→                        $gratuityProvision = round($basicDa * 4.81 / 100, 2);
   610→                    }
   611→
   612→                    // CTC
   613→                    $ctc = round($grossWithOT + $employerContribution + $bonusProvision + $gratuityProvision, 2);
   614→
   615→                    // Minimum wage check (Audit Issue #14)
   616→                    $minWage = $this->db->fetch(
   617→                        "SELECT mw.minimum_wage FROM minimum_wages mw
   618→                         WHERE (mw.state = :state OR mw.state = 'All')
   619→                         AND mw.designation LIKE CONCAT('%', :cat, '%')
   620→                         AND mw.effective_from <= :dt
   621→                         ORDER BY mw.effective_from DESC LIMIT 1",
   622→                        ['state' => ($emp['unit_state'] ?? ''), 'cat' => ($emp['worker_category'] ?? ''), 'dt' => $period['end_date']]
   623→                    );
   624→                    if ($minWage && $grossEarnings < floatval($minWage['minimum_wage']) && $paidDays > 0) {

   627→                            'employee_name' => $emp['full_name'],
   628→                            'type' => 'Below Minimum Wage',
   629→                            'message' => 'Gross ₹' . number_format($grossEarnings, 2) . ' < min wage ₹' . number_format($minWage['minimum_wage'], 2) . ' for ' . ($emp['worker_category'] ?? 'N/A')
   630→                        ];
   631→                    }
   632→
   633→                    // Insert or update payroll record - matching database schema with basic_da column
   634→                    $this->db->query(
   635→                        "INSERT INTO payroll (
   636→                            payroll_period_id, employee_id, unit_id,

   638→                            basic_da, hra, leave_encashment, bonus_encashment, washing_allowance,
   639→                            overtime_amount, extra_days_amount, gross_earnings,
   640→                            pf_employee, esi_employee, professional_tax, lwf_employee,
   641→                            salary_advance, loan_emi, office_deduction, total_deductions,
   642→                            pf_employer, eps_employer, edlis_employer, epf_admin_charges, esi_employer,
   643→                            bonus_provision, gratuity_provision,
   644→                            total_employer_contribution,
   645→                            net_pay, gross_salary, ctc, status,

   650→                            :total_days, :paid_days, :unpaid_days, :ot_hours,

   698→                        [

   806→    /**
   807→     * Hold salary for specific employees
   808→     * 
   809→     * @param int $periodId The payroll period ID
   810→     * @param array $employeeCodes Array of employee codes
   811→     * @param string $reason Reason for holding salary
   812→     * @return array Result with success status
   813→     */
   814→    public function holdSalary($periodId, $employeeCodes, $reason = '') {
   815→        // Check if period is frozen
   816→        $period = $this->db->fetch(
   817→            "SELECT status FROM payroll_periods WHERE id = :id",
   818→            ['id' => $periodId]
   819→        );
   820→
   821→        if (!$period || in_array($period['status'], ['Frozen', 'Locked', 'Paid'])) {
   822→            return ['success' => false, 'message' => 'Cannot hold salary for this period.'];
   823→        }
   824→
   825→        $placeholders = implode(',', array_fill(0, count($employeeCodes), '?'));
   826→        $params = array_merge([$periodId], $employeeCodes);
   827→
   828→        $this->db->query(
   829→            "UPDATE payroll SET 
   830→                salary_hold = 1, 
   831→                hold_reason = ?, 
   832→                hold_date = CURDATE(),
   833→                status = 'Hold'
   834→            WHERE payroll_period_id = ? AND employee_id IN ($placeholders)",
   835→            array_merge([$reason], $params)
   836→        );
   837→
   838→        // Update hold count in period
   839→        $holdCount = $this->db->fetch(
   840→            "SELECT COUNT(*) as count FROM payroll WHERE payroll_period_id = ? AND salary_hold = 1",
   841→            [$periodId]
   842→        );
   843→
   844→        $this->db->update('payroll_periods', [
   845→            'hold_count' => $holdCount['count']
   846→        ], SQL_WHERE_ID, ['id' => $periodId]);
   847→
   848→        return ['success' => true, 'message' => 'Salary hold applied successfully.'];
   849→    }

   967→     * @param string $paymentDate The payment date
   968→     * @param array $excludeHeld Whether to exclude held salaries
   969→     * @return array Result with success status
   970→     */
   971→    public function markAsPaid($periodId, $paymentDate = null, $excludeHeld = true) {
   972→        if (!$paymentDate) {
   973→            $paymentDate = date('Y-m-d');
   974→        }
   975→
   976→        // Check if period is frozen

   982→        if (!$period) {
   983→            return ['success' => false, 'message' => 'Payroll period not found.'];

   988→        }
   989→
   990→        // Update period

   992→            'status' => 'Paid',

   999→            : "payroll_period_id = :period_id";
  1000→
  1001→        $this->db->query(
  1002→            "UPDATE payroll SET status = 'Paid', payment_status = 'Paid' WHERE $whereClause",
  1003→            ['period_id' => $periodId]
  1004→        );
  1005→
  1006→        return ['success' => true, 'message' => 'Payroll marked as paid.'];

  1254→    /**

  1256→     * Default slabs - can be overridden based on employee's work state
  1257→     * Common PT slabs across India (monthly)
  1258→     */
  1259→    private function calculatePT($gross, $state = 'GJ') {
  1260→        // Read from professional_tax_rates DB table instead of hardcoded slabs
  1261→        try {
  1262→            $rate = $this->db->fetch(
  1263→                "SELECT ptr.pt_amount
  1264→                 FROM professional_tax_rates ptr
  1265→                 JOIN states s ON ptr.state_id = s.id
  1266→                 WHERE (s.state_code = :state OR s.state_name = :state)
  1267→                   AND ptr.salary_from <= :gross
  1268→                   AND (ptr.salary_to IS NULL OR ptr.salary_to >= :gross)
  1269→                   AND ptr.is_active = 1
  1270→                 ORDER BY ptr.salary_from DESC
  1271→                 LIMIT 1",
  1272→                ['state' => $state, 'gross' => $gross]
  1273→            );
  1274→            return $rate ? (float)$rate['pt_amount'] : 0;
  1275→        } catch (Exception $e) {
  1276→            // Fallback to simple default if DB query fails
  1277→            if ($gross > 12000) return 200;
  1278→            return 0;
  1279→        }
  1280→    }
