     1→<?php
     2→/**

     5→ *

     8→ * - Minimum wage revision arrears
     9→ * - Correction arrears (mistakes in previous payroll)
    10→ * - Missing payment arrears
    11→ * 
    12→ * Routed through index.php - no need to re-include config/database
    13→ */
    14→

    16→

    19→
    20→// Handle arrear calculation
    21→if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
    22→    
    23→    if ($_POST['action'] === 'calculate') {
    24→        $employeeId = sanitize($_POST['employee_id']);
    25→        $arrearType = sanitize($_POST['arrear_type']);
    26→        $fromMonth = intval($_POST['from_month']);
    27→        $fromYear = intval($_POST['from_year']);
    28→        $toMonth = intval($_POST['to_month']);
    29→        $toYear = intval($_POST['to_year']);
    30→        $reason = sanitize($_POST['reason'] ?? '');
    31→        
    32→        // Get employee details
    33→        $employee = $db->fetch(
    34→            "SELECT e.*, ess.basic_da, ess.hra, ess.gross_salary, ess.effective_from
    35→             FROM employees e
    36→             LEFT JOIN employee_salary_structures ess ON e.id = ess.employee_id 
    37→                AND (ess.effective_to IS NULL OR ess.effective_to >= CURDATE())
    38→             WHERE e.id = :id",
    39→            ['id' => $employeeId]
    40→        );
    41→        
    42→        if (!$employee) {
    43→            setFlash('error', 'Employee not found');
    44→            redirect(ARREARS_PAGE_URL);
    45→        }
    46→        
    47→        // Calculate months between from and to
    48→        $startDate = new DateTime("$fromYear-$fromMonth-01");
    49→        $endDate = new DateTime("$toYear-$toMonth-01");
    50→        $interval = $startDate->diff($endDate);
    51→        $monthsCount = ($interval->y * 12) + $interval->m + 1;
    52→        
    53→        // Get old and new salary based on arrear type
    54→        $oldBasicDa = 0;
    55→        $newBasicDa = floatval($employee['basic_da'] ?? $employee['basic_salary'] ?? 0);
    56→        $oldHRA = 0;
    57→        $newHRA = floatval($employee['hra'] ?? 0);
    58→        $difference = 0;
    59→        
    60→        if ($arrearType === 'salary_revision') {
    61→            // Get previous salary structure
    62→            $prevSalary = $db->fetch(
    63→                "SELECT * FROM employee_salary_structures 
    64→                 WHERE employee_id = :id AND effective_to IS NOT NULL 
    65→                 ORDER BY effective_to DESC LIMIT 1",
    66→                ['id' => $employeeId]
    67→            );
    68→            
    69→            if ($prevSalary) {
    70→                $oldBasicDa = floatval($prevSalary['basic_da']);
    71→                $oldHRA = floatval($prevSalary['hra']);
    72→            }
    73→            
    74→            $difference = (($newBasicDa - $oldBasicDa) + ($newHRA - $oldHRA)) * $monthsCount;
    75→        } elseif ($arrearType === 'minimum_wage') {
    76→            // Get minimum wage revision difference
    77→            $oldWage = floatval($_POST['old_wage'] ?? 0);
    78→            $newWage = floatval($_POST['new_wage'] ?? $newBasicDa);
    79→            $difference = ($newWage - $oldWage) * $monthsCount;
    80→        } elseif ($arrearType === 'correction') {
    81→            // Direct arrear amount
    82→            $difference = floatval($_POST['arrear_amount'] ?? 0);
    83→            $monthsCount = 1;
    84→        }
    85→        
    86→        // Calculate statutory deductions on arrear
    87→        $pfArrear = 0;
    88→        $esiArrear = 0;
    89→        $ptArrear = 0;
    90→        
    91→        if ($employee['is_pf_applicable'] && $difference > 0) {
    92→            $pfArrear = $difference * 0.12; // 12% PF
    93→        }
    94→        
    95→        if ($employee['is_esi_applicable'] && ($employee['gross_salary'] ?? 0) <= 21000) {
    96→            $esiArrear = $difference * 0.0075; // 0.75% ESI
    97→        }
    98→        
    99→        // PT on arrear (typically not applicable on arrears)
   100→        $ptArrear = 0;
   101→        
   102→        $netArrear = $difference - $pfArrear - $esiArrear - $ptArrear;
   103→        
   104→        // Save arrear record
   105→        $arrearData = [
   106→            'employee_id' => $employeeId,
   107→            'arrear_type' => $arrearType,
   108→            'from_month' => $fromMonth,
   109→            'from_year' => $fromYear,
   110→            'to_month' => $toMonth,
   111→            'to_year' => $toYear,
   112→            'months_count' => $monthsCount,
   113→            'old_basic' => $oldBasicDa,
   114→            'new_basic' => $newBasicDa,
   115→            'old_hra' => $oldHRA,
   116→            'new_hra' => $newHRA,
   117→            'gross_arrear' => $difference,
   118→            'pf_arrear' => $pfArrear,
   119→            'esi_arrear' => $esiArrear,
   120→            'pt_arrear' => $ptArrear,
   121→            'net_arrear' => $netArrear,
   122→            'reason' => $reason,
   123→            'status' => 'pending',
   124→            'created_by' => $_SESSION['user_id'],
   125→            'created_at' => date(DATETIME_FORMAT_DB)
   126→        ];
   127→        
   128→        try {
   129→            $arrearId = $db->insert('employee_arrears', $arrearData);
   130→            setFlash('success', "Arrear calculated successfully. Net Arrear: ₹" . number_format($netArrear, 2));
   131→            redirect(ARREARS_PAGE_URL);
   132→        } catch (Exception $e) {
   133→            setFlash('error', 'Error saving arrear: ' . $e->getMessage());
   134→        }
   135→    }
   136→

   140→        $paymentYear = intval($_POST['payment_year']);
   141→        
   142→        // Get arrear details
   143→        $arrear = $db->fetch(
   144→            "SELECT * FROM employee_arrears WHERE id = :id",
   145→            ['id' => $arrearId]
   146→        );
   147→        
   148→        if (!$arrear) {
   149→            setFlash('error', 'Arrear not found');
   150→            redirect(ARREARS_PAGE_URL);
   151→        }
   152→        
   153→        // Get or create payroll period
   154→        $period = $db->fetch(
   155→            "SELECT * FROM payroll_periods WHERE month = :month AND year = :year",
   156→            ['month' => $paymentMonth, 'year' => $paymentYear]
   157→        );
   158→        
   159→        if (!$period) {
   160→            // Create payroll period
   161→            $startDate = date('Y-m-01', strtotime("$paymentYear-$paymentMonth-01"));
   162→            $endDate = date('Y-m-t', strtotime("$paymentYear-$paymentMonth-01"));
   163→            
   164→            $periodId = $db->insert('payroll_periods', [
   165→                'period_name' => date('F Y', strtotime("$paymentYear-$paymentMonth-01")),
   166→                'month' => $paymentMonth,
   167→                'year' => $paymentYear,
   168→                'start_date' => $startDate,
   169→                'end_date' => $endDate,
   170→                'status' => 'draft',
   171→                'created_by' => $_SESSION['user_id'],
   172→                'created_at' => date(DATETIME_FORMAT_DB)
   173→            ]);
   174→        } else {
   175→            $periodId = $period['id'];
   176→        }
   177→        
   178→        // Add arrear to payroll
   179→        $existingPayroll = $db->fetch(
   180→            "SELECT * FROM payroll WHERE payroll_period_id = :period_id AND employee_id = :emp_id",
   181→            ['period_id' => $periodId, 'emp_id' => $arrear['employee_id']]
   182→        );
   183→        
   184→        if ($existingPayroll) {
   185→            // Update existing payroll
   186→            $db->update('payroll', [
   187→                'arrears' => $existingPayroll['arrears'] + $arrear['gross_arrear'],
   188→                'pf_employee' => $existingPayroll['pf_employee'] + $arrear['pf_arrear'],
   189→                'esi_employee' => $existingPayroll['esi_employee'] + $arrear['esi_arrear'],
   190→                'net_salary' => $existingPayroll['net_salary'] + $arrear['net_arrear'],
   191→                'updated_at' => date(DATETIME_FORMAT_DB)
   192→            ], 'id = :id', ['id' => $existingPayroll['id']]);
   193→        } else {
   194→            // Create new payroll record with arrear
   195→            $db->insert('payroll', [
   196→                'payroll_period_id' => $periodId,
   197→                'employee_id' => $arrear['employee_id'],
   198→                'arrears' => $arrear['gross_arrear'],
   199→                'pf_employee' => $arrear['pf_arrear'],
   200→                'esi_employee' => $arrear['esi_arrear'],
   201→                'net_salary' => $arrear['net_arrear'],
   202→                'payment_status' => 'pending',
   203→                'created_at' => date(DATETIME_FORMAT_DB)
   204→            ]);
   205→        }
   206→        
   207→        // Update arrear status
   208→        $db->update('employee_arrears', [
   209→            'status' => 'approved',
   210→            'approved_by' => $_SESSION['user_id'],
   211→            'approved_at' => date(DATETIME_FORMAT_DB),
   212→            'payment_period_id' => $periodId
   213→        ], 'id = :id', ['id' => $arrearId]);
   214→        
   215→        setFlash('success', 'Arrear approved and added to payroll');
   216→        redirect(ARREARS_PAGE_URL);
   217→    }
   218→}
