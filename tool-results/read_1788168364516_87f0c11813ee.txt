    54→                ['status' => STATUS_PENDING . '%']

    56→            $counts['pending'] = (int)($result['count'] ?? 0);
    57→            
    58→            // Inactive employees
    59→            $result = $this->db->fetch(
    60→                "SELECT COUNT(*) as count FROM employees WHERE status = :status",
    61→                ['status' => STATUS_INACTIVE]

    63→            $counts['inactive'] = (int)($result['count'] ?? 0);
    64→            
    65→            // Left employees
    66→            $result = $this->db->fetch(
    67→                "SELECT COUNT(*) as count FROM employees WHERE date_of_leaving IS NOT NULL"

    78→    // Get all employees with filters
    79→    public function getAll($filters = [], $page = 1, $perPage = DEFAULT_PAGE_SIZE) {
    80→        // Enforce integer types for pagination (defense-in-depth)
    81→        $page = max(1, (int)$page);
    82→        $perPage = max(1, min(500, (int)$perPage));
    83→
    84→        $baseColumns = $this->getBaseColumns();
    85→        
    86→        $sql = "SELECT $baseColumns,
    87→                       ess.basic_da, ess.hra, ess.leave_encashment, ess.bonus_encashment,
    88→                       ess.washing_allowance, ess.gross_salary,
    89→                       ess.pf_applicable, ess.esi_applicable, ess.pt_applicable,
    90→                       c.name as client_name_display,
    91→                       u.name as unit_name_display
    92→                FROM employees e
    93→                LEFT JOIN (
    94→                    SELECT employee_id, basic_da, hra, leave_encashment, bonus_encashment,
    95→                           washing_allowance, gross_salary, pf_applicable, esi_applicable, pt_applicable
    96→                    FROM employee_salary_structures
    97→                    WHERE effective_to IS NULL OR effective_to >= CURDATE()
    98→                    GROUP BY employee_id
    99→                ) ess ON e.id = ess.employee_id
   100→                LEFT JOIN clients c ON e.client_id = c.id
   101→                LEFT JOIN units u ON e.unit_id = u.id
   102→                WHERE 1=1";
   103→        $params = [];
   104→
   105→        // Status filter - use strtolower for case-insensitive comparison
   106→        if (!empty($filters['status'])) {
   107→            $statusLower = strtolower($filters['status']);
   108→            if ($statusLower === 'active' || $statusLower === 'approved') {
   109→                $sql .= " AND e.status = :status";
   110→                $params['status'] = STATUS_APPROVED;
   111→            } elseif ($statusLower === 'pending') {
   112→                $sql .= " AND e.status LIKE :status";
   113→                $params['status'] = STATUS_PENDING . '%';
   114→            } else {
   115→                $sql .= " AND e.status = :status";
   116→                $params['status'] = $filters['status'];
   117→            }
   118→        }
   119→
   120→        // Client filter
   121→        if (!empty($filters['client_id'])) {
   122→            $sql .= " AND e.client_id = :client_id";
   123→            $params['client_id'] = $filters['client_id'];
   124→        }
   125→
   126→        // Unit filter
   127→        if (!empty($filters['unit_id'])) {
   128→            $sql .= " AND e.unit_id = :unit_id";
   129→            $params['unit_id'] = $filters['unit_id'];
   130→        }
   131→
   132→        // Worker category filter
   133→        if (!empty($filters['worker_category'])) {
   134→            $sql .= " AND e.worker_category = :worker_category";
   135→            $params['worker_category'] = $filters['worker_category'];
   136→        }
   137→
   138→        // Search filter - use unique parameter names for PDO with emulated prepares disabled
   139→        if (!empty($filters['search'])) {
   140→            $sql .= " AND (e.employee_code LIKE :search1
   141→                      OR e.full_name LIKE :search2
   142→                      OR e.mobile_number LIKE :search3
   143→                      OR e.aadhaar_number LIKE :search4)";
   144→            $searchTerm = '%' . $filters['search'] . '%';
   145→            $params['search1'] = $searchTerm;
   146→            $params['search2'] = $searchTerm;
   147→            $params['search3'] = $searchTerm;
   148→            $params['search4'] = $searchTerm;
   149→        }
   150→
   151→        // Count total
   152→        $countSql = "SELECT COUNT(*) as total FROM ($sql) as t";
   153→        $total = $this->db->fetchColumn($countSql, $params);
   154→
   155→        // Pagination
   156→        $offset = ($page - 1) * $perPage;
   157→        $sql .= " ORDER BY e.created_at DESC LIMIT $perPage OFFSET $offset";
   158→
   159→        $employees = $this->db->fetchAll($sql, $params);
   160→
   161→        return [
   162→            'data' => $employees,
   163→            'total' => $total ?: 0,
   164→            'page' => $page,
   165→            'per_page' => $perPage,
   166→            'total_pages' => ceil(($total ?: 0) / $perPage)
   167→        ];
   168→    }

   259→    // Create new employee
   260→    public function create($data) {
   261→        // Generate employee code if not provided
   262→        if (empty($data['employee_code'])) {
   263→            $data['employee_code'] = $this->generateEmployeeCode();
   264→        }
   265→        
   266→        // Set default status
   267→        $data['status'] = $data['status'] ?? STATUS_PENDING_HR;
   268→        // Set default app_role to 'employee' if not provided
   269→        $data['app_role'] = $data['app_role'] ?? 'employee';
   270→        $data['created_at'] = date(DATETIME_FORMAT_DB);
   271→
   272→        // Map form fields to database columns
   273→        $dbData = $this->mapFormDataToDb($data);
   274→
   275→        try {
   276→            $this->db->beginTransaction();
   277→            
   278→            // Insert employee
   279→            $this->db->insert('employees', $dbData);
   280→            
   281→            // Get the newly created employee ID
   282→            $employeeId = $this->db->lastInsertId();
   283→            
   284→            // Insert salary structure if any salary/statutory data provided
   285→            $hasSalaryData = !empty($data['basic_da']) || !empty($data['basic_salary']) ||
   286→                isset($data['pf_applicable']) || isset($data['esi_applicable']) ||
   287→                isset($data['pt_applicable']) || isset($data['lwf_applicable']) ||
   288→                isset($data['bonus_applicable']) || isset($data['gratuity_applicable']) ||
   289→                isset($data['overtime_applicable']);
   290→            
   291→            if ($hasSalaryData && $employeeId) {
   292→                $salaryData = $this->buildSalaryData($data, $employeeId);
   293→                $this->db->insert('employee_salary_structures', $salaryData);
   294→            }
   295→            
   296→            $this->db->commit();
   297→            
   298→            // Send notification for pending employee registration
   299→            if (strpos($data['status'], 'pending') !== false) {
   300→                $notification = new Notification();
   301→                $notification->notifyNewEmployeeRegistration([
   302→                    'id' => $employeeId,
   303→                    'full_name' => $data['full_name'] ?? '',
   304→                    'employee_code' => $data['employee_code']
   305→                ]);
   306→            }
   307→            
   308→            return ['success' => true, 'message' => MSG_RECORD_CREATED, 'employee_id' => $employeeId];
   309→        } catch (Exception $e) {
   310→            $this->db->rollBack();
   311→            return ['success' => false, 'message' => 'Failed to create employee: ' . $e->getMessage()];
   312→        }
   313→    }

   315→    // Map form data to database columns
   316→    private function mapFormDataToDb($data) {
   317→        $mapped = [];
   318→        
   319→        // Direct mapping - form fields match database columns (removed client_name/unit_name as they don't exist)
   320→        $directFields = [
   321→            'full_name', 'father_name', 'employee_code', 'mobile_number', 'alternate_mobile',
   322→            'email', 'gender', 'date_of_birth', 'marital_status', 'blood_group',
   323→            'aadhaar_number', 'uan_number', 'esic_number',
   324→            'address', 'pin_code', 'state', 'district',
   325→            'bank_name', 'account_number', 'ifsc_code', 'account_holder_name',
   326→            'client_id', 'unit_id', 'app_role',
   327→            'designation', 'department',
   328→            'worker_category', 'employment_type', 'date_of_joining', 'probation_period',
   329→            'nominee_name', 'nominee_relationship', 'nominee_dob', 'nominee_contact',
   330→            'emergency_contact_name', 'emergency_contact_relation',
   331→            'status', 'profile_pic_url', 'profile_pic_cropped_url',
   332→            'aadhaar_front_url', 'aadhaar_back_url', 'bank_document_url'
   333→        ];
   334→        
   335→        foreach ($directFields as $field) {
   336→            if (isset($data[$field])) {
   337→                $mapped[$field] = $data[$field];
   338→            }
   339→        }
   340→        
   341→        // Handle middle_name -> father_name mapping (for backward compatibility)
   342→        if (isset($data['middle_name']) && !isset($data['father_name'])) {
   343→            $mapped['father_name'] = $data['middle_name'];
   344→        }
   345→        
   346→        // Look up client_id from client_name if not provided (for import compatibility)
   347→        if (!empty($data['client_name']) && empty($data['client_id'])) {
   348→            $client = $this->db->fetch(
   349→                "SELECT id FROM clients WHERE name = :name LIMIT 1",
   350→                ['name' => $data['client_name']]
   351→            );
   352→            if ($client) {
   353→                $mapped['client_id'] = $client['id'];
   354→            }
   355→        }
   356→        
   357→        // Look up unit_id from unit_name if not provided (for import compatibility)
   358→        if (!empty($data['unit_name']) && empty($data['unit_id'])) {
   359→            $unit = $this->db->fetch(
   360→                "SELECT id FROM units WHERE name = :name LIMIT 1",
   361→                ['name' => $data['unit_name']]
   362→            );
   363→            if ($unit) {
   364→                $mapped['unit_id'] = $unit['id'];
   365→            }
   366→        }
   367→        
   368→        return $mapped;
   369→    }

   371→    // Update employee
   372→    public function update($id, $data) {
   373→        $employee = $this->getById($id);
   374→
   375→        if (!$employee) {
   376→            return ['success' => false, 'message' => MSG_RECORD_NOT_FOUND];
   377→        }
   378→
   379→        // Map form fields to database columns
   380→        $dbData = $this->mapFormDataToDb($data);
   381→        unset($dbData['id'], $dbData['employee_code'], $dbData['created_at']);
   382→        $dbData['updated_at'] = date(DATETIME_FORMAT_DB);
   383→
   384→        // Extract salary data - include all statutory fields even if salary is not set
   385→        $salaryData = [];
   386→        // Check if any salary or statutory field is present
   387→        $hasSalaryData = isset($data['basic_da']) || isset($data['basic_salary']) ||
   388→            isset($data['pf_applicable']) || isset($data['esi_applicable']) ||
   389→            isset($data['pt_applicable']) || isset($data['lwf_applicable']) ||
   390→            isset($data['bonus_applicable']) || isset($data['gratuity_applicable']);
   391→
   392→        if ($hasSalaryData) {
   393→            $salaryData = [
   394→                'basic_da' => floatval($data['basic_da'] ?? $data['basic_salary'] ?? 0),
   395→                'hra' => floatval($data['hra'] ?? 0),
   396→                'leave_encashment' => floatval($data['leave_encashment'] ?? 0),
   397→                'bonus_encashment' => floatval($data['bonus_encashment'] ?? 0),
   398→                'washing_allowance' => floatval($data['washing_allowance'] ?? 0),
   399→                'gross_salary' => floatval($data['gross_salary'] ?? 0),
   400→                'pf_applicable' => !empty($data['pf_applicable']) ? 1 : 0,
   401→                'esi_applicable' => !empty($data['esi_applicable']) ? 1 : 0,
   402→                'pt_applicable' => !empty($data['pt_applicable']) ? 1 : 0,
   403→                'lwf_applicable' => !empty($data['lwf_applicable']) ? 1 : 0,
   404→                'bonus_applicable' => !empty($data['bonus_applicable']) ? 1 : 0,
   405→                'gratuity_applicable' => !empty($data['gratuity_applicable']) ? 1 : 0,
   406→                'overtime_applicable' => !empty($data['overtime_applicable']) ? 1 : 0,
   407→            ];
   408→        }
   409→
   410→        try {
   411→            $this->db->beginTransaction();
   412→            
   413→            // Update employee
   414→            $this->db->update('employees', $dbData, SQL_WHERE_ID, ['id' => $id]);
   415→            
   416→            // Update salary structure if provided and table exists
   417→            if (!empty($salaryData)) {
   418→                // Check if salary structure exists
   419→                $existing = $this->db->fetch(
   420→                    "SELECT id FROM employee_salary_structures WHERE employee_id = :id AND (effective_to IS NULL OR effective_to >= CURDATE())",
   421→                    ['id' => $id]
   422→                );
   423→                
   424→                if ($existing) {
   425→                    $this->db->update('employee_salary_structures', $salaryData, 'id = :id', ['id' => $existing['id']]);
   426→                } else {
   427→                    $salaryData['employee_id'] = $id;
   428→                    $salaryData['effective_from'] = date(DATE_FORMAT_DB);
   429→                    $this->db->insert('employee_salary_structures', $salaryData);
   430→                }
   431→            }
   432→            
   433→            $this->db->commit();
   434→            return ['success' => true, 'message' => MSG_RECORD_UPDATED];
   435→        } catch (Exception $e) {
   436→            $this->db->rollBack();
   437→            return ['success' => false, 'message' => 'Failed to update employee: ' . $e->getMessage()];
   438→        }
   439→    }

   498→    // Get employees for payroll processing
   499→    public function getActiveForPayroll($clientId = null, $unitId = null) {
   500→        $sql = "SELECT e.id, e.employee_code, e.full_name, e.father_name, e.date_of_joining, 
   501→                       c.name as client_name_display, 
   502→                       u.name as unit_name_display, e.worker_category,
   503→                       ess.basic_da, ess.hra, ess.leave_encashment, ess.bonus_encashment,
   504→                       ess.washing_allowance, ess.gross_salary, ess.pf_applicable, ess.esi_applicable,

   579→    // Import employees from array
   580→    public function importFromData($data, $skipDuplicates = true) {
   581→        $imported = 0;
   582→        $errors = [];
   583→        $duplicates = [];
   584→
   585→        foreach ($data as $row) {
   586→            try {
   587→                // Check for existing employee by mobile or aadhaar
   588→                if (!empty($row['mobile_number'])) {
   589→                    $exists = $this->db->fetch(
   590→                        "SELECT id FROM employees WHERE mobile_number = :mobile",
   591→                        ['mobile' => $row['mobile_number']]
   592→                    );
   593→                    if ($exists && $skipDuplicates) {
   594→                        $duplicates[] = $row['mobile_number'];
   595→                        continue;
   596→                    }
   597→                }
   598→
   599→                $result = $this->create($row);
   600→                if ($result['success']) {
   601→                    $imported++;
   602→                } else {
   603→                    $errors[] = $result['message'];
   604→                }
   605→            } catch (Exception $e) {
   606→                $errors[] = "Error importing row: " . $e->getMessage();
   607→            }
   608→        }
   609→
   610→        return [
   611→            'success' => true,
   612→            'imported' => $imported,
   613→            'duplicates' => count($duplicates),
   614→            'errors' => $errors
   615→        ];
   616→    }
