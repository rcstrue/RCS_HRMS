   566→function markDirty(empId, fieldKey, td) {
   567→    var cellKey = empId + '|' + fieldKey;
   568→    var original = originalData[empId];
   569→    var input = td.querySelector('.bulk-input');
   570→    var currentValue;

   572→        currentValue = input.checked ? '1' : '0';
   573→    } else {
   574→        currentValue = input ? input.value : '';
   575→    }

   577→    if (currentValue !== originalValue) {
   578→        dirtyCells.add(cellKey);
   579→        td.classList.add('cell-changed');
   580→    } else {
   581→        dirtyCells.delete(cellKey);
   582→        td.classList.remove('cell-changed');

   603→function saveAllChanges() {
   604→    if (dirtyCells.size === 0) return;
   605→    var uniqueEmps = new Set();
   606→    dirtyCells.forEach(function(key) { uniqueEmps.add(key.split('|')[0]); });
   607→    var msg = 'Update ' + uniqueEmps.size + ' employee(s) with ' + dirtyCells.size + ' changed field(s). Are you sure?';
   608→    if (!confirm(msg)) return;
   609→
   610→    var btn = document.getElementById('btnSaveAll');
   611→    btn.disabled = true;
   612→    btn.innerHTML = '<i class="bi bi-spinner me-1"></i><span class="spinner-border spinner-border-sm me-1"></span>Saving...';
   613→
   614→    var employeesData = {};
   615→    dirtyCells.forEach(function(cellKey) {
   616→        var parts = cellKey.split('|');
   617→        var empId = parts[0];
   618→        var fieldName = parts[1];
   619→        var td = document.querySelector('td[data-emp-id="' + empId + '"][data-field="' + fieldName + '"]');
   620→        if (!td) return;
   621→        var input = td.querySelector('.bulk-input');
   622→        if (!input) return;
   623→        if (!employeesData[empId]) employeesData[empId] = { id: parseInt(empId), fields: {} };
   624→        if (input.type === 'checkbox') {
   625→            employeesData[empId].fields[fieldName] = input.checked ? '1' : '0';
   626→        } else if (input.type === 'number' || (ALL_COLUMNS[fieldName] && ALL_COLUMNS[fieldName].type === 'number')) {
   627→            employeesData[empId].fields[fieldName] = parseFloat(input.value) || 0;
   628→        } else {
   629→            employeesData[empId].fields[fieldName] = input.value;
   630→        }
   631→    });
   632→
   633→    fetch('index.php?page=api/bulk-edit', {
   634→        method: 'POST',
   635→        headers: { 'Content-Type': 'application/json' },
   636→        body: JSON.stringify({ employees: Object.values(employeesData) })
   637→    })
   638→    .then(function(r) { return r.json(); })
   639→    .then(function(data) {
   640→        if (data.success) {
   641→            Object.keys(employeesData).forEach(function(empId) {
   642→                Object.keys(employeesData[empId].fields).forEach(function(key) {
   643→                    originalData[empId][key] = employeesData[empId].fields[key];
   644→                });
   645→            });
   646→            dirtyCells.clear();
   647→            document.querySelectorAll('.cell-changed').forEach(function(el) { el.classList.remove('cell-changed'); });
   648→            updateSaveButton();
   649→            showSaveStatus('success', data.message || 'Updated ' + data.updated + ' employee(s)');
   650→        } else {
   651→            showSaveStatus('error', data.message || 'Failed to save changes');
   652→            btn.disabled = false;
   653→            updateSaveButton();
   654→        }
   655→    })
   656→    .catch(function(err) {
   657→        showSaveStatus('error', 'Network error. Please try again.');
   658→        btn.disabled = false;
   659→        updateSaveButton();
   660→    });
   661→}

   663→function undoAllChanges() {
   664→    if (dirtyCells.size === 0) return;
   665→    if (!confirm('Undo all changes? This will restore all cells to their original values.')) return;
   666→    dirtyCells.forEach(function(cellKey) {
   667→        var parts = cellKey.split('|');
   668→        var empId = parts[0];
   669→        var fieldName = parts[1];
   670→        var td = document.querySelector('td[data-emp-id="' + empId + '"][data-field="' + fieldName + '"]');
   671→        if (!td) return;
   672→        var input = td.querySelector('.bulk-input');
   673→        if (!input) return;
   674→        var original = originalData[empId][fieldName];
   675→        if (input.type === 'checkbox') {
   676→            input.checked = !!(original == 1 || original === '1' || original === true);
   677→        } else {
   678→            input.value = original || '';
   679→        }
   680→        td.classList.remove('cell-changed');
   681→    });
   682→    dirtyCells.clear();
   683→    updateSaveButton();
   684→    showSaveStatus('info', 'All changes undone.');
   685→}
