   229→   229→   229→     * 
   230→   230→   230→     * @param int $periodId The payroll period ID
   231→   231→   231→     * @param array $filters Optional filters for selective processing
   232→   232→   232→     *                       - client_id: Process only this client's employees
   233→   233→   233→     *                       - unit_id: Process only this unit's employees
   234→   234→   234→     *                       - employee_codes: Array of specific employee codes
   235→   235→   235→     *                       - recalculate_dirty: Only recalculate dirty records
   236→   236→   236→     * @return array Result with success status and processed count
   237→   237→   237→     */
   238→   238→   238→    public function processPayroll($periodId, $filters = []) {
   239→   239→   239→        // Get period info
   240→   240→   240→        $period = $this->db->fetch(
   241→   241→   241→            "SELECT * FROM payroll_periods WHERE id = :id",
   242→   242→   242→            ['id' => $periodId]
   243→   243→   243→        );
   244→   244→   244→
   245→   245→   245→        if (!$period) {
   246→   246→   246→            return ['success' => false, 'message' => 'Payroll period not found.'];
   247→   247→   247→        }
   248→   248→   248→
   249→   249→   249→        // Check if period is frozen
   250→   250→   250→        if ($period['status'] === 'Frozen' || $period['status'] === 'Locked') {
   251→   251→   251→            return ['success' => false, 'message' => 'Payroll period is frozen. Cannot process.'];
   252→   252→   252→        }
   253→   253→   253→
   254→   254→   254→        // Ensure payroll table has extra_days_amount column
   255→   255→   255→        try {
   256→   256→   256→            $checkColumn = $this->db->fetch("SHOW COLUMNS FROM payroll LIKE 'extra_days_amount'");
   257→   257→   257→            if (!$checkColumn) {
   258→   258→   258→                $this->db->exec("ALTER TABLE payroll ADD COLUMN extra_days_amount DECIMAL(10,2) DEFAULT 0.00 AFTER overtime_amount");
   259→   259→   259→            }
   260→   260→   260→        } catch (Exception $e) {
   261→   261→   261→            // Column might already exist or table might not have records
   262→   262→   262→        }
   263→   263→   263→
   264→   264→   264→        // Build employee query with filters
   265→   265→   265→        $sql = "SELECT e.id, e.employee_code, e.full_name, e.date_of_joining,
   266→   266→   266→                       e.date_of_leaving, e.client_id, e.unit_id, e.worker_category,
   267→   267→   267→                       e.bank_name, e.account_number, e.ifsc_code,
   268→   268→   268→                       c.name as client_name, u.name as unit_name,
   269→   269→   269→                       ess.basic_da, ess.hra, ess.leave_encashment, ess.bonus_encashment, ess.washing_allowance,
   270→   270→   270→                       ess.gross_salary, ess.pf_applicable, ess.esi_applicable,
   271→   271→   271→                       ess.pt_applicable, ess.lwf_applicable, ess.overtime_applicable,
   272→   272→   272→                       ess.bonus_applicable, ess.gratuity_applicable,
   273→   273→   273→                       usf.pay_days_type, usf.ot_calculation_type, usf.ot_calculation_on, usf.ot_hours_per_day,
   274→   274→   274→                       u.state as unit_state
   275→   275→   275→                FROM employees e
   276→   276→   276→                INNER JOIN employee_salary_structures ess ON e.id = ess.employee_id
   277→   277→   277→                    AND (ess.effective_to IS NULL OR ess.effective_to >= CURDATE())
   278→   278→   278→                LEFT JOIN clients c ON e.client_id = c.id
   279→   279→   279→                LEFT JOIN units u ON e.unit_id = u.id
   280→   280→   280→                LEFT JOIN unit_salary_formulas usf ON usf.unit_id = e.unit_id
   281→   281→   281→                    AND usf.is_active = 1
   282→   282→   282→                    AND (usf.effective_to IS NULL OR usf.effective_to >= CURDATE())
   283→   283→   283→                WHERE e.status = 'approved'";
   284→   284→   284→
   285→   285→   285→        $params = [];
   286→   286→   286→
   287→   287→   287→        // Apply filters for selective processing
   288→   288→   288→        if (!empty($filters['client_id'])) {
   289→   289→   289→            $sql .= " AND e.client_id = :client_id";
   290→   290→   290→            $params['client_id'] = $filters['client_id'];
   291→   291→   291→        }
   292→   292→   292→
   293→   293→   293→        if (!empty($filters['unit_id'])) {
   294→   294→   294→            $sql .= " AND e.unit_id = :unit_id";
   295→   295→   295→            $params['unit_id'] = $filters['unit_id'];
   296→   296→   296→        }
   297→   297→   297→
   298→   298→   298→        // Employee codes filter - use named params to avoid mixing with positional
   299→   299→   299→        if (!empty($filters['employee_codes']) && is_array($filters['employee_codes'])) {
   300→   300→   300→            $empPlaceholders = [];
   301→   301→   301→            $idx = 1;
   302→   302→   302→            foreach ($filters['employee_codes'] as $code) {
   303→   303→   303→                $key = 'emp_code_' . $idx++;
   304→   304→   304→                $empPlaceholders[] = ':' . $key;
   305→   305→   305→                $params[$key] = $code;
   306→   306→   306→            }
   307→   307→   307→            $sql .= " AND e.employee_code IN (" . implode(',', $empPlaceholders) . ")";
   308→   308→   308→        }
   309→   309→   309→
   310→   310→   310→        // Only recalculate dirty records if specified
   311→   311→   311→        if (!empty($filters['recalculate_dirty'])) {
   312→   312→   312→            $sql .= " AND EXISTS (
   313→   313→   313→                SELECT 1 FROM payroll p 
   314→   314→   314→                WHERE p.employee_id = e.employee_code 
   315→   315→   315→                AND p.payroll_period_id = :period_id_check
   316→   316→   316→                AND p.payroll_dirty = 1
   317→   317→   317→            )";
   318→   318→   318→            $params['period_id_check'] = $periodId;
   319→   319→   319→        }
   320→   320→   320→
   321→   321→   321→        $employees = $this->db->fetchAll($sql, $params);
   322→   322→   322→
   323→   323→   323→        $processed = 0;
   324→   324→   324→        $errors = [];
   325→   325→   325→        $exceptions = [];
   326→   326→   326→        $totalGross = 0;
   327→   327→   327→        $totalNet = 0;
   328→   328→   328→        $totalPF = 0;
   329→   329→   329→        $totalESI = 0;
   330→   330→   330→
   331→   331→   331→        $this->db->beginTransaction();
   332→   332→   332→
   333→   333→   333→        try {
   334→   334→   334→            foreach ($employees as $emp) {
   335→   335→   335→                try {
   336→   336→   336→                    // Check for exceptions
   337→   337→   337→                    $exceptionType = null;
   338→   338→   338→                    
   339→   339→   339→                    // Check if employee joined after period end
   340→   340→   340→                    $periodEndDate = $period['end_date'];
   341→   341→   341→                    if ($emp['date_of_joining'] && strtotime($emp['date_of_joining']) > strtotime($periodEndDate)) {
   342→   342→   342→                        continue;
   343→   343→   343→                    }
   344→   344→   344→
   345→   345→   345→                    // Check for missing salary structure
   346→   346→   346→                    if (empty($emp['basic_da']) && empty($emp['gross_salary'])) {
   347→   347→   347→                        $exceptionType = 'Undefined Salary';
   348→   348→   348→                        $exceptions[] = [
   349→   349→   349→                            'employee_id' => $emp['employee_code'],
   350→   350→   350→                            'employee_name' => $emp['full_name'],
   351→   351→   351→                            'type' => 'Undefined Salary',
   352→   352→   352→                            'message' => 'No salary structure defined'
   353→   353→   353→                        ];
   354→   354→   354→                    }
   355→   355→   355→
   356→   356→   356→                    // Check for missing bank details
   357→   357→   357→                    if (empty($emp['bank_name']) || empty($emp['account_number']) || empty($emp['ifsc_code'])) {
   358→   358→   358→                        if (!$exceptionType) {
   359→   359→   359→                            $exceptionType = 'Missing Bank Details';
   360→   360→   360→                        }
   361→   361→   361→                        $exceptions[] = [
   362→   362→   362→                            'employee_id' => $emp['employee_code'],
   363→   363→   363→                            'employee_name' => $emp['full_name'],
   364→   364→   364→                            'type' => 'Missing Bank Details',
   365→   365→   365→                            'message' => 'Bank account details incomplete'
   366→   366→   366→                        ];
   367→   367→   367→                    }
   368→   368→   368→
   369→   369→   369→                    // Get attendance summary — prefer unit_id match, fallback to any unit (backward compatible)
   370→   370→   370→                    $attendance = $this->db->fetch(
   371→   371→   371→                        "SELECT
   372→   372→   372→                            total_present as present_days,
   373→   373→   373→                            total_extra,
   374→   374→   374→                            overtime_hours,
   375→   375→   375→                            total_wo as weekly_offs,
   376→   376→   376→                            total_paid_days
   377→   377→   377→                        FROM attendance_summary
   378→   378→   378→                        WHERE employee_id = :emp_id
   379→   379→   379→                        AND month = :month
   380→   380→   380→                        AND year = :year
   381→   381→   381→                        ORDER BY unit_id = :unit_id DESC
   382→   382→   382→                        LIMIT 1",
   383→   383→   383→                        ['emp_id' => $emp['id'], 'unit_id' => $emp['unit_id'], 'month' => $period['month'], 'year' => $period['year']]
   384→   384→   384→                    );
   385→   385→   385→
   386→   386→   386→                    // Calculate total days based on unit's pay_days_type setting
   387→   387→   387→                    $payDaysType = $emp['pay_days_type'] ?? 'actual';
   388→   388→   388→                    switch ($payDaysType) {
