    46→// ══════════════════════════════════════════════════════════════
    47→// EditProfilePage — Clean edit form, single Save Changes button
    48→// ══════════════════════════════════════════════════════════════
    49→//
    50→// Design:
    51→//   - All fields render as simple input fields (no per-field buttons)
    52→//   - A small badge next to each label shows "Saves directly" (green)
    53→//     or "Needs approval" (amber) based on resolveEffectiveRule()
    54→//   - One "Save Changes" button at the bottom handles everything:
    55→//     → Free fields are saved directly via onSaveFreeFields
    56→//     → Approval-needed fields are submitted as change requests
    57→//     → A summary toast tells the user what happened
    58→//   - Profile photo: upload button (handled by Save for approval)
    59→
    60→interface EditProfilePageProps {
    61→  employee: Employee;
    62→  pendingChangeRequests?: ChangeRequest[];
    63→  onSaveFreeFields: (fields: Record<string, string | null>) => Promise<{ success: boolean; error?: string }>;
    64→  onSubmitChangeRequest: (data: { field_name: string; old_value: string; new_value: string; reason?: string }) => Promise<{ success: boolean; error?: string }>;
    65→  onBack: () => void;
    66→}
    67→
    68→// Section icon mapping
    69→const SECTION_ICONS: Record<string, React.ElementType> = {
    70→  personal: User,
    71→  employment: Briefcase,
    72→  address: MapPin,
    73→  emergency: Phone,
    74→  nominee: UserCheck,
    75→  sensitive: CreditCard,
    76→  documents: FileText,
    77→};
    78→
    79→export default function EditProfilePage({
    80→  employee,
    81→  pendingChangeRequests = [],
    82→  onSaveFreeFields,
    83→  onSubmitChangeRequest,
    84→  onBack,
    85→}: EditProfilePageProps) {
    86→  const [saving, setSaving] = useState(false);
    87→  const [formValues, setFormValues] = useState<Record<string, string>>({});
    88→  const [uploadingPhoto, setUploadingPhoto] = useState(false);
    89→  const [pendingPhotoUrl, setPendingPhotoUrl] = useState<string | null>(null);
    90→  const fileInputRef = useRef<HTMLInputElement>(null);
    91→
    92→  // ── KYC Document image uploads ──
    93→  // Track pending uploaded URLs for image fields (aadhaar_front_url, aadhaar_back_url, bank_document_url)
    94→  const [pendingImageUrls, setPendingImageUrls] = useState<Record<string, string>>({});
    95→  const [uploadingImageKey, setUploadingImageKey] = useState<string | null>(null);
    96→  const imageInputRefs = useRef<Record<string, HTMLInputElement | null>>({});
    97→
    98→  const pendingFieldNames = useMemo(() => {
    99→    const set = new Set<string>();
   100→    for (const r of pendingChangeRequests) {
   101→      if (r.status === 'pending') set.add(r.field_name);
   102→    }
   103→    return set;
   104→  }, [pendingChangeRequests]);
   105→
   106→  const getValue = (key: string): string => {
   107→    const val = (employee as unknown as Record<string, unknown>)[key];
   108→    return (val as string) || '';
   109→  };
   110→
   111→  // Initialize form with current values for all editable fields
   112→  useEffect(() => {
   113→    const initial: Record<string, string> = {};
   114→    for (const f of FIELD_RULES) {
   115→      if (f.rule !== 'readonly') {
   116→        initial[f.key] = getValue(f.key);
   117→      }
   118→    }
   119→    setFormValues(initial);
   120→  }, [employee]); // eslint-disable-line react-hooks/exhaustive-deps
   121→
   122→  // Only show sections that have at least one editable field
   123→  const sectionsWithFields = useMemo(() => {
   124→    return FIELD_SECTIONS.map(sectionDef => {
   125→      const fields = FIELD_RULES.filter(f => f.section === sectionDef.key && f.rule !== 'readonly');
   126→      return { ...sectionDef, fields };
   127→    }).filter(s => s.fields.length > 0);
   128→  }, []);
   129→
   130→  const updateField = (key: string, value: string) => {
   131→    setFormValues(prev => ({ ...prev, [key]: value }));
   132→  };
   133→
   134→  // Count changes for the save button badge
   135→  const changedCount = useMemo(() => {
   136→    let count = 0;
   137→    for (const f of FIELD_RULES) {
   138→      if (f.rule === 'readonly') continue;
   139→      if (f.inputType === 'photo') continue; // handled separately below
   140→      if (f.inputType === 'image') continue; // handled separately below
   141→      const current = getValue(f.key);
   142→      const newVal = formValues[f.key] ?? '';
   143→      if (newVal !== current) count++;
   144→    }
   145→    // Photo change
   146→    const currentPhoto = getValue('profile_pic_url');
   147→    if (pendingPhotoUrl && pendingPhotoUrl !== currentPhoto) count++;
   148→    // Document image changes
   149→    for (const f of FIELD_RULES) {
   150→      if (f.inputType !== 'image') continue;
   151→      const current = getValue(f.key);
   152→      const pending = pendingImageUrls[f.key];
   153→      if (pending && pending !== current) count++;
   154→    }
   155→    return count;
   156→  }, [formValues, employee, pendingPhotoUrl, pendingImageUrls]); // eslint-disable-line react-hooks/exhaustive-deps
   157→
   158→  // ── Single Save Changes handler ──
   159→  // Auto-routes: free fields → direct save, approval fields → change requests
   160→  const handleSave = async () => {
   161→    setSaving(true);
   162→    try {
   163→      const freeChanged: Record<string, string | null> = {};
   164→      const approvalFields: { field: FieldRule; newVal: string }[] = [];
   165→
   166→      for (const f of FIELD_RULES) {
   167→        if (f.rule === 'readonly') continue;
   168→        if (f.inputType === 'photo') continue; // handle photo separately
   169→        if (f.inputType === 'image') continue; // handle image uploads separately
   170→
   171→        const current = getValue(f.key);
   172→        const newVal = formValues[f.key] ?? '';
   173→        if (newVal === current) continue; // no change
   174→
   175→        const effectiveRule = resolveEffectiveRule(f.rule, current);
   176→
   177→        if (effectiveRule === 'free') {
   178→          freeChanged[f.key] = newVal || null;
   179→        } else if (effectiveRule === 'admin_approval') {
   180→          if (!newVal) continue; // skip empty values for approval fields
   181→          approvalFields.push({ field: f, newVal });
   182→        }
   183→      }
   184→
   185→      // Handle photo (always needs approval)
   186→      const currentPhoto = getValue('profile_pic_url');
   187→      if (pendingPhotoUrl && pendingPhotoUrl !== currentPhoto) {
   188→        approvalFields.push({
   189→          field: FIELD_RULES.find(f => f.key === 'profile_pic_url')!,
   190→          newVal: pendingPhotoUrl,
   191→        });
   192→      }
   193→
   194→      // Handle KYC document image uploads (always need approval)
   195→      for (const f of FIELD_RULES) {
   196→        if (f.inputType !== 'image') continue;
   197→        const current = getValue(f.key);
   198→        const pending = pendingImageUrls[f.key];
   199→        if (pending && pending !== current) {
   200→          approvalFields.push({ field: f, newVal: pending });
   201→        }
   202→      }
   203→
   204→      const totalChanges = Object.keys(freeChanged).length + approvalFields.length;
   205→      if (totalChanges === 0) {
   206→        toast.info('No changes to save.');
   207→        setSaving(false);
   208→        return;
   209→      }
   210→
   211→      // 1. Save free fields directly
   212→      let savedCount = 0;
   213→      if (Object.keys(freeChanged).length > 0) {
   214→        const result = await onSaveFreeFields(freeChanged);
   215→        if (result.success) {
   216→          savedCount = Object.keys(freeChanged).length;
   217→        } else {
   218→          toast.error(result.error || 'Failed to save some fields');
   219→          setSaving(false);
   220→          return;
   221→        }
   222→      }
   223→
   224→      // 2. Submit change requests for approval-needed fields
   225→      let requestCount = 0;
   226→      let requestErrors = 0;
   227→      for (const { field, newVal } of approvalFields) {
   228→        if (pendingFieldNames.has(field.key)) continue; // already pending
   229→        const result = await onSubmitChangeRequest({
   230→          field_name: field.key,
   231→          old_value: getValue(field.key),
   232→          new_value: newVal,
   233→        });
   234→        if (result.success) {
   235→          requestCount++;
   236→        } else {
   237→          requestErrors++;
   238→        }
   239→      }
   240→
   241→      // 3. Summary toast
   242→      if (requestErrors > 0) {
   243→        toast.error(`Some change requests failed. ${savedCount} field${savedCount !== 1 ? 's' : ''} saved, ${requestCount} request${requestCount !== 1 ? 's' : ''} submitted.`);
   244→      } else if (savedCount > 0 && requestCount > 0) {
   245→        toast.success(`${savedCount} field${savedCount !== 1 ? 's' : ''} saved directly, ${requestCount} change request${requestCount !== 1 ? 's' : ''} submitted for approval.`);
   246→      } else if (savedCount > 0) {
   247→        toast.success('Profile updated successfully');
   248→      } else if (requestCount > 0) {
   249→        toast.success(`${requestCount} change request${requestCount !== 1 ? 's' : ''} submitted for HR approval.`);
   250→      }
   251→
   252→      onBack();
   253→    } catch {
   254→      toast.error('An unexpected error occurred');
   255→    } finally {
   256→      setSaving(false);
   257→    }
   258→  };
   259→
   260→  // ── Profile Photo Upload ──
   261→  const handlePhotoSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   262→    const file = e.target.files?.[0];
   263→    if (!file) return;
   264→
   265→    if (!file.type.startsWith('image/')) {
   266→      toast.error('Please select an image file');
   267→      return;
   268→    }
   269→    if (file.size > 5 * 1024 * 1024) {
   270→      toast.error('Image must be less than 5MB');
   271→      return;
   272→    }
   273→
   274→    setUploadingPhoto(true);
   275→    try {
   276→      const formData = new FormData();
   277→      formData.append('photo', file);
   278→      formData.append('employee_id', String(employee.id));
   279→
   280→      const API_BASE = (import.meta as Record<string, Record<string, string>>).env?.VITE_API_BASE_URL ?? '';
   281→      const API_KEY = (import.meta as Record<string, Record<string, string>>).env?.VITE_API_KEY ?? '';
   282→      const token = typeof window !== 'undefined' ? localStorage.getItem('ess_token') || '' : '';
   283→
   284→      const resp = await fetch(`${API_BASE}/api/ess/upload.php`, {
   285→        method: 'POST',
   286→        headers: { 'X-API-KEY': API_KEY, 'Authorization': `Bearer ${token}` },
   287→        body: formData,
   288→      });
   289→
   290→      const json = await resp.json();
   291→      if (json.success && json.data?.url) {
   292→        setPendingPhotoUrl(json.data.url);
   293→        setFormValues(prev => ({ ...prev, profile_pic_url: json.data.url }));
   294→        toast.success('Photo uploaded. Will be submitted for approval when you Save.');
   295→      } else {
   296→        toast.error(json.error || 'Upload failed');
   297→      }
   298→    } catch {
   299→      toast.error('Upload failed. Please try again.');
   300→    } finally {
   301→      setUploadingPhoto(false);
   302→      if (fileInputRef.current) fileInputRef.current.value = '';
   303→    }
   304→  };
   305→
   306→  // ── KYC Document Image Upload ──
   307→  const handleImageSelect = async (fieldKey: string, folder: string, filename: string, e: React.ChangeEvent<HTMLInputElement>) => {
   308→    const file = e.target.files?.[0];
   309→    if (!file) return;
   310→
   311→    if (!file.type.startsWith('image/')) {
   312→      toast.error('Please select an image file');
   313→      return;
   314→    }
   315→    if (file.size > 5 * 1024 * 1024) {
   316→      toast.error('Image must be less than 5MB');
   317→      return;
   318→    }
   319→
   320→    setUploadingImageKey(fieldKey);
   321→    try {
   322→      // Convert file to base64 for upload-base64 endpoint
   323→      const base64Data = await new Promise<string>((resolve, reject) => {
   324→        const reader = new FileReader();
   325→        reader.onload = () => resolve(reader.result as string);
   326→        reader.onerror = reject;
   327→        reader.readAsDataURL(file);
   328→      });
   329→
   330→      const { url, error } = await uploadBase64Image(base64Data, filename, folder);
   331→      if (url) {
   332→        setPendingImageUrls(prev => ({ ...prev, [fieldKey]: url }));
   333→        setFormValues(prev => ({ ...prev, [fieldKey]: url }));
   334→        toast.success('Document uploaded. Will be submitted for approval when you Save.');
   335→      } else {
   336→        toast.error(error || 'Upload failed');
   337→      }
   338→    } catch {
   339→      toast.error('Upload failed. Please try again.');
   340→    } finally {
   341→      setUploadingImageKey(null);
   342→      // Reset file input
   343→      const inputRef = imageInputRefs.current[fieldKey];
   344→      if (inputRef) inputRef.value = '';
   345→    }
   346→  };
   347→
   348→  // Render input based on field type
   349→  const renderInput = (
   350→    field: FieldRule,
   351→    value: string,
   352→    onChange: (v: string) => void,
   353→    disabled: boolean,
   354→  ) => {
   355→    const inputId = `field-${field.key}`;
   356→
   357→    if (field.inputType === 'select' && field.options) {
   358→      const validValue = (value && field.options.includes(value)) ? value : '';
   359→      return (
   360→        <Select value={validValue} onValueChange={onChange} disabled={disabled}>
   361→          <SelectTrigger id={inputId}>
   362→            <SelectValue placeholder={`Select ${field.label.toLowerCase()}`} />
   363→          </SelectTrigger>
   364→          <SelectContent>
   365→            {field.options.map(opt => (
   366→              <SelectItem key={opt} value={opt}>{opt}</SelectItem>
   367→            ))}
   368→          </SelectContent>
   369→        </Select>
   370→      );
   371→    }
   372→
   373→    if (field.inputType === 'textarea') {
   374→      return (
   375→        <Textarea
   376→          id={inputId}
   377→          placeholder={`Enter ${field.label.toLowerCase()}`}
   378→          value={value}
   379→          onChange={e => onChange(e.target.value)}
   380→          rows={3}
   381→          disabled={disabled}
   382→        />
   383→      );
   384→    }
   385→
   386→    if (field.inputType === 'date') {
   387→      return (
   388→        <Input
   389→          id={inputId}
   390→          type="date"
   391→          value={value}
   392→          onChange={e => onChange(e.target.value)}
   393→          disabled={disabled}
   394→        />
   395→      );
   396→    }
   397→
   398→    if (field.inputType === 'tel') {
   399→      return (
   400→        <Input
   401→          id={inputId}
   402→          type="tel"
   403→          placeholder={`Enter ${field.label.toLowerCase()}`}
   404→          value={value}
   405→          onChange={e => onChange(e.target.value)}
   406→          disabled={disabled}
   407→        />
   408→      );
   409→    }
   410→
   411→    if (field.inputType === 'email') {
   412→      return (
   413→        <Input
   414→          id={inputId}
   415→          type="email"
   416→          placeholder={`Enter ${field.label.toLowerCase()}`}
   417→          value={value}
   418→          onChange={e => onChange(e.target.value)}
   419→          disabled={disabled}
   420→        />
   421→      );
   422→    }
   423→
   424→    return (
   425→      <Input
   426→        id={inputId}
   427→        type="text"
   428→        placeholder={`Enter ${field.label.toLowerCase()}`}
   429→        value={value}
   430→        onChange={e => onChange(e.target.value)}
   431→        disabled={disabled}
   432→      />
   433→    );
   434→  };
   435→
   436→  // Render a single field row — clean, no per-field buttons
   437→  const renderField = (field: FieldRule) => {
   438→    if (field.rule === 'readonly') return null;
   439→
   440→    const hasPending = pendingFieldNames.has(field.key);
   441→    const effectiveRule = resolveEffectiveRule(field.rule, getValue(field.key));
   442→    const current = getValue(field.key);
   443→    const newVal = formValues[field.key] ?? '';
   444→    const isChanged = newVal !== current;
   445→
   446→    // ── Special: Profile Photo ──
   447→    if (field.inputType === 'photo') {
   448→      const currentPhoto = current;
   449→      const displayPhoto = pendingPhotoUrl || currentPhoto;
   450→      const photoChanged = !!pendingPhotoUrl && pendingPhotoUrl !== currentPhoto;
   451→
   452→      return (
   453→        <div key={field.key} className="space-y-2">
   454→          <div className="flex items-center justify-between">
   455→            <Label className="text-sm font-medium">{field.label}</Label>
   456→            {hasPending ? (
   457→              <Badge variant="outline" className="bg-amber-50 text-amber-600 border-amber-200 text-[10px] px-1.5 py-0">
   458→                <Clock className="w-3 h-3 mr-0.5" /> Pending
   459→              </Badge>
   460→            ) : (
   461→              <Badge variant="outline" className="bg-amber-50 text-amber-600 border-amber-200 text-[10px] px-1.5 py-0">
   462→                Needs approval
   463→              </Badge>
   464→            )}
   465→          </div>
   466→          <div className="flex items-center gap-4">
   467→            <div className="w-16 h-16 rounded-full bg-gray-100 border-2 border-gray-200 overflow-hidden flex-shrink-0">
   468→              {displayPhoto ? (
   469→                <img
   470→                  src={getFileUrl(displayPhoto)}
   471→                  alt="Profile"
   472→                  className="w-full h-full object-cover"
   473→                  onError={e => { (e.target as HTMLImageElement).style.display = 'none'; }}
   474→                />
   475→              ) : (
   476→                <div className="w-full h-full flex items-center justify-center text-gray-300">
   477→                  <User className="w-7 h-7" />
   478→                </div>
   479→              )}
   480→            </div>
   481→            <div className="flex-1 space-y-1.5">
   482→              {!hasPending && (
   483→                <>
   484→                  <Button
   485→                    type="button"
   486→                    variant="outline"
   487→                    size="sm"
   488→                    className="w-full border-dashed"
   489→                    onClick={() => fileInputRef.current?.click()}
   490→                    disabled={uploadingPhoto}
   491→                  >
   492→                    {uploadingPhoto ? (
   493→                      <Loader2 className="w-3.5 h-3.5 animate-spin mr-1.5" />
   494→                    ) : (
   495→                      <Camera className="w-3.5 h-3.5 mr-1.5" />
   496→                    )}
   497→                    {uploadingPhoto ? 'Uploading...' : 'Change Photo'}
   498→                  </Button>
   499→                  <input
   500→                    ref={fileInputRef}
   501→                    type="file"
   502→                    accept="image/*"
   503→                    className="hidden"
   504→                    onChange={handlePhotoSelect}
   505→                  />
   506→                </>
   507→              )}
   508→              {photoChanged && (
   509→                <p className="text-xs text-emerald-600 flex items-center gap-1">
   510→                  <CheckCircle2 className="w-3 h-3" /> New photo ready
   511→                </p>
   512→              )}
   513→            </div>
   514→          </div>
   515→        </div>
   516→      );
   517→    }
   518→
   519→    // ── Special: KYC Document Image ──
   520→    if (field.inputType === 'image') {
   521→      const currentUrl = current;
   522→      const pendingUrl = pendingImageUrls[field.key];
   523→      const displayUrl = pendingUrl || currentUrl;
   524→      const imageChanged = !!pendingUrl && pendingUrl !== currentUrl;
   525→      const isUploading = uploadingImageKey === field.key;
   526→      const folder = field.uploadFolder || 'documents';
   527→      const filename = field.uploadFilename || `${field.key}.jpg`;
   528→
   529→      return (
   530→        <div key={field.key} className="space-y-2">
   531→          <div className="flex items-center justify-between">
   532→            <Label className="text-sm font-medium">{field.label}</Label>
   533→            {hasPending ? (
   534→              <Badge variant="outline" className="bg-amber-50 text-amber-600 border-amber-200 text-[10px] px-1.5 py-0">
   535→                <Clock className="w-3 h-3 mr-0.5" /> Pending
   536→              </Badge>
   537→            ) : (
   538→              <Badge variant="outline" className="bg-amber-50 text-amber-600 border-amber-200 text-[10px] px-1.5 py-0">
   539→                Needs approval
   540→              </Badge>
   541→            )}
   542→          </div>
   543→          <div className="flex items-center gap-3">
   544→            {/* Document thumbnail */}
   545→            <div className="w-20 h-14 rounded-lg bg-gray-50 border-2 border-dashed border-gray-200 overflow-hidden flex-shrink-0 flex items-center justify-center">
   546→              {displayUrl ? (
   547→                <img
   548→                  src={getFileUrl(displayUrl)}
   549→                  alt={field.label}
   550→                  className="w-full h-full object-cover"
   551→                  onError={e => { (e.target as HTMLImageElement).style.display = 'none'; (e.target as HTMLImageElement).parentElement!.innerHTML = '<div class="w-full h-full flex items-center justify-center text-gray-300"><svg class="w-6 h-6" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="1.5" d="M4 16l4.586-4.586a2 2 0 012.828 0L16 16m-2-2l1.586-1.586a2 2 0 012.828 0L20 14m-6-6h.01M6 20h12a2 2 0 002-2V6a2 2 0 00-2-2H6a2 2 0 00-2 2v12a2 2 0 002 2z"/></svg></div>'; }}
   552→                />
   553→              ) : (
   554→                <ImageIcon className="w-6 h-6 text-gray-300" />
   555→              )}
   556→            </div>
   557→            <div className="flex-1 space-y-1.5">
   558→              {!hasPending && (
   559→                <>
   560→                  <Button
   561→                    type="button"
   562→                    variant="outline"
   563→                    size="sm"
   564→                    className="w-full border-dashed"
   565→                    onClick={() => imageInputRefs.current[field.key]?.click()}
   566→                    disabled={isUploading}

   568→                    {isUploading ? (
   569→                      <Loader2 className="w-3.5 h-3.5 animate-spin mr-1.5" />

   571→                      <Upload className="w-3.5 h-3.5 mr-1.5" />

   575→                    {isUploading ? 'Uploading...' : currentUrl ? 'Replace' : 'Upload'}

   578→                    ref={el => { imageInputRefs.current[field.key] = el; }}
   579→                    type="file"

   582→                    onChange={e => handleImageSelect(field.key, folder, filename, e)}

   588→                  <CheckCircle2 className="w-3 h-3" /> New document ready

   590→              )}
   591→              {currentUrl && !displayUrl?.startsWith('pending') && (
   592→                <p className="text-[11px] text-gray-400 truncate">

   594→                </p>
   595→              )}
   596→            </div>
   597→          </div>
   598→        </div>
   599→      );
   600→    }
   601→
   602→    // ── All other fields — clean single input with badge ──
   603→    const isApproval = effectiveRule === 'admin_approval';
   604→
   605→    return (
   606→      <div key={field.key} className="space-y-1">
   607→        <div className="flex items-center justify-between gap-2">

   609→          <div className="flex items-center gap-1 shrink-0">

   611→              <Badge variant="outline" className="bg-amber-50 text-amber-600 border-amber-200 text-[10px] px-1.5 py-0">
   612→                <Clock className="w-3 h-3 mr-0.5" /> Pending

   614→            ) : isApproval ? (
   615→              <Badge variant="outline" className="bg-amber-50 text-amber-600 border-amber-200 text-[10px] px-1.5 py-0">
   616→                Needs approval
   617→              </Badge>

   619→              <Badge variant="outline" className="bg-emerald-50 text-emerald-600 border-emerald-200 text-[10px] px-1.5 py-0">
   620→                Saves directly

   622→            )}
   623→          </div>
   624→        </div>
   625→        {renderInput(
   626→          field,
   627→          formValues[field.key] || '',
   628→          (v) => updateField(field.key, v),
   629→          hasPending, // disable input if change is already pending
   630→        )}
   631→        {/* Show current value hint for approval fields that already have data */}
   632→        {isApproval && current && !hasPending && (
   633→          <p className="text-[11px] text-gray-400">
   634→            Current: <span className="text-gray-500 font-medium">{current}</span>

   636→        )}
   637→      </div>

   639→  };
   640→
   641→  return (
   642→    <div className="space-y-4 pb-6">
   643→      <PageHeader title="Edit Profile" subtitle="Update employee information" onBack={onBack} />
   644→
   645→      {sectionsWithFields.map(section => {
   646→        const Icon = SECTION_ICONS[section.key] || User;
   647→
   648→        return (
   649→          <Card key={section.key} className="border-0 shadow-sm">
   650→            <CardContent className="p-4 space-y-3">
   651→              <h3 className="text-sm font-semibold text-gray-500 uppercase tracking-wider flex items-center gap-2">
   652→                <Icon className="w-4 h-4" />
   653→                {section.label}
   654→              </h3>
   655→
   656→              {section.fields.map(field => renderField(field))}
   657→            </CardContent>
   658→          </Card>
   659→        );
   660→      })}
   661→
   662→      {/* Info box explaining the two save paths */}
   663→      <div className="flex items-start gap-2.5 px-1 text-xs text-gray-500">
   664→        <AlertCircle className="w-4 h-4 text-gray-400 shrink-0 mt-0.5" />
   665→        <p>
   666→          Fields marked <span className="text-emerald-600 font-medium">"Saves directly"</span> update immediately.
   667→          Fields marked <span className="text-amber-600 font-medium">"Needs approval"</span> will be submitted to HR for review — original values stay unchanged until approved.
   668→        </p>
   669→      </div>
   670→
   671→      {/* Bottom Buttons */}
   672→      <div className="flex gap-3 pt-2">
   673→        <Button
   674→          variant="outline"
   675→          className="flex-1"
   676→          onClick={onBack}
   677→          disabled={saving}
   678→        >
   679→          <X className="w-4 h-4 mr-1.5" />
   680→          Cancel
   681→        </Button>
   682→        <Button
   683→          className="flex-1 bg-emerald-600 hover:bg-emerald-700 text-white"
   684→          onClick={handleSave}
   685→          disabled={saving || changedCount === 0}
   686→        >
   687→          {saving ? (
   688→            <Loader2 className="w-4 h-4 animate-spin mr-1.5" />
   689→          ) : (
   690→            <Save className="w-4 h-4 mr-1.5" />
   691→          )}
   692→          {saving
   693→            ? 'Saving...'
   694→            : changedCount > 0
   695→              ? `Save Changes (${changedCount})`
   696→              : 'Save Changes'
   697→          }
   698→        </Button>
   699→      </div>
   700→    </div>
   701→  );
   702→}
