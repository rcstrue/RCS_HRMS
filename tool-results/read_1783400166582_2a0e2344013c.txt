    69→            $currentMonth = date('n');

   290→    // Get compliance calendar
   291→    public function getComplianceCalendar($month = null, $year = null) {
   292→        $sql = "SELECT cc.*, s.state_name,
   293→                CASE
   294→                    WHEN cc.due_date < CURDATE() THEN 'Overdue'
   295→                    WHEN cc.due_date <= DATE_ADD(CURDATE(), INTERVAL 7 DAY) THEN 'Due Soon'
   296→                    ELSE 'Upcoming'

   299→                LEFT JOIN states s ON cc.state_id = s.id

   301→        $params = [];
   302→
   303→        if ($month && $year) {
   304→            $sql .= " AND MONTH(cc.due_date) = :month AND YEAR(cc.due_date) = :year";

   306→            $params['year'] = $year;
   307→        } elseif ($year) {
   308→            $sql .= " AND YEAR(cc.due_date) = :year";
   309→            $params['year'] = $year;
   310→        }
   311→
   312→        $sql .= " ORDER BY cc.due_date ASC";
   313→
   314→        return $this->db->fetchAll($sql, $params);

   420→    // Calculate PT for given salary and state
   421→    public function calculatePT($grossSalary, $stateId, $gender = 'All') {
   422→        $rate = $this->db->fetch(
   423→            "SELECT pt_amount FROM professional_tax_rates
   424→             WHERE state_id = :state_id
   425→             AND is_active = 1
   426→             AND effective_from <= CURDATE()
   427→             AND salary_from <= :gross
   428→             AND (salary_to IS NULL OR salary_to >= :gross)
   429→             AND (gender_specific = 'All' OR gender_specific = :gender)
   430→             ORDER BY effective_from DESC, salary_from DESC
   431→             LIMIT 1",
   432→            ['state_id' => $stateId, 'gross' => $grossSalary, 'gender' => $gender]
   433→        );
   434→
   435→        return $rate ? (float)$rate['pt_amount'] : 0;
   436→    }

   498→    // Get notifications
   499→    public function getNotifications($limit = 10) {
   500→        $notifications = [];
   501→
   502→        try {
   503→            // Check for pending filings
   504→            $pending = $this->db->fetchAll(
   505→                "SELECT * FROM compliance_filings
   506→                 WHERE status = 'Pending'
   507→                 ORDER BY due_date ASC
   508→                 LIMIT :limit",
   509→                ['limit' => $limit]
   510→            );
   511→
   512→            foreach ($pending as $p) {
   513→                $isOverdue = strtotime($p['due_date']) < time();
   514→                $notifications[] = [
   515→                    'type' => $isOverdue ? 'danger' : 'warning',
   516→                    'title' => $p['compliance_type'] . ' Filing ' . ($isOverdue ? 'Overdue' : 'Pending'),
   517→                    'description' => sprintf(
   518→                        '%s return for %s/%s is %s',
   519→                        $p['compliance_type'],
   520→                        str_pad($p['filing_period_month'], 2, '0', STR_PAD_LEFT),
   521→                        $p['filing_period_year'],
   522→                        $isOverdue ? 'overdue' : 'pending'
   523→                    ),
   524→                    'created_at' => $p['due_date']
   525→                ];
   526→            }
   527→
   528→            // Check for wage updates
   529→            $wageUpdates = $this->db->fetchAll(
   530→                "SELECT s.state_name, MAX(mw.effective_from) as last_update
   531→                 FROM minimum_wages mw
   532→                 JOIN states s ON mw.state_id = s.id
   533→                 WHERE mw.is_active = 1
   534→                 GROUP BY s.id
   535→                 HAVING last_update < DATE_SUB(CURDATE(), INTERVAL 6 MONTH)
   536→                 LIMIT 5"
   537→            );
   538→
   539→            foreach ($wageUpdates as $w) {
   540→                $notifications[] = [
   541→                    'type' => 'info',
   542→                    'title' => 'Minimum Wage Update Needed',
   543→                    'description' => $w['state_name'] . ' minimum wages may need updating. Last update: ' . $w['last_update'],
   544→                    'created_at' => date('Y-m-d')
   545→                ];
   546→            }
   547→
   548→        } catch (Exception $e) {
   549→            error_log('Compliance getNotifications error: ' . $e->getMessage());
   550→        }
   551→
   552→        return array_slice($notifications, 0, $limit);
   553→    }

   555→    // Get monthly summary
   556→    public function getMonthlySummary($month, $year) {
   557→        $summary = [
   558→            'pf_employee' => 0,
   559→            'pf_employer' => 0,
   560→            'eps_employer' => 0,
   561→            'esi_employee' => 0,
   562→            'esi_employer' => 0,
   563→            'professional_tax' => 0,
   564→            'lwf_employee' => 0,
   565→            'lwf_employer' => 0
   566→        ];
   567→
   568→        try {
   569→            $period = $this->db->fetch(
   570→                SQL_GET_PAYROLL_PERIOD,
   571→                ['month' => $month, 'year' => $year]
   572→            );
   573→
   574→            if ($period) {
   575→                $totals = $this->db->fetch(
   576→                    "SELECT
   577→                        SUM(pf_employee) as pf_employee,
   578→                        SUM(pf_employer) as pf_employer,
   579→                        SUM(eps_employer) as eps_employer,
   580→                        SUM(esi_employee) as esi_employee,
   581→                        SUM(esi_employer) as esi_employer,
   582→                        SUM(professional_tax) as professional_tax,
   583→                        SUM(lwf_employee) as lwf_employee,
   584→                        SUM(lwf_employer) as lwf_employer
   585→                     FROM payroll
   586→                     WHERE payroll_period_id = :period_id",
   587→                    ['period_id' => $period['id']]
   588→                );
   589→
   590→                if ($totals) {

   659→        try {
   660→            return $this->db->fetchAll(
   661→                "SELECT e.employee_code, e.full_name, e.father_name, e.designation, e.worker_category,
   662→                        COALESCE(a.present_days, 0) as present_days,
   663→                        COALESCE(a.absent_days, 0) as absent_days,
   664→                        COALESCE(a.weekly_offs, 0) as weekly_offs,
   665→                        COALESCE(a.holidays, 0) as holidays,
   666→                        COALESCE(a.total_working_days, 0) as total_working_days,
   667→                        COALESCE(a.overtime_hours, 0) as total_overtime_hours
   668→                 FROM employees e
   669→                 LEFT JOIN attendance a ON e.id = a.employee_id AND MONTH(a.attendance_date) = :month AND YEAR(a.attendance_date) = :year
   670→                 WHERE e.unit_id = :unit_id
   671→                 AND e.status = 'approved'
   672→                 ORDER BY e.employee_code",
   673→                ['unit_id' => $unitId, 'month' => $month, 'year' => $year]
   674→            );
   675→        } catch (Exception $e) {

   677→            return $this->db->fetchAll(
   678→                "SELECT e.employee_code, e.full_name, e.father_name, e.designation, e.worker_category,
   679→                        0 as present_days, 0 as absent_days, 0 as weekly_offs, 0 as holidays, 0 as total_working_days, 0 as total_overtime_hours
   680→                 FROM employees e
   681→                 WHERE e.unit_id = :unit_id
   682→                 AND e.status = 'approved'
   683→                 ORDER BY e.employee_code",
   684→                ['unit_id' => $unitId]
   685→            );
   686→        }
   687→    }
