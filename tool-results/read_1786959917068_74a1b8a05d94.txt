     4→/**

    14→
    15→$pageTitle = 'Change Request Approvals';
    16→

    18→try {

    29→        reviewed_by INT NULL,

    37→
    38→// ─── POST Actions ────────────────────────────────────────────────────────────
    39→
    40→if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
    41→    // CSRF check
    42→    if (!validateCSRFToken($_POST['csrf_token'] ?? '')) {
    43→        setFlash('error', 'Invalid request. Please refresh the page and try again.');
    44→        redirect($_SERVER['REQUEST_URI'] ?? 'index.php?page=employee/change-requests');
    45→    }
    46→    $action = sanitize($_POST['action']);
    47→
    48→    // ── Whitelist of fields that can be updated on approval ──
    49→    $APPROVAL_FIELDS = [
    50→        'full_name', 'father_name', 'date_of_birth', 'gender',
    51→        'designation', 'department', 'profile_pic_url',
    52→    ];
    53→
    54→    // ── 1. Approve ─────────────────────────────────────────────
    55→    if ($action === 'approve') {
    56→        $id = (int) ($_POST['id'] ?? 0);
    57→        if ($id > 0) {
    58→            $request = $db->fetch(
    59→                "SELECT r.*, e.full_name AS emp_name, e.email, e.mobile_number,
    60→                        e.employee_code, e.designation
    61→                 FROM employee_change_requests r
    62→                 JOIN employees e ON e.id = r.employee_id
    63→                 WHERE r.id = :id AND r.status = 'pending'
    64→                 LIMIT 1",
    65→                ['id' => $id]
    66→            );
    67→
    68→            if ($request) {
    69→                $field = $request['field_name'];
    70→                $newValue = $request['new_value'];

    72→                // Apply the change to employee record if it's a whitelisted field
    73→                if (in_array($field, $APPROVAL_FIELDS, true)) {

    76→                            'employees',
    77→                            [$field => $newValue],
    78→                            'id = :id',

    82→                        setFlash('error', 'Failed to update employee record: ' . $e->getMessage());

    85→                }
    86→
    87→                // Mark request as approved
    88→                $db->update(
    89→                    'employee_change_requests',
    90→                    [
    91→                        'status'      => 'approved',
    92→                        'reviewed_at' => date('Y-m-d H:i:s'),

    94→                    ],

    96→                    ['id' => $id]

   113→                            <div style='padding:25px;background:#f9fafb;border:1px solid #e5e7eb;'>
   114→                                <p>Dear {$request['emp_name']},</p>
   115→                                <p>Your change request has been <strong style='color:#10b981;'>APPROVED</strong>.</p>
   116→                                <table style='width:100%;border-collapse:collapse;margin:15px 0;'>

   118→                                        <td style='padding:8px;border:1px solid #e5e7eb;'>{$fieldLabel}</td></tr>

   121→                                    <tr><td style='padding:8px;border:1px solid #e5e7eb;background:#f3f4f6;font-weight:bold;'>New Value</td>

   151→                setFlash('error', 'Request not found or already processed.');
   152→            }

   199→                            <div style='background:#ef4444;padding:20px;text-align:center;'>
   200→                                <h2 style='color:#fff;margin:0;'>RCS True Facilities Pvt Ltd</h2>
   201→                            </div>
   202→                            <div style='padding:25px;background:#f9fafb;border:1px solid #e5e7eb;'>

   207→                                        <td style='padding:8px;border:1px solid #e5e7eb;'>{$fieldLabel}</td></tr>
   208→                                    <tr><td style='padding:8px;border:1px solid #e5e7eb;background:#f3f4f6;font-weight:bold;'>Requested Value</td>
   209→                                        <td style='padding:8px;border:1px solid #e5e7eb;'>" . htmlspecialchars($request['new_value']) . "</td></tr>
   210→                                    <tr><td style='padding:8px;border:1px solid #e5e7eb;background:#f3f4f6;font-weight:bold;'>Rejection Reason</td>
   211→                                        <td style='padding:8px;border:1px solid #e5e7eb;'>" . htmlspecialchars($reason) . "</td></tr>
   212→                                </table>
   213→                                <p>If you believe this was an error, please contact HR.</p>
   214→                            </div>
   215→                            <p style='text-align:center;color:#9ca3af;font-size:12px;'>This is an automated message from RCS HRMS Pro.</p>

   218→                    }

   220→                    // WhatsApp

   222→                    if (!empty($empMobile)) {
   223→                        $fieldLabel = ucfirst(str_replace('_', ' ', $field));

   225→                        $notif->sendWhatsApp($empMobile, $waMsg);
   226→                    }
   227→                } catch (Exception $e) {
   228→                    error_log('[Change Request Rejection] Notification error: ' . $e->getMessage());
   229→                }
   230→
   231→                if (function_exists('logActivity')) {
   232→                    logActivity('reject_change_request', 'employee', $request['employee_id'],
   233→                        "Rejected change request #{$id}: {$field} — " . substr($reason, 0, 50));
   234→                }
   235→
   236→                setFlash('success', 'Change request rejected. Employee has been notified.');
   237→            } else {
   238→                setFlash('error', 'Request not found or already processed.');
   239→            }
   240→        } elseif ($id > 0 && $reason === '') {
   241→            setFlash('danger', 'Rejection reason is required.');
   242→        }
   243→        redirect('index.php?page=employee/change-requests');
   244→    }
   245→
   246→    // ── 3. Bulk approve ───────────────────────────────────────
   247→    if ($action === 'bulk_approve') {
   248→        $ids = $_POST['selected_ids'] ?? [];
   249→        if (!empty($ids) && is_array($ids)) {
   250→            $APPROVAL_FIELDS = [
   251→                'full_name', 'father_name', 'date_of_birth', 'gender',
   252→                'designation', 'department', 'profile_pic_url',
   253→            ];
   254→            $count = 0;
   255→            $errors = 0;
   256→
   257→            foreach ($ids as $id) {
   258→                $id = (int) $id;
   259→                if ($id <= 0) continue;
   260→
   261→                $request = $db->fetch(
   262→                    "SELECT r.* FROM employee_change_requests r
   263→                     WHERE r.id = :id AND r.status = 'pending'
   264→                     LIMIT 1",
   265→                    ['id' => $id]
   266→                );
   267→
   268→                if (!$request) continue;
   269→
   270→                $field = $request['field_name'];
   271→                $newValue = $request['new_value'];
   272→
   273→                if (in_array($field, $APPROVAL_FIELDS, true)) {
   274→                    try {
   275→                        $db->update('employees', [$field => $newValue], 'id = :id', ['id' => $request['employee_id']]);
   276→                    } catch (Exception $e) {
   277→                        $errors++;
   278→                        continue;
   279→                    }
   280→                }
   281→
   282→                $db->update(
   283→                    'employee_change_requests',
   284→                    [
   285→                        'status'      => 'approved',
   286→                        'reviewed_at' => date('Y-m-d H:i:s'),
   287→                        'reviewed_by' => $_SESSION['user_id'] ?? null,
   288→                    ],
   289→                    'id = :id',
   290→                    ['id' => $id]
   291→                );
   292→                $count++;
   293→            }
   294→
   295→            $msg = "{$count} change request(s) approved.";
   296→            if ($errors > 0) $msg .= " ({$errors} error(s))";
   297→            setFlash('success', $msg);
   298→        } else {
   299→            setFlash('warning', 'No requests selected.');
   300→        }
   301→        redirect('index.php?page=employee/change-requests');
   302→    }
   303→}

   311→
   312→$where  = ['1=1'];
   313→$params = [];
   314→
   315→if ($fStatus !== 'all' && in_array($fStatus, ['pending', 'approved', 'rejected'], true)) {
   316→    $where[]  = 'r.status = :status';
   317→    $params['status'] = $fStatus;

   319→

   323→    $params['search2'] = "%{$fSearch}%";
   324→    $params['search3'] = "%{$fSearch}%";

   327→$whereSql = implode(' AND ', $where);

   329→// ─── Summary counts (independent of filtered query) ────────────────────────────
   330→
   331→$pendingCount  = 0;
   332→$approvedCount = 0;
   333→$rejectedCount = 0;
   334→try {
   335→    $pendingCount  = (int)$db->fetchColumn("SELECT COUNT(*) FROM employee_change_requests WHERE status = 'pending'") ?: 0;
   336→    $approvedCount = (int)$db->fetchColumn("SELECT COUNT(*) FROM employee_change_requests WHERE status = 'approved'") ?: 0;
   337→    $rejectedCount = (int)$db->fetchColumn("SELECT COUNT(*) FROM employee_change_requests WHERE status = 'rejected'") ?: 0;
   338→} catch (Exception $e) {
   339→    error_log('[change-requests] Count queries failed: ' . $e->getMessage());

   341→
   342→// ─── Fetch requests ─────────────────────────────────────────────────────────
   343→
   344→$requests = [];
   345→$dbError = '';

   347→// Strategy 1: Full query with all JOINs
   348→try {
   349→    $requests = $db->fetchAll(

   377→             FROM employee_change_requests r
   378→             JOIN employees e ON e.id = r.employee_id

   380→             WHERE {$whereSql}
   381→             ORDER BY
   382→                CASE WHEN r.status = 'pending' THEN 0 ELSE 1 END,
   383→                r.created_at DESC",

   401→                    CASE WHEN r.status = 'pending' THEN 0 ELSE 1 END,
   402→                    r.created_at DESC",

   404→            );
   405→        } catch (Exception $e3) {

   666→<!-- ── Page-specific JS ────────────────────────────────────────────────── -->
   667→<script>
   668→(function() {
   669→    // Select all checkbox
   670→    const selectAll = document.getElementById('selectAll');
   671→    if (selectAll) {
   672→        selectAll.addEventListener('change', function() {
   673→            document.querySelectorAll('.row-checkbox').forEach(cb => cb.checked = this.checked);
   674→            updateSelectedCount();
   675→        });
   676→    }
   677→
   678→    // Individual checkboxes
   679→    document.querySelectorAll('.row-checkbox').forEach(cb => {
   680→        cb.addEventListener('change', updateSelectedCount);
   681→    });
   682→
   683→    function updateSelectedCount() {
   684→        const count = document.querySelectorAll('.row-checkbox:checked').length;
   685→        const el = document.getElementById('selectedCount');
   686→        if (el) el.textContent = count > 0 ? count + ' selected' : '';
   687→    }
   688→
   689→    // Approve single request
   690→    window.approveRequest = function(id) {
   691→        if (!confirm('Are you sure you want to approve this change request? The employee record will be updated.')) return;
   692→
   693→        const form = document.createElement('form');
   694→        form.method = 'POST';
   695→        form.action = '?page=employee/change-requests<?= $fStatus !== 'all' ? '&status=' . $fStatus : '' ?>';
   696→
   697→        form.appendChild(createInput('action', 'approve'));
   698→        form.appendChild(createInput('id', id));
   699→        form.appendChild(createInput('csrf_token', '<?= htmlspecialchars(getCSRFToken()) ?>'));
   700→
   701→        document.body.appendChild(form);
   702→        form.submit();
   703→    };
   704→
   705→    // Show reject modal
   706→    window.showRejectModal = function(id) {
   707→        document.getElementById('rejectId').value = id;
   708→        document.getElementById('rejection_reason').value = '';
   709→        new bootstrap.Modal(document.getElementById('rejectModal')).show();
   710→    };
   711→
   712→    // Bulk approve
   713→    window.bulkApprove = function() {
   714→        const checked = document.querySelectorAll('.row-checkbox:checked');
   715→        if (checked.length === 0) {
   716→            alert('Please select at least one request.');
   717→            return;
   718→        }
   719→        if (!confirm('Approve ' + checked.length + ' selected change request(s)? Employee records will be updated.')) return;
   720→        document.getElementById('bulkForm').submit();
   721→    };
   722→
   723→    function createInput(name, value) {
   724→        const input = document.createElement('input');
   725→        input.type = 'hidden';
   726→        input.name = name;
   727→        input.value = value;
   728→        return input;
   729→    }
   730→})();
   731→</script>
