    62→// ── Build filter URL helper ───
    63→function buildFilterUrl($overrides = []) {
    64→    $params = [
    65→        'client_id'  => $GLOBALS['filterClient'] ?? 0,
    66→        'unit_id'    => $GLOBALS['filterUnit'] ?? 0,
    67→        'month'      => $GLOBALS['filterMonth'] ?? 0,
    68→        'year'       => $GLOBALS['filterYear'] ?? 0,
    69→        'status'     => $GLOBALS['filterStatus'] ?? '',
    70→        'employee_id'=> $GLOBALS['filterEmployee'] ?? 0,
    71→    ];
    72→    $params = array_merge($params, $overrides);
    73→    $parts = [];
    74→    foreach ($params as $k => $v) {
    75→        if ($v !== '' && $v !== 0) {
    76→            $parts[] = $k . '=' . urlencode($v);
    77→        }
    78→    }
    79→    return $parts ? '&' . implode('&', $parts) : '';
    80→}
