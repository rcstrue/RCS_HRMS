   505→   505→function updateCounts() {
   506→   506→    var allChecks = document.querySelectorAll('.unitCheck');
   507→   507→    var selectedTotal = 0;
   508→   508→    var cityCounts = {};
   509→   509→    
   510→   510→    allChecks.forEach(function(cb) {
   511→   511→        var city = cb.getAttribute('data-city');
   512→   512→        if (!cityCounts[city]) cityCounts[city] = {total: 0, checked: 0};
   513→   513→        cityCounts[city].total++;
   514→   514→        if (cb.checked) {
   515→   515→            selectedTotal++;
   516→   516→            cityCounts[city].checked++;
   517→   517→        }
   518→   518→    });
   519→   519→    
   520→   520→    var countEl = document.getElementById('selectedCount');
   521→   521→    if (countEl) countEl.textContent = selectedTotal;
   522→   522→    
   523→   523→    // Update city checkboxes and counts
   524→   524→    document.querySelectorAll('.cityCheck').forEach(function(cityCb) {
   525→   525→        var city = cityCb.getAttribute('data-city');
   526→   526→        if (cityCounts[city]) {
   527→   527→            var info = cityCounts[city];
   528→   528→            cityCb.checked = (info.checked === info.total);
   529→   529→            var label = cityCb.nextElementSibling;
   530→   530→            if (label) {
   531→   531→                var smallEl = label.querySelector('small');
   532→   532→                if (smallEl) smallEl.textContent = info.checked + '/' + info.total;
   533→   533→            }
   534→   534→        }
   535→   535→    });
   536→   536→}
