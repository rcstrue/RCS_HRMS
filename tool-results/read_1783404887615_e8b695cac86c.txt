    28→    // Check existing session
    29→    private function checkSession() {
    30→        if (isset($_SESSION['user_id'])) {
    31→            $user = $this->db->fetch(
    32→                "SELECT u.id, u.username, u.email, u.role_id, u.first_name, u.last_name, u.is_active,
    33→                        r.role_name, r.role_code
    34→                 FROM users u 
    35→                 LEFT JOIN roles r ON u.role_id = r.id
    36→                 WHERE u.id = :id AND u.is_active = 1",
    37→                ['id' => $_SESSION['user_id']]
    38→            );
    39→            
    40→            if ($user) {
    41→                $this->user = $user;
    42→            }
    43→        }
    44→    }

    46→    // Login user
    47→    public function login($username, $password, $remember = false) {
    48→        // ── Rate limiting / lockout check ──
    49→        $ip = $_SERVER['REMOTE_ADDR'] ?? '0.0.0.0';
    50→        $lockout = $this->_checkLoginLockout($username, $ip);
    51→        if ($lockout) {
    52→            return ['success' => false, 'error' => $lockout];
    53→        }
    54→
    55→        // Get user with role info
    56→        $user = $this->db->fetch(
    57→            "SELECT u.id, u.username, u.email, u.password, u.role_id, u.first_name, u.last_name, u.is_active,
    58→                    r.role_name, r.role_code
    59→             FROM users u 
    60→             LEFT JOIN roles r ON u.role_id = r.id 
    61→             WHERE (u.username = :username OR u.email = :email)",
    62→            ['username' => $username, 'email' => $username]
    63→        );
    64→        
    65→        // Check if user exists
    66→        if (!$user) {
    67→            $this->_recordFailedLogin($username, $ip);
    68→            return ['success' => false, 'error' => 'Invalid username or password.'];
    69→        }
    70→        
    71→        // Check if active
    72→        if (empty($user['is_active'])) {
    73→            return ['success' => false, 'error' => 'Account is inactive.'];
    74→        }
    75→        
    76→        // Verify password
    77→        if (!password_verify($password, $user['password'])) {
    78→            $this->_recordFailedLogin($username, $ip);
    79→            return ['success' => false, 'error' => 'Invalid username or password.'];
    80→        }
    81→
    82→        // Clear failed attempts on successful login
    83→        $this->_clearFailedLogins($username, $ip);
    84→        
    85→        // Set session variables
    86→        $_SESSION['user_id'] = $user['id'];
    87→        $_SESSION['username'] = $user['username'];
    88→        $_SESSION['email'] = $user['email'];
    89→        $_SESSION['role_id'] = $user['role_id'];
    90→        $_SESSION['role_code'] = $user['role_code'] ?? 'worker';
    91→        $_SESSION['role_name'] = $user['role_name'] ?? 'User';
    92→        $_SESSION['first_name'] = $user['first_name'] ?? '';
    93→        $_SESSION['last_name'] = $user['last_name'] ?? '';
    94→        
    95→        // Update last login
    96→        try {
    97→            $this->db->update('users', [
    98→                'last_login' => date('Y-m-d H:i:s')
    99→            ], SQL_WHERE_USER_ID, ['id' => $user['id']]);
   100→        } catch (Exception $e) {
   101→            // Ignore if last_login column doesn't exist
   102→        }
   103→
   104→        // Audit log — login
   105→        if (function_exists('audit_log')) {
   106→            audit_log('login', 'auth', $user['id'], "User '{$user['username']}' logged in");
   107→        }
   108→
   109→        $this->user = $user;
   110→        
   111→        return [
   112→            'success' => true,
   113→            'message' => 'Login successful.',
   114→            'user' => $user
   115→        ];
   116→    }

   118→    // Logout user
   119→    public function logout() {
   120→        // Audit log — logout (session still active at this point)
   121→        if (function_exists('audit_log') && isset($_SESSION['user_id'])) {
   122→            $uid = $_SESSION['user_id'];
   123→            $uname = $_SESSION['username'] ?? 'unknown';
   124→            audit_log('logout', 'auth', $uid, "User '{$uname}' logged out");
   125→        }
   126→        $_SESSION = [];
   127→        session_destroy();
   128→        $this->user = null;
   129→    }

   156→    // Check if user has role level or higher
   157→    public function hasRoleLevel($role) {
   158→        if (!isset($_SESSION['role_code'])) {
   159→            return false;
   160→        }
   161→        
   162→        $userLevel = $this->roleHierarchy[$_SESSION['role_code']] ?? 0;
   163→        $requiredLevel = $this->roleHierarchy[$role] ?? 0;
   164→        
   165→        return $userLevel >= $requiredLevel;
   166→    }

   168→    // Change password
   169→    public function changePassword($userId, $currentPassword, $newPassword) {
   170→        $user = $this->db->fetch(
   171→            "SELECT password FROM users WHERE id = :id",
   172→            ['id' => $userId]
   173→        );
   174→        
   175→        if (!$user || !password_verify($currentPassword, $user['password'])) {
   176→            return ['success' => false, 'message' => 'Current password is incorrect.'];
   177→        }
   178→        
   179→        $hashedPassword = password_hash($newPassword, PASSWORD_BCRYPT, ['cost' => 12]);
   180→        
   181→        $this->db->update('users', [
   182→            'password' => $hashedPassword
   183→        ], SQL_WHERE_USER_ID, ['id' => $userId]);
   184→        
   185→        return ['success' => true, 'message' => 'Password changed successfully.'];
   186→    }

   188→    // Create new user
   189→    public function createUser($data) {
   190→        $exists = $this->db->fetch(
   191→            "SELECT id FROM users WHERE username = :username OR email = :email",
   192→            ['username' => $data['username'], 'email' => $data['email']]
   193→        );
   194→        
   195→        if ($exists) {
   196→            return ['success' => false, 'message' => 'Username or email already exists.'];
   197→        }
   198→        
   199→        $hashedPassword = password_hash($data['password'], PASSWORD_BCRYPT, ['cost' => 12]);
   200→        
   201→        $userId = $this->db->insert('users', [
   202→            'username' => $data['username'],
   203→            'email' => $data['email'],
   204→            'password' => $hashedPassword,
   205→            'role_id' => $data['role_id'] ?? 5,
   206→            'first_name' => $data['first_name'] ?? '',
   207→            'last_name' => $data['last_name'] ?? '',
   208→            'is_active' => 1

   211→        if ($userId) {
   212→            return ['success' => true, 'message' => 'User created successfully.', 'user_id' => $userId];

   215→        return ['success' => false, 'message' => 'Failed to create user.'];
   216→    }

   233→    // Get all users (admin/hr)
   234→    public function getUsers($filters = []) {
   235→        $sql = "SELECT u.id, u.username, u.email, u.role_id, u.is_active, u.last_login, 
   236→                       u.created_at, u.first_name, u.last_name, r.role_name, r.role_code
   237→                FROM users u
   238→                LEFT JOIN roles r ON u.role_id = r.id
   239→                WHERE 1=1";
   240→        $params = [];
   241→        
   242→        if (!empty($filters['role_id'])) {
   243→            $sql .= " AND u.role_id = :role_id";
   244→            $params['role_id'] = $filters['role_id'];
   245→        }
   246→        
   247→        if (isset($filters['is_active'])) {
   248→            $sql .= " AND u.is_active = :is_active";
   249→            $params['is_active'] = $filters['is_active'];
   250→        }
   251→        
   252→        $sql .= " ORDER BY u.created_at DESC";
   253→        
   254→        return $this->db->fetchAll($sql, $params);
   255→    }

   278→    // Update user
   279→    public function updateUser($userId, $data) {
   280→        $allowedFields = ['email', 'role_id', 'first_name', 'last_name', 'phone', 'is_active'];
   281→        $updateData = [];
   282→        
   283→        foreach ($allowedFields as $field) {
   284→            if (isset($data[$field])) {
   285→                $updateData[$field] = $data[$field];
   286→            }
   287→        }
   288→        
   289→        if (empty($updateData)) {
   290→            return ['success' => false, 'message' => 'No data to update.'];
   291→        }
   292→        
   293→        $result = $this->db->update('users', $updateData, SQL_WHERE_USER_ID, ['id' => $userId]);
   294→        
   295→        return ['success' => true, 'message' => 'User updated successfully.'];
   296→    }

   298→    // Delete user
   299→    public function deleteUser($userId) {
   300→        // Prevent deleting self
   301→        if (isset($_SESSION['user_id']) && $userId == $_SESSION['user_id']) {
   302→            return ['success' => false, 'message' => 'Cannot delete your own account.'];
   303→        }
   304→        
   305→        $result = $this->db->delete('users', SQL_WHERE_USER_ID, ['id' => $userId]);
   306→        
   307→        return ['success' => true, 'message' => 'User deleted successfully.'];
   308→    }

   451→    /**
   452→     * Create menu permissions table if not exists (enhanced version with submenus and actions)
   453→     */
   454→    public function ensureMenuPermissionsTable() {
   455→        try {
   456→            // Check if table exists with new structure
   457→            $checkColumn = $this->db->fetch("SHOW COLUMNS FROM role_menu_permissions LIKE 'submenu_key'");
   458→            
   459→            if (!$checkColumn) {
   460→                // Table exists but needs migration - add new columns
   461→                try {
   462→                    $this->db->query("ALTER TABLE role_menu_permissions 
   463→                        ADD COLUMN submenu_key VARCHAR(100) DEFAULT NULL AFTER menu_key,
   464→                        ADD COLUMN can_view TINYINT(1) DEFAULT 1 AFTER is_visible,
   465→                        ADD COLUMN can_add TINYINT(1) DEFAULT 0 AFTER can_view,
   466→                        ADD COLUMN can_edit TINYINT(1) DEFAULT 0 AFTER can_add,
   467→                        ADD COLUMN can_delete TINYINT(1) DEFAULT 0 AFTER can_edit,

   963→    /**
   964→     * Check if the username+IP is currently locked out.
   965→     * Returns an error message string if locked, or null if allowed.
   966→     */
   967→    private function _checkLoginLockout($username, $ip): ?string
   968→    {
   969→        $this->_ensureLoginAttemptsTable();
   970→
   971→        $row = $this->db->fetch(
   972→            "SELECT attempts, locked_until FROM login_attempts WHERE username = :u AND ip = :i",
   973→            ['u' => $username, 'i' => $ip]
   974→        );
   975→
   976→        if (!$row) {

   978→        }
   979→
   980→        // If there is a lockout window, check if it has expired

   990→                }
   991→                return "Account temporarily locked due to too many failed login attempts. Try again in {$mins} minute" . ($mins !== 1 ? 's' : '') . ".";
   992→            }
   993→            // Lockout expired — reset
   994→            $this->_clearFailedLogins($username, $ip);
   995→        }
   996→
   997→        return null;
   998→    }
